TypeScript 装饰器 (Decorators)

什么是装饰器？

装饰器是一种特殊类型的声明，它能够被附加到类声明、方法、访问符、属性或参数上。装饰器使用 @expression 这种形式，其中 expression 必须求值为一个函数，它会在运行时被调用，被装饰的声明信息做为参数传入。

启用装饰器

在 TypeScript 中使用装饰器需要启用实验性功能：

// tsconfig.json
{
  "compilerOptions": {
    "target": "ES5",
    "experimentalDecorators": true,
    "emitDecoratorMetadata": true
  }
}

// 简单的类装饰器
function sealed(constructor: Function) {
    Object.seal(constructor);
    Object.seal(constructor.prototype);
}

@sealed
class Greeter {
    greeting: string;
    
    constructor(message: string) {
        this.greeting = message;
    }
    
    greet() {
        return "Hello, " + this.greeting;
    }
}

// 带参数的类装饰器
function classDecorator<T extends { new(...args: any[]): {} }>(constructor: T) {
    return class extends constructor {
        newProperty = "new property";
        hello = "override";
    };
}

@classDecorator
class Greeter2 {
    property = "property";
    hello: string;
    
    constructor(m: string) {
        this.hello = m;
    }
}

console.log(new Greeter2("world")); // 包含 newProperty 和被覆盖的 hello

基本方法装饰器

// 方法装饰器
function enumerable(value: boolean) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        descriptor.enumerable = value;
    };
}

function log(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;
    
    descriptor.value = function (...args: any[]) {
        console.log(`Calling ${propertyKey} with arguments:`, args);
        const result = originalMethod.apply(this, args);
        console.log(`${propertyKey} returned:`, result);
        return result;
    };
    
    return descriptor;
}

class Calculator {
    @enumerable(false)
    @log
    add(a: number, b: number): number {
        return a + b;
    }
    
    @log
    multiply(a: number, b: number): number {
        return a * b;
    }
}

const calc = new Calculator();
calc.add(2, 3);      // 会打印日志
calc.multiply(4, 5); // 会打印日志

性能监控装饰器

// 性能监控装饰器
function measure(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;
    
    descriptor.value = function (...args: any[]) {
        const start = performance.now();
        const result = originalMethod.apply(this, args);
        const end = performance.now();
        
        console.log(`${propertyKey} execution time: ${end - start} milliseconds`);
        return result;
    };
    
    return descriptor;
}

// 缓存装饰器
function cache(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;
    const cacheMap = new Map();
    
    descriptor.value = function (...args: any[]) {
        const key = JSON.stringify(args);
        
        if (cacheMap.has(key)) {
            console.log(`Cache hit for ${propertyKey}`);
            return cacheMap.get(key);
        }
        
        const result = originalMethod.apply(this, args);
        cacheMap.set(key, result);
        console.log(`Cache miss for ${propertyKey}, result cached`);
        
        return result;
    };
    
    return descriptor;
}

class MathService {
    @measure
    @cache
    fibonacci(n: number): number {
        if (n <= 1) return n;
        return this.fibonacci(n - 1) + this.fibonacci(n - 2);
    }
    
    @measure
    factorial(n: number): number {
        if (n <= 1) return 1;
        return n * this.factorial(n - 1);
    }
}

const mathService = new MathService();
console.log(mathService.fibonacci(10)); // 第一次计算
console.log(mathService.fibonacci(10)); // 使用缓存

基本属性装饰器

// 属性装饰器
function format(formatString: string) {
    return function (target: any, propertyKey: string) {
        let value: string;
        
        const getter = function () {
            return value;
        };
        
        const setter = function (newVal: string) {
            value = formatString.replace('%s', newVal);
        };
        
        Object.defineProperty(target, propertyKey, {
            get: getter,
            set: setter,
            enumerable: true,
            configurable: true
        });
    };
}

// 验证装饰器
function validate(validationFn: (value: any) => boolean, errorMessage: string) {
    return function (target: any, propertyKey: string) {
        let value: any;
        
        const getter = function () {
            return value;
        };
        
        const setter = function (newVal: any) {
            if (!validationFn(newVal)) {
                throw new Error(`${propertyKey}: ${errorMessage}`);
            }
            value = newVal;
        };
        
        Object.defineProperty(target, propertyKey, {
            get: getter,
            set: setter,
            enumerable: true,
            configurable: true
        });
    };
}

class User {
    @format("Hello, %s!")
    greeting: string;
    
    @validate((value: string) => value.length >= 2, "Name must be at least 2 characters")
    name: string;
    
    @validate((value: number) => value >= 0 && value <= 120, "Age must be between 0 and 120")
    age: number;
    
    constructor(name: string, age: number) {
        this.name = name;
        this.age = age;
        this.greeting = name;
    }
}

const user = new User("Alice", 25);
console.log(user.greeting); // "Hello, Alice!"

// user.age = 150; // 抛出错误：Age must be between 0 and 120

自动序列化装饰器

// 序列化装饰器
function serializable(target: any, propertyKey: string) {
    // 标记属性为可序列化
    if (!target.constructor._serializableProperties) {
        target.constructor._serializableProperties = [];
    }
    target.constructor._serializableProperties.push(propertyKey);
}

function exclude(target: any, propertyKey: string) {
    // 标记属性为排除序列化
    if (!target.constructor._excludeProperties) {
        target.constructor._excludeProperties = [];
    }
    target.constructor._excludeProperties.push(propertyKey);
}

// 序列化类装饰器
function Serializable<T extends { new(...args: any[]): {} }>(constructor: T) {
    return class extends constructor {
        toJSON() {
            const obj: any = {};
            const serializableProps = (constructor as any)._serializableProperties || [];
            const excludeProps = (constructor as any)._excludeProperties || [];
            
            for (const key in this) {
                if (this.hasOwnProperty(key)) {
                    if (serializableProps.length > 0) {
                        // 如果有指定可序列化属性，只序列化这些属性
                        if (serializableProps.includes(key)) {
                            obj[key] = (this as any)[key];
                        }
                    } else {
                        // 否则序列化所有属性，除了被排除的
                        if (!excludeProps.includes(key)) {
                            obj[key] = (this as any)[key];
                        }
                    }
                }
            }
            
            return obj;
        }
        
        static fromJSON(json: string) {
            const data = JSON.parse(json);
            const instance = new constructor();
            Object.assign(instance, data);
            return instance;
        }
    };
}

@Serializable
class Product {
    @serializable
    id: number;
    
    @serializable
    name: string;
    
    @serializable
    price: number;
    
    @exclude
    internalId: string;
    
    createdAt: Date;
    
    constructor(id: number, name: string, price: number) {
        this.id = id;
        this.name = name;
        this.price = price;
        this.internalId = `internal_${id}`;
        this.createdAt = new Date();
    }
}

const product = new Product(1, "Laptop", 999);
console.log(JSON.stringify(product)); // 只包含 id, name, price

// 参数装饰器
function required(target: any, propertyKey: string | symbol, parameterIndex: number) {
    const existingRequiredParameters: number[] = Reflect.getOwnMetadata("required", target, propertyKey) || [];
    existingRequiredParameters.push(parameterIndex);
    Reflect.defineMetadata("required", existingRequiredParameters, target, propertyKey);
}

function validate(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const method = descriptor.value;
    
    descriptor.value = function (...args: any[]) {
        const requiredParameters: number[] = Reflect.getOwnMetadata("required", target, propertyKey) || [];
        
        for (const parameterIndex of requiredParameters) {
            if (parameterIndex >= args.length || args[parameterIndex] === undefined || args[parameterIndex] === null) {
                throw new Error(`Missing required argument at index ${parameterIndex}`);
            }
        }
        
        return method.apply(this, args);
    };
}

class UserService {
    @validate
    createUser(@required name: string, @required email: string, age?: number) {
        console.log(`Creating user: ${name}, ${email}, ${age || 'N/A'}`);
    }
    
    @validate
    updateUser(@required id: number, name?: string, email?: string) {
        console.log(`Updating user ${id}: ${name || 'unchanged'}, ${email || 'unchanged'}`);
    }
}

const userService = new UserService();
userService.createUser("Alice", "alice@example.com"); // 正常
// userService.createUser("Bob"); // 抛出错误：Missing required argument at index 1

// 类型转换装饰器
function toNumber(target: any, propertyKey: string | symbol, parameterIndex: number) {
    const existingConvertParameters: number[] = Reflect.getOwnMetadata("convert:number", target, propertyKey) || [];
    existingConvertParameters.push(parameterIndex);
    Reflect.defineMetadata("convert:number", existingConvertParameters, target, propertyKey);
}

function toString(target: any, propertyKey: string | symbol, parameterIndex: number) {
    const existingConvertParameters: number[] = Reflect.getOwnMetadata("convert:string", target, propertyKey) || [];
    existingConvertParameters.push(parameterIndex);
    Reflect.defineMetadata("convert:string", existingConvertParameters, target, propertyKey);
}

function autoConvert(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const method = descriptor.value;
    
    descriptor.value = function (...args: any[]) {
        const numberConversions: number[] = Reflect.getOwnMetadata("convert:number", target, propertyKey) || [];
        const stringConversions: number[] = Reflect.getOwnMetadata("convert:string", target, propertyKey) || [];
        
        // 转换为数字
        for (const parameterIndex of numberConversions) {
            if (parameterIndex < args.length) {
                args[parameterIndex] = Number(args[parameterIndex]);
            }
        }
        
        // 转换为字符串
        for (const parameterIndex of stringConversions) {
            if (parameterIndex < args.length) {
                args[parameterIndex] = String(args[parameterIndex]);
            }
        }
        
        return method.apply(this, args);
    };
}

class CalculatorService {
    @autoConvert
    add(@toNumber a: any, @toNumber b: any): number {
        return a + b;
    }
    
    @autoConvert
    concat(@toString a: any, @toString b: any): string {
        return a + b;
    }
}

const calcService = new CalculatorService();
console.log(calcService.add("5", "3"));      // 8 (自动转换为数字)
console.log(calcService.concat(123, 456));   // "123456" (自动转换为字符串)

访问器装饰器

// 访问器装饰器
function configurable(value: boolean) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        descriptor.configurable = value;
    };
}

function logAccess(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalGetter = descriptor.get;
    const originalSetter = descriptor.set;
    
    if (originalGetter) {
        descriptor.get = function () {
            console.log(`Getting ${propertyKey}`);
            return originalGetter.call(this);
        };
    }
    
    if (originalSetter) {
        descriptor.set = function (value: any) {
            console.log(`Setting ${propertyKey} to ${value}`);
            originalSetter.call(this, value);
        };
    }
}

class Rectangle {
    private _width: number = 0;
    private _height: number = 0;
    
    @configurable(false)
    @logAccess
    get width(): number {
        return this._width;
    }
    
    set width(value: number) {
        if (value < 0) {
            throw new Error("Width cannot be negative");
        }
        this._width = value;
    }
    
    @logAccess
    get height(): number {
        return this._height;
    }
    
    set height(value: number) {
        if (value < 0) {
            throw new Error("Height cannot be negative");
        }
        this._height = value;
    }
    
    get area(): number {
        return this._width * this._height;
    }
}

const rect = new Rectangle();
rect.width = 10;  // 打印：Setting width to 10
rect.height = 5;  // 打印：Setting height to 5
console.log(rect.width);  // 打印：Getting width，然后输出 10

// 依赖注入装饰器
const container = new Map<string, any>();

function Injectable(name: string) {
    return function <T extends { new(...args: any[]): {} }>(constructor: T) {
        container.set(name, constructor);
        return constructor;
    };
}

function Inject(name: string) {
    return function (target: any, propertyKey: string | symbol, parameterIndex: number) {
        const existingTokens = Reflect.getOwnMetadata("inject:tokens", target) || [];
        existingTokens[parameterIndex] = name;
        Reflect.defineMetadata("inject:tokens", existingTokens, target);
    };
}

function createInstance<T>(constructor: new (...args: any[]) => T): T {
    const tokens = Reflect.getOwnMetadata("inject:tokens", constructor) || [];
    const injections = tokens.map((token: string) => {
        const dependency = container.get(token);
        return dependency ? new dependency() : undefined;
    });
    
    return new constructor(...injections);
}

// 服务定义
@Injectable("logger")
class Logger {
    log(message: string) {
        console.log(`[LOG] ${message}`);
    }
}

@Injectable("database")
class Database {
    connect() {
        console.log("Connected to database");
    }
    
    query(sql: string) {
        console.log(`Executing: ${sql}`);
        return [];
    }
}

@Injectable("userService")
class UserService {
    constructor(
        @Inject("logger") private logger: Logger,
        @Inject("database") private database: Database
    ) {}
    
    getUsers() {
        this.logger.log("Getting users");
        this.database.connect();
        return this.database.query("SELECT * FROM users");
    }
}

// 使用依赖注入
const userService = createInstance(UserService);
userService.getUsers();

API 路由装饰器

// 路由装饰器系统
interface RouteInfo {
    method: string;
    path: string;
    handler: string;
}

const routes: RouteInfo[] = [];

function Controller(basePath: string) {
    return function <T extends { new(...args: any[]): {} }>(constructor: T) {
        const controllerRoutes = Reflect.getOwnMetadata("routes", constructor.prototype) || [];
        
        for (const route of controllerRoutes) {
            routes.push({
                ...route,
                path: basePath + route.path
            });
        }
        
        return constructor;
    };
}

function Get(path: string) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        addRoute(target, "GET", path, propertyKey);
    };
}

function Post(path: string) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        addRoute(target, "POST", path, propertyKey);
    };
}

function Put(path: string) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        addRoute(target, "PUT", path, propertyKey);
    };
}

function Delete(path: string) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        addRoute(target, "DELETE", path, propertyKey);
    };
}

function addRoute(target: any, method: string, path: string, handler: string) {
    const existingRoutes = Reflect.getOwnMetadata("routes", target) || [];
    existingRoutes.push({ method, path, handler });
    Reflect.defineMetadata("routes", existingRoutes, target);
}

// 中间件装饰器
function UseMiddleware(middleware: Function) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        const originalMethod = descriptor.value;
        
        descriptor.value = function (...args: any[]) {
            // 执行中间件
            middleware.apply(this, args);
            // 执行原方法
            return originalMethod.apply(this, args);
        };
    };
}

// 身份验证中间件
function authenticate(req: any, res: any, next: any) {
    console.log("Authenticating user...");
    // 身份验证逻辑
    next();
}

function authorize(roles: string[]) {
    return function (req: any, res: any, next: any) {
        console.log(`Checking authorization for roles: ${roles.join(", ")}`);
        // 授权逻辑
        next();
    };
}

// 控制器定义
@Controller("/api/users")
class UserController {
    @Get("/")
    getAllUsers() {
        return { message: "Getting all users" };
    }
    
    @Get("/:id")
    getUserById() {
        return { message: "Getting user by ID" };
    }
    
    @Post("/")
    @UseMiddleware(authenticate)
    createUser() {
        return { message: "Creating user" };
    }
    
    @Put("/:id")
    @UseMiddleware(authenticate)
    @UseMiddleware(authorize(["admin", "moderator"]))
    updateUser() {
        return { message: "Updating user" };
    }
    
    @Delete("/:id")
    @UseMiddleware(authenticate)
    @UseMiddleware(authorize(["admin"]))
    deleteUser() {
        return { message: "Deleting user" };
    }
}

数据验证装饰器

// 验证装饰器系统
interface ValidationRule {
    property: string;
    constraints: any[];
    validator: Function;
    message: string;
}

function IsRequired(message: string = "Field is required") {
    return function (target: any, propertyKey: string) {
        addValidationRule(target, propertyKey, [], (value: any) => {
            return value !== undefined && value !== null && value !== "";
        }, message);
    };
}

function IsEmail(message: string = "Invalid email format") {
    return function (target: any, propertyKey: string) {
        addValidationRule(target, propertyKey, [], (value: string) => {
            const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            return emailRegex.test(value);
        }, message);
    };
}

function MinLength(min: number, message?: string) {
    return function (target: any, propertyKey: string) {
        addValidationRule(target, propertyKey, [min], (value: string) => {
            return value && value.length >= min;
        }, message || `Minimum length is ${min}`);
    };
}

function MaxLength(max: number, message?: string) {
    return function (target: any, propertyKey: string) {
        addValidationRule(target, propertyKey, [max], (value: string) => {
            return !value || value.length <= max;
        }, message || `Maximum length is ${max}`);
    };
}

function IsNumber(message: string = "Must be a number") {
    return function (target: any, propertyKey: string) {
        addValidationRule(target, propertyKey, [], (value: any) => {
            return typeof value === "number" && !isNaN(value);
        }, message);
    };
}

function Range(min: number, max: number, message?: string) {
    return function (target: any, propertyKey: string) {
        addValidationRule(target, propertyKey, [min, max], (value: number) => {
            return value >= min && value <= max;
        }, message || `Value must be between ${min} and ${max}`);
    };
}

function addValidationRule(
    target: any,
    property: string,
    constraints: any[],
    validator: Function,
    message: string
) {
    const existingRules: ValidationRule[] = Reflect.getOwnMetadata("validation:rules", target) || [];
    existingRules.push({ property, constraints, validator, message });
    Reflect.defineMetadata("validation:rules", existingRules, target);
}

// 验证函数
function validate(obj: any): { isValid: boolean; errors: string[] } {
    const rules: ValidationRule[] = Reflect.getOwnMetadata("validation:rules", obj) || [];
    const errors: string[] = [];
    
    for (const rule of rules) {
        const value = obj[rule.property];
        if (!rule.validator(value)) {
            errors.push(`${rule.property}: ${rule.message}`);
        }
    }
    
    return {
        isValid: errors.length === 0,
        errors
    };
}

// 使用验证装饰器
class CreateUserDto {
    @IsRequired()
    @MinLength(2)
    @MaxLength(50)
    name: string;
    
    @IsRequired()
    @IsEmail()
    email: string;
    
    @IsRequired()
    @IsNumber()
    @Range(18, 120)
    age: number;
    
    @MinLength(8)
    password: string;
    
    constructor(name: string, email: string, age: number, password: string) {
        this.name = name;
        this.email = email;
        this.age = age;
        this.password = password;
    }
}

// 测试验证
const validUser = new CreateUserDto("Alice", "alice@example.com", 25, "password123");
const invalidUser = new CreateUserDto("", "invalid-email", 15, "123");

console.log("Valid user:", validate(validUser));
console.log("Invalid user:", validate(invalidUser));

    };

总结

TypeScript 装饰器提供了强大的元编程能力：

  1. 类装饰器：修改或扩展类的行为
  2. 方法装饰器：增强方法功能（日志、缓存、验证等）
  3. 属性装饰器：控制属性的行为
  4. 参数装饰器：处理方法参数
  5. 访问器装饰器：增强 getter/setter

装饰器在现代框架（如 Angular、NestJS）中广泛使用，是构建企业级应用的重要工具。
