装饰器 (Decorators)

TypeScript | 2025.07.30

装饰器是 TypeScript 的一个实验性特性，提供了一种为类声明和成员添加元数据和修改行为的方式。装饰器使用 @expression 的形式，可以应用于类、方法、属性、参数等。

启用装饰器

tsconfig.json 配置

{
  "compilerOptions": {
    "target": "ES5",
    "experimentalDecorators": true,
    "emitDecoratorMetadata": true
  }
}

类装饰器

基本类装饰器

function sealed(constructor: Function) {
    Object.seal(constructor);
    Object.seal(constructor.prototype);
}

@sealed
class Greeter {
    greeting: string;
    
    constructor(message: string) {
        this.greeting = message;
    }
    
    greet() {
        return "Hello, " + this.greeting;
    }
}

console.log(new Greeter("world").greet()); // "Hello, world"

装饰器工厂

function classDecorator<T extends { new(...args: any[]): {} }>(constructor: T) {
    return class extends constructor {
        newProperty = "new property";
        hello = "override";
    }
}

@classDecorator
class Greeter2 {
    property = "property";
    hello: string;
    
    constructor(m: string) {
        this.hello = m;
    }
}

const greeter2 = new Greeter2("world");
console.log((greeter2 as any).newProperty); // "new property"

带参数的类装饰器

function Component(config: { selector: string; template: string }) {
    return function <T extends { new(...args: any[]): {} }>(constructor: T) {
        return class extends constructor {
            selector = config.selector;
            template = config.template;
            
            render() {
                console.log(`Rendering ${this.selector}: ${this.template}`);
            }
        }
    }
}

@Component({
    selector: 'app-user',
    template: '<div>User Component</div>'
})
class UserComponent {
    name: string = 'User';
}

const userComponent = new UserComponent();
(userComponent as any).render(); // Rendering app-user: <div>User Component</div>

方法装饰器

基本方法装饰器

function enumerable(value: boolean) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        descriptor.enumerable = value;
    };
}

class Greeter3 {
    greeting: string;
    
    constructor(message: string) {
        this.greeting = message;
    }
    
    @enumerable(false)
    greet() {
        return "Hello, " + this.greeting;
    }
}

const greeter3 = new Greeter3("world");
console.log(Object.keys(greeter3)); // ['greeting'] - greet 方法不可枚举

日志装饰器

function log(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;
    
    descriptor.value = function (...args: any[]) {
        console.log(`Calling ${propertyKey} with arguments:`, args);
        const result = originalMethod.apply(this, args);
        console.log(`${propertyKey} returned:`, result);
        return result;
    };
    
    return descriptor;
}

class Calculator {
    @log
    add(a: number, b: number): number {
        return a + b;
    }
    
    @log
    multiply(a: number, b: number): number {
        return a * b;
    }
}

const calc = new Calculator();
calc.add(2, 3);      // 输出日志
calc.multiply(4, 5); // 输出日志

性能监控装饰器

function measure(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;
    
    descriptor.value = function (...args: any[]) {
        const start = performance.now();
        const result = originalMethod.apply(this, args);
        const end = performance.now();
        console.log(`${propertyKey} execution time: ${end - start}ms`);
        return result;
    };
    
    return descriptor;
}

class DataProcessor {
    @measure
    processLargeDataset(data: number[]): number {
        // 模拟耗时操作
        return data.reduce((sum, num) => sum + num, 0);
    }
}

const processor = new DataProcessor();
const data = Array.from({ length: 1000000 }, (_, i) => i);
processor.processLargeDataset(data);

属性装饰器

基本属性装饰器

function format(formatString: string) {
    return function (target: any, propertyKey: string) {
        let value: string;
        
        const getter = function () {
            return value;
        };
        
        const setter = function (newVal: string) {
            value = formatString.replace('%s', newVal);
        };
        
        Object.defineProperty(target, propertyKey, {
            get: getter,
            set: setter,
            enumerable: true,
            configurable: true
        });
    };
}

class Greeter4 {
    @format("Hello, %s!")
    name: string = "";
}

const greeter4 = new Greeter4();
greeter4.name = "world";
console.log(greeter4.name); // "Hello, world!"

验证装饰器

function validate(validationFn: (value: any) => boolean, errorMessage: string) {
    return function (target: any, propertyKey: string) {
        let value: any;
        
        Object.defineProperty(target, propertyKey, {
            get: function () {
                return value;
            },
            set: function (newValue: any) {
                if (!validationFn(newValue)) {
                    throw new Error(`${propertyKey}: ${errorMessage}`);
                }
                value = newValue;
            },
            enumerable: true,
            configurable: true
        });
    };
}

class User {
    @validate((value: string) => value.length >= 3, "Name must be at least 3 characters")
    name: string = "";
    
    @validate((value: number) => value >= 0 && value <= 120, "Age must be between 0 and 120")
    age: number = 0;
    
    @validate((value: string) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value), "Invalid email format")
    email: string = "";
}

const user = new User();
user.name = "Alice";  // 正确
user.age = 25;        // 正确
user.email = "alice@example.com"; // 正确

try {
    user.name = "Al";  // 抛出错误
} catch (error) {
    console.error(error.message); // "name: Name must be at least 3 characters"
}

参数装饰器

基本参数装饰器

function required(target: any, propertyKey: string, parameterIndex: number) {
    const existingRequiredParameters: number[] = Reflect.getOwnMetadata('required', target, propertyKey) || [];
    existingRequiredParameters.push(parameterIndex);
    Reflect.defineMetadata('required', existingRequiredParameters, target, propertyKey);
}

function validate2(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const method = descriptor.value;
    
    descriptor.value = function (...args: any[]) {
        const requiredParameters: number[] = Reflect.getOwnMetadata('required', target, propertyKey) || [];
        
        for (const parameterIndex of requiredParameters) {
            if (args[parameterIndex] === undefined || args[parameterIndex] === null) {
                throw new Error(`Parameter at index ${parameterIndex} is required`);
            }
        }
        
        return method.apply(this, args);
    };
}

class UserService {
    @validate2
    createUser(@required name: string, @required email: string, age?: number) {
        console.log(`Creating user: ${name}, ${email}, ${age}`);
    }
}

const userService = new UserService();
userService.createUser("Alice", "alice@example.com", 25); // 正确

try {
    userService.createUser("Bob", null as any); // 抛出错误
} catch (error) {
    console.error(error.message);
}

多个装饰器的应用

function first() {
    console.log("first(): factory evaluated");
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        console.log("first(): called");
    };
}

function second() {
    console.log("second(): factory evaluated");
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        console.log("second(): called");
    };
}

class ExampleClass {
    @first()
    @second()
    method() {
        console.log("method called");
    }
}

// 输出顺序：
// first(): factory evaluated
// second(): factory evaluated
// second(): called
// first(): called

实例演示：完整的装饰器系统

// 元数据存储
const metadataStorage = new Map<string, any>();

// 路由装饰器
function Controller(basePath: string) {
    return function <T extends { new(...args: any[]): {} }>(constructor: T) {
        Reflect.defineMetadata('basePath', basePath, constructor);
        return constructor;
    };
}

function Get(path: string) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        Reflect.defineMetadata('path', path, target, propertyKey);
        Reflect.defineMetadata('method', 'GET', target, propertyKey);
    };
}

function Post(path: string) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        Reflect.defineMetadata('path', path, target, propertyKey);
        Reflect.defineMetadata('method', 'POST', target, propertyKey);
    };
}

// 参数装饰器
function Body(target: any, propertyKey: string, parameterIndex: number) {
    const existingBodyParameters: number[] = Reflect.getOwnMetadata('body', target, propertyKey) || [];
    existingBodyParameters.push(parameterIndex);
    Reflect.defineMetadata('body', existingBodyParameters, target, propertyKey);
}

function Param(paramName: string) {
    return function (target: any, propertyKey: string, parameterIndex: number) {
        const existingParams: { [key: number]: string } = Reflect.getOwnMetadata('params', target, propertyKey) || {};
        existingParams[parameterIndex] = paramName;
        Reflect.defineMetadata('params', existingParams, target, propertyKey);
    };
}

// 中间件装饰器
function UseMiddleware(middleware: Function) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        const existingMiddleware: Function[] = Reflect.getOwnMetadata('middleware', target, propertyKey) || [];
        existingMiddleware.push(middleware);
        Reflect.defineMetadata('middleware', existingMiddleware, target, propertyKey);
    };
}

// 验证装饰器
function ValidateBody(validationSchema: any) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        const originalMethod = descriptor.value;
        
        descriptor.value = function (...args: any[]) {
            const bodyParameters: number[] = Reflect.getOwnMetadata('body', target, propertyKey) || [];
            
            for (const paramIndex of bodyParameters) {
                const body = args[paramIndex];
                if (!validateSchema(body, validationSchema)) {
                    throw new Error('Validation failed');
                }
            }
            
            return originalMethod.apply(this, args);
        };
    };
}

// 缓存装饰器
function Cache(ttl: number = 60000) {
    const cache = new Map<string, { value: any; timestamp: number }>();
    
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        const originalMethod = descriptor.value;
        
        descriptor.value = function (...args: any[]) {
            const cacheKey = `${target.constructor.name}.${propertyKey}:${JSON.stringify(args)}`;
            const cached = cache.get(cacheKey);
            
            if (cached && Date.now() - cached.timestamp < ttl) {
                console.log(`Cache hit for ${cacheKey}`);
                return cached.value;
            }
            
            const result = originalMethod.apply(this, args);
            cache.set(cacheKey, { value: result, timestamp: Date.now() });
            console.log(`Cache miss for ${cacheKey}`);
            
            return result;
        };
    };
}

// 异步重试装饰器
function Retry(maxAttempts: number = 3, delay: number = 1000) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        const originalMethod = descriptor.value;
        
        descriptor.value = async function (...args: any[]) {
            let lastError: Error;
            
            for (let attempt = 1; attempt <= maxAttempts; attempt++) {
                try {
                    return await originalMethod.apply(this, args);
                } catch (error) {
                    lastError = error as Error;
                    console.log(`Attempt ${attempt} failed: ${error.message}`);
                    
                    if (attempt < maxAttempts) {
                        await new Promise(resolve => setTimeout(resolve, delay));
                    }
                }
            }
            
            throw lastError!;
        };
    };
}

// 权限检查装饰器
function RequireRole(role: string) {
    return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
        const originalMethod = descriptor.value;
        
        descriptor.value = function (...args: any[]) {
            // 假设从某处获取当前用户
            const currentUser = getCurrentUser();
            
            if (!currentUser || !currentUser.roles.includes(role)) {
                throw new Error(`Access denied. Required role: ${role}`);
            }
            
            return originalMethod.apply(this, args);
        };
    };
}

// 示例控制器
@Controller('/api/users')
class UserController {
    private users: any[] = [
        { id: 1, name: 'Alice', email: 'alice@example.com' },
        { id: 2, name: 'Bob', email: 'bob@example.com' }
    ];
    
    @Get('/')
    @Cache(30000)
    @UseMiddleware(authMiddleware)
    getAllUsers() {
        console.log('Fetching all users...');
        return this.users;
    }
    
    @Get('/:id')
    @Cache(60000)
    getUserById(@Param('id') id: string) {
        console.log(`Fetching user with id: ${id}`);
        return this.users.find(user => user.id === parseInt(id));
    }
    
    @Post('/')
    @ValidateBody({ name: 'string', email: 'string' })
    @RequireRole('admin')
    @Retry(3, 500)
    async createUser(@Body body: { name: string; email: string }) {
        console.log('Creating user:', body);
        
        // 模拟可能失败的异步操作
        if (Math.random() < 0.7) {
            throw new Error('Database connection failed');
        }
        
        const newUser = {
            id: this.users.length + 1,
            ...body
        };
        
        this.users.push(newUser);
        return newUser;
    }
}

// 辅助函数
function validateSchema(data: any, schema: any): boolean {
    // 简单的验证逻辑
    for (const [key, type] of Object.entries(schema)) {
        if (typeof data[key] !== type) {
            return false;
        }
    }
    return true;
}

function authMiddleware(req: any, res: any, next: Function) {
    console.log('Auth middleware executed');
    next();
}

function getCurrentUser() {
    // 模拟获取当前用户
    return {
        id: 1,
        name: 'Admin',
        roles: ['admin', 'user']
    };
}

// 路由注册器
class RouteRegistry {
    static registerController(controller: any) {
        const basePath = Reflect.getMetadata('basePath', controller);
        const instance = new controller();
        
        const methods = Object.getOwnPropertyNames(controller.prototype)
            .filter(method => method !== 'constructor');
        
        for (const methodName of methods) {
            const path = Reflect.getMetadata('path', instance, methodName);
            const httpMethod = Reflect.getMetadata('method', instance, methodName);
            const middleware = Reflect.getMetadata('middleware', instance, methodName) || [];
            
            if (path && httpMethod) {
                console.log(`Registered route: ${httpMethod} ${basePath}${path}`);
                console.log(`Middleware: ${middleware.length} functions`);
            }
        }
    }
}

// 注册路由
RouteRegistry.registerController(UserController);
