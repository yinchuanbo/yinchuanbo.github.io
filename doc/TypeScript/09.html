TypeScript | 2025.07.31

联合类型 (Union Types)

基本联合类型

// 基本联合类型
type StringOrNumber = string | number;

function formatValue(value: StringOrNumber): string {
    if (typeof value === "string") {
        return value.toUpperCase();
    } else {
        return value.toString();
    }
}

console.log(formatValue("hello")); // "HELLO"
console.log(formatValue(42));      // "42"

// 多个类型的联合
type Status = "loading" | "success" | "error";
type ID = string | number;

interface User {
    id: ID;
    name: string;
    status: Status;
}

联合类型的类型守卫

// 使用 typeof 进行类型守卫
function processValue(value: string | number | boolean): string {
    if (typeof value === "string") {
        // TypeScript 知道这里 value 是 string
        return value.toLowerCase();
    } else if (typeof value === "number") {
        // TypeScript 知道这里 value 是 number
        return value.toFixed(2);
    } else {
        // TypeScript 知道这里 value 是 boolean
        return value ? "true" : "false";
    }
}

// 使用 in 操作符进行类型守卫
interface Bird {
    fly(): void;
    layEggs(): void;
}

interface Fish {
    swim(): void;
    layEggs(): void;
}

type Pet = Bird | Fish;

function getPetAction(pet: Pet): string {
    if ("fly" in pet) {
        // TypeScript 知道这里 pet 是 Bird
        return "This pet can fly";
    } else {
        // TypeScript 知道这里 pet 是 Fish
        return "This pet can swim";
    }
}

交叉类型 (Intersection Types)

基本交叉类型

// 基本交叉类型
interface Person {
    name: string;
    age: number;
}

interface Employee {
    employeeId: string;
    department: string;
}

type PersonEmployee = Person & Employee;

    name: "Alice",

    employeeId: "EMP001",

混入 (Mixins) 模式

// 混入模式
class Timestamped {
    timestamp = Date.now();
}

class Activatable {
    isActive = false;
    
    activate() {
        this.isActive = true;
    }
    
    deactivate() {
        this.isActive = false;
    }
}

// 创建混入类型
type TimestampedActivatable = Timestamped & Activatable;

// 混入函数
function applyMixins(derivedCtor: any, baseCtors: any[]) {
    baseCtors.forEach(baseCtor => {
        Object.getOwnPropertyNames(baseCtor.prototype).forEach(name => {
            if (name !== 'constructor') {
                derivedCtor.prototype[name] = baseCtor.prototype[name];
            }
        });
    });
}

class SmartObject implements TimestampedActivatable {
    // 需要声明混入的属性和方法
    timestamp!: number;
    isActive!: boolean;
    activate!: () => void;
    deactivate!: () => void;
    
    constructor() {
        // 初始化逻辑
    }
}

// 应用混入
applyMixins(SmartObject, [Timestamped, Activatable]);

const smartObj = new SmartObject();
smartObj.activate();
console.log(smartObj.isActive); // true
console.log(smartObj.timestamp); // 时间戳

条件类型 (Conditional Types)

基本条件类型

// 基本条件类型语法：T extends U ? X : Y
type IsString<T> = T extends string ? true : false;

type Test1 = IsString<string>;  // true
type Test2 = IsString<number>;  // false
type Test3 = IsString<"hello">; // true

// 更实用的条件类型
type NonNullable<T> = T extends null | undefined ? never : T;

type Example1 = NonNullable<string | null>;      // string
type Example2 = NonNullable<number | undefined>; // number
type Example3 = NonNullable<null>;               // never

分布式条件类型

// 分布式条件类型
type ToArray<T> = T extends any ? T[] : never;

type StrArray = ToArray<string>;           // string[]
type NumArray = ToArray<number>;           // number[]
type UnionArray = ToArray<string | number>; // string[] | number[]

// 过滤类型
type Filter<T, U> = T extends U ? T : never;

type Filtered = Filter<string | number | boolean, string | number>; // string | number

// 排除类型
type Exclude<T, U> = T extends U ? never : T;

type Remaining = Exclude<string | number | boolean, string>; // number | boolean

条件类型中的 infer

// 使用 infer 推断类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : any;

function getString(): string { return ""; }
function getNumber(): number { return 0; }

type StringReturn = ReturnType<typeof getString>; // string
type NumberReturn = ReturnType<typeof getNumber>; // number

// 推断数组元素类型
type ArrayElement<T> = T extends (infer U)[] ? U : never;

type StringElement = ArrayElement<string[]>; // string
type NumberElement = ArrayElement<number[]>; // number

// 推断 Promise 的值类型
type PromiseValue<T> = T extends Promise<infer U> ? U : T;

type AsyncString = PromiseValue<Promise<string>>; // string
type SyncString = PromiseValue<string>;           // string

// 推断函数参数类型
type Parameters<T> = T extends (...args: infer P) => any ? P : never;

function example(a: string, b: number, c: boolean): void {}

type ExampleParams = Parameters<typeof example>; // [string, number, boolean]

type Getters<T> = {

模板字面量类型

// 模板字面量类型
type EventName<T extends string> = `on${Capitalize<T>}`;

type ClickEvent = EventName<"click">;   // "onClick"
type HoverEvent = EventName<"hover">;   // "onHover"

// 组合模板字面量类型
type HTTPMethod = "GET" | "POST" | "PUT" | "DELETE";
type Endpoint = "users" | "products" | "orders";

type APIRoute = `/${Endpoint}`;
type APICall = `${HTTPMethod} ${APIRoute}`;

type UserRoutes = APICall; 
// "GET /users" | "POST /users" | "PUT /users" | "DELETE /users" |
// "GET /products" | "POST /products" | "PUT /products" | "DELETE /products" |
// "GET /orders" | "POST /orders" | "PUT /orders" | "DELETE /orders"

// 解析模板字面量类型
type ParseRoute<T extends string> = T extends `/${infer Path}` ? Path : never;

type ParsedRoute = ParseRoute<"/users/123">; // "users/123"

类型守卫 (Type Guards)

自定义类型守卫

// 自定义类型守卫函数
function isString(value: any): value is string {
    return typeof value === "string";
}

function isNumber(value: any): value is number {
    return typeof value === "number";
}

function processValue(value: unknown): string {
    if (isString(value)) {
        // TypeScript 知道这里 value 是 string
        return value.toUpperCase();
    } else if (isNumber(value)) {
        // TypeScript 知道这里 value 是 number
        return value.toString();
    } else {
        return "Unknown type";
    }
}

// 复杂对象的类型守卫
interface Cat {
    name: string;
    meow(): void;
}

interface Dog {
    name: string;
    bark(): void;
}

function isCat(animal: Cat | Dog): animal is Cat {
    return "meow" in animal;
}

function makeSound(animal: Cat | Dog): void {
    if (isCat(animal)) {
        animal.meow(); // TypeScript 知道这是 Cat
    } else {
        animal.bark(); // TypeScript 知道这是 Dog
    }
}

断言函数 (Assertion Functions)

// 断言函数
function assertIsNumber(value: any): asserts value is number {
    if (typeof value !== "number") {
        throw new Error("Expected number");
    }
}

function processNumber(value: unknown): number {
    assertIsNumber(value);
    // 在这里 TypeScript 知道 value 是 number
    return value * 2;
}

// 更复杂的断言函数
function assertIsUser(obj: any): asserts obj is User {
    if (!obj || typeof obj !== "object") {
        throw new Error("Expected object");
    }
    if (typeof obj.id !== "number") {
        throw new Error("Expected id to be number");
    }
    if (typeof obj.name !== "string") {
        throw new Error("Expected name to be string");
    }
    if (typeof obj.email !== "string") {
        throw new Error("Expected email to be string");
    }
}

function processUser(data: unknown): void {
    assertIsUser(data);
    // 在这里 TypeScript 知道 data 是 User
    console.log(`Processing user: ${data.name}`);
}

实际应用示例

API 响应处理

// API 响应类型
interface ApiSuccess<T> {
    success: true;
    data: T;
}

interface ApiError {
    success: false;
    error: string;
    code: number;
}

type ApiResponse<T> = ApiSuccess<T> | ApiError;

// 类型守卫
function isApiSuccess<T>(response: ApiResponse<T>): response is ApiSuccess<T> {
    return response.success === true;
}

// 使用示例
async function fetchUser(id: number): Promise<User | null> {
    const response: ApiResponse<User> = await fetch(`/api/users/${id}`)
        .then(res => res.json());
    
    if (isApiSuccess(response)) {
        return response.data; // TypeScript 知道这里有 data 属性
    } else {
        console.error(`API Error: ${response.error} (Code: ${response.code})`);
        return null;
    }
}

状态管理

// 状态类型
interface LoadingState {
    status: "loading";
}

interface SuccessState<T> {
    status: "success";
    data: T;
}

interface ErrorState {
    status: "error";
    error: string;
}

type AsyncState<T> = LoadingState | SuccessState<T> | ErrorState;

// 状态处理函数
function handleAsyncState<T>(state: AsyncState<T>): string {
    switch (state.status) {
        case "loading":
            return "Loading...";
        case "success":
            return `Data loaded: ${JSON.stringify(state.data)}`;
        case "error":
            return `Error: ${state.error}`;
        default:
            // 确保所有情况都被处理
            const _exhaustiveCheck: never = state;
            return _exhaustiveCheck;
    }
}

// 状态创建函数
function createLoadingState(): LoadingState {
    return { status: "loading" };
}

function createSuccessState<T>(data: T): SuccessState<T> {
    return { status: "success", data };
}

function createErrorState(error: string): ErrorState {
    return { status: "error", error };
}

表单验证

// 验证结果类型
interface ValidationSuccess {
    valid: true;
}

interface ValidationError {
    valid: false;
    errors: string[];
}

type ValidationResult = ValidationSuccess | ValidationError;

// 验证器类型
type Validator<T> = (value: T) => ValidationResult;

// 组合验证器
function combineValidators<T>(...validators: Validator<T>[]): Validator<T> {
    return (value: T): ValidationResult => {
        const errors: string[] = [];
        
        for (const validator of validators) {
            const result = validator(value);
            if (!result.valid) {
                errors.push(...result.errors);
            }
        }
        
        return errors.length === 0 
            ? { valid: true }
            : { valid: false, errors };
    };
}

// 具体验证器
const required: Validator<string> = (value) => 
    value.trim().length > 0 
        ? { valid: true }
        : { valid: false, errors: ["Field is required"] };

const minLength = (min: number): Validator<string> => (value) =>
    value.length >= min
        ? { valid: true }
        : { valid: false, errors: [`Minimum length is ${min}`] };

const email: Validator<string> = (value) => {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return emailRegex.test(value)
        ? { valid: true }
        : { valid: false, errors: ["Invalid email format"] };
};

// 使用示例
const emailValidator = combineValidators(required, email);
const passwordValidator = combineValidators(required, minLength(8));

const emailResult = emailValidator("user@example.com");
const passwordResult = passwordValidator("123");

console.log(emailResult);    // { valid: true }
console.log(passwordResult); // { valid: false, errors: ["Minimum length is 8"] }

最佳实践

1. 优先使用联合类型而不是枚举

// 推荐：联合类型
type Theme = "light" | "dark" | "auto";

// 某些情况下使用枚举
enum Permission {
    Read = 1,
    Write = 2,
    Execute = 4
}

2. 使用类型守卫确保类型安全

// 好的做法
function processData(data: unknown): void {
    if (isUser(data)) {
        // 安全地使用 data
        console.log(data.name);
    }
}

// 避免类型断言
function processDataBad(data: unknown): void {
    const user = data as User; // 不安全
    console.log(user.name);
}

3. 合理使用条件类型

// 创建有用的工具类型
type NonNullable<T> = T extends null | undefined ? never : T;
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : any;

// 避免过度复杂的条件类型
type OverlyComplex<T> = T extends string 
    ? T extends `${infer A}${infer B}` 
        ? A extends "a" 
            ? B extends "b" 
                ? true 
                : false 
            : false 
        : false 
    : false; // 太复杂，难以理解和维护

总结

TypeScript 的高级类型系统提供了强大的工具：

  1. 联合类型：表示多种可能的类型
  2. 交叉类型：组合多个类型的属性
  3. 条件类型：基于条件选择类型
  4. 映射类型：转换现有类型
  5. 类型守卫：运行时类型检查

掌握这些高级类型特性可以编写更加类型安全和灵活的代码。
