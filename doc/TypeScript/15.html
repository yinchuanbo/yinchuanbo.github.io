TypeScript 声明文件 (Declaration Files)

什么是声明文件？

声明文件（.d.ts）是 TypeScript 的类型定义文件，用于为 JavaScript 库提供类型信息。它们不包含实际的实现代码，只包含类型声明。

声明文件的作用

1. 为 JavaScript 库提供类型支持

// 没有声明文件时
const $ = require('jquery'); // any 类型
$('#myElement').fadeIn(); // 没有类型检查和智能提示

// 有声明文件时
import * as $ from 'jquery'; // 有完整的类型信息
$('#myElement').fadeIn(500); // 完整的类型检查和智能提示

2. 模块增强

// 扩展现有模块的类型
declare module 'express' {
  interface Request {
    user?: {
      id: number;
      name: string;
    };
  }
}

// 现在可以安全使用扩展的属性
app.use((req, res, next) => {
  if (req.user) {
    console.log(`User: ${req.user.name}`);
  }
  next();
});

基本声明语法

变量声明

// 声明全局变量
declare var myGlobalVar: string;
declare let myGlobalLet: number;
declare const myGlobalConst: boolean;

// 使用示例
console.log(myGlobalVar); // TypeScript 知道这是 string 类型

函数声明

// 声明全局函数
declare function myGlobalFunction(param: string): number;

// 函数重载声明
declare function createElement(tagName: 'div'): HTMLDivElement;
declare function createElement(tagName: 'span'): HTMLSpanElement;
declare function createElement(tagName: string): HTMLElement;

// 使用示例
const div = createElement('div'); // 类型为 HTMLDivElement
const span = createElement('span'); // 类型为 HTMLSpanElement
const element = createElement('p'); // 类型为 HTMLElement

类声明

// 声明类
declare class MyClass {
  constructor(value: string);
  
  // 属性声明
  public property: string;
  private _privateProperty: number;
  static staticProperty: boolean;
  
  // 方法声明
  method(param: number): string;
  static staticMethod(): void;
}

// 使用示例
const instance = new MyClass('hello');
console.log(instance.property);
MyClass.staticMethod();

接口和类型声明

// 声明接口
declare interface GlobalInterface {
  id: number;
  name: string;
  optional?: boolean;
}

// 声明类型别名
declare type GlobalType = string | number | boolean;

// 声明枚举
declare enum GlobalEnum {
  VALUE1,
  VALUE2,
  VALUE3
}

模块声明

外部模块声明

// 声明整个模块
declare module 'my-library' {
  export interface Config {
    apiUrl: string;
    timeout: number;
  }
  
  export class Client {
    constructor(config: Config);
    request(endpoint: string): Promise<any>;
  }
  
  export function createClient(config: Config): Client;
  
  // 默认导出
  export default Client;
}

// 使用示例
import MyLibrary, { createClient, Config } from 'my-library';

const config: Config = {
  apiUrl: 'https://api.example.com',
  timeout: 5000
};

const client = createClient(config);

高级声明技巧

条件类型声明

// 条件类型声明
declare module 'conditional-library' {
  export type ApiResponse<T> = T extends string 
    ? { message: T } 
    : T extends number 
    ? { code: T } 
    : { data: T };
  
  export function makeRequest<T>(input: T): Promise<ApiResponse<T>>;
}

// 使用示例
import { makeRequest } from 'conditional-library';

makeRequest('hello').then(response => {
  // response 类型为 { message: string }
  console.log(response.message);
});

makeRequest(404).then(response => {
  // response 类型为 { code: number }
  console.log(response.code);
});

泛型约束声明

// 泛型约束声明
declare module 'generic-library' {
  interface Identifiable {
    id: string | number;
  }
  
  export class Repository<T extends Identifiable> {
    constructor();
    
    findById(id: T['id']): Promise<T | null>;
    save(entity: T): Promise<T>;
    delete(id: T['id']): Promise<void>;
  }
  
  export function createRepository<T extends Identifiable>(): Repository<T>;
}

// 使用示例
import { createRepository } from 'generic-library';

interface User {
  id: number;
  name: string;
  email: string;
}

const userRepo = createRepository<User>();
userRepo.findById(1).then(user => {
  if (user) {
    console.log(user.name); // TypeScript 知道 user 的类型
  }
});

模块增强声明

// 增强现有模块

为第三方库创建声明文件

// types/my-chart-library.d.ts
declare module 'my-chart-library' {
  export interface ChartOptions {
    type: 'line' | 'bar' | 'pie';
    data: ChartData;
    options?: {
      responsive?: boolean;
      maintainAspectRatio?: boolean;
      plugins?: {
        legend?: {
          display?: boolean;
          position?: 'top' | 'bottom' | 'left' | 'right';
        };
      };
    };
  }
  
  export interface ChartData {
    labels: string[];
    datasets: Dataset[];
  }
  
  export interface Dataset {
    label: string;
    data: number[];
    backgroundColor?: string | string[];
    borderColor?: string | string[];
    borderWidth?: number;
  }
  
  export class Chart {
    constructor(canvas: HTMLCanvasElement, config: ChartOptions);
    
    update(): void;
    destroy(): void;
    resize(): void;
    
    data: ChartData;
    options: ChartOptions['options'];
  }
  
  export default Chart;
}

// 使用示例
import Chart, { ChartOptions } from 'my-chart-library';

const chartConfig: ChartOptions = {
  type: 'line',
  data: {
    labels: ['Jan', 'Feb', 'Mar'],
    datasets: [{
      label: 'Sales',
      data: [10, 20, 30],
      borderColor: 'blue'
    }]
  },
  options: {
    responsive: true,
    plugins: {
      legend: {
        display: true,
        position: 'top'
      }
    }
  }
};

const canvas = document.getElementById('chart') as HTMLCanvasElement;
const chart = new Chart(canvas, chartConfig);

window.myGlobalLibrary.init({ debug: true });

声明文件的组织

// types/index.d.ts - 主声明文件
export * from './api';
export * from './components';
export * from './utils';

// types/api.d.ts - API 相关声明
export interface ApiResponse<T = any> {
  success: boolean;
  data: T;
  message?: string;
  errors?: string[];
}

export interface PaginatedResponse<T> extends ApiResponse<T[]> {
  pagination: {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
  };
}

export interface ApiClient {
  get<T>(url: string): Promise<ApiResponse<T>>;
  post<T>(url: string, data: any): Promise<ApiResponse<T>>;
  put<T>(url: string, data: any): Promise<ApiResponse<T>>;
  delete(url: string): Promise<ApiResponse<void>>;
}

// types/components.d.ts - 组件相关声明
export interface ComponentProps {
  className?: string;
  children?: React.ReactNode;
}

export interface ButtonProps extends ComponentProps {
  variant?: 'primary' | 'secondary' | 'danger';
  size?: 'small' | 'medium' | 'large';
  disabled?: boolean;
  onClick?: (event: React.MouseEvent) => void;
}

export interface ModalProps extends ComponentProps {
  isOpen: boolean;
  onClose: () => void;
  title?: string;
}

// types/utils.d.ts - 工具函数声明
export interface DateUtils {
  format(date: Date, pattern: string): string;
  parse(dateString: string, pattern: string): Date;
  addDays(date: Date, days: number): Date;
  diffInDays(date1: Date, date2: Date): number;
}

export interface ValidationUtils {
  isEmail(email: string): boolean;
  isPhoneNumber(phone: string): boolean;
  isStrongPassword(password: string): boolean;
}

2. 版本化声明文件

// types/v1/api.d.ts
declare module 'my-api-v1' {
  export interface User {
    id: number;
    name: string;
  }
  
  export function getUser(id: number): Promise<User>;
}

    id: string; // 改为 string

    email: string; // 新增字段
  }

  export function getUserByEmail(email: string): Promise<User>; // 新增方法
}

3. 渐进式类型化

// 开始时使用宽松的类型
declare module 'legacy-library' {
  export function doSomething(input: any): any;
}

// 逐步完善类型定义
declare module 'legacy-library' {
  export interface Input {
    type: string;
    value: any;
  }
  
  export interface Output {
    result: any;
    status: 'success' | 'error';
  }
  
  export function doSomething(input: Input): Output;
}

// 最终的完整类型定义
declare module 'legacy-library' {
  export interface StringInput {
    type: 'string';
    value: string;
  }
  
  export interface NumberInput {
    type: 'number';
    value: number;
  }
  
  export type Input = StringInput | NumberInput;
  
  export interface SuccessOutput {
    result: string | number;
    status: 'success';
  }
  
  export interface ErrorOutput {
    result: null;
    status: 'error';
    message: string;
  }
  
  export type Output = SuccessOutput | ErrorOutput;
  
  export function doSomething(input: Input): Output;
}

总结

声明文件是 TypeScript 生态系统的重要组成部分：

  1. 类型安全：为 JavaScript 库提供类型信息
  2. 开发体验：提供智能提示和错误检查
  3. 模块增强：扩展现有模块的功能
  4. 渐进式采用：支持从 JavaScript 逐步迁移

掌握声明文件有助于：

  • 更好地使用第三方库
  • 为团队提供类型定义
  • 维护大型项目的类型一致性
  • 提高代码的可维护性
