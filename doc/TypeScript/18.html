TypeScript 映射类型 (Mapped Types)

什么是映射类型？

映射类型是 TypeScript 中的一种高级类型特性，它允许基于现有类型创建新类型。映射类型通过遍历现有类型的属性，对每个属性应用转换规则来生成新的类型。

基本语法

// 基本映射类型语法
type MappedType<T> = {
  [K in keyof T]: T[K]
}

// 带修饰符的映射类型
type MappedTypeWithModifiers<T> = {
  readonly [K in keyof T]?: T[K]
}

基础映射类型

简单属性映射

interface User {
  id: number;
  name: string;
  email: string;
  age: number;
}

// 基本映射：复制所有属性
type UserCopy<T> = {
  [K in keyof T]: T[K]
};

type CopiedUser = UserCopy<User>;
// 结果：{ id: number; name: string; email: string; age: number; }

// 将所有属性转为字符串类型
type StringifyProperties<T> = {
  [K in keyof T]: string;
};

type StringifiedUser = StringifyProperties<User>;
// 结果：{ id: string; name: string; email: string; age: string; }

// 将所有属性转为可选
type OptionalProperties<T> = {
  [K in keyof T]?: T[K];
};

type OptionalUser = OptionalProperties<User>;
// 结果：{ id?: number; name?: string; email?: string; age?: number; }

属性修饰符操作

interface RequiredUser {
  id: number;
  name: string;
  email?: string;
  phone?: string;
}

// 移除可选修饰符（变为必需）
type RemoveOptional<T> = {
  [K in keyof T]-?: T[K];
};

type AllRequiredUser = RemoveOptional<RequiredUser>;
// 结果：{ id: number; name: string; email: string; phone: string; }

// 添加只读修饰符
type AddReadonly<T> = {
  readonly [K in keyof T]: T[K];
};

type ReadonlyUser = AddReadonly<User>;
// 结果：{ readonly id: number; readonly name: string; ... }

// 移除只读修饰符
type RemoveReadonly<T> = {
  -readonly [K in keyof T]: T[K];
};

interface ReadonlyConfig {
  readonly apiUrl: string;
  readonly timeout: number;
}

type MutableConfig = RemoveReadonly<ReadonlyConfig>;
// 结果：{ apiUrl: string; timeout: number; }

高级映射类型

键名转换

// 键名前缀映射
type PrefixKeys<T, P extends string> = {
  [K in keyof T as `${P}${string & K}`]: T[K];
};

type PrefixedUser = PrefixKeys<User, 'user_'>;
// 结果：{ user_id: number; user_name: string; user_email: string; user_age: number; }

// 键名后缀映射
type SuffixKeys<T, S extends string> = {
  [K in keyof T as `${string & K}${S}`]: T[K];
};

type SuffixedUser = SuffixKeys<User, '_field'>;
// 结果：{ id_field: number; name_field: string; email_field: string; age_field: number; }

// 驼峰命名转下划线
type CamelToSnake<S extends string> = S extends `${infer T}${infer U}`
  ? `${T extends Capitalize<T> ? '_' : ''}${Lowercase<T>}${CamelToSnake<U>}`
  : S;

type SnakeCaseKeys<T> = {
  [K in keyof T as CamelToSnake<string & K>]: T[K];
};

interface CamelCaseUser {
  userId: number;
  userName: string;
  userEmail: string;
}

type SnakeCaseUser = SnakeCaseKeys<CamelCaseUser>;
// 结果：{ user_id: number; user_name: string; user_email: string; }

条件映射

// 基于类型的条件映射
type StringKeysOnly<T> = {
  [K in keyof T]: T[K] extends string ? T[K] : never;
};

interface MixedTypes {
  id: number;
  name: string;
  email: string;
  age: number;
  active: boolean;
}

type StringProperties = StringKeysOnly<MixedTypes>;
// 结果：{ id: never; name: string; email: string; age: never; active: never; }

// 过滤特定类型的属性
type FilterByType<T, U> = {
  [K in keyof T]: T[K] extends U ? K : never;
}[keyof T];

type StringKeys = FilterByType<MixedTypes, string>;
// 结果：'name' | 'email'

type NumberKeys = FilterByType<MixedTypes, number>;
// 结果：'id' | 'age'

// 提取特定类型的属性
type PickByType<T, U> = Pick<T, FilterByType<T, U>>;

type StringOnlyProperties = PickByType<MixedTypes, string>;
// 结果：{ name: string; email: string; }

深度映射

// 深度可选映射
type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P];
};

interface NestedUser {
  id: number;
  profile: {
    name: string;
    avatar: string;
    settings: {
      theme: string;
      notifications: boolean;
    };
  };
  preferences: {
    language: string;
    timezone: string;
  };
}

type PartialNestedUser = DeepPartial<NestedUser>;
// 所有嵌套属性都变为可选

// 深度只读映射
type DeepReadonly<T> = {
  readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P];
};

type ReadonlyNestedUser = DeepReadonly<NestedUser>;
// 所有嵌套属性都变为只读

// 深度可变映射
type DeepMutable<T> = {
  -readonly [P in keyof T]: T[P] extends object ? DeepMutable<T[P]> : T[P];
};

实际应用示例

API 响应类型转换

// 原始 API 响应类型
interface ApiUser {
  id: number;
  first_name: string;
  last_name: string;
  email_address: string;
  phone_number: string;
  created_at: string;
  updated_at: string;
}

// 转换为前端使用的类型
type CamelCaseProperty<S extends string> = S extends `${infer P1}_${infer P2}${infer P3}`
  ? `${P1}${Capitalize<CamelCaseProperty<`${P2}${P3}`>>}`
  : S;

type CamelCaseKeys<T> = {
  [K in keyof T as CamelCaseProperty<string & K>]: T[K];
};

type FrontendUser = CamelCaseKeys<ApiUser>;
// 结果：{
//   id: number;
//   firstName: string;
//   lastName: string;
//   emailAddress: string;
//   phoneNumber: string;
//   createdAt: string;
//   updatedAt: string;
// }

// API 响应转换函数
function transformApiResponse<T>(data: T): CamelCaseKeys<T> {
  const result = {} as CamelCaseKeys<T>;
  
  for (const key in data) {
    const camelKey = key.replace(/_([a-z])/g, (_, letter) => 
      letter.toUpperCase()
    ) as keyof CamelCaseKeys<T>;
    
    result[camelKey] = data[key] as any;
  }
  
  return result;
}

数据库模型映射

// 数据库实体基类
interface BaseEntity {
  id: number;
  createdAt: Date;
  updatedAt: Date;
}

// 用户实体
interface UserEntity extends BaseEntity {
  name: string;
  email: string;
  password: string;
  role: 'admin' | 'user';
}

// 创建输入类型（排除自动生成的字段）
type CreateInput<T extends BaseEntity> = Omit<T, keyof BaseEntity>;

type CreateUserInput = CreateInput<UserEntity>;
// 结果：{ name: string; email: string; password: string; role: 'admin' | 'user'; }

// 更新输入类型（所有字段可选，排除 id）
type UpdateInput<T extends BaseEntity> = Partial<Omit<T, 'id'>>;

type UpdateUserInput = UpdateInput<UserEntity>;
// 结果：{
//   name?: string;
//   email?: string;
//   password?: string;
//   role?: 'admin' | 'user';
//   createdAt?: Date;
//   updatedAt?: Date;
// }

// 公开信息类型（排除敏感字段）
type PublicInfo<T, K extends keyof T> = Omit<T, K>;

type PublicUserInfo = PublicInfo<UserEntity, 'password'>;
// 结果：{ id: number; createdAt: Date; updatedAt: Date; name: string; email: string; role: 'admin' | 'user'; }

// 数据库查询选项
type SelectOptions<T> = {
  select?: (keyof T)[];
  where?: Partial<T>;
  orderBy?: {
    [K in keyof T]?: 'asc' | 'desc';
  };
};

type UserSelectOptions = SelectOptions<UserEntity>;

事件系统类型

// 事件定义
interface EventMap {
  'user:created': { user: User };
  'user:updated': { user: User; changes: Partial<User> };
  'user:deleted': { userId: number };
  'post:published': { post: Post; author: User };
  'comment:added': { comment: Comment; post: Post };
}

// 事件监听器类型
type EventListener<T> = (data: T) => void | Promise<void>;

// 事件系统类型
type EventListeners<T> = {
  [K in keyof T]: EventListener<T[K]>[];
};

type EventEmitter<T> = {
  [K in keyof T as `on${Capitalize<string & K>}`]: (listener: EventListener<T[K]>) => void;
} & {
  [K in keyof T as `emit${Capitalize<string & K>}`]: (data: T[K]) => void;
} & {
  [K in keyof T as `off${Capitalize<string & K>}`]: (listener: EventListener<T[K]>) => void;
};

type UserEventEmitter = EventEmitter<EventMap>;
// 生成所有事件的 on/emit/off 方法

// 事件系统实现
class TypedEventEmitter<T extends Record<string, any>> {
  private listeners: EventListeners<T> = {} as EventListeners<T>;

  on<K extends keyof T>(event: K, listener: EventListener<T[K]>): void {
    if (!this.listeners[event]) {
      this.listeners[event] = [];
    }
    this.listeners[event].push(listener);
  }

  emit<K extends keyof T>(event: K, data: T[K]): void {
    const eventListeners = this.listeners[event];
    if (eventListeners) {
      eventListeners.forEach(listener => listener(data));
    }
  }

  off<K extends keyof T>(event: K, listener: EventListener<T[K]>): void {
    const eventListeners = this.listeners[event];
    if (eventListeners) {
      const index = eventListeners.indexOf(listener);
      if (index > -1) {
        eventListeners.splice(index, 1);
      }
    }
  }
}

const eventEmitter = new TypedEventEmitter<EventMap>();

// 类型安全的事件使用
eventEmitter.on('user:created', (data) => {
  console.log(`User created: ${data.user.name}`);
});

eventEmitter.emit('user:created', {
  user: { id: 1, name: 'Alice', email: 'alice@example.com' }
});

映射类型最佳实践

1. 使用辅助类型提高可读性

// 不好的做法：复杂的内联映射
type ComplexMapping<T> = {
  [K in keyof T as T[K] extends string 
    ? `str_${string & K}` 
    : T[K] extends number 
    ? `num_${string & K}` 
    : `other_${string & K}`]: T[K] extends string 
    ? string 
    : T[K] extends number 
    ? number 
    : any;
};

// 好的做法：分解为多个辅助类型
type GetKeyPrefix<T> = T extends string 
  ? 'str_' 
  : T extends number 
  ? 'num_' 
  : 'other_';

type GetValueType<T> = T extends string 
  ? string 
  : T extends number 
  ? number 
  : any;

type PrefixedMapping<T> = {
  [K in keyof T as `${GetKeyPrefix<T[K]>}${string & K}`]: GetValueType<T[K]>;
};

2. 提供默认类型参数

// 提供合理的默认值
type ApiResponse<T = any, E = string> = {
  data: T;
  error?: E;
  loading: boolean;
};

type MappedApiResponses<T extends Record<string, any>> = {
  [K in keyof T]: ApiResponse<T[K]>;
};

// 使用示例
interface ApiEndpoints {
  users: User[];
  posts: Post[];
}

type ApiResponseMap = MappedApiResponses<ApiEndpoints>;
// 结果：{
//   users: ApiResponse<User[]>;
//   posts: ApiResponse<Post[]>;
// }

3. 组合映射类型

// 组合多个映射操作
type TransformEntity<T extends BaseEntity> = 
  PublicInfo<CamelCaseKeys<T>, 'password' | 'createdAt' | 'updatedAt'>;

// 批量转换
type TransformEntities<T extends Record<string, BaseEntity>> = {
  [K in keyof T]: TransformEntity<T[K]>;
};

interface DatabaseEntities {
  user: UserEntity;
  post: PostEntity;
  comment: CommentEntity;
}

type PublicEntities = TransformEntities<DatabaseEntities>;

4. 性能优化

// 避免过深的递归
type DeepPartialOptimized<T, Depth extends number = 5> = Depth extends 0
  ? T
  : {
      [P in keyof T]?: T[P] extends object 
        ? DeepPartialOptimized<T[P], Prev<Depth>>
        : T[P];
    };

// 辅助类型：递减数字
type Prev<T extends number> = T extends 5 ? 4
  : T extends 4 ? 3
  : T extends 3 ? 2
  : T extends 2 ? 1
  : T extends 1 ? 0
  : never;

// 使用缓存避免重复计算
type CachedMapping<T> = T extends infer U ? MappedType<U> : never;

总结

映射类型是 TypeScript 类型系统的强大特性：

  1. 类型转换：基于现有类型创建新类型
  2. 修饰符操作：添加或移除可选、只读修饰符
  3. 键名转换：支持键名的动态转换
  4. 条件映射：结合条件类型实现复杂逻辑

掌握映射类型有助于：

  • 创建灵活的类型系统
  • 减少重复的类型定义
  • 实现类型安全的数据转换
  • 构建强大的工具类型
