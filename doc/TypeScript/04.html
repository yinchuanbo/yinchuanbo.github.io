类 (Class)

TypeScript | 2025.07.30

TypeScript 中的类是基于 ES6 类语法的扩展，提供了更强大的类型检查和面向对象编程特性。类可以包含属性、方法、构造函数，并支持继承、访问修饰符等高级特性。

基本类语法

简单类定义

class Greeter {
    greeting: string;
    
    constructor(message: string) {
        this.greeting = message;
    }
    
    greet() {
        return "Hello, " + this.greeting;
    }
}

let greeter = new Greeter("world");
console.log(greeter.greet()); // "Hello, world"

属性初始化

class Person {
    // 直接初始化
    name: string = "Unknown";
    age: number = 0;
    
    // 构造函数中初始化
    email: string;
    
    constructor(email: string) {
        this.email = email;
    }
}

继承

基本继承

class Animal {
    name: string;
    
    constructor(theName: string) {
        this.name = theName;
    }
    
    move(distanceInMeters: number = 0) {
        console.log(`${this.name} moved ${distanceInMeters}m.`);
    }
}

class Dog extends Animal {
    breed: string;
    
    constructor(name: string, breed: string) {
        super(name); // 调用父类构造函数
        this.breed = breed;
    }
    
    bark() {
        console.log('Woof! Woof!');
    }
    
    move(distanceInMeters = 5) {
        console.log("Running...");
        super.move(distanceInMeters); // 调用父类方法
    }
}

const dog = new Dog("Rex", "German Shepherd");
dog.bark(); // Woof! Woof!
dog.move(10); // Running... Rex moved 10m.

方法重写

class Bird extends Animal {
    constructor(name: string) {
        super(name);
    }
    
    move(distanceInMeters = 10) {
        console.log("Flying...");
        super.move(distanceInMeters);
    }
}

const bird = new Bird("Eagle");
bird.move(20); // Flying... Eagle moved 20m.

访问修饰符

public（公共）

class PublicExample {
    public name: string; // 默认为 public
    
    public constructor(name: string) {
        this.name = name;
    }
    
    public greet() {
        return `Hello, ${this.name}`;
    }
}

const example = new PublicExample("Alice");
console.log(example.name); // 可以访问
console.log(example.greet()); // 可以访问

private（私有）

class PrivateExample {
    private secret: string;
    
    constructor(secret: string) {
        this.secret = secret;
    }
    
    private getSecret() {
        return this.secret;
    }
    
    public revealSecret() {
        return this.getSecret(); // 类内部可以访问私有成员
    }
}

const example = new PrivateExample("top secret");
// console.log(example.secret); // 错误：无法访问私有属性
// console.log(example.getSecret()); // 错误：无法访问私有方法
console.log(example.revealSecret()); // 正确：通过公共方法访问

protected（受保护）

class ProtectedExample {
    protected name: string;
    
    constructor(name: string) {
        this.name = name;
    }
    
    protected getName() {
        return this.name;
    }
}

class ExtendedExample extends ProtectedExample {
    constructor(name: string) {
        super(name);
    }
    
    public displayName() {
        return this.getName(); // 子类可以访问受保护成员
    }
}

const extended = new ExtendedExample("Bob");
console.log(extended.displayName()); // 正确
// console.log(extended.name); // 错误：无法从外部访问受保护成员

参数属性

class ParameterProperties {
    // 参数属性：在构造函数参数前添加访问修饰符
    constructor(
        private name: string,
        public age: number,
        protected email: string,
        readonly id: number
    ) {
        // 不需要手动赋值，TypeScript 会自动创建属性并赋值
    }
    
    getInfo() {
        return `${this.name} (${this.age}) - ${this.email}`;
    }
}

const person = new ParameterProperties("Alice", 25, "alice@example.com", 1);
console.log(person.age); // 25
// console.log(person.name); // 错误：私有属性
console.log(person.getInfo()); // Alice (25) - alice@example.com

存取器 (Getters/Setters)

class Employee {
    private _fullName: string = "";
    private _department: string = "";
    
    get fullName(): string {
        return this._fullName;
    }
    
    set fullName(newName: string) {
        if (newName && newName.length > 0) {
            this._fullName = newName;
        } else {
            throw new Error("Name cannot be empty");
        }
    }
    
    get department(): string {
        return this._department;
    }
    
    set department(dept: string) {
        this._department = dept.toUpperCase();
    }
}

const employee = new Employee();
employee.fullName = "Bob Smith";
employee.department = "engineering";

console.log(employee.fullName); // "Bob Smith"
console.log(employee.department); // "ENGINEERING"

静态属性和方法

class MathUtils {
    static PI: number = 3.14159;
    static instanceCount: number = 0;
    
    constructor() {
        MathUtils.instanceCount++;
    }
    
    static calculateCircleArea(radius: number): number {
        return MathUtils.PI * radius * radius;
    }
    
    static getInstanceCount(): number {
        return MathUtils.instanceCount;
    }
}

console.log(MathUtils.PI); // 3.14159
console.log(MathUtils.calculateCircleArea(5)); // 78.53975

const math1 = new MathUtils();
const math2 = new MathUtils();
console.log(MathUtils.getInstanceCount()); // 2

抽象类

abstract class Shape {
    abstract name: string;
    
    constructor(protected color: string) {}
    
    // 抽象方法：必须在子类中实现
    abstract calculateArea(): number;
    abstract draw(): void;
    
    // 具体方法：可以在子类中使用
    getColor(): string {
        return this.color;
    }
    
    displayInfo(): void {
        console.log(`${this.name} - Color: ${this.color}, Area: ${this.calculateArea()}`);
    }
}

class Circle extends Shape {
    name: string = "Circle";
    
    constructor(color: string, private radius: number) {
        super(color);
    }
    
    calculateArea(): number {
        return Math.PI * this.radius * this.radius;
    }
    
    draw(): void {
        console.log(`Drawing a ${this.color} circle with radius ${this.radius}`);
    }
}

class Rectangle extends Shape {
    name: string = "Rectangle";
    
    constructor(color: string, private width: number, private height: number) {
        super(color);
    }
    
    calculateArea(): number {
        return this.width * this.height;
    }
    
    draw(): void {
        console.log(`Drawing a ${this.color} rectangle ${this.width}x${this.height}`);
    }
}

// const shape = new Shape("red"); // 错误：无法实例化抽象类

const circle = new Circle("red", 5);
const rectangle = new Rectangle("blue", 10, 8);

circle.displayInfo(); // Circle - Color: red, Area: 78.53981633974483
rectangle.displayInfo(); // Rectangle - Color: blue, Area: 80

circle.draw(); // Drawing a red circle with radius 5
rectangle.draw(); // Drawing a blue rectangle 10x8

实例演示：图书管理系统

// 抽象基类
abstract class LibraryItem {
    protected static nextId: number = 1;
    public readonly id: number;
    protected _isAvailable: boolean = true;
    
    constructor(
        public readonly title: string,
        public readonly author: string,
        protected publishYear: number
    ) {
        this.id = LibraryItem.nextId++;
    }
    
    // 抽象方法
    abstract getItemType(): string;
    abstract calculateLateFee(daysLate: number): number;
    
    // 存取器
    get isAvailable(): boolean {
        return this._isAvailable;
    }
    
    get age(): number {
        return new Date().getFullYear() - this.publishYear;
    }
    
    // 公共方法
    borrow(): boolean {
        if (this._isAvailable) {
            this._isAvailable = false;
            console.log(`${this.title} has been borrowed.`);
            return true;
        }
        console.log(`${this.title} is not available.`);
        return false;
    }
    
    return(): void {
        this._isAvailable = true;
        console.log(`${this.title} has been returned.`);
    }
    
    getInfo(): string {
        return `[${this.id}] ${this.title} by ${this.author} (${this.publishYear})`;
    }
}

// 具体实现类
class Book extends LibraryItem {
    constructor(
        title: string,
        author: string,
        publishYear: number,
        private pages: number,
        private isbn: string
    ) {
        super(title, author, publishYear);
    }
    
    getItemType(): string {
        return "Book";
    }
    
    calculateLateFee(daysLate: number): number {
        return daysLate * 0.5; // $0.5 per day
    }
    
    getPages(): number {
        return this.pages;
    }
    
    getISBN(): string {
        return this.isbn;
    }
}

class DVD extends LibraryItem {
    constructor(
        title: string,
        author: string, // director
        publishYear: number,
        private duration: number, // in minutes
        private genre: string
    ) {
        super(title, author, publishYear);
    }
    
    getItemType(): string {
        return "DVD";
    }
    
    calculateLateFee(daysLate: number): number {
        return daysLate * 1.0; // $1.0 per day
    }
    
    getDuration(): number {
        return this.duration;
    }
    
    getGenre(): string {
        return this.genre;
    }
}

class Magazine extends LibraryItem {
    constructor(
        title: string,
        author: string, // publisher
        publishYear: number,
        private issueNumber: number,
        private month: string
    ) {
        super(title, author, publishYear);
    }
    
    getItemType(): string {
        return "Magazine";
    }
    
    calculateLateFee(daysLate: number): number {
        return daysLate * 0.25; // $0.25 per day
    }
    
    getIssueInfo(): string {
        return `Issue ${this.issueNumber} - ${this.month} ${this.publishYear}`;
    }
}

// 图书馆管理类
class Library {
    private items: LibraryItem[] = [];
    private static instance: Library;
    
    private constructor() {}
    
    // 单例模式
    static getInstance(): Library {
        if (!Library.instance) {
            Library.instance = new Library();
        }
        return Library.instance;
    }
    
    addItem(item: LibraryItem): void {
        this.items.push(item);
        console.log(`Added ${item.getItemType()}: ${item.getInfo()}`);
    }
    
    findItemById(id: number): LibraryItem | undefined {
        return this.items.find(item => item.id === id);
    }
    
    findItemsByAuthor(author: string): LibraryItem[] {
        return this.items.filter(item => 
            item.author.toLowerCase().includes(author.toLowerCase())
        );
    }
    
    getAvailableItems(): LibraryItem[] {
        return this.items.filter(item => item.isAvailable);
    }
    
    displayAllItems(): void {
        console.log("\n=== Library Catalog ===");
        this.items.forEach(item => {
            const status = item.isAvailable ? "Available" : "Borrowed";
            console.log(`${item.getInfo()} - ${item.getItemType()} [${status}]`);
        });
    }
    
    calculateTotalLateFees(daysLate: number): number {
        return this.items
            .filter(item => !item.isAvailable)
            .reduce((total, item) => total + item.calculateLateFee(daysLate), 0);
    }
}

// 使用示例
const library = Library.getInstance();

// 添加图书馆物品
const book1 = new Book("TypeScript Handbook", "Microsoft", 2021, 300, "978-1234567890");
const book2 = new Book("Clean Code", "Robert Martin", 2008, 464, "978-0132350884");
const dvd1 = new DVD("The Matrix", "Wachowski Sisters", 1999, 136, "Sci-Fi");
const magazine1 = new Magazine("National Geographic", "National Geographic Society", 2024, 3, "March");

library.addItem(book1);
library.addItem(book2);
library.addItem(dvd1);
library.addItem(magazine1);

// 显示所有物品
library.displayAllItems();

// 借阅物品
console.log("\n=== Borrowing Items ===");
book1.borrow();
dvd1.borrow();

// 显示可用物品
console.log("\n=== Available Items ===");
const availableItems = library.getAvailableItems();
availableItems.forEach(item => console.log(item.getInfo()));

// 计算滞纳金
console.log("\n=== Late Fees ===");
const totalLateFees = library.calculateTotalLateFees(5); // 5 days late
console.log(`Total late fees for 5 days: $${totalLateFees.toFixed(2)}`);

// 归还物品
console.log("\n=== Returning Items ===");
book1.return();
dvd1.return();

这个例子展示了 TypeScript 类的高级特性，包括抽象类、继承、访问修饰符、静态方法、单例模式等，构建了一个完整的图书馆管理系统。
