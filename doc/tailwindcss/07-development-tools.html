PostCSS 集成

基本 PostCSS 配置

// postcss.config.js
module.exports = {
  plugins: {
    tailwindcss: {},
    autoprefixer: {},
  },
}

高级 PostCSS 配置

// postcss.config.js
const path = require('path')

module.exports = {
  plugins: [
    require('tailwindcss')(path.resolve(__dirname, 'tailwind.config.js')),
    require('autoprefixer'),
    require('cssnano')({
      preset: 'default',
    }),
    // 其他 PostCSS 插件
    require('postcss-import'),
    require('postcss-nested'),
    require('postcss-custom-properties'),
  ],
}

条件配置

// postcss.config.js
module.exports = (ctx) => ({
  plugins: [
    require('tailwindcss'),
    require('autoprefixer'),
    ctx.env === 'production' && require('cssnano')({
      preset: ['default', {
        discardComments: {
          removeAll: true,
        },
      }]
    }),
  ].filter(Boolean),
})

预处理器集成

Sass/SCSS 集成

// styles/main.scss
@tailwind base;
@tailwind components;
@tailwind utilities;

// 自定义 Sass 变量和 mixins
$primary-color: theme('colors.blue.500');

@mixin button-style {
  @apply px-4 py-2 rounded font-medium;
}

.custom-button {
  @include button-style;
  background-color: $primary-color;
}

注意事项：

  • Sass 在 Tailwind 之前处理
  • 不能在 Sass 函数中使用 theme() 函数
  • 需要额外的构建步骤

性能优化工具

PurgeCSS 集成

// postcss.config.js
module.exports = {
  plugins: [
    require('tailwindcss'),
    require('autoprefixer'),
    process.env.NODE_ENV === 'production' && require('@fullhuman/postcss-purgecss')({
      content: [
        './src/**/*.html',
        './src/**/*.js',
      ],
      defaultExtractor: content => content.match(/[\w-/:]+(?<!:)/g) || []
    }),
  ].filter(Boolean),
}
