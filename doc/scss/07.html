SCSS运算符

SCSS | 2025.07.30

SCSS支持多种运算符，包括数学运算、比较运算、逻辑运算等，让样式计算更加灵活。

数学运算符

基本数学运算

// 加法运算
$base-size: 16px;
$large-size: $base-size + 4px; // 20px
$margin: 1rem + 0.5rem; // 1.5rem

.title {
  font-size: $large-size;
  margin: $margin;
}

// 减法运算
$container-width: 1200px;
$sidebar-width: 300px;
$content-width: $container-width - $sidebar-width; // 900px

.content {
  width: $content-width;
}

// 乘法运算
$base-padding: 0.5rem;
$double-padding: $base-padding * 2; // 1rem
$grid-columns: 12;
$column-width: 100% / $grid-columns; // 8.33333%

.column {
  width: $column-width;
  padding: $double-padding;
}

// 除法运算（需要括号或变量）
$total-width: 960px;
$columns: 4;
$column-width: ($total-width / $columns); // 240px

// 或者使用math.div函数（推荐）
@use 'sass:math';
$precise-width: math.div($total-width, $columns);

单位运算

// 相同单位运算
$width: 100px + 50px; // 150px
$height: 10rem - 2rem; // 8rem

// 不同单位运算（需要兼容）
$mixed: 1rem + 16px; // 需要确保单位兼容

// 百分比运算
$half: 50%;
$quarter: $half / 2; // 25%
$three-quarters: $half + $quarter; // 75%

.grid-item {
  width: $quarter;
  
  &.half {
    width: $half;
  }
  
  &.three-quarters {
    width: $three-quarters;
  }
}

颜色运算

$base-color: #333;
$light-color: $base-color + #222; // #555555
$dark-color: $base-color - #111; // #222222

// 使用颜色函数更安全
$primary: #007bff;
$primary-light: lighten($primary, 10%);
$primary-dark: darken($primary, 10%);
$primary-transparent: rgba($primary, 0.8);

.button {
  background-color: $primary;
  
  &:hover {
    background-color: $primary-dark;
  }
  
  &.light {
    background-color: $primary-light;
  }
  
  &.transparent {
    background-color: $primary-transparent;
  }
}

比较运算符

$mobile-width: 768px;
$tablet-width: 1024px;
$desktop-width: 1200px;

// 比较运算符：==, !=, <, >, <=, >=
@mixin responsive-font($screen-size) {
  @if $screen-size >= $desktop-width {
    font-size: 1.5rem;
  } @else if $screen-size >= $tablet-width {
    font-size: 1.25rem;
  } @else if $screen-size >= $mobile-width {
    font-size: 1.125rem;
  } @else {
    font-size: 1rem;
  }
}

// 字符串比较
$theme: 'dark';

.header {
  @if $theme == 'dark' {
    background-color: #333;
    color: white;
  } @else if $theme == 'light' {
    background-color: white;
    color: #333;
  }
}

逻辑运算符

$is-mobile: true;
$is-dark-theme: false;
$show-sidebar: true;

// and 运算符
@if $is-mobile and $show-sidebar {
  .sidebar {
    position: fixed;
    transform: translateX(-100%);
    
    &.open {
      transform: translateX(0);
    }
  }
}

// or 运算符
@if $is-mobile or $show-sidebar {
  .main-content {
    padding-left: 0;
  }
}

// not 运算符
@if not $is-dark-theme {
  .page {
    background-color: white;
    color: #333;
  }
}

字符串运算

$prefix: 'app';
$component: 'button';
$modifier: 'primary';

// 字符串连接
$class-name: $prefix + '-' + $component; // 'app-button'
$full-class: $class-name + '--' + $modifier; // 'app-button--primary'

// 使用插值更清晰
$interpolated-class: '#{$prefix}-#{$component}--#{$modifier}';

.#{$interpolated-class} {
  background-color: #007bff;
  color: white;
}

// 路径拼接
$image-path: '/assets/images/';
$icon-name: 'search';
$icon-extension: '.svg';

.search-icon {
  background-image: url('#{$image-path}#{$icon-name}#{$icon-extension}');
}

列表运算

$margins: 0.5rem, 1rem, 1.5rem, 2rem;
$colors: #ff0000, #00ff00, #0000ff;

// 列表索引（从1开始）
$first-margin: nth($margins, 1); // 0.5rem
$last-color: nth($colors, length($colors)); // #0000ff

// 列表操作
$new-margins: append($margins, 2.5rem); // 添加元素
$joined-list: join($margins, $colors); // 合并列表

@each $margin in $margins {
  .margin-#{index($margins, $margin)} {
    margin: $margin;
  }
}

Map运算

$breakpoints: (
  mobile: 768px,
  tablet: 1024px,
  desktop: 1200px
);

$colors: (
  primary: #007bff,
  secondary: #6c757d,
  success: #28a745,
  danger: #dc3545
);

// Map操作
$mobile-bp: map-get($breakpoints, mobile); // 768px
$primary-color: map-get($colors, primary); // #007bff

// 检查键是否存在
@if map-has-key($colors, warning) {
  .warning {
    color: map-get($colors, warning);
  }
} @else {
  .warning {
    color: #ffc107; // 默认值
  }
}

// 遍历Map
@each $name, $color in $colors {
  .text-#{$name} {
    color: $color;
  }
  
  .bg-#{$name} {
    background-color: $color;
  }
}

实际应用实例

响应式网格系统

$grid-columns: 12;
$grid-gutter: 2rem;
$container-max-width: 1200px;

// 计算列宽
@function column-width($columns) {
  @return (100% / $grid-columns) * $columns;
}

// 计算偏移
@function column-offset($columns) {
  @return (100% / $grid-columns) * $columns;
}

.container {
  max-width: $container-max-width;
  margin: 0 auto;
  padding: 0 ($grid-gutter / 2);
}

.row {
  display: flex;
  margin: 0 (-$grid-gutter / 2);
}

// 生成列类
@for $i from 1 through $grid-columns {
  .col-#{$i} {
    width: column-width($i);
    padding: 0 ($grid-gutter / 2);
  }
  
  .offset-#{$i} {
    margin-left: column-offset($i);
  }
}

动态主题系统

$themes: (
  light: (
    background: #ffffff,
    text: #333333,
    primary: #007bff,
    secondary: #6c757d
  ),
  dark: (
    background: #1a1a1a,
    text: #ffffff,
    primary: #0d6efd,
    secondary: #6c757d
  )
);

@mixin theme-colors($theme-name) {
  $theme: map-get($themes, $theme-name);
  
  background-color: map-get($theme, background);
  color: map-get($theme, text);
  
  .btn-primary {
    background-color: map-get($theme, primary);
  }
  
  .btn-secondary {
    background-color: map-get($theme, secondary);
  }
}

.theme-light {
  @include theme-colors(light);
}

.theme-dark {
  @include theme-colors(dark);
}

动态间距系统

$base-spacing: 1rem;
$spacing-scale: 1.5;

// 生成间距值
@function spacing($multiplier) {
  @return $base-spacing * pow($spacing-scale, $multiplier - 1);
}

// 生成间距类
@for $i from 1 through 6 {
  $space: spacing($i);
  
  .m-#{$i} { margin: $space; }
  .mt-#{$i} { margin-top: $space; }
  .mr-#{$i} { margin-right: $space; }
  .mb-#{$i} { margin-bottom: $space; }
  .ml-#{$i} { margin-left: $space; }
  
  .p-#{$i} { padding: $space; }
  .pt-#{$i} { padding-top: $space; }
  .pr-#{$i} { padding-right: $space; }
  .pb-#{$i} { padding-bottom: $space; }
  .pl-#{$i} { padding-left: $space; }
}
