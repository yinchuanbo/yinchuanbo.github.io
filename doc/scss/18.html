高级断点系统

多维断点管理

// 复杂断点系统
$breakpoints: (
  // 移动设备
  mobile: (
    portrait: (min-width: 320px, max-width: 479px, orientation: portrait),
    landscape: (min-width: 480px, max-width: 767px, orientation: landscape)
  ),
  // 平板设备
  tablet: (
    portrait: (min-width: 768px, max-width: 1023px, orientation: portrait),
    landscape: (min-width: 1024px, max-width: 1199px, orientation: landscape)
  ),
  // 桌面设备
  desktop: (
    small: (min-width: 1200px, max-width: 1399px),
    medium: (min-width: 1400px, max-width: 1799px),
    large: (min-width: 1800px)
  ),
  // 特殊设备
  watch: (max-width: 319px),
  tv: (min-width: 1920px)
);

// 高级响应式混合器
@mixin respond-to($device, $variant: null) {
  $device-map: map-get($breakpoints, $device);
  
  @if not $device-map {
    @error "Device '#{$device}' not found in breakpoints map.";
  }
  
  @if $variant {
    $variant-map: map-get($device-map, $variant);
    @if not $variant-map {
      @error "Variant '#{$variant}' not found for device '#{$device}'.";
    }
    $device-map: $variant-map;
  }
  
  $query: '';
  @each $property, $value in $device-map {
    @if $query != '' {
      $query: $query + ' and ';
    }
    $query: $query + '(#{$property}: #{$value})';
  }
  
  @media #{$query} {
    @content;
  }
}

// 使用示例
.responsive-component {
  padding: 1rem;
  
  @include respond-to(mobile, portrait) {
    padding: 0.5rem;
    font-size: 0.875rem;
  }
  
  @include respond-to(tablet, landscape) {
    padding: 1.5rem;
    display: flex;
  }
  
  @include respond-to(desktop, large) {
    padding: 3rem;
    max-width: 1600px;
    margin: 0 auto;
  }
}

// 特性检测混合器
@mixin supports($feature) {
  @supports #{$feature} {
    @content;
  }
}

@mixin supports-not($feature) {
  @supports not #{$feature} {
    @content;
  }
}

.modern-grid {

      }
    }
  }

流体设计系统

流体字体和间距

// 流体计算函数
@function fluid-size($min-size, $max-size, $min-width: 320px, $max-width: 1200px) {
  $slope: ($max-size - $min-size) / ($max-width - $min-width);
  $intersection: (-$min-width * $slope) + $min-size;
  
  @return clamp(#{$min-size}, #{$intersection} + #{$slope * 100vw}, #{$max-size});
}

// 流体间距系统
$fluid-spacing: (
  xs: (min: 0.5rem, max: 1rem),
  sm: (min: 1rem, max: 1.5rem),
  md: (min: 1.5rem, max: 2.5rem),
  lg: (min: 2rem, max: 4rem),
  xl: (min: 3rem, max: 6rem),
  2xl: (min: 4rem, max: 8rem)
);

@function fluid-space($size) {
  $space-map: map-get($fluid-spacing, $size);
  $min: map-get($space-map, min);
  $max: map-get($space-map, max);
  
  @return fluid-size($min, $max);
}

// 流体字体系统
$fluid-typography: (
  xs: (min: 0.75rem, max: 0.875rem),
  sm: (min: 0.875rem, max: 1rem),
  base: (min: 1rem, max: 1.125rem),
  lg: (min: 1.125rem, max: 1.25rem),
  xl: (min: 1.25rem, max: 1.5rem),
  2xl: (min: 1.5rem, max: 2rem),
  3xl: (min: 1.875rem, max: 2.5rem),
  4xl: (min: 2.25rem, max: 3rem),
  5xl: (min: 3rem, max: 4rem)
);

@function fluid-font($size) {
  $font-map: map-get($fluid-typography, $size);
  $min: map-get($font-map, min);
  $max: map-get($font-map, max);
  
  @return fluid-size($min, $max);
}

// 流体网格配置
$grid-config: (
  columns: 12,
  min-column-width: 250px,
  max-column-width: 1fr,
  gap: (min: 1rem, max: 2rem)
);

// 流体网格混合器
@mixin fluid-grid($columns: null, $min-width: 250px, $gap: null) {
  display: grid;
  
  @if $columns {
    grid-template-columns: repeat($columns, 1fr);
  } @else {
    grid-template-columns: repeat(auto-fit, minmax($min-width, 1fr));
  }
  
  @if $gap {
    gap: $gap;
  } @else {
    gap: fluid-space(md);
  }
}
