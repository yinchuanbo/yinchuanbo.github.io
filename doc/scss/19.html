import { defineConfig } from 'vite';

export default defineConfig({
  css: {
    preprocessorOptions: {
      scss: {
        // 全局导入
        additionalData: `
          @import "@/styles/abstracts/variables";
          @import "@/styles/abstracts/mixins";
        `,
        // 包含路径
        includePaths: [
          resolve(__dirname, 'src/styles'),
          resolve(__dirname, 'node_modules')
        ]
      }
    },
    // PostCSS配置
    postcss: {
      plugins: [
        require('autoprefixer'),
        require('cssnano')({
          preset: 'default'
        })
      ]
    },
    // 开发时的源映射
    devSourcemap: true
  },
  resolve: {
    alias: {
      '@': resolve(__dirname, 'src'),
      '@styles': resolve(__dirname, 'src/styles')
    }
  },
  build: {
    // CSS代码分割
    cssCodeSplit: true,
    // 生成源映射
    sourcemap: true,
    rollupOptions: {
      output: {
        // CSS文件命名
        assetFileNames: (assetInfo) => {
          if (assetInfo.name.endsWith('.css')) {
            return 'css/[name].[hash][extname]';
          }
          return 'assets/[name].[hash][extname]';
        }
      }
    }
  }
});

Gulp工作流

// gulpfile.js
const gulp = require('gulp');
const sass = require('gulp-sass')(require('sass'));
const postcss = require('gulp-postcss');
const autoprefixer = require('autoprefixer');
const cssnano = require('cssnano');
const sourcemaps = require('gulp-sourcemaps');
const rename = require('gulp-rename');
const browserSync = require('browser-sync').create();
const del = require('del');

// 路径配置
const paths = {
  styles: {
    src: 'src/styles/**/*.scss',
    dest: 'dist/css/',
    main: 'src/styles/main.scss'
  },
  html: {
    src: 'src/**/*.html',
    dest: 'dist/'
  }
};

// 清理任务
function clean() {
  return del(['dist']);
}

// SCSS编译任务
function styles() {
  return gulp.src(paths.styles.main)
    .pipe(sourcemaps.init())
    .pipe(sass({
      includePaths: ['node_modules'],
      outputStyle: 'expanded'
    }).on('error', sass.logError))
    .pipe(postcss([
      autoprefixer(),
      cssnano()
    ]))
    .pipe(sourcemaps.write('.'))
    .pipe(gulp.dest(paths.styles.dest))
    .pipe(browserSync.stream());
}

// 生产环境样式任务
function stylesProd() {
  return gulp.src(paths.styles.main)
    .pipe(sass({
      includePaths: ['node_modules'],
      outputStyle: 'compressed'
    }).on('error', sass.logError))
    .pipe(postcss([
      autoprefixer(),
      cssnano({
        preset: ['default', {
          discardComments: { removeAll: true }
        }]
      })
    ]))
    .pipe(rename({ suffix: '.min' }))
    .pipe(gulp.dest(paths.styles.dest));
}

// HTML任务
function html() {
  return gulp.src(paths.html.src)
    .pipe(gulp.dest(paths.html.dest))
    .pipe(browserSync.stream());
}

// 监听任务
function watch() {
  browserSync.init({
    server: {
      baseDir: './dist'
    }
  });
  
  gulp.watch(paths.styles.src, styles);
  gulp.watch(paths.html.src, html);
}

// 任务导出
exports.clean = clean;
exports.styles = styles;
exports.stylesProd = stylesProd;
exports.html = html;
exports.watch = watch;
exports.build = gulp.series(clean, gulp.parallel(stylesProd, html));
exports.dev = gulp.series(clean, gulp.parallel(styles, html), watch);
exports.default = exports.dev;

自动化工作流

GitHub Actions CI/CD

# .github/workflows/ci.yml
name: CI/CD Pipeline

on:
  push:
    branches: [ main, develop ]
  pull_request:
    branches: [ main ]

jobs:
  lint-and-test:
    runs-on: ubuntu-latest
    
    steps:
    - name: Checkout code
      uses: actions/checkout@v3
      
    - name: Setup Node.js
      uses: actions/setup-node@v3
      with:
        node-version: '18'
        cache: 'npm'
        
    - name: Install dependencies
      run: npm ci
      
    - name: Lint SCSS
      run: npm run lint:scss
      
    - name: Run tests
      run: npm run test
      
    - name: Build project
      run: npm run build
      
    - name: Upload build artifacts
      uses: actions/upload-artifact@v3
      with:
        name: build-files
        path: dist/
        
  deploy:
    needs: lint-and-test
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'
    
    steps:
    - name: Download build artifacts
      uses: actions/download-artifact@v3
      with:
        name: build-files
        path: dist/
        
    - name: Deploy to production
      run: |
        # 部署脚本
        echo "Deploying to production..."

# SCSS Design System

## 项目结构

src/styles/ ├── abstracts/ # 变量、函数、混合器 ├── base/ # 重置、基础样式 ├── components/ # 组件样式 ├── layout/ # 布局样式 ├── pages/ # 页面特定样式 ├── themes/ # 主题样式 └── main.scss # 主入口文件

- 使用kebab-case命名
- 使用语义化名称
- 提供默认值

$border-radius-sm: 0.25rem !default;
