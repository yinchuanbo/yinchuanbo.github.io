创建自定义CSS框架

框架基础结构

// framework/_variables.scss
// 设计令牌
$colors: (
  primary: (
    50: #eff6ff,
    100: #dbeafe,
    500: #3b82f6,
    600: #2563eb,
    900: #1e3a8a
  ),
  gray: (
    50: #f9fafb,
    100: #f3f4f6,
    500: #6b7280,
    600: #4b5563,
    900: #111827
  )
);

$typography: (
  font-families: (
    sans: ('Inter', system-ui, sans-serif),
    serif: ('Merriweather', Georgia, serif),
    mono: ('Fira Code', Consolas, monospace)
  ),
  font-sizes: (
    xs: 0.75rem,
    sm: 0.875rem,
    base: 1rem,
    lg: 1.125rem,
    xl: 1.25rem,
    2xl: 1.5rem,
    3xl: 1.875rem,
    4xl: 2.25rem
  ),
  font-weights: (
    light: 300,
    normal: 400,
    medium: 500,
    semibold: 600,
    bold: 700
  )
);

$spacing: (
  0: 0,
  1: 0.25rem,
  2: 0.5rem,
  3: 0.75rem,
  4: 1rem,
  5: 1.25rem,
  6: 1.5rem,
  8: 2rem,
  10: 2.5rem,
  12: 3rem,
  16: 4rem,
  20: 5rem
);

$breakpoints: (
  sm: 640px,
  md: 768px,
  lg: 1024px,
  xl: 1280px,
  2xl: 1536px
);

// framework/_functions.scss
@function color($name, $variant: 500) {
  $color-map: map-get($colors, $name);
  @return map-get($color-map, $variant);
}

@function font-family($name) {
  $families: map-get($typography, font-families);
  @return map-get($families, $name);
}

@function font-size($name) {
  $sizes: map-get($typography, font-sizes);
  @return map-get($sizes, $name);
}

@function font-weight($name) {
  $weights: map-get($typography, font-weights);
  @return map-get($weights, $name);
}

@function spacing($name) {
  @return map-get($spacing, $name);
}

// framework/_mixins.scss
@mixin respond-to($breakpoint) {
  $size: map-get($breakpoints, $breakpoint);
  @media (min-width: #{$size}) {
    @content;
  }
}

@mixin button-variant($bg-color, $text-color: white) {
  background-color: $bg-color;
  color: $text-color;
  border: 1px solid $bg-color;
  
  &:hover {
    background-color: darken($bg-color, 8%);
    border-color: darken($bg-color, 10%);
  }
  
  &:focus {
    box-shadow: 0 0 0 3px rgba($bg-color, 0.3);
  }
  
  &:active {
    background-color: darken($bg-color, 12%);
  }
}

@mixin card($padding: spacing(6), $shadow: true) {
  background-color: white;
  border-radius: 0.5rem;
  padding: $padding;
  
  @if $shadow {
    box-shadow: 0 4px 6px -1px rgba(0, 0, 0, 0.1);
  }
}

// framework/_components.scss
// 按钮组件
.btn {
  display: inline-flex;
  align-items: center;
  justify-content: center;
  padding: spacing(2) spacing(4);
  font-size: font-size(base);
  font-weight: font-weight(medium);
  border-radius: 0.375rem;
  border: 1px solid transparent;
  cursor: pointer;
  transition: all 0.2s ease;
  text-decoration: none;
  
  &:focus {
    outline: none;
  }
  
  &.btn-primary {
    @include button-variant(color(primary));
  }
  
  &.btn-secondary {
    @include button-variant(color(gray, 600));
  }
  
  &.btn-sm {
    padding: spacing(1) spacing(3);
    font-size: font-size(sm);
  }
  
  &.btn-lg {
    padding: spacing(3) spacing(6);
    font-size: font-size(lg);
  }
}

// 卡片组件
.card {
  @include card;
  
  &.card-sm {
    @include card(spacing(4));
  }
  
  &.card-lg {
    @include card(spacing(8));
  }
}

// 网格系统
.container {
  width: 100%;
  margin: 0 auto;
  padding: 0 spacing(4);
  
  @include respond-to(sm) { max-width: 640px; }
  @include respond-to(md) { max-width: 768px; }
  @include respond-to(lg) { max-width: 1024px; }
  @include respond-to(xl) { max-width: 1280px; }
}

.row {
  display: flex;
  flex-wrap: wrap;
  margin: 0 -#{spacing(2)};
}

.col {
  padding: 0 spacing(2);
  flex: 1;
  
  @for $i from 1 through 12 {
    &.col-#{$i} {
      flex: 0 0 percentage($i / 12);
      max-width: percentage($i / 12);
    }
  }
}

这些示例展示了如何：

  1. 自定义现有框架：通过变量覆盖和组件扩展
  2. 混合使用：结合SCSS的强大功能和框架的便利性
  3. 创建自定义框架：构建符合项目需求的设计系统
  4. 保持一致性：在不同框架间维护设计一致性

选择合适的集成方式取决于项目需求、团队偏好和维护成本考虑。
