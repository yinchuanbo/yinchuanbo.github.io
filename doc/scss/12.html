错误处理和调试

SCSS | 2025.07.30

SCSS提供了多种错误处理和调试工具，帮助开发者编写更健壮的样式代码。

@error指令

基本错误处理

// 参数验证
@mixin button($size: medium) {
  $valid-sizes: small, medium, large;
  
  @if not index($valid-sizes, $size) {
    @error "Invalid button size '#{$size}'. Valid sizes are: #{$valid-sizes}";
  }
  
  @if $size == small {
    padding: 0.25rem 0.5rem;
    font-size: 0.875rem;
  } @else if $size == medium {
    padding: 0.5rem 1rem;
    font-size: 1rem;
  } @else if $size == large {
    padding: 0.75rem 1.5rem;
    font-size: 1.125rem;
  }
}

// 使用示例
.btn {
  @include button(medium); // ✅ 正确
  // @include button(extra-large); // ❌ 会抛出错误
}

类型检查

// 类型验证函数
@function validate-type($value, $expected-type, $name: 'value') {
  $actual-type: type-of($value);
  
  @if $actual-type != $expected-type {
    @error "#{$name} must be of type #{$expected-type}, got #{$actual-type}";
  }
  
  @return $value;
}

// 数值范围验证
@function validate-range($value, $min, $max, $name: 'value') {
  @if type-of($value) != number {
    @error "#{$name} must be a number";
  }
  
  @if $value < $min or $value > $max {
    @error "#{$name} must be between #{$min} and #{$max}, got #{$value}";
  }
  
  @return $value;
}

// 使用验证函数
@mixin opacity($value) {
  $value: validate-type($value, number, 'opacity');
  $value: validate-range($value, 0, 1, 'opacity');
  
  opacity: $value;
}

.transparent {
  @include opacity(0.5); // ✅ 正确
  // @include opacity(1.5); // ❌ 会抛出错误
  // @include opacity('half'); // ❌ 会抛出错误
}

依赖检查

// 检查必需的变量
@function require-variable($variable-name) {
  @if not variable-exists($variable-name) {
    @error "Required variable '#{$variable-name}' is not defined";
  }
}

// 检查必需的混合器
@function require-mixin($mixin-name) {
  @if not mixin-exists($mixin-name) {
    @error "Required mixin '#{$mixin-name}' is not defined";
  }
}

// 使用依赖检查
@mixin themed-component {
  @if not variable-exists('theme-colors') {
    @error "themed-component requires $theme-colors map to be defined";
  }
  
  @if not map-has-key($theme-colors, primary) {
    @error "$theme-colors map must contain 'primary' key";
  }
  
  background-color: map-get($theme-colors, primary);
}

配置警告

$enable-flex: true !default;
$enable-grid: false !default;

@mixin layout-container {
  @if not $enable-flex and not $enable-grid {
    @warn "Neither flexbox nor grid is enabled. Layout may not work as expected.";
  }
  
  @if $enable-flex and $enable-grid {
    @warn "Both flexbox and grid are enabled. This may cause conflicts.";
  }
  
  @if $enable-flex {
    display: flex;
  } @else if $enable-grid {
    display: grid;
  } @else {
    display: block;
  }
}

@mixin css-grid($columns: 1fr) {

    @warn "CSS Grid has limited support in IE11. Consider using flexbox fallback.";
  }

    // IE11 fallback

条件调试

$debug-mode: false !default;

@mixin debug-log($message) {
  @if $debug-mode {
    @debug $message;
  }
}

@function get-color($name) {
  @include debug-log("Getting color: #{$name}");
  
  $colors: (
    primary: #007bff,
    secondary: #6c757d,
    success: #28a745
  );
  
  @if not map-has-key($colors, $name) {
    @include debug-log("Color '#{$name}' not found, using default");
    @return #333;
  }
  
  $color: map-get($colors, $name);
  @include debug-log("Returning color: #{$color}");
  
  @return $color;
}

性能调试

@function benchmark($name) {
  @debug "Starting benchmark: #{$name}";
  // 在实际应用中，这里可能会有时间戳逻辑
  @return $name;
}

@function end-benchmark($name) {
  @debug "Ending benchmark: #{$name}";
  @return $name;
}

// 使用性能调试
@mixin expensive-calculation($iterations: 100) {
  $benchmark: benchmark('expensive-calculation');
  
  $result: 0;
  @for $i from 1 through $iterations {
    $result: $result + ($i * 2);
  }
  
  $end: end-benchmark($benchmark);
  
  // 使用计算结果
  width: #{$result}px;
}

错误恢复策略

默认值策略

@function safe-get($map, $key, $default: null) {
  @if type-of($map) != map {
    @warn "Expected map, got #{type-of($map)}. Using default value.";
    @return $default;
  }
  
  @if map-has-key($map, $key) {
    @return map-get($map, $key);
  } @else {
    @warn "Key '#{$key}' not found in map. Using default value.";
    @return $default;
  }
}

// 使用安全获取函数
$colors: (
  primary: #007bff,
  secondary: #6c757d
);

.component {
  color: safe-get($colors, primary, #333);
  background: safe-get($colors, tertiary, #f8f9fa); // 使用默认值
}

类型转换策略

@function to-number($value) {
  @if type-of($value) == number {
    @return $value;
  }
  
  @if type-of($value) == string {
    // 尝试解析字符串中的数字
    @if str-index($value, 'px') {
      @return str-slice($value, 1, str-index($value, 'px') - 1) * 1px;
    }
    @if str-index($value, 'rem') {
      @return str-slice($value, 1, str-index($value, 'rem') - 1) * 1rem;
    }
  }
  
  @warn "Cannot convert '#{$value}' to number. Using 0.";
  @return 0;
}

@function safe-divide($dividend, $divisor) {
  $dividend: to-number($dividend);
  $divisor: to-number($divisor);
  
  @if $divisor == 0 {
    @error "Division by zero is not allowed";
  }
  
  @return $dividend / $divisor;
}

渐进增强策略

@mixin progressive-enhancement($feature, $fallback: null) {
  @if $fallback {
    #{$fallback};
  }
  
  @supports (#{$feature}) {
    #{$feature};
  }
}

// 使用渐进增强
.grid-container {
  @include progressive-enhancement(
    $feature: (display: grid),
    $fallback: (display: flex, flex-wrap: wrap)
  );
  
  grid-template-columns: repeat(auto-fit, minmax(250px, 1fr));
  gap: 1rem;
  
  // Flexbox fallback
  & > * {
    flex: 1 1 250px;
    margin: 0.5rem;
  }
}

测试和验证

单元测试风格的验证

@function test-function($function-name, $input, $expected) {
  $result: call($function-name, $input);
  
  @if $result != $expected {
    @error "Test failed for #{$function-name}: expected #{$expected}, got #{$result}";
  } @else {
    @debug "✓ Test passed for #{$function-name}";
  }
  
  @return true;
}

// 测试示例
@function double($value) {
  @return $value * 2;
}

// 运行测试
$test1: test-function('double', 5, 10);
$test2: test-function('double', 2.5, 5);

集成测试

@mixin test-suite($name) {
  @debug "Running test suite: #{$name}";
  @content;
  @debug "Test suite completed: #{$name}";
}

@include test-suite('Color Functions') {
  $test-color: #007bff;
  
  @if lightness($test-color) < 50% {
    @debug "✓ Color is dark";
  } @else {
    @debug "✓ Color is light";
  }
  
  @if saturation($test-color) > 50% {
    @debug "✓ Color is saturated";
  } @else {
    @debug "✓ Color is desaturated";
  }
}
