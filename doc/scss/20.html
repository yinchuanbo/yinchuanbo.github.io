@use和@forward的高级用法

// _config.scss - 配置模块
$prefix: 'ui' !default;
$enable-dark-mode: true !default;
$enable-animations: true !default;

// 配置验证
@if not index(('ui', 'app', 'site'), $prefix) {
  @error "Invalid prefix '#{$prefix}'. Must be one of: ui, app, site";
}

// _theme.scss - 主题模块
@use 'config';

$themes: (
  light: (
    background: #ffffff,
    text: #333333,
    primary: #007bff
  ),
  dark: (
    background: #1a1a1a,
    text: #ffffff,
    primary: #66b3ff
  )
) !default;

@function get-theme-color($theme, $color) {
  $theme-map: map-get($themes, $theme);
  @if not $theme-map {
    @error "Theme '#{$theme}' not found";
  }
  @return map-get($theme-map, $color);
}

// 主题混合器
@mixin theme-colors($theme-name) {
  $theme: map-get($themes, $theme-name);
  
  @each $property, $value in $theme {
    --color-#{$property}: #{$value};
  }
}

// _components.scss - 组件模块
@use 'config';
@use 'theme';

// 使用命名空间
.#{config.$prefix}-button {
  background-color: var(--color-primary, #{theme.get-theme-color(light, primary)});
  
  @if config.$enable-animations {
    transition: all 0.3s ease;
  }
}

// _index.scss - 统一导出
@forward 'config';
@forward 'theme';
@forward 'components';

// 重新导出时添加前缀
@forward 'utilities' as util-*;

// 条件导出
@forward 'dark-theme' if config.$enable-dark-mode;

模块系统的依赖管理

// _dependencies.scss - 依赖管理
$loaded-modules: () !default;

@function is-module-loaded($module-name) {
  @return index($loaded-modules, $module-name) != null;
}

@mixin register-module($module-name) {
  @if is-module-loaded($module-name) {
    @warn "Module '#{$module-name}' is already loaded";
  } @else {
    $loaded-modules: append($loaded-modules, $module-name) !global;
  }
}

@mixin require-module($module-name) {
  @if not is-module-loaded($module-name) {
    @error "Required module '#{$module-name}' is not loaded";
  }
}

// _base.scss
@use 'dependencies';
@include dependencies.register-module('base');

// _components.scss
@use 'dependencies';
@include dependencies.require-module('base');
@include dependencies.register-module('components');

// CSS自定义属性生成器
@mixin generate-css-vars($map, $prefix: '') {
  @each $key, $value in $map {
    @if type-of($value) == map {
      @include generate-css-vars($value, '#{$prefix}#{$key}-');
    } @else {
      --#{$prefix}#{$key}: #{$value};
    }
  }
}

CSS Grid和Subgrid

// 现代网格系统
@mixin modern-grid($columns: auto-fit, $min-width: 250px, $gap: 1rem) {
  display: grid;
  gap: $gap;
  
  @if $columns == auto-fit {
    grid-template-columns: repeat(auto-fit, minmax($min-width, 1fr));
  } @else if $columns == auto-fill {
    grid-template-columns: repeat(auto-fill, minmax($min-width, 1fr));
  } @else {
    grid-template-columns: repeat($columns, 1fr);
  }
}

// Subgrid支持
@mixin subgrid($axis: both) {
  @supports (grid-template-rows: subgrid) {
    @if $axis == both {
      grid-template-rows: subgrid;
      grid-template-columns: subgrid;
    } @else if $axis == rows {
      grid-template-rows: subgrid;
    } @else if $axis == columns {
      grid-template-columns: subgrid;
    }
  }
}

// 复杂网格布局
.magazine-layout {
  @include modern-grid(auto-fit, 300px, 2rem);
  
  .article {
    display: grid;
    grid-template-rows: auto 1fr auto;
    
    @include subgrid(rows);
    
    .article-header {
      grid-row: 1;
    }
    
    .article-content {
      grid-row: 2;
    }
    
    .article-footer {
      grid-row: 3;
    }
  }
  
  // 特色文章跨越多列
  .featured-article {
    grid-column: 1 / -1;
    
    @media (min-width: 1200px) {
      grid-column: span 2;
    }
  }
}

微交互设计

// 微交互混合器
@mixin micro-interaction($trigger: hover, $effect: scale, $intensity: 1.05) {
  transition: all 0.2s ease;
  
  @if $trigger == hover {
    &:hover {
      @include apply-micro-effect($effect, $intensity);
    }
  } @else if $trigger == focus {
    &:focus {
      @include apply-micro-effect($effect, $intensity);
    }
  } @else if $trigger == active {
    &:active {
      @include apply-micro-effect($effect, $intensity);
    }
  }
}

@mixin apply-micro-effect($effect, $intensity) {
  @if $effect == scale {
    transform: scale($intensity);
  } @else if $effect == lift {
    transform: translateY(-#{$intensity}px);
    box-shadow: 0 #{$intensity * 2}px #{$intensity * 4}px rgba(0, 0, 0, 0.1);
  } @else if $effect == glow {
    box-shadow: 0 0 #{$intensity * 10}px rgba(0, 123, 255, 0.3);
  } @else if $effect == rotate {
    transform: rotate(#{$intensity}deg);
  }
}

动态样式加载

// 条件样式加载
@mixin load-if-needed($feature, $fallback: null) {
  @if $feature {
    @content;
  } @else if $fallback {
    #{$fallback};
  }
}

// 功能检测样式
.modern-features {
  // 基础样式
  display: block;
  
  // 支持Grid时的样式
  @include load-if-needed($supports-grid) {
    display: grid;
    grid-template-columns: repeat(auto-fit, minmax(250px, 1fr));
  }
  
  // 支持Flexbox时的回退
  @include load-if-needed($supports-flexbox and not $supports-grid) {
    display: flex;
    flex-wrap: wrap;
  }
}
