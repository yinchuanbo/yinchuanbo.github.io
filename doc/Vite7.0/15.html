unplugin-vue-components

npm install unplugin-vue-components -D

// vite.config.js
import { defineConfig } from 'vite'
import Components from 'unplugin-vue-components/vite'
import { ElementPlusResolver, AntDesignVueResolver } from 'unplugin-vue-components/resolvers'

export default defineConfig({
  plugins: [
    Components({
      // 自动导入组件的目录
      dirs: ['src/components'],
      
      // 组件的有效文件扩展名
      extensions: ['vue', 'jsx', 'tsx'],
      
      // 搜索子目录
      deep: true,
      
      // 解析器
      resolvers: [
        // Element Plus
        ElementPlusResolver({
          // 自动导入样式
          importStyle: 'sass'
        }),
        
        // Ant Design Vue
        AntDesignVueResolver({
          importStyle: false
        })
      ],
      
      // 包含的文件
      include: [/\.vue$/, /\.vue\?vue/, /\.jsx$/, /\.tsx$/],
      
      // 排除的文件
      exclude: [/[\\/]node_modules[\\/]/, /[\\/]\.git[\\/]/, /[\\/]\.nuxt[\\/]/],
      
      // 生成类型声明
      dts: true,
      
      // 类型声明文件位置
      dtsPath: 'types/components.d.ts',
      
      // 自定义组件解析
      resolvers: [
        // 自定义解析器
        (componentName) => {
          if (componentName.startsWith('My'))
            return { name: componentName, from: '@/components' }
        }
      ]
    })
  ]
})

<!-- 无需手动导入，自动识别并导入组件 -->
<template>
  <div>
    <!-- Element Plus 组件 -->
    <el-button type="primary">按钮</el-button>
    <el-input v-model="input" placeholder="请输入内容" />
    
    <!-- Ant Design Vue 组件 -->
    <a-button type="primary">按钮</a-button>
    <a-input v-model:value="input" placeholder="请输入内容" />
    
    <!-- 自定义组件 -->
    <MyCustomComponent />
    <UserCard :user="user" />
  </div>
</template>

vite-plugin-pages

npm install vite-plugin-pages -D

// vite.config.js
import { defineConfig } from 'vite'
import Pages from 'vite-plugin-pages'

export default defineConfig({
  plugins: [
    Pages({
      // 页面目录
      dirs: [
        { dir: 'src/pages', baseRoute: '' },
        { dir: 'src/features/**/pages', baseRoute: 'features' }
      ],
      
      // 文件扩展名
      extensions: ['vue', 'js', 'ts', 'jsx', 'tsx'],
      
      // 排除文件
      exclude: ['**/components/**', '**/*.test.*'],
      
      // 路由块解析
      routeBlockLang: 'yaml',
      
      // 导入模式
      importMode: 'async', // 'sync' | 'async'
      
      // 解析器
      resolver: 'vue',
      
      // 自定义路由
      extendRoute(route, parent) {
        if (route.path === '/') {
          // 首页添加别名
          route.alias = ['/home', '/index']
        }
        
        if (route.name === 'about') {
          // 添加元信息
          route.meta = { requiresAuth: false }
        }
        
        return route
      },
      
      // 路由生成回调
      onRoutesGenerated: (routes) => {
        console.log('Generated routes:', routes)
      }
    })
  ]
})

vite-plugin-compression

npm install vite-plugin-compression -D

// vite.config.js
import { defineConfig } from 'vite'
import { compression } from 'vite-plugin-compression'

export default defineConfig({
  plugins: [
    // Gzip 压缩
    compression({
      // 压缩算法
      algorithm: 'gzip',
      
      // 文件扩展名
      ext: '.gz',
      
      // 压缩阈值，小于此大小的文件不压缩
      threshold: 1024,
      
      // 压缩比率，小于此比率的文件不压缩
      compressionRatio: 0.8,
      
      // 包含的文件
      include: /\.(js|mjs|json|css|html)$/i,
      
      // 排除的文件
      exclude: /\.(br|gz)$/i,
      
      // 删除原文件
      deleteOriginFile: false,
      
      // 压缩选项
      compressionOptions: {
        level: 9
      }
    }),
    
    // Brotli 压缩
    compression({
      algorithm: 'brotliCompress',
      ext: '.br',
      threshold: 1024,
      compressionRatio: 0.8,
      include: /\.(js|mjs|json|css|html)$/i,
      compressionOptions: {
        level: 11
      }
    })
  ]
})

// mock/user.js
import { Random } from 'mockjs'

const users = Array.from({ length: 50 }, (_, index) => ({
  id: index + 1,
  name: Random.cname(),
  email: Random.email(),
  avatar: Random.image('100x100', Random.color(), '#FFF', 'Avatar'),
  role: Random.pick(['admin', 'user', 'editor']),
  status: Random.pick(['active', 'inactive']),
  createdAt: Random.datetime(),
  updatedAt: Random.datetime()
}))

export default [
  {
    url: '/api/users',
    method: 'get',
    response: ({ query }) => {
      const { page = 1, pageSize = 10, search = '' } = query
      
      let filteredUsers = users
      if (search) {
        filteredUsers = users.filter(user => 
          user.name.includes(search) || user.email.includes(search)
        )
      }
      
      const start = (page - 1) * pageSize
      const end = start + parseInt(pageSize)
      const list = filteredUsers.slice(start, end)
      
      return {
        code: 200,
        message: 'success',
        data: {
          list,
          total: filteredUsers.length,
          page: parseInt(page),
          pageSize: parseInt(pageSize)
        }
      }
    }
  },
  
  {
    url: '/api/users/:id',
    method: 'get',
    response: ({ params }) => {
      const user = users.find(u => u.id === parseInt(params.id))
      
      if (!user) {
        return {
          code: 404,
          message: 'User not found'
        }
      }
      
      return {
        code: 200,
        message: 'success',
        data: user
      }
    }
  }
]

插件组合示例

完整的开发环境配置

// vite.config.js
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import Components from 'unplugin-vue-components/vite'
import AutoImport from 'unplugin-auto-import/vite'
import { ElementPlusResolver } from 'unplugin-vue-components/resolvers'
import Pages from 'vite-plugin-pages'
import Layouts from 'vite-plugin-layouts'
import WindiCSS from 'vite-plugin-windicss'
import { viteMockServe } from 'vite-plugin-mock'
import { compression } from 'vite-plugin-compression'
import { ViteImageOptimize } from 'vite-plugin-imagemin'
import progress from 'vite-plugin-progress'

export default defineConfig(({ command, mode }) => {
  const plugins = [
    // 基础框架
    vue(),
    
    // 自动导入
    AutoImport({
      imports: ['vue', 'vue-router', 'pinia'],
      resolvers: [ElementPlusResolver()],
      dts: true
    }),
    
    // 组件自动导入
    Components({
      resolvers: [ElementPlusResolver()],
      dts: true
    }),
    
    // 文件路由
    Pages({
      dirs: 'src/pages',
      extensions: ['vue']
    }),
    
    // 布局系统
    Layouts(),
    
    // CSS 框架
    WindiCSS(),
    
    // 构建进度
    progress()
  ]
  
  // 开发环境插件
  if (command === 'serve') {
    plugins.push(
      viteMockServe({
        mockPath: 'mock',
        enable: true,
        logger: true
      })
    )
  }
  
  // 生产环境插件
  if (command === 'build') {
    plugins.push(
      // 图片优化
      ViteImageOptimize({
        mozjpeg: { quality: 80 },
        pngquant: { quality: [0.65, 0.8] }
      }),
      
      // 文件压缩
      compression({
        algorithm: 'gzip',
        threshold: 1024
      })
    )
  }
  
  return {
    plugins,
    resolve: {
      alias: {
        '@': path.resolve(__dirname, 'src')
      }
    }
  }
})

总结

  • 丰富的 UI 框架集成方案

合理选择和组合这些社区插件，能够显著提升开发效率和项目质量，构建出功能完整的现代前端应用。
