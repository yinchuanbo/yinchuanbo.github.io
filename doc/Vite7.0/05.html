静态资源处理

Vite7.0 | 2025.07.31

Vite 提供了强大的静态资源处理能力，支持多种资源类型的导入、优化和处理。本章将详细介绍如何在 Vite 中处理各种静态资源。

资源导入方式

基本资源导入

图片优化配置

// vite.config.js
import { defineConfig } from 'vite'
import { ViteImageOptimize } from 'vite-plugin-imagemin'

export default defineConfig({
  plugins: [
    ViteImageOptimize({
      // JPEG 优化
      mozjpeg: {
        quality: 80,
        progressive: true
      },
      // PNG 优化
      pngquant: {
        quality: [0.65, 0.8],
        speed: 4
      },
      // SVG 优化
      svgo: {
        plugins: [
          { name: 'removeViewBox', active: false },
          { name: 'removeEmptyAttrs', active: false },
          { name: 'removeUselessStrokeAndFill', active: false }
        ]
      },
      // WebP 转换
      webp: {
        quality: 75
      }
    })
  ],
  
  build: {
    // 小于 4kb 的图片将被内联为 base64
    assetsInlineLimit: 4096
  }
})

动态字体加载

// src/utils/fontLoader.js
export async function loadFont(fontName, fontUrl) {
  const font = new FontFace(fontName, `url(${fontUrl})`)
  
  try {
    await font.load()
    document.fonts.add(font)
    return true
  } catch (error) {
    console.error('Font loading failed:', error)
    return false
  }
}

// 使用示例
import customFontUrl from '@/assets/fonts/custom-font.woff2'

  await loadFont('CustomFont', customFontUrl)
}

Web Workers

基础 Worker 使用

// src/workers/calculator.js
self.onmessage = function(e) {
  const { operation, numbers } = e.data
  
  let result
  switch (operation) {
    case 'add':
      result = numbers.reduce((sum, num) => sum + num, 0)
      break
    case 'multiply':
      result = numbers.reduce((product, num) => product * num, 1)
      break
    default:
      result = 0
  }
  
  self.postMessage({ result })
}

// src/components/Calculator.jsx
import CalculatorWorker from '../workers/calculator.js?worker'

function Calculator() {
  const [result, setResult] = useState(0)
  const workerRef = useRef()
  
  useEffect(() => {
    workerRef.current = new CalculatorWorker()
    
    workerRef.current.onmessage = (e) => {
      setResult(e.data.result)
    }
    
    return () => {
      workerRef.current?.terminate()
    }
  }, [])
  
  const calculate = (operation, numbers) => {
    workerRef.current?.postMessage({ operation, numbers })
  }
  
  return (
    <div>
      <button onClick={() => calculate('add', [1, 2, 3, 4, 5])}>
        Calculate Sum
      </button>
      <p>Result: {result}</p>
    </div>
  )
}

共享 Worker

// src/workers/shared-worker.js
const connections = []

self.onconnect = function(e) {
  const port = e.ports[0]
  connections.push(port)
  
  port.onmessage = function(e) {
    // 广播消息给所有连接
    connections.forEach(conn => {
      if (conn !== port) {
        conn.postMessage(e.data)
      }
    })
  }
}

资源优化策略

资源内联配置

// vite.config.js
export default defineConfig({
  build: {
    assetsInlineLimit: 4096, // 4kb 以下内联
    
    rollupOptions: {
      output: {
        assetFileNames: (assetInfo) => {
          const info = assetInfo.name.split('.')
          const ext = info[info.length - 1]
          
          // 根据文件类型分类存放
          if (/\.(png|jpe?g|gif|svg|webp)$/i.test(assetInfo.name)) {
            return `images/[name]-[hash].${ext}`
          }
          if (/\.(woff2?|eot|ttf|otf)$/i.test(assetInfo.name)) {
            return `fonts/[name]-[hash].${ext}`
          }
          if (/\.(css)$/i.test(assetInfo.name)) {
            return `styles/[name]-[hash].${ext}`
          }
          return `assets/[name]-[hash].${ext}`
        }
      }
    }
  }
})

资源压缩

import { defineConfig } from 'vite'
import { ViteImageOptimize } from 'vite-plugin-imagemin'
import { compression } from 'vite-plugin-compression'

export default defineConfig({
  plugins: [
    // 图片压缩
    ViteImageOptimize({
      gifsicle: { optimizationLevel: 7 },
      mozjpeg: { quality: 80 },
      pngquant: { quality: [0.65, 0.8] }

      algorithm: 'gzip',
      ext: '.gz'
    }),

    })
  ]
})

故障排除

常见资源问题

// 1. 资源路径问题
// ❌ 错误的相对路径
import logo from '../../../assets/logo.png'

// ✅ 使用别名
import logo from '@/assets/logo.png'

// 2. 动态导入路径问题
// ❌ 完全动态的路径
const module = await import(dynamicPath)

// ✅ 部分静态路径
const module = await import(`./modules/${moduleName}.js`)

// 3. 资源未找到问题
// 检查文件是否存在
import.meta.glob('./assets/images/*.png', { eager: true })

资源加载错误处理

// 图片加载错误处理
function ImageWithFallback({ src, fallback, alt, ...props }) {
  const [imgSrc, setImgSrc] = useState(src)
  
  const handleError = () => {
    setImgSrc(fallback)
  }
  
  return (
    <img 
      src={imgSrc} 
      alt={alt} 
      onError={handleError}
      {...props} 
    />
  )
}
