Vite7.0 | 2025.07.31

// 使用示例 - Button.jsx
import { defineComponent } from 'vue'

export default defineComponent({
  props: {
    type: {
      type: String,
      default: 'default'
    },
    size: {
      type: String,
      default: 'medium'
    }
  },
  
  emits: ['click'],
  
  setup(props, { emit, slots }) {
    const handleClick = (event) => {
      emit('click', event)
    }
    
    return () => (
      <button
        class={[
          'btn',
          `btn--${props.type}`,
          `btn--${props.size}`
        ]}
        onClick={handleClick}
      >
        {slots.default?.()}
      </button>
    )
  }
})

插件组合最佳实践

完整配置示例

// vite.config.js
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import vueJsx from '@vitejs/plugin-vue-jsx'
import legacy from '@vitejs/plugin-legacy'
import { VitePWA } from 'vite-plugin-pwa'
import { visualizer } from 'rollup-plugin-visualizer'
import eslint from 'vite-plugin-eslint'
import { viteMockServe } from 'vite-plugin-mock'

export default defineConfig(({ command, mode }) => {
  const plugins = [
    // 基础框架插件
    vue({
      script: {
        defineModel: true,
        propsDestructure: true
      }
    }),
    vueJsx()
  ]
  
  // 开发环境插件
  if (command === 'serve') {
    plugins.push(
      eslint({
        include: ['src/**/*.{js,jsx,ts,tsx,vue}'],
        cache: true
      }),
      viteMockServe({
        mockPath: 'mock',
        enable: true,
        logger: true
      })
    )
  }
  
  // 生产环境插件
  if (command === 'build') {
    plugins.push(
      legacy({
        targets: ['defaults', 'not IE 11']
      }),
      VitePWA({
        registerType: 'autoUpdate',
        workbox: {
          globPatterns: ['**/*.{js,css,html,ico,png,svg}']
        }
      })
    )
    
    // 分析模式
    if (mode === 'analyze') {
      plugins.push(
        visualizer({
          filename: 'dist/stats.html',
          open: true,
          gzipSize: true,
          brotliSize: true
        })
      )
    }
  }
  
  return {
    plugins,
    
    // 其他配置
    resolve: {
      alias: {
        '@': path.resolve(__dirname, 'src')
      }
    },
    
    build: {
      rollupOptions: {
        output: {
          manualChunks: {
            vendor: ['vue', 'vue-router'],
            utils: ['lodash-es', 'axios']
          }
        }
      }
    },
    
    test: {
      environment: 'jsdom',
      globals: true
    }
  }
})

插件配置管理

// config/plugins.js
export const createPlugins = (options = {}) => {
  const {
    command,
    mode,
    enableLegacy = true,
    enablePWA = true,
    enableMock = true,
    enableAnalyzer = false
  } = options
  
  const plugins = []
  
  // 基础插件
  plugins.push(
    vue(),
    vueJsx()
  )
  
  // 条件插件
  if (command === 'serve') {
    plugins.push(eslint())
    
    if (enableMock) {
      plugins.push(viteMockServe({ enable: true }))
    }
  }
  
  if (command === 'build') {
    if (enableLegacy) {
      plugins.push(legacy())
    }
    
    if (enablePWA) {
      plugins.push(VitePWA())
    }
    
    if (enableAnalyzer) {
      plugins.push(visualizer())
    }
  }
  
  return plugins
}
