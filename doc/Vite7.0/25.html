// src/utils/hmrDebugger.js
class HMRDebugger {
  constructor() {
    this.enabled = import.meta.env.DEV
    this.logs = []
    this.maxLogs = 100
    
    if (this.enabled && import.meta.hot) {
      this.setupHMRListeners()
    }
  }
  
  setupHMRListeners() {
    // 监听 HMR 事件
    import.meta.hot.on('vite:beforeUpdate', (payload) => {
      this.log('beforeUpdate', payload)
    })
    
    import.meta.hot.on('vite:afterUpdate', (payload) => {
      this.log('afterUpdate', payload)
    })
    
    import.meta.hot.on('vite:beforeFullReload', (payload) => {
      this.log('beforeFullReload', payload)
    })
    
    import.meta.hot.on('vite:error', (payload) => {
      this.log('error', payload)
    })
    
    import.meta.hot.on('vite:invalidate', (payload) => {
      this.log('invalidate', payload)
    })
  }
  
  log(event, payload) {
    const logEntry = {
      timestamp: Date.now(),
      event,
      payload: JSON.parse(JSON.stringify(payload))
    }
    
    this.logs.push(logEntry)
    
    // 限制日志数量
    if (this.logs.length > this.maxLogs) {
      this.logs.shift()
    }
    
    console.group(`🔥 HMR ${event}`)
    console.log('Timestamp:', new Date(logEntry.timestamp).toISOString())
    console.log('Payload:', logEntry.payload)
    console.groupEnd()
  }
  
  getLogs() {
    return this.logs
  }
  
  clearLogs() {
    this.logs = []
  }
  
  // 导出日志
  exportLogs() {
    const data = JSON.stringify(this.logs, null, 2)
    const blob = new Blob([data], { type: 'application/json' })
    const url = URL.createObjectURL(blob)
    
    const a = document.createElement('a')
    a.href = url
    a.download = `hmr-logs-${Date.now()}.json`
    a.click()
    
    URL.revokeObjectURL(url)
  }
}

if (import.meta.hot) {
  // 接受自身更新
  import.meta.hot.accept((newModule) => {
    console.log('DebugComponent updated:', newModule)
    updateCount.value++
    lastUpdate.value = new Date().toISOString()
  })
  
  // 处理数据保持
  import.meta.hot.dispose((data) => {
    console.log('DebugComponent disposing, saving data:', {
      updateCount: updateCount.value,
      lastUpdate: lastUpdate.value
    })
    data.updateCount = updateCount.value
    data.lastUpdate = lastUpdate.value
  })
  
  // 恢复数据
  if (import.meta.hot.data) {
    console.log('DebugComponent restoring data:', import.meta.hot.data)
    updateCount.value = import.meta.hot.data.updateCount || 0
    lastUpdate.value = import.meta.hot.data.lastUpdate
  }
}

// scripts/analyze-deps.js
import { build } from 'vite'
import { visualizer } from 'rollup-plugin-visualizer'

async function analyzeDependencies() {
  console.log('🔍 Analyzing dependencies...')
  
  try {
    const result = await build({
      plugins: [
        visualizer({
          filename: 'dist/stats.html',
          open: true,
          gzipSize: true,
          brotliSize: true,
          template: 'treemap' // 'treemap' | 'sunburst' | 'network'
        })
      ],
      
      build: {
        rollupOptions: {
          plugins: [
            {
              name: 'dependency-analyzer',
              generateBundle(options, bundle) {
                const analysis = {
                  chunks: {},
                  dependencies: {},
                  totalSize: 0
                }
                
                Object.entries(bundle).forEach(([fileName, chunk]) => {
                  if (chunk.type === 'chunk') {
                    const size = chunk.code.length
                    analysis.totalSize += size
                    
                    analysis.chunks[fileName] = {
                      size,
                      modules: Object.keys(chunk.modules || {}),
                      imports: chunk.imports || [],
                      dynamicImports: chunk.dynamicImports || []
                    }
                    
                    // 分析依赖
                    Object.keys(chunk.modules || {}).forEach(moduleId => {
                      if (moduleId.includes('node_modules')) {
                        const packageName = extractPackageName(moduleId)
                        if (!analysis.dependencies[packageName]) {
                          analysis.dependencies[packageName] = {
                            size: 0,
                            chunks: []
                          }
                        }
                        analysis.dependencies[packageName].size += size
                        analysis.dependencies[packageName].chunks.push(fileName)
                      }
                    })
                  }
                })
                
                // 输出分析结果
                console.log('\n📊 Dependency Analysis:')
                console.log(`Total Size: ${Math.round(analysis.totalSize / 1024)}KB`)
                
                console.log('\n📦 Top Dependencies:')
                Object.entries(analysis.dependencies)
                  .sort(([,a], [,b]) => b.size - a.size)
                  .slice(0, 10)
                  .forEach(([name, info]) => {
                    console.log(`  ${name}: ${Math.round(info.size / 1024)}KB`)
                  })
                
                // 保存详细分析
                this.emitFile({
                  type: 'asset',
                  fileName: 'dependency-analysis.json',
                  source: JSON.stringify(analysis, null, 2)
                })
              }
            }
          ]
        }
      }
    })
    
    console.log('✅ Analysis complete!')
  } catch (error) {
    console.error('❌ Analysis failed:', error)
  }
}

function extractPackageName(moduleId) {
  const match = moduleId.match(/node_modules\/(@[^\/]+\/[^\/]+|[^\/]+)/)
  return match ? match[1] : 'unknown'
}

// scripts/build-performance.js
import { build } from 'vite'
import { performance } from 'perf_hooks'

class BuildPerformanceAnalyzer {
  constructor() {
    this.timings = new Map()
    this.memoryUsage = []
  }
  
  startTimer(name) {
    this.timings.set(name, performance.now())
  }
  
  endTimer(name) {
    const startTime = this.timings.get(name)
    if (startTime) {
      const duration = performance.now() - startTime
      console.log(`⏱️ ${name}: ${duration.toFixed(2)}ms`)
      return duration
    }
  }
  
  recordMemory(label) {
    const usage = process.memoryUsage()
    this.memoryUsage.push({
      label,
      timestamp: Date.now(),
      ...usage
    })
    
    console.log(`💾 ${label}:`, {
      rss: `${Math.round(usage.rss / 1024 / 1024)}MB`,
      heapUsed: `${Math.round(usage.heapUsed / 1024 / 1024)}MB`,
      heapTotal: `${Math.round(usage.heapTotal / 1024 / 1024)}MB`
    })
  }
  
  async analyzeBuild() {
    console.log('🚀 Starting build performance analysis...')
    
    this.recordMemory('Build Start')
    this.startTimer('Total Build')
    
    try {
      await build({
        plugins: [
          {
            name: 'performance-analyzer',
            buildStart() {
              analyzer.startTimer('Build Start')
              analyzer.recordMemory('Build Started')
            },
            
            buildEnd() {
              analyzer.endTimer('Build Start')
              analyzer.recordMemory('Build Ended')
            },
            
            generateBundle() {
              analyzer.startTimer('Generate Bundle')
              analyzer.recordMemory('Generate Bundle')
            },
            
            writeBundle() {
              analyzer.endTimer('Generate Bundle')
              analyzer.recordMemory('Write Bundle')
            }
          }
        ]
      })
      
      this.endTimer('Total Build')
      this.recordMemory('Build Complete')
      
      this.generateReport()
    } catch (error) {
      console.error('❌ Build failed:', error)
    }
  }
  
  generateReport() {
    console.log('\n📊 Build Performance Report:')
    
    // 内存使用趋势
    console.log('\n💾 Memory Usage:')
    this.memoryUsage.forEach(usage => {
      console.log(`  ${usage.label}: ${Math.round(usage.heapUsed / 1024 / 1024)}MB`)
    })
    
    // 性能建议
    console.log('\n💡 Performance Suggestions:')
    const maxMemory = Math.max(...this.memoryUsage.map(u => u.heapUsed))
    if (maxMemory > 1024 * 1024 * 1024) { // 1GB
      console.log('  - Consider increasing Node.js memory limit: --max-old-space-size=4096')
    }
    
    if (this.timings.has('Total Build')) {
      const totalTime = performance.now() - this.timings.get('Total Build')
      if (totalTime > 60000) { // 1 minute
        console.log('  - Build time is slow, consider:')
        console.log('    * Enabling parallel processing')
        console.log('    * Optimizing dependencies')
        console.log('    * Using faster minifier (esbuild)')
      }
    }
  }
}

// vite.config.js
export default defineConfig({
  plugins: [
    {
      name: 'dev-performance-monitor',
      configureServer(server) {
        const requestTimes = new Map()
        
        // 监控请求性能
        server.middlewares.use((req, res, next) => {
          const start = Date.now()
          const url = req.url
          
          // 记录开始时间
          requestTimes.set(req, start)
          
          // 监听响应结束
          res.on('finish', () => {
            const duration = Date.now() - start
            
            // 记录慢请求
            if (duration > 1000) { // 超过1秒
              console.warn(`🐌 Slow request: ${url} (${duration}ms)`)
            }
            
            // 记录所有请求（可选）
            if (process.env.DEBUG_REQUESTS) {
              console.log(`📡 ${req.method} ${url} - ${duration}ms`)
            }
            
            requestTimes.delete(req)
          })
          
          next()
        })
        
        // 定期报告性能统计
        setInterval(() => {
          const activeRequests = requestTimes.size
          if (activeRequests > 0) {
            console.log(`📊 Active requests: ${activeRequests}`)
          }
        }, 10000) // 每10秒
      }
    }
  ]
})

// src/utils/browserDebugger.js
class BrowserDebugger {
  constructor() {
    this.enabled = import.meta.env.DEV
    
    if (this.enabled) {
      this.setupConsoleCommands()
      this.setupPerformanceMonitoring()
    }
  }
  
  setupConsoleCommands() {
    // 暴露调试命令到控制台
    window.__VITE_DEBUG__ = {
      // 重新加载页面
      reload: () => {
        if (import.meta.hot) {
          import.meta.hot.send('vite:invalidate', { path: '*' })
        } else {
          location.reload()
        }
      },
      
      // 清除缓存
      clearCache: () => {
        if ('caches' in window) {
          caches.keys().then(names => {
            names.forEach(name => caches.delete(name))
          })
        }
        localStorage.clear()
        sessionStorage.clear()
        console.log('✅ Cache cleared')
      },
      
      // 显示模块信息
      showModules: () => {
        if (import.meta.hot) {
          console.table(Object.keys(import.meta.hot.data || {}))
        }
      },
      
      // 性能信息
      performance: () => {
        const navigation = performance.getEntriesByType('navigation')[0]
        const paint = performance.getEntriesByType('paint')
        
        console.group('🚀 Performance Metrics')
        console.log('DOM Content Loaded:', navigation.domContentLoadedEventEnd - navigation.navigationStart, 'ms')
        console.log('Load Complete:', navigation.loadEventEnd - navigation.navigationStart, 'ms')
        
        paint.forEach(entry => {
          console.log(`${entry.name}:`, entry.startTime, 'ms')
        })
        console.groupEnd()
      }
    }
    
    console.log('🛠️ Debug commands available: window.__VITE_DEBUG__')
  }
  
  setupPerformanceMonitoring() {
    // 监控长任务
    if ('PerformanceObserver' in window) {
      const observer = new PerformanceObserver((list) => {
        list.getEntries().forEach(entry => {
          if (entry.duration > 50) { // 超过50ms的长任务
            console.warn('🐌 Long task detected:', {
              duration: entry.duration,
              startTime: entry.startTime,
              name: entry.name
            })
          }
        })
      })
      
      observer.observe({ entryTypes: ['longtask'] })
    }
    
    // 监控内存使用
    if ('memory' in performance) {
      setInterval(() => {
        const memory = performance.memory
        const used = Math.round(memory.usedJSHeapSize / 1024 / 1024)
        const total = Math.round(memory.totalJSHeapSize / 1024 / 1024)
        const limit = Math.round(memory.jsHeapSizeLimit / 1024 / 1024)
        
        if (used / limit > 0.9) { // 内存使用超过90%
          console.warn('🚨 High memory usage:', { used, total, limit })
        }
      }, 30000) // 每30秒检查一次
    }
  }
}

// scripts/health-check.js
import fs from 'fs'
import path from 'path'
import { createRequire } from 'module'

const require = createRequire(import.meta.url)

class HealthChecker {
  constructor() {
    this.issues = []
    this.warnings = []
  }
  
  async runChecks() {
    console.log('🏥 Running Vite health checks...\n')
    
    this.checkNodeVersion()
    this.checkPackageJson()
    this.checkViteConfig()
    this.checkDependencies()
    this.checkFileStructure()
    this.checkEnvironment()
    
    this.generateReport()
  }
  
  checkNodeVersion() {
    const nodeVersion = process.version
    const majorVersion = parseInt(nodeVersion.slice(1).split('.')[0])
    
    if (majorVersion < 16) {
      this.issues.push(`Node.js version ${nodeVersion} is too old. Vite requires Node.js 16+`)
    } else if (majorVersion < 18) {
      this.warnings.push(`Node.js version ${nodeVersion} works but 18+ is recommended`)
    } else {
      console.log('✅ Node.js version:', nodeVersion)
    }
  }
  
  checkPackageJson() {
    try {
      const packageJson = JSON.parse(fs.readFileSync('package.json', 'utf-8'))
      
      // 检查 type 字段
      if (packageJson.type !== 'module') {
        this.warnings.push('Consider setting "type": "module" in package.json for better ES module support')
      }
      
      // 检查 Vite 版本
      const viteVersion = packageJson.devDependencies?.vite || packageJson.dependencies?.vite
      if (!viteVersion) {
        this.issues.push('Vite is not listed in dependencies')
      } else {
        console.log('✅ Vite version:', viteVersion)
      }
      
      // 检查脚本
      const scripts = packageJson.scripts || {}
      if (!scripts.dev && !scripts.serve) {
        this.warnings.push('No dev script found in package.json')
      }
      if (!scripts.build) {
        this.warnings.push('No build script found in package.json')
      }
      
    } catch (error) {
      this.issues.push('Cannot read package.json: ' + error.message)
    }
  }
  
  checkViteConfig() {
    const configFiles = [
      'vite.config.js',
      'vite.config.ts',
      'vite.config.mjs',
      'vite.config.cjs'
    ]
    
    const existingConfig = configFiles.find(file => fs.existsSync(file))
    
    if (existingConfig) {
      console.log('✅ Vite config found:', existingConfig)
      
      try {
        const configContent = fs.readFileSync(existingConfig, 'utf-8')
        
        // 检查常见配置问题
        if (configContent.includes('require(') && existingConfig.endsWith('.js')) {
          this.warnings.push('Using require() in .js config file. Consider using .cjs extension or ES modules')
        }
        
        if (!configContent.includes('defineConfig')) {
          this.warnings.push('Consider using defineConfig() for better TypeScript support')
        }
        
      } catch (error) {
        this.issues.push(`Cannot read config file ${existingConfig}: ${error.message}`)
      }
    } else {
      this.warnings.push('No Vite config file found. Using default configuration.')
    }
  }
  
  checkDependencies() {
    try {
      const packageJson = JSON.parse(fs.readFileSync('package.json', 'utf-8'))
      const allDeps = {
        ...packageJson.dependencies,
        ...packageJson.devDependencies
      }
      
      // 检查常见冲突
      const conflicts = [
        ['@vitejs/plugin-vue', '@vitejs/plugin-vue2'],
        ['@vitejs/plugin-react', '@vitejs/plugin-react-swc']
      ]
      
      conflicts.forEach(([dep1, dep2]) => {
        if (allDeps[dep1] && allDeps[dep2]) {
          this.issues.push(`Conflicting dependencies: ${dep1} and ${dep2}`)
        }
      })
      
      // 检查 peer dependencies
      const peerDepWarnings = this.checkPeerDependencies(allDeps)
      this.warnings.push(...peerDepWarnings)
      
    } catch (error) {
      this.issues.push('Cannot analyze dependencies: ' + error.message)
    }
  }
  
  checkPeerDependencies(deps) {
    const warnings = []
    
    // Vue 相关检查
    if (deps['@vitejs/plugin-vue'] && !deps.vue) {
      warnings.push('Vue plugin found but Vue is not installed')
    }
    
    // React 相关检查
    if (deps['@vitejs/plugin-react'] && (!deps.react || !deps['react-dom'])) {
      warnings.push('React plugin found but React/ReactDOM is not installed')
    }
    
    return warnings
  }
  
  checkFileStructure() {
    // 检查入口文件
    const entryFiles = ['index.html', 'src/main.js', 'src/main.ts', 'src/App.vue', 'src/App.jsx']
    const missingFiles = entryFiles.filter(file => !fs.existsSync(file))
    
    if (missingFiles.length === entryFiles.length) {
      this.issues.push('No entry files found. Expected index.html and src/main.js or similar')
    } else {
      console.log('✅ Entry files found')
    }
    
    // 检查 public 目录
    if (!fs.existsSync('public')) {
      this.warnings.push('No public directory found')
    }
    
    // 检查 node_modules
    if (!fs.existsSync('node_modules')) {
      this.issues.push('node_modules not found. Run npm install or yarn install')
    }
  }
  
  checkEnvironment() {
    // 检查环境变量
    const envFiles = ['.env', '.env.local', '.env.development', '.env.production']
    const existingEnvFiles = envFiles.filter(file => fs.existsSync(file))
    
    if (existingEnvFiles.length > 0) {
      console.log('✅ Environment files found:', existingEnvFiles.join(', '))
      
      // 检查环境变量格式
      existingEnvFiles.forEach(file => {
        try {
          const content = fs.readFileSync(file, 'utf-8')
          const lines = content.split('\n').filter(line => line.trim() && !line.startsWith('#'))
          
          lines.forEach((line, index) => {
            if (!line.includes('=')) {
              this.warnings.push(`Invalid env var format in ${file}:${index + 1}: ${line}`)
            } else {
              const [key] = line.split('=')
              if (!key.startsWith('VITE_') && key !== 'NODE_ENV') {
                this.warnings.push(`Env var ${key} in ${file} won't be available in client code (must start with VITE_)`)
              }
            }
          })
        } catch (error) {
          this.warnings.push(`Cannot read ${file}: ${error.message}`)
        }
      })
    }
  }
  
  generateReport() {
    console.log('\n📋 Health Check Report:')
    
    if (this.issues.length === 0 && this.warnings.length === 0) {
      console.log('🎉 All checks passed! Your Vite setup looks good.')
      return
    }
    
    if (this.issues.length > 0) {
      console.log('\n❌ Issues (must fix):')
      this.issues.forEach(issue => console.log(`  • ${issue}`))
    }
    
    if (this.warnings.length > 0) {
      console.log('\n⚠️ Warnings (recommended to fix):')
      this.warnings.forEach(warning => console.log(`  • ${warning}`))
    }
    
    console.log('\n💡 For more help, visit: https://vitejs.dev/guide/troubleshooting.html')
  }
}

const checker = new HealthChecker()
checker.runChecks()
