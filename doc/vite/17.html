插件 API 详解

Vite7.0 | 2025.07.31

Vite 插件 API 基于 Rollup 插件系统，并扩展了开发服务器相关的功能。本章将详细介绍 Vite 插件 API 的各个方面，帮助你深入理解插件的工作机制。

const configResolvedPlugin = () => {

      }

        console.warn('Missing @ alias configuration')
      }

    // 在其他钩子中使用配置

      if (isProduction && code.includes('console.log')) {
        // 生产环境移除 console.log
        return {
          code: code.replace(/console\.log\([^)]*\);?/g, ''),
          map: null
        }
      }
    }
  }
}

// 配置开发服务器的钩子
const serverConfigPlugin = () => ({
  name: 'server-config-plugin',
  configureServer(server) {
    // 添加中间件
    server.middlewares.use('/api', (req, res, next) => {
      if (req.url.startsWith('/api/mock')) {
        // 处理 mock API
        res.setHeader('Content-Type', 'application/json')
        res.end(JSON.stringify({ message: 'Mock API response' }))
      } else {
        next()
      }
    })
    
    // 代理配置
    server.middlewares.use('/proxy', proxy('http://localhost:8080', {
      changeOrigin: true,
      pathRewrite: {
        '^/proxy': ''
      }
    }))
    
    // 静态文件服务
    server.middlewares.use('/static', express.static('public'))
    
    // WebSocket 处理
    server.ws.on('custom-event', (data, client) => {
      console.log('Received custom event:', data)
      
      // 广播给所有客户端
      server.ws.clients.forEach(client => {
        client.send(JSON.stringify({
          type: 'custom-response',
          data: 'Response from server'
        }))
      })
    })
    
    // 服务器启动后的回调
    server.httpServer?.once('listening', () => {
      const address = server.httpServer.address()
      console.log(`Server running at http://localhost:${address.port}`)
    })
  }
})

handleHotUpdate 钩子

// 处理热更新的钩子
const hotUpdatePlugin = () => ({
  name: 'hot-update-plugin',
  handleHotUpdate(ctx) {
    const { file, timestamp, server } = ctx
    
    // 特殊文件处理
    if (file.endsWith('.env')) {
      // 环境变量文件变化，重启服务器
      console.log('Environment file changed, restarting server...')
      server.restart()
      return []
    }
    
    if (file.endsWith('.config.js')) {
      // 配置文件变化，全量刷新
      console.log('Config file changed, full reload...')
      server.ws.send({
        type: 'full-reload'
      })
      return []
    }
    
    if (file.includes('/api/')) {
      // API 文件变化，通知客户端
      server.ws.send({
        type: 'custom',
        event: 'api-changed',
        data: { file, timestamp }
      })
      
      // 不进行默认的热更新
      return []
    }
    
    // 自定义模块更新逻辑
    if (file.endsWith('.vue')) {
      const modules = ctx.modules
      
      // 过滤需要更新的模块
      const filteredModules = modules.filter(mod => {
        // 只更新直接依赖此文件的模块
        return mod.file === file
      })
      
      if (filteredModules.length > 0) {
        console.log(`Hot updating ${filteredModules.length} modules for ${file}`)
      }
      
      return filteredModules
    }
    
    // 返回 undefined 使用默认行为
    return undefined
  }
})

resolveId 钩子

// 模块解析钩子
const resolvePlugin = () => ({
  name: 'resolve-plugin',
  resolveId(id, importer, options) {
    // 虚拟模块处理
    if (id.startsWith('virtual:')) {
      return id
    }
    
    // 别名处理
    if (id.startsWith('@/')) {
      return path.resolve(__dirname, 'src', id.slice(2))
    }
    
    // 条件解析
    if (id === 'my-lib') {
      if (options.isEntry) {
        // 入口文件使用完整版本
        return path.resolve(__dirname, 'node_modules/my-lib/dist/full.js')
      } else {
        // 非入口文件使用精简版本
        return path.resolve(__dirname, 'node_modules/my-lib/dist/lite.js')
      }
    }
    
    // 环境相关解析
    if (id.endsWith('.dev.js') && process.env.NODE_ENV === 'production') {
      // 生产环境使用生产版本
      return id.replace('.dev.js', '.prod.js')
    }
    
    // 外部模块处理
    if (id.startsWith('http://') || id.startsWith('https://')) {
      // 标记为外部模块
      return { id, external: true }
    }
    
    // 返回 null 表示使用默认解析
    return null
  }
})

load 钩子

// 模块加载钩子
const loadPlugin = () => {
  const cache = new Map()
  
  return {
    name: 'load-plugin',
    load(id) {
      // 虚拟模块加载
      if (id.startsWith('virtual:')) {
        const moduleName = id.slice(8) // 移除 'virtual:' 前缀
        
        switch (moduleName) {
          case 'env':
            return `export default ${JSON.stringify(process.env)}`
          
          case 'config':
            return `export default ${JSON.stringify({
              apiUrl: process.env.VITE_API_URL,
              version: process.env.npm_package_version
            })}`
          
          case 'build-info':
            return `
              export const buildTime = "${new Date().toISOString()}"
              export const gitCommit = "${process.env.GIT_COMMIT || 'unknown'}"
              export const nodeVersion = "${process.version}"
            `
        }
      }
      
      // 缓存处理
      if (cache.has(id)) {
        return cache.get(id)
      }
      
      // 特殊文件处理
      if (id.endsWith('.txt')) {
        const content = fs.readFileSync(id, 'utf-8')
        const result = `export default ${JSON.stringify(content)}`
        cache.set(id, result)
        return result
      }
      
      if (id.endsWith('.csv')) {
        const content = fs.readFileSync(id, 'utf-8')
        const rows = content.split('\n').map(row => row.split(','))
        const result = `export default ${JSON.stringify(rows)}`
        cache.set(id, result)
        return result
      }
      
      // 远程模块加载
      if (id.startsWith('remote:')) {
        const url = id.slice(7)
        return fetch(url).then(res => res.text())
      }
      
      // 返回 null 使用默认加载
      return null
    }
  }
}

transform 钩子

// 代码转换钩子
const transformPlugin = () => {
  const sourceMapCache = new Map()
  
  return {
    name: 'transform-plugin',
    transform(code, id) {
      // 跳过 node_modules
      if (id.includes('node_modules')) {
        return null
      }
      
      let transformedCode = code
      let hasChanges = false
      const sourcemap = this.getCombinedSourcemap()
      
      // TypeScript 装饰器转换
      if (id.endsWith('.ts') && code.includes('@')) {
        transformedCode = transformDecorators(transformedCode)
        hasChanges = true
      }
      
      // JSX 转换
      if (id.endsWith('.jsx') && !id.includes('.vue')) {
        transformedCode = transformJSX(transformedCode)
        hasChanges = true
      }
      
      // 环境变量替换
      const envVars = Object.entries(process.env)
        .filter(([key]) => key.startsWith('VITE_'))
      
      envVars.forEach(([key, value]) => {
        const pattern = new RegExp(`import\\.meta\\.env\\.${key}`, 'g')
        if (pattern.test(transformedCode)) {
          transformedCode = transformedCode.replace(pattern, JSON.stringify(value))
          hasChanges = true
        }
      })
      
      // 开发环境注入
      if (process.env.NODE_ENV === 'development') {
        if (transformedCode.includes('console.error')) {
          // 开发环境增强错误信息
          transformedCode = transformedCode.replace(
            /console\.error\(([^)]+)\)/g,
            `console.error('[${id}]', $1)`
          )
          hasChanges = true
        }
      }
      
      // 生产环境优化
      if (process.env.NODE_ENV === 'production') {
        // 移除调试代码
        transformedCode = transformedCode.replace(/debugger;?/g, '')
        transformedCode = transformedCode.replace(/console\.(log|debug|info)\([^)]*\);?/g, '')
        hasChanges = true
      }
      
      // 返回转换结果
      if (hasChanges) {
        return {
          code: transformedCode,
          map: sourcemap
        }
      }
      
      return null
    }
  }
}

// 辅助函数
function transformDecorators(code) {
  // 简单的装饰器转换示例
  return code.replace(/@(\w+)/g, '/* @$1 */')
}

function transformJSX(code) {
  // 简单的 JSX 转换示例
  return code.replace(/<(\w+)([^>]*)>/g, 'React.createElement("$1", {$2})')
}

generateBundle 钩子

// 构建包生成钩子
const generateBundlePlugin = () => ({
  name: 'generate-bundle-plugin',
  generateBundle(options, bundle) {
    // 分析构建包
    const chunks = Object.values(bundle).filter(chunk => chunk.type === 'chunk')
    const assets = Object.values(bundle).filter(asset => asset.type === 'asset')
    
    console.log(`Generated ${chunks.length} chunks and ${assets.length} assets`)
    
    // 添加构建信息文件
    const buildInfo = {
      buildTime: new Date().toISOString(),
      chunks: chunks.map(chunk => ({
        fileName: chunk.fileName,
        size: chunk.code.length,
        modules: Object.keys(chunk.modules || {}).length
      })),
      assets: assets.map(asset => ({
        fileName: asset.fileName,
        size: asset.source.length
      })),
      totalSize: chunks.reduce((sum, chunk) => sum + chunk.code.length, 0) +
                 assets.reduce((sum, asset) => sum + asset.source.length, 0)
    }
    
    // 添加到构建包中
    this.emitFile({
      type: 'asset',
      fileName: 'build-info.json',
      source: JSON.stringify(buildInfo, null, 2)
    })
    
    // 修改现有文件
    Object.keys(bundle).forEach(fileName => {
      const chunk = bundle[fileName]
      
      if (chunk.type === 'chunk' && chunk.isEntry) {
        // 在入口文件顶部添加版本信息
        chunk.code = `/* Build: ${buildInfo.buildTime} */\n${chunk.code}`
      }
      
      if (chunk.type === 'asset' && fileName.endsWith('.css')) {
        // 在 CSS 文件顶部添加注释
        chunk.source = `/* Generated at ${buildInfo.buildTime} */\n${chunk.source}`
      }
    })
    
    // 生成依赖图
    const dependencyGraph = generateDependencyGraph(chunks)
    this.emitFile({
      type: 'asset',
      fileName: 'dependency-graph.json',
      source: JSON.stringify(dependencyGraph, null, 2)
    })
  }
})

function generateDependencyGraph(chunks) {
  const graph = {}
  
  chunks.forEach(chunk => {
    graph[chunk.fileName] = {
      imports: chunk.imports || [],
      dynamicImports: chunk.dynamicImports || [],
      modules: Object.keys(chunk.modules || {})
    }
  })
  
  return graph
}

插件上下文方法

const contextPlugin = () => ({
  name: 'context-plugin',
  
  buildStart() {
    // 添加监听文件
    this.addWatchFile('config/app.json')
    this.addWatchFile('src/types/global.d.ts')
  },
  
  load(id) {
    if (id.endsWith('.special')) {
      // 发出文件
      const referenceId = this.emitFile({
        type: 'asset',
        fileName: 'generated.txt',
        source: 'This is a generated file'
      })
      
      // 获取文件名
      const fileName = this.getFileName(referenceId)
      
      return `export const generatedFile = "${fileName}"`
    }
  },
  
  transform(code, id) {
    if (code.includes('import.meta.SPECIAL')) {
      // 解析模块
      const resolved = this.resolve('./utils', id)
      
      if (resolved) {
        return {
          code: code.replace(
            'import.meta.SPECIAL',
            `"${resolved.id}"`
          ),
          map: null
        }
      }
    }
  },
  
  generateBundle() {
    // 获取模块信息
    const moduleIds = this.getModuleIds()
    
    moduleIds.forEach(id => {
      const moduleInfo = this.getModuleInfo(id)
      
      if (moduleInfo && moduleInfo.isEntry) {
        console.log(`Entry module: ${id}`)
        console.log(`Dependencies: ${moduleInfo.importedIds.length}`)
      }
    })
  }
})

错误处理

const errorHandlingPlugin = () => ({
  name: 'error-handling-plugin',
  
  load(id) {
    try {
      if (id.endsWith('.risky')) {
        // 可能出错的操作
        return processRiskyFile(id)
      }
    } catch (error) {
      // 发出警告
      this.warn(`Failed to process ${id}: ${error.message}`)
      
      // 或者发出错误
      this.error(`Critical error in ${id}: ${error.message}`)
    }
  },
  
  transform(code, id) {
    // 验证代码
    if (!isValidCode(code)) {
      // 发出错误并停止构建
      this.error(`Invalid code in ${id}`, {
        id,
        loc: { line: 1, column: 1 },
        frame: code.split('\n').slice(0, 5).join('\n')
      })
    }
    
    // 发出警告但继续构建
    if (hasDeprecatedAPI(code)) {
      this.warn(`Deprecated API usage in ${id}`, {
        id,
        message: 'This API will be removed in the next version'
      })
    }
  }
})

function processRiskyFile(id) {
  // 模拟可能出错的文件处理
  if (Math.random() < 0.1) {
    throw new Error('Random processing error')
  }
  return `export default "processed-${id}"`
}

function isValidCode(code) {
  // 简单的代码验证
  try {
    new Function(code)
    return true
  } catch {
    return false
  }
}

function hasDeprecatedAPI(code) {
  return code.includes('deprecatedFunction')
}

// 共享状态插件
const sharedStatePlugin = () => {
  const state = new Map()
  
  return {
    name: 'shared-state-plugin',
    
    configResolved() {
      // 在全局对象上暴露状态
      global.__PLUGIN_STATE__ = state
    },
    
    buildStart() {
      state.set('buildStartTime', Date.now())
    },
    
    generateBundle() {
      const startTime = state.get('buildStartTime')
      const duration = Date.now() - startTime
      console.log(`Build duration: ${duration}ms`)
    }
  }
}

// 消费状态插件
const consumerPlugin = () => ({
  name: 'consumer-plugin',
  
  transform(code, id) {
    const state = global.__PLUGIN_STATE__
    
    if (state && state.has('buildStartTime')) {
      const startTime = state.get('buildStartTime')
      const elapsed = Date.now() - startTime
      
      // 在代码中注入构建时间
      if (code.includes('__BUILD_ELAPSED__')) {
        return {
          code: code.replace('__BUILD_ELAPSED__', elapsed.toString()),
          map: null
        }
      }
    }
  }
})

插件依赖管理

// 依赖管理插件
const dependencyPlugin = () => ({
  name: 'dependency-plugin',
  
  // 确保在其他插件之前执行
  enforce: 'pre',
  
  configResolved(config) {
    // 检查依赖插件
    const requiredPlugins = ['vue', 'typescript']
    const availablePlugins = config.plugins.map(p => p.name)
    
    const missing = requiredPlugins.filter(name => 
      !availablePlugins.includes(name)
    )
    
    if (missing.length > 0) {
      throw new Error(`Missing required plugins: ${missing.join(', ')}`)
    }
  }
})

// 插件工厂，确保依赖顺序
const createPluginSuite = (options = {}) => {
  return [
    // 基础插件
    sharedStatePlugin(),
    dependencyPlugin(),
    
    // 功能插件
    consumerPlugin(),
    
    // 确保在最后执行的插件
    {
      name: 'cleanup-plugin',
      enforce: 'post',
      writeBundle() {
        // 清理全局状态
        delete global.__PLUGIN_STATE__
      }
    }
  ]
}

总结

Vite 插件 API 提供了：

  • 完整的构建生命周期钩子
  • 强大的代码转换能力
  • 灵活的模块解析机制
  • 丰富的上下文方法和错误处理

深入理解这些 API，能够帮助你开发出功能强大、性能优异的 Vite 插件，充分发挥 Vite 的扩展能力。
