迁移脚本

// scripts/migrate-to-vite.js
import fs from 'fs'
import path from 'path'

class WebpackToViteMigrator {
  constructor(projectPath) {
    this.projectPath = projectPath
    this.webpackConfig = null
    this.packageJson = null
  }
  
  async migrate() {
    console.log('🚀 Starting Webpack to Vite migration...')
    
    // 1. 分析现有配置
    await this.analyzeProject()
    
    // 2. 备份现有文件
    await this.backupFiles()
    
    // 3. 更新 package.json
    await this.updatePackageJson()
    
    // 4. 创建 Vite 配置
    await this.createViteConfig()
    
    // 5. 更新入口文件
    await this.updateEntryFile()
    
    // 6. 处理静态资源
    await this.handleStaticAssets()
    
    // 7. 更新环境变量
    await this.updateEnvironmentVariables()
    
    // 8. 生成迁移报告
    await this.generateMigrationReport()
    
    console.log('✅ Migration completed!')
  }
  
  async analyzeProject() {
    console.log('📊 Analyzing project structure...')
    
    // 读取 package.json
    const packagePath = path.join(this.projectPath, 'package.json')
    this.packageJson = JSON.parse(fs.readFileSync(packagePath, 'utf-8'))
    
    // 读取 webpack 配置
    const webpackConfigPath = path.join(this.projectPath, 'webpack.config.js')
    if (fs.existsSync(webpackConfigPath)) {
      // 这里需要动态导入和解析 webpack 配置
      console.log('Found webpack.config.js')
    }
    
    // 分析项目依赖
    this.analyzeDependencies()
  }
  
  analyzeDependencies() {
    const deps = this.packageJson.dependencies || {}
    const devDeps = this.packageJson.devDependencies || {}
    
    // 检查框架
    if (deps.vue || devDeps.vue) {
      this.framework = 'vue'
    } else if (deps.react || devDeps.react) {
      this.framework = 'react'
    }
    
    // 检查需要替换的依赖
    this.dependenciesToReplace = {
      'webpack': 'vite',
      'webpack-cli': null,
      'webpack-dev-server': null,
      'vue-loader': '@vitejs/plugin-vue',
      'babel-loader': null, // Vite 内置支持
      'css-loader': null,
      'style-loader': null,
      'file-loader': null,
      'url-loader': null,
      'html-webpack-plugin': null
    }
  }
  
  async backupFiles() {
    console.log('💾 Creating backup...')
    
    const backupDir = path.join(this.projectPath, '.migration-backup')
    if (!fs.existsSync(backupDir)) {
      fs.mkdirSync(backupDir)
    }
    
    // 备份关键文件
    const filesToBackup = [
      'package.json',
      'webpack.config.js',
      'babel.config.js',
      '.babelrc'
    ]
    
    filesToBackup.forEach(file => {
      const filePath = path.join(this.projectPath, file)
      if (fs.existsSync(filePath)) {
        const backupPath = path.join(backupDir, file)
        fs.copyFileSync(filePath, backupPath)
        console.log(`  Backed up ${file}`)
      }
    })
  }
  
  async updatePackageJson() {
    console.log('📦 Updating package.json...')
    
    // 更新脚本
    this.packageJson.scripts = {
      ...this.packageJson.scripts,
      dev: 'vite',
      build: 'vite build',
      preview: 'vite preview'
    }
    
    // 移除 webpack 相关依赖
    Object.keys(this.dependenciesToReplace).forEach(dep => {
      delete this.packageJson.dependencies?.[dep]
      delete this.packageJson.devDependencies?.[dep]
    })
    
    // 添加 Vite 依赖
    this.packageJson.devDependencies = {
      ...this.packageJson.devDependencies,
      vite: '^5.0.0'
    }
    
    // 添加框架插件
    if (this.framework === 'vue') {
      this.packageJson.devDependencies['@vitejs/plugin-vue'] = '^5.0.0'
    } else if (this.framework === 'react') {
      this.packageJson.devDependencies['@vitejs/plugin-react'] = '^4.0.0'
    }
    
    // 写入更新后的 package.json
    const packagePath = path.join(this.projectPath, 'package.json')
    fs.writeFileSync(packagePath, JSON.stringify(this.packageJson, null, 2))
  }
  
  async createViteConfig() {
    console.log('⚙️ Creating Vite configuration...')
    
    const viteConfig = this.generateViteConfig()
    const configPath = path.join(this.projectPath, 'vite.config.js')
    fs.writeFileSync(configPath, viteConfig)
  }
  
  generateViteConfig() {
    let config = `import { defineConfig } from 'vite'\n`
    
    // 添加插件导入
    if (this.framework === 'vue') {
      config += `import vue from '@vitejs/plugin-vue'\n`
    } else if (this.framework === 'react') {
      config += `import react from '@vitejs/plugin-react'\n`
    }
    
    config += `import { resolve } from 'path'\n\n`
    
    config += `export default defineConfig({\n`
    config += `  plugins: [\n`
    
    if (this.framework === 'vue') {
      config += `    vue()\n`
    } else if (this.framework === 'react') {
      config += `    react()\n`
    }
    
    config += `  ],\n\n`
    
    // 添加解析配置
    config += `  resolve: {\n`
    config += `    alias: {\n`
    config += `      '@': resolve(__dirname, 'src')\n`
    config += `    }\n`
    config += `  },\n\n`
    
    // 添加服务器配置
    config += `  server: {\n`
    config += `    port: 8080\n`
    config += `  }\n`
    
    config += `})\n`
    
    return config
  }
  
  async updateEntryFile() {
    console.log('📄 Updating entry file...')
    
    // 检查是否需要创建 index.html
    const indexPath = path.join(this.projectPath, 'index.html')
    if (!fs.existsSync(indexPath)) {
      const htmlTemplate = this.generateIndexHtml()
      fs.writeFileSync(indexPath, htmlTemplate)
    }
  }
  
  generateIndexHtml() {
    return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Vite App</title>
</head>
<body>
  <div id="app"></div>
  <script type="module" src="/src/main.js"></script>
</body>
</html>`
  }
  
  async handleStaticAssets() {
    console.log('🖼️ Handling static assets...')
    
    // 检查 public 目录
    const publicDir = path.join(this.projectPath, 'public')
    if (!fs.existsSync(publicDir)) {
      fs.mkdirSync(publicDir)
    }
    
    // 移动静态资源到 public 目录
    // 这里需要根据具体项目结构来处理
  }
  
  async updateEnvironmentVariables() {
    console.log('🌍 Updating environment variables...')
    
    // 检查现有的环境变量文件
    const envFiles = ['.env', '.env.local', '.env.development', '.env.production']
    
    envFiles.forEach(envFile => {
      const envPath = path.join(this.projectPath, envFile)
      if (fs.existsSync(envPath)) {
        let content = fs.readFileSync(envPath, 'utf-8')
        
        // 将环境变量转换为 Vite 格式
        content = content.replace(/^(?!VITE_)([A-Z_]+)=/gm, 'VITE_$1=')
        
        fs.writeFileSync(envPath, content)
        console.log(`  Updated ${envFile}`)
      }
    })
  }
  
  async generateMigrationReport() {
    console.log('📋 Generating migration report...')
    
    const report = {
      timestamp: new Date().toISOString(),
      framework: this.framework,
      migratedFiles: [
        'package.json',
        'vite.config.js',
        'index.html'
      ],
      removedDependencies: Object.keys(this.dependenciesToReplace),
      addedDependencies: ['vite', `@vitejs/plugin-${this.framework}`],
      nextSteps: [
        'Run npm install to install new dependencies',
        'Test the development server with npm run dev',
        'Update any remaining webpack-specific code',
        'Test the build process with npm run build'
      ],
      potentialIssues: [
        'Dynamic imports may need adjustment',
        'Asset handling might require updates',
        'Environment variables need VITE_ prefix'
      ]
    }
    
    const reportPath = path.join(this.projectPath, 'MIGRATION_REPORT.md')
    const reportContent = this.formatReport(report)
    fs.writeFileSync(reportPath, reportContent)
    
    console.log('📋 Migration report saved to MIGRATION_REPORT.md')
  }
  
  formatReport(report) {
    return `# Webpack to Vite Migration Report

Generated: ${report.timestamp}

## Summary
- Framework: ${report.framework}
- Migration completed successfully ✅

## Changes Made

### Files Modified
${report.migratedFiles.map(file => `- ${file}`).join('\n')}

### Dependencies Removed
${report.removedDependencies.map(dep => `- ${dep}`).join('\n')}

### Dependencies Added
${report.addedDependencies.map(dep => `- ${dep}`).join('\n')}

## Next Steps
${report.nextSteps.map(step => `1. ${step}`).join('\n')}

## Potential Issues
${report.potentialIssues.map(issue => `⚠️ ${issue}`).join('\n')}

## Rollback Instructions
If you need to rollback:
1. Restore files from .migration-backup/
2. Run npm install to restore original dependencies
3. Remove vite.config.js and index.html if they were created

## Support
If you encounter issues, check:
- Vite documentation: https://vitejs.dev/
- Migration guide: https://vitejs.dev/guide/migration.html
`
  }
}

功能对比测试

// scripts/compare-builds.js
import { execSync } from 'child_process'
import fs from 'fs'
import path from 'path'

class BuildComparator {
  constructor() {
    this.results = {
      vite: {},
      webpack: {}
    }
  }
  
  async compare() {
    console.log('🔍 Comparing Vite and Webpack builds...')
    
    // 测试构建时间
    await this.compareBuildTime()
    
    // 测试构建大小
    await this.compareBuildSize()
    
    // 测试开发服务器启动时间
    await this.compareDevServerStartup()
    
    // 生成对比报告
    this.generateReport()
  }
  
  async compareBuildTime() {
    console.log('⏱️ Comparing build times...')
    
    // Vite 构建
    const viteStart = Date.now()
    execSync('npm run build:vite', { stdio: 'pipe' })
    this.results.vite.buildTime = Date.now() - viteStart
    
    // Webpack 构建
    const webpackStart = Date.now()
    execSync('npm run build:webpack', { stdio: 'pipe' })
    this.results.webpack.buildTime = Date.now() - webpackStart
  }
  
  async compareBuildSize() {
    console.log('📦 Comparing build sizes...')
    
    // 分析 Vite 构建大小
    this.results.vite.buildSize = this.calculateDirSize('dist')
    
    // 分析 Webpack 构建大小
    this.results.webpack.buildSize = this.calculateDirSize('build')
  }
  
  calculateDirSize(dirPath) {
    let totalSize = 0
    
    function traverse(currentPath) {
      const files = fs.readdirSync(currentPath)
      
      files.forEach(file => {
        const filePath = path.join(currentPath, file)
        const stats = fs.statSync(filePath)
        
        if (stats.isDirectory()) {
          traverse(filePath)
        } else {
          totalSize += stats.size
        }
      })
    }
    
    if (fs.existsSync(dirPath)) {
      traverse(dirPath)
    }
    
    return totalSize
  }
  
  generateReport() {
    const report = `
# Build Comparison Report

## Build Time
- Vite: ${this.results.vite.buildTime}ms
- Webpack: ${this.results.webpack.buildTime}ms
- Improvement: ${((this.results.webpack.buildTime - this.results.vite.buildTime) / this.results.webpack.buildTime * 100).toFixed(1)}%

## Build Size
- Vite: ${(this.results.vite.buildSize / 1024 / 1024).toFixed(2)}MB
- Webpack: ${(this.results.webpack.buildSize / 1024 / 1024).toFixed(2)}MB
- Difference: ${((this.results.vite.buildSize - this.results.webpack.buildSize) / this.results.webpack.buildSize * 100).toFixed(1)}%

## Recommendation
${this.results.vite.buildTime < this.results.webpack.buildTime ? '✅ Vite shows better build performance' : '⚠️ Webpack shows better build performance'}
    `
    
    fs.writeFileSync('BUILD_COMPARISON.md', report)
    console.log('📋 Comparison report saved to BUILD_COMPARISON.md')
  }
}

const validationChecklist = [
  {
    name: '开发服务器启动',
    test: () => testDevServer(),
    critical: true
  },
  {
    name: '热更新功能',
    test: () => testHMR(),

  },
  {
    name: '生产构建',
    test: () => testProductionBuild(),

  },
  {
    name: '静态资源处理',
    test: () => testAssetHandling(),
    critical: false
  },
  {
    name: '环境变量',
    test: () => testEnvironmentVariables(),
    critical: false
  },
  {
    name: '代理配置',
    test: () => testProxyConfig(),
    critical: false
  }
]

async function runValidation() {
  console.log('🧪 Running migration validation...')
  
  const results = []
  
  for (const check of validationChecklist) {
    try {
      const result = await check.test()
      results.push({
        name: check.name,
        status: 'passed',
        critical: check.critical,
        result
      })
      console.log(`✅ ${check.name}: PASSED`)
    } catch (error) {
      results.push({
        name: check.name,
        status: 'failed',
        critical: check.critical,
        error: error.message
      })
      console.log(`❌ ${check.name}: FAILED - ${error.message}`)
    }
  }
  
  // 生成验证报告
  generateValidationReport(results)
}

function generateValidationReport(results) {
  const passed = results.filter(r => r.status === 'passed').length
  const failed = results.filter(r => r.status === 'failed').length
  const criticalFailed = results.filter(r => r.status === 'failed' && r.critical).length
  
  const report = `# Migration Validation Report

## Summary
- Total Tests: ${results.length}
- Passed: ${passed}
- Failed: ${failed}
- Critical Failures: ${criticalFailed}

## Test Results
${results.map(result => `
### ${result.name}
- Status: ${result.status.toUpperCase()}
- Critical: ${result.critical ? 'Yes' : 'No'}
${result.error ? `- Error: ${result.error}` : ''}
`).join('')}

## Recommendation
${criticalFailed === 0 ? '✅ Migration successful - ready for production' : '⚠️ Critical issues found - review before production'}
  `
  
  fs.writeFileSync('VALIDATION_REPORT.md', report)
}
