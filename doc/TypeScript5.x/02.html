TypeScript 基本类型

TypeScript | 2025.07.30

TypeScript 提供了丰富的类型系统，包括基本类型、复合类型和高级类型。理解这些类型是掌握 TypeScript 的基础。

基本类型

数字类型 (number)

let decimal: number = 6;
let hex: number = 0xf00d;
let binary: number = 0b1010;
let octal: number = 0o744;
let big: bigint = 100n;

字符串类型 (string)

let color: string = "blue";
let fullName: string = `Bob Bobbington`;
let age: number = 37;
let sentence: string = `Hello, my name is ${fullName}.
I'll be ${age + 1} years old next month.`;

布尔类型 (boolean)

let isDone: boolean = false;
let isActive: boolean = true;

数组类型 (Array)

// 方式一：类型[]
let list: number[] = [1, 2, 3];
let names: string[] = ["Alice", "Bob", "Charlie"];

// 方式二：Array<类型>
let numbers: Array<number> = [1, 2, 3];
let fruits: Array<string> = ["apple", "banana", "orange"];

元组类型 (Tuple)

// 声明一个元组类型
let x: [string, number];
x = ["hello", 10]; // 正确
// x = [10, "hello"]; // 错误

// 访问元组元素
console.log(x[0].substring(1)); // 正确
// console.log(x[1].substring(1)); // 错误，number 没有 substring 方法

// 越界访问
x[3] = "world"; // 错误，元组只有 2 个元素

枚举类型 (enum)

// 数字枚举
enum Color {
    Red,
    Green,
    Blue,
}
let c: Color = Color.Green; // 1

// 手动设置枚举值
enum Color2 {
    Red = 1,
    Green,
    Blue,
}
let colorName: string = Color2[2]; // "Green"

// 字符串枚举
enum Direction {
    Up = "UP",
    Down = "DOWN",
    Left = "LEFT",
    Right = "RIGHT",
}

// 异构枚举（不推荐）
enum BooleanLikeHeterogeneousEnum {
    No = 0,
    Yes = "YES",
}

Any 类型

let notSure: any = 4;
notSure = "maybe a string instead";
notSure = false; // 也可以是布尔值

// any 类型的数组
let list: any[] = [1, true, "free"];
list[1] = 100;

Void 类型

// 函数没有返回值
function warnUser(): void {
    console.log("This is my warning message");
}

// void 类型的变量只能赋值 undefined 或 null
let unusable: void = undefined;

Null 和 Undefined

let u: undefined = undefined;
let n: null = null;

// 默认情况下，null 和 undefined 是所有类型的子类型
let num: number = undefined; // 在严格模式下会报错

Never 类型

// 返回 never 的函数必须存在无法达到的终点
function error(message: string): never {
    throw new Error(message);
}

// 推断的返回值类型为 never
function fail() {
    return error("Something failed");
}

// 返回 never 的函数必须存在无法达到的终点
function infiniteLoop(): never {
    while (true) {
    }
}

Object 类型

declare function create(o: object | null): void;

create({ prop: 0 }); // 正确
create(null); // 正确

// create(42); // 错误
// create("string"); // 错误
// create(false); // 错误
// create(undefined); // 错误

类型断言

尖括号语法

let someValue: any = "this is a string";
let strLength: number = (<string>someValue).length;

as 语法（推荐）

let someValue: any = "this is a string";
let strLength: number = (someValue as string).length;

实例演示：用户信息管理

// 定义用户状态枚举
enum UserStatus {
    Active = "ACTIVE",
    Inactive = "INACTIVE",
    Pending = "PENDING",
    Suspended = "SUSPENDED"
}

// 定义用户角色枚举
enum UserRole {
    Admin,
    User,
    Guest
}

// 用户信息接口
interface UserInfo {
    id: number;
    name: string;
    email: string;
    age: number;
    isVerified: boolean;
    status: UserStatus;
    role: UserRole;
    tags: string[];
    metadata: [string, any]; // 元组类型
}

// 创建用户实例
const user: UserInfo = {
    id: 1,
    name: "张三",
    email: "zhangsan@example.com",
    age: 28,
    isVerified: true,
    status: UserStatus.Active,
    role: UserRole.User,
    tags: ["developer", "typescript"],
    metadata: ["lastLogin", new Date()]
};

// 用户管理函数
function getUserStatusText(status: UserStatus): string {
    switch (status) {
        case UserStatus.Active:
            return "用户活跃";
        case UserStatus.Inactive:
            return "用户不活跃";
        case UserStatus.Pending:
            return "待审核";
        case UserStatus.Suspended:
            return "已暂停";
        default:
            return "未知状态";
    }
}

// 类型断言示例
function processUserData(data: any): UserInfo {
    // 假设我们确定 data 是 UserInfo 类型
    const userInfo = data as UserInfo;
    return userInfo;
}

console.log(`用户 ${user.name} 的状态是：${getUserStatusText(user.status)}`);

这个例子展示了如何在实际项目中使用 TypeScript 的基本类型，包括枚举、接口、元组等，构建一个类型安全的用户信息管理系统。
