接口 (Interface)

TypeScript | 2025.07.31

接口是 TypeScript 的核心特性之一，用于定义对象的结构和契约。接口描述了对象应该具有的属性和方法。

基本接口定义

简单接口

interface Person {
    name: string;
    age: number;
}

function greet(person: Person) {
    return `Hello, ${person.name}!`;
}

let user = { name: "Alice", age: 25 };
console.log(greet(user));

可选属性

interface Config {
    color?: string;
    width?: number;
    height?: number;
}

function createSquare(config: Config): { color: string; area: number } {
    let newSquare = { color: "white", area: 100 };
    
    if (config.color) {
        newSquare.color = config.color;
    }
    if (config.width) {
        newSquare.area = config.width * config.width;
    }
    
    return newSquare;
}

let mySquare = createSquare({ color: "black" });

函数类型接口

函数签名

interface SearchFunc {
    (source: string, subString: string): boolean;
}

let mySearch: SearchFunc;
mySearch = function(source: string, subString: string): boolean {
    let result = source.search(subString);
    return result > -1;
};

// 参数名不需要匹配
mySearch = function(src: string, sub: string): boolean {
    let result = src.search(sub);
    return result > -1;
};

构造函数类型

interface ClockConstructor {
    new (hour: number, minute: number): ClockInterface;
}

interface ClockInterface {
    tick(): void;
}

class DigitalClock implements ClockInterface {
    constructor(h: number, m: number) { }
    tick() {
        console.log("beep beep");
    }
}

function createClock(ctor: ClockConstructor, hour: number, minute: number): ClockInterface {
    return new ctor(hour, minute);
}

let digital = createClock(DigitalClock, 12, 17);

继承接口

interface Shape {
    color: string;
}

interface PenStroke {
    penWidth: number;

混合类型

一个对象可以同时做为函数和对象使用：

interface Counter {
    (start: number): string;
    interval: number;
    reset(): void;
}

function getCounter(): Counter {
    let counter = (function (start: number) { }) as Counter;
    counter.interval = 123;
    counter.reset = function () { };
    return counter;
}

let c = getCounter();
c(10);
c.reset();
c.interval = 5.0;

泛型接口

interface GenericIdentityFn<T> {
    (arg: T): T;
}

function identity<T>(arg: T): T {
    return arg;
}

let myIdentity: GenericIdentityFn<number> = identity;

事件处理接口

interface EventListener<T = any> {
    (event: T): void;
}

interface CustomEvent<T = any> {
    type: string;
    data: T;
    timestamp: number;
}

interface EventEmitter {
    on<T>(event: string, listener: EventListener<CustomEvent<T>>): void;
    emit<T>(event: string, data: T): void;
    off(event: string, listener: EventListener): void;
}

class SimpleEventEmitter implements EventEmitter {
    private listeners: { [key: string]: EventListener[] } = {};

    on<T>(event: string, listener: EventListener<CustomEvent<T>>): void {
        if (!this.listeners[event]) {
            this.listeners[event] = [];
        }
        this.listeners[event].push(listener);
    }

    emit<T>(event: string, data: T): void {
        if (this.listeners[event]) {
            const customEvent: CustomEvent<T> = {
                type: event,
                data,
                timestamp: Date.now()
            };
            this.listeners[event].forEach(listener => listener(customEvent));
        }
    }

    off(event: string, listener: EventListener): void {
        if (this.listeners[event]) {
            this.listeners[event] = this.listeners[event].filter(l => l !== listener);
        }
    }
}
