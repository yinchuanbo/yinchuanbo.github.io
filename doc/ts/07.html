泛型 (Generics)

TypeScript | 2025.07.31

什么是泛型？

泛型是 TypeScript 中一个强大的特性，它允许我们在定义函数、接口或类的时候，不预先指定具体的类型，而在使用的时候再指定类型。泛型使得代码更加灵活和可重用。

基本泛型语法

泛型函数

// 基本泛型函数
function identity<T>(arg: T): T {
    return arg;
}

// 使用方式1：显式指定类型
let output1 = identity<string>("myString");

// 使用方式2：类型推断
let output2 = identity("myString");

// 数组泛型函数
function getArrayLength<T>(arr: T[]): number {
    return arr.length;
}

console.log(getArrayLength<number>([1, 2, 3])); // 3
console.log(getArrayLength(["a", "b", "c"])); // 3

泛型接口

// 泛型接口
interface GenericIdentityFn<T> {
    (arg: T): T;
}

function identity<T>(arg: T): T {
    return arg;
}

let myIdentity: GenericIdentityFn<number> = identity;

// 更复杂的泛型接口
interface Pair<T, U> {
    first: T;
    second: U;
}

let pair: Pair<string, number> = {
    first: "hello",
    second: 42
};

泛型类

// 泛型类
class GenericNumber<T> {
    zeroValue: T;
    add: (x: T, y: T) => T;
    
    constructor(zeroValue: T, addFn: (x: T, y: T) => T) {
        this.zeroValue = zeroValue;
        this.add = addFn;
    }
}

// 数字版本
let myGenericNumber = new GenericNumber<number>(0, (x, y) => x + y);
console.log(myGenericNumber.add(5, 10)); // 15

// 字符串版本
let myGenericString = new GenericNumber<string>("", (x, y) => x + y);
console.log(myGenericString.add("Hello ", "World")); // "Hello World"

泛型约束

基本约束

// 约束泛型必须有 length 属性
interface Lengthwise {
    length: number;
}

function loggingIdentity<T extends Lengthwise>(arg: T): T {
    console.log(arg.length); // 现在我们知道它有 length 属性
    return arg;
}

// 正确使用
loggingIdentity("hello"); // 字符串有 length 属性
loggingIdentity([1, 2, 3]); // 数组有 length 属性
loggingIdentity({ length: 10, value: 3 }); // 对象有 length 属性

// 错误使用
// loggingIdentity(3); // Error: 数字没有 length 属性

在泛型约束中使用类型参数

// 使用 keyof 约束
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
    return obj[key];
}

let person = { name: "Alice", age: 30, city: "New York" };

let name = getProperty(person, "name"); // string
let age = getProperty(person, "age");   // number
// let invalid = getProperty(person, "salary"); // Error: 'salary' 不存在

条件约束

// 条件类型约束
type NonNullable<T> = T extends null | undefined ? never : T;

type A = NonNullable<string | null>; // string
type B = NonNullable<number | undefined>; // number
type C = NonNullable<null>; // never

// 更复杂的条件约束
type ApiResponse<T> = T extends string 
    ? { message: T } 
    : T extends number 
    ? { code: T } 
    : { data: T };

type StringResponse = ApiResponse<string>; // { message: string }
type NumberResponse = ApiResponse<number>; // { code: number }
type ObjectResponse = ApiResponse<{ id: number }>; // { data: { id: number } }

多个泛型参数

// 多个泛型参数
function merge<T, U>(obj1: T, obj2: U): T & U {
    return { ...obj1, ...obj2 };
}

let merged = merge(
    { name: "Alice", age: 30 },
    { city: "New York", country: "USA" }
);
// merged 的类型是 { name: string; age: number; city: string; country: string; }

// 泛型函数重载
function reverse<T>(items: T[]): T[];
function reverse(items: string): string;
function reverse<T>(items: T[] | string): T[] | string {
    if (typeof items === "string") {
        return items.split("").reverse().join("");
    }
    return items.slice().reverse();
}

console.log(reverse([1, 2, 3])); // [3, 2, 1]
console.log(reverse("hello")); // "olleh"

泛型工具类型

内置工具类型

// Partial<T> - 将所有属性变为可选
interface User {
    id: number;
    name: string;
    email: string;
}

type PartialUser = Partial<User>;
// 等价于: { id?: number; name?: string; email?: string; }

// Required<T> - 将所有属性变为必需
type RequiredUser = Required<PartialUser>;
// 等价于: { id: number; name: string; email: string; }

// Pick<T, K> - 选择指定属性
type UserSummary = Pick<User, "id" | "name">;
// 等价于: { id: number; name: string; }

// Omit<T, K> - 排除指定属性
type UserWithoutId = Omit<User, "id">;
// 等价于: { name: string; email: string; }

// Record<K, T> - 创建键值对类型
type UserRoles = Record<string, boolean>;
// 等价于: { [key: string]: boolean; }

自定义工具类型

// 深度只读
type DeepReadonly<T> = {
    readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P];
};

interface NestedObject {
    a: {
        b: {
            c: number;
        };
    };
}

type ReadonlyNested = DeepReadonly<NestedObject>;
// 所有嵌套属性都变为只读

// 可空类型
type Nullable<T> = T | null;
type Optional<T> = T | undefined;
type Maybe<T> = T | null | undefined;

// 函数参数类型提取
type Parameters<T extends (...args: any) => any> = T extends (...args: infer P) => any ? P : never;
type ReturnType<T extends (...args: any) => any> = T extends (...args: any) => infer R ? R : any;

function example(a: string, b: number): boolean {
    return true;
}

type ExampleParams = Parameters<typeof example>; // [string, number]
type ExampleReturn = ReturnType<typeof example>; // boolean

实际应用示例

通用数据获取器

// 通用 API 响应类型
interface ApiResponse<T> {
    data: T;
    status: number;
    message: string;
}

// 通用数据获取函数
async function fetchData<T>(url: string): Promise<ApiResponse<T>> {
    const response = await fetch(url);
    return response.json();
}

// 使用示例
interface User {
    id: number;
    name: string;
    email: string;
}

// 类型安全的数据获取

通用状态管理

// 通用状态类型
interface State<T> {
    data: T | null;
    loading: boolean;
    error: string | null;
}

// 状态管理类
class StateManager<T> {
    private state: State<T> = {
        data: null,
        loading: false,
        error: null
    };

    private listeners: Array<(state: State<T>) => void> = [];

    getState(): State<T> {
        return { ...this.state };
    }

    setState(newState: Partial<State<T>>): void {
        this.state = { ...this.state, ...newState };
        this.notifyListeners();
    }

    subscribe(listener: (state: State<T>) => void): () => void {
        this.listeners.push(listener);
        return () => {
            const index = this.listeners.indexOf(listener);
            if (index > -1) {
                this.listeners.splice(index, 1);
            }
        };
    }

    private notifyListeners(): void {
        this.listeners.forEach(listener => listener(this.state));
    }

    async loadData(dataLoader: () => Promise<T>): Promise<void> {
        this.setState({ loading: true, error: null });
        try {
            const data = await dataLoader();
            this.setState({ data, loading: false });
        } catch (error) {
            this.setState({ 
                loading: false, 
                error: error instanceof Error ? error.message : "Unknown error" 
            });
        }
    }
}

// 使用示例
const userManager = new StateManager<User[]>();
const productManager = new StateManager<Product[]>();

userManager.subscribe(state => {
    console.log("User state changed:", state);
});

await userManager.loadData(() => fetchData<User[]>("/api/users").then(res => res.data));

最佳实践

1. 合理命名泛型参数

// 好的命名
interface Repository<TEntity, TKey> {
    findById(id: TKey): Promise<TEntity | null>;
    save(entity: TEntity): Promise<TEntity>;
    delete(id: TKey): Promise<void>;
}

// 避免单字母命名（除非是简单情况）
interface Mapper<TInput, TOutput> {
    map(input: TInput): TOutput;
}

2. 使用默认泛型参数

// 提供默认类型
interface ApiOptions<T = any> {
    url: string;
    method?: string;
    data?: T;
    headers?: Record<string, string>;
}

// 使用时可以省略类型参数
const options1: ApiOptions = { url: "/api/data" };
const options2: ApiOptions<User> = { url: "/api/users", data: user };

3. 避免过度泛型化

// 过度泛型化 - 不推荐
function processData<T, U, V, W>(
    data: T, 
    processor: (item: T) => U, 
    validator: (item: U) => V,
    formatter: (item: V) => W
): W {
    return formatter(validator(processor(data)));
}

// 更好的方式 - 分步处理
function processData<T, U>(data: T, processor: (item: T) => U): U {
    return processor(data);
}

function validateData<T, U>(data: T, validator: (item: T) => U): U {
    return validator(data);
}

总结

泛型是 TypeScript 中最强大的特性之一，它提供了：

  1. 类型安全：在编译时检查类型错误
  2. 代码重用：一套代码适用于多种类型
  3. 性能优化：避免运行时类型检查
  4. 更好的 IDE 支持：自动补全和类型提示

掌握泛型对于编写高质量的 TypeScript 代码至关重要，它是构建大型应用程序的基础。
