编译性能监控

// scripts/build-performance.ts
import { performance } from 'perf_hooks';
import * as ts from 'typescript';

interface CompilationMetrics {
  totalTime: number;
  fileCount: number;
  errorCount: number;
  warningCount: number;
  memoryUsage: NodeJS.MemoryUsage;
}

function measureCompilation(configPath: string): CompilationMetrics {
  const startTime = performance.now();
  const startMemory = process.memoryUsage();

  // 读取配置
  const configFile = ts.readConfigFile(configPath, ts.sys.readFile);
  const parsedConfig = ts.parseJsonConfigFileContent(
    configFile.config,
    ts.sys,
    process.cwd()
  );

  // 创建程序
  const program = ts.createProgram({
    rootNames: parsedConfig.fileNames,
    options: parsedConfig.options
  });

  // 获取诊断信息
  const diagnostics = ts.getPreEmitDiagnostics(program);
  const emitResult = program.emit();

  const endTime = performance.now();
  const endMemory = process.memoryUsage();

  return {
    totalTime: endTime - startTime,
    fileCount: program.getSourceFiles().length,
    errorCount: diagnostics.filter(d => d.category === ts.DiagnosticCategory.Error).length,
    warningCount: diagnostics.filter(d => d.category === ts.DiagnosticCategory.Warning).length,
    memoryUsage: {
      rss: endMemory.rss - startMemory.rss,
      heapTotal: endMemory.heapTotal - startMemory.heapTotal,
      heapUsed: endMemory.heapUsed - startMemory.heapUsed,
      external: endMemory.external - startMemory.external,
      arrayBuffers: endMemory.arrayBuffers - startMemory.arrayBuffers
    }
  };
}

console.log(`文件数量: ${metrics.fileCount}`);

// 对象池模式
class ObjectPool<T> {
  private pool: T[] = [];
  private createFn: () => T;
  private resetFn: (obj: T) => void;

  constructor(createFn: () => T, resetFn: (obj: T) => void, initialSize: number = 10) {
    this.createFn = createFn;
    this.resetFn = resetFn;
    
    // 预填充对象池
    for (let i = 0; i < initialSize; i++) {
      this.pool.push(this.createFn());
    }
  }

  acquire(): T {
    return this.pool.pop() || this.createFn();
  }

  release(obj: T): void {
    this.resetFn(obj);
    this.pool.push(obj);
  }

  size(): number {
    return this.pool.length;
  }
}

// 使用示例
interface Point {
  x: number;
  y: number;
}

const pointPool = new ObjectPool<Point>(
  () => ({ x: 0, y: 0 }),
  (point) => { point.x = 0; point.y = 0; },
  100
);

function calculateDistance(points: Point[]): number {
  let total = 0;
  
  for (let i = 0; i < points.length - 1; i++) {
    const current = points[i];
    const next = points[i + 1];
    
    const dx = next.x - current.x;
    const dy = next.y - current.y;
    total += Math.sqrt(dx * dx + dy * dy);
  }
  
  return total;
}

函数优化

// 记忆化装饰器
function memoize<T extends (...args: any[]) => any>(
  fn: T,
  keyGenerator?: (...args: Parameters<T>) => string
): T {
  const cache = new Map<string, ReturnType<T>>();
  
  const memoized = (...args: Parameters<T>): ReturnType<T> => {
    const key = keyGenerator ? keyGenerator(...args) : JSON.stringify(args);
    
    if (cache.has(key)) {
      return cache.get(key)!;
    }
    
    const result = fn(...args);
    cache.set(key, result);
    return result;
  };
  
  return memoized as T;
}

// 使用示例
const expensiveCalculation = memoize((n: number): number => {
  console.log(`计算 ${n} 的阶乘`);
  if (n <= 1) return 1;
  return n * expensiveCalculation(n - 1);
});

// 防抖和节流
function debounce<T extends (...args: any[]) => any>(
  func: T,
  delay: number
): (...args: Parameters<T>) => void {
  let timeoutId: NodeJS.Timeout;
  
  return (...args: Parameters<T>) => {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => func(...args), delay);
  };
}

function throttle<T extends (...args: any[]) => any>(
  func: T,
  limit: number
): (...args: Parameters<T>) => void {
  let inThrottle: boolean;
  
  return (...args: Parameters<T>) => {
    if (!inThrottle) {
      func(...args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);
    }
  };
}

异步操作优化

// 并发控制
class ConcurrencyController {
  private running = 0;
  private queue: Array<() => Promise<any>> = [];

  constructor(private maxConcurrency: number) {}

  async execute<T>(task: () => Promise<T>): Promise<T> {
    return new Promise((resolve, reject) => {
      this.queue.push(async () => {
        try {
          const result = await task();
          resolve(result);
        } catch (error) {
          reject(error);
        }
      });
      
      this.processQueue();
    });
  }

  private async processQueue(): Promise<void> {
    if (this.running >= this.maxConcurrency || this.queue.length === 0) {
      return;
    }

    this.running++;
    const task = this.queue.shift()!;
    
    try {
      await task();
    } finally {
      this.running--;
      this.processQueue();
    }
  }
}

// 批处理操作
class BatchProcessor<T, R> {
  private batch: T[] = [];
  private timeoutId: NodeJS.Timeout | null = null;

  constructor(
    private processBatch: (items: T[]) => Promise<R[]>,
    private batchSize: number = 10,
    private batchTimeout: number = 100
  ) {}

  async add(item: T): Promise<R> {
    return new Promise((resolve, reject) => {
      this.batch.push(item);
      
      const index = this.batch.length - 1;
      
      // 设置超时处理
      if (this.timeoutId) {
        clearTimeout(this.timeoutId);
      }
      
      this.timeoutId = setTimeout(() => {
        this.flush().then(results => {
          resolve(results[index]);
        }).catch(reject);
      }, this.batchTimeout);
      
      // 检查是否达到批处理大小
      if (this.batch.length >= this.batchSize) {
        if (this.timeoutId) {
          clearTimeout(this.timeoutId);
          this.timeoutId = null;
        }
        
        this.flush().then(results => {
          resolve(results[index]);
        }).catch(reject);
      }
    });
  }

  private async flush(): Promise<R[]> {
    if (this.batch.length === 0) {
      return [];
    }

    const currentBatch = this.batch.splice(0);
    return this.processBatch(currentBatch);
  }
}

// 事件监听器管理
class EventManager {
  private listeners = new Map<string, Set<Function>>();
  private abortController = new AbortController();

  addEventListener<K extends keyof WindowEventMap>(
    type: K,
    listener: (event: WindowEventMap[K]) => void,
    options?: AddEventListenerOptions
  ): void {
    // 使用 AbortController 统一管理
    const signal = this.abortController.signal;
    window.addEventListener(type, listener, { ...options, signal });
    
    // 记录监听器
    if (!this.listeners.has(type)) {
      this.listeners.set(type, new Set());
    }
    this.listeners.get(type)!.add(listener);
  }

  removeEventListener<K extends keyof WindowEventMap>(
    type: K,
    listener: (event: WindowEventMap[K]) => void
  ): void {
    window.removeEventListener(type, listener);
    
    const typeListeners = this.listeners.get(type);
    if (typeListeners) {
      typeListeners.delete(listener);
      if (typeListeners.size === 0) {
        this.listeners.delete(type);
      }
    }
  }

  cleanup(): void {
    // 移除所有监听器
    this.abortController.abort();
    this.listeners.clear();
  }
}

// 虚拟滚动实现
interface VirtualScrollOptions {
  itemHeight: number;
  containerHeight: number;
  overscan?: number;
}

class VirtualScroll<T> {
  private items: T[] = [];
  private scrollTop = 0;
  private options: Required<VirtualScrollOptions>;

  constructor(options: VirtualScrollOptions) {
    this.options = {
      overscan: 5,
      ...options
    };
  }

  setItems(items: T[]): void {
    this.items = items;
  }

  setScrollTop(scrollTop: number): void {
    this.scrollTop = scrollTop;
  }

  getVisibleRange(): { start: number; end: number; items: T[] } {
    const { itemHeight, containerHeight, overscan } = this.options;
    
    const visibleStart = Math.floor(this.scrollTop / itemHeight);
    const visibleEnd = Math.min(
      visibleStart + Math.ceil(containerHeight / itemHeight),
      this.items.length - 1
    );

    const start = Math.max(0, visibleStart - overscan);
    const end = Math.min(this.items.length - 1, visibleEnd + overscan);

    return {
      start,
      end,
      items: this.items.slice(start, end + 1)
    };
  }

  getTotalHeight(): number {
    return this.items.length * this.options.itemHeight;
  }

  getOffsetY(index: number): number {
    return index * this.options.itemHeight;
  }
}

// 分页数据加载
class PaginatedDataLoader<T> {
  private cache = new Map<number, T[]>();
  private loading = new Set<number>();

  constructor(
    private fetchPage: (page: number, size: number) => Promise<T[]>,
    private pageSize: number = 20
  ) {}

  async getPage(page: number): Promise<T[]> {
    // 检查缓存
    if (this.cache.has(page)) {
      return this.cache.get(page)!;
    }

    // 检查是否正在加载
    if (this.loading.has(page)) {
      return new Promise((resolve) => {
        const checkLoading = () => {
          if (!this.loading.has(page)) {
            resolve(this.cache.get(page) || []);
          } else {
            setTimeout(checkLoading, 10);
          }
        };
        checkLoading();
      });
    }

    // 开始加载
    this.loading.add(page);
    
    try {
      const data = await this.fetchPage(page, this.pageSize);
      this.cache.set(page, data);
      return data;
    } finally {
      this.loading.delete(page);
    }
  }

  clearCache(): void {
    this.cache.clear();
  }

  preloadPages(currentPage: number, range: number = 2): void {
    for (let i = Math.max(0, currentPage - range); i <= currentPage + range; i++) {
      if (!this.cache.has(i) && !this.loading.has(i)) {
        this.getPage(i).catch(() => {
          // 预加载失败时静默处理
        });
      }
    }
  }
}

class PerformanceMonitor {

        }
      });
      
      navObserver.observe({ entryTypes: ['navigation'] });
      this.observers.push(navObserver);
    }

    // 监控资源加载
    const resourceObserver = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.entryType === 'resource') {
          const resourceEntry = entry as PerformanceResourceTiming;
          this.recordMetric(`resource.${resourceEntry.initiatorType}`, resourceEntry.duration);
        }
      }
    });
    
    resourceObserver.observe({ entryTypes: ['resource'] });
    this.observers.push(resourceObserver);
  }

  // 记录自定义指标
  recordMetric(name: string, value: number): void {
    if (!this.metrics.has(name)) {
      this.metrics.set(name, []);
    }
    this.metrics.get(name)!.push(value);
  }

  // 测量函数执行时间
  measure<T>(name: string, fn: () => T): T {
    const start = performance.now();
    const result = fn();
    const end = performance.now();
    this.recordMetric(name, end - start);
    return result;
  }

  // 测量异步函数执行时间
  async measureAsync<T>(name: string, fn: () => Promise<T>): Promise<T> {
    const start = performance.now();
    const result = await fn();
    const end = performance.now();
    this.recordMetric(name, end - start);
    return result;
  }

  // 获取指标统计
  getMetricStats(name: string): { avg: number; min: number; max: number; count: number } | null {
    const values = this.metrics.get(name);
    if (!values || values.length === 0) {
      return null;
    }

    const sum = values.reduce((a, b) => a + b, 0);
    return {
      avg: sum / values.length,
      min: Math.min(...values),
      max: Math.max(...values),
      count: values.length
    };
  }

  // 清理资源
  cleanup(): void {
    this.observers.forEach(observer => observer.disconnect());
    this.observers = [];
    this.metrics.clear();
  }
}

内存使用分析

// 内存使用监控
class MemoryMonitor {
  private intervalId: NodeJS.Timeout | null = null;
  private samples: Array<{ timestamp: number; usage: MemoryInfo }> = [];

  start(interval: number = 5000): void {
    if (this.intervalId) {
      return;
    }

    this.intervalId = setInterval(() => {
      if ('memory' in performance) {
        this.samples.push({
          timestamp: Date.now(),
          usage: (performance as any).memory
        });

        // 保持最近 100 个样本
        if (this.samples.length > 100) {
          this.samples.shift();
        }
      }
    }, interval);
  }

  stop(): void {
    if (this.intervalId) {
      clearInterval(this.intervalId);
      this.intervalId = null;
    }
  }

  getMemoryTrend(): Array<{ timestamp: number; heapUsed: number; heapTotal: number }> {
    return this.samples.map(sample => ({
      timestamp: sample.timestamp,
      heapUsed: sample.usage.usedJSHeapSize,
      heapTotal: sample.usage.totalJSHeapSize
    }));
  }

  detectMemoryLeaks(): boolean {
    if (this.samples.length < 10) {
      return false;
    }

    const recent = this.samples.slice(-10);
    const trend = recent.map(s => s.usage.usedJSHeapSize);
    
    // 简单的内存泄漏检测：连续增长
    let increasing = 0;
    for (let i = 1; i < trend.length; i++) {
      if (trend[i] > trend[i - 1]) {
        increasing++;
      }
    }

    return increasing > 7; // 70% 的样本都在增长
  }

  getCurrentMemoryUsage(): MemoryInfo | null {
    if ('memory' in performance) {
      return (performance as any).memory;
    }
    return null;
  }
}

总结

TypeScript 性能优化是一个多层面的工程：

  1. 编译性能：通过合理的 TypeScript 配置和项目结构优化编译速度
  2. 类型系统：避免过度复杂的类型计算，合理使用类型缓存
  3. 运行时性能：使用对象池、记忆化、虚拟滚动等技术优化执行效率
  4. 内存管理：预防内存泄漏，优化大数据处理
  5. 构建优化：通过代码分割、压缩等技术减少包体积
  6. 监控分析：建立性能监控体系，持续优化

性能优化是一个持续的过程，需要根据具体的应用场景和需求进行针对性的优化。
