枚举 (Enums)

TypeScript | 2025.07.31

什么是枚举？

枚举（Enum）是 TypeScript 中的一种数据类型，它允许我们定义一组命名的常量。枚举使代码更加可读和可维护，特别适用于表示一组相关的常量值。

数字枚举

基本数字枚举

// 基本数字枚举
enum Direction {
    Up,    // 0
    Down,  // 1
    Left,  // 2
    Right  // 3
}

console.log(Direction.Up);    // 0
console.log(Direction.Down);  // 1
console.log(Direction.Left);  // 2
console.log(Direction.Right); // 3

// 使用枚举
function move(direction: Direction) {
    switch (direction) {
        case Direction.Up:
            console.log("Moving up");
            break;
        case Direction.Down:
            console.log("Moving down");
            break;
        case Direction.Left:
            console.log("Moving left");
            break;
        case Direction.Right:
            console.log("Moving right");
            break;
    }
}

move(Direction.Up); // "Moving up"

计算成员和常量成员

常量成员

enum FileAccess {
    // 常量成员
    None,
    Read = 1 << 1,
    Write = 1 << 2,
    ReadWrite = Read | Write,
    
    // 计算成员
    G = "123".length
}

console.log(FileAccess.None);      // 0
console.log(FileAccess.Read);      // 2
console.log(FileAccess.Write);     // 4
console.log(FileAccess.ReadWrite); // 6
console.log(FileAccess.G);         // 3

位标志枚举

// 位标志枚举
enum Permission {
    None = 0,
    Read = 1 << 0,    // 1
    Write = 1 << 1,   // 2
    Execute = 1 << 2, // 4
    Delete = 1 << 3,  // 8
    
    // 组合权限
    ReadWrite = Read | Write,           // 3
    ReadExecute = Read | Execute,       // 5
    All = Read | Write | Execute | Delete // 15
}

// 权限检查函数
function hasPermission(userPermissions: Permission, requiredPermission: Permission): boolean {
    return (userPermissions & requiredPermission) === requiredPermission;
}

// 添加权限
function addPermission(current: Permission, newPermission: Permission): Permission {
    return current | newPermission;
}

// 移除权限
function removePermission(current: Permission, permissionToRemove: Permission): Permission {
    return current & ~permissionToRemove;
}

// 使用示例
let userPerms = Permission.Read | Permission.Write;
console.log(hasPermission(userPerms, Permission.Read));  // true
console.log(hasPermission(userPerms, Permission.Execute)); // false

userPerms = addPermission(userPerms, Permission.Execute);
console.log(hasPermission(userPerms, Permission.Execute)); // true

userPerms = removePermission(userPerms, Permission.Write);
console.log(hasPermission(userPerms, Permission.Write)); // false

字面量枚举

// 字面量枚举
enum ShapeKind {
    Circle,
    Square
}

interface Circle {
    kind: ShapeKind.Circle;
    radius: number;
}

interface Square {
    kind: ShapeKind.Square;
    sideLength: number;
}

type Shape = Circle | Square;

function getArea(shape: Shape): number {
    switch (shape.kind) {
        case ShapeKind.Circle:
            return Math.PI * shape.radius ** 2;
        case ShapeKind.Square:
            return shape.sideLength ** 2;
        default:
            // TypeScript 知道这里永远不会执行
            const _exhaustiveCheck: never = shape;
            return _exhaustiveCheck;
    }
}

实际应用示例

状态机

// 订单状态枚举
enum OrderStatus {
    Pending = "pending",
    Processing = "processing",
    Shipped = "shipped",
    Delivered = "delivered",
    Cancelled = "cancelled",
    Returned = "returned"
}

// 状态转换映射
const statusTransitions: Record<OrderStatus, OrderStatus[]> = {
    [OrderStatus.Pending]: [OrderStatus.Processing, OrderStatus.Cancelled],
    [OrderStatus.Processing]: [OrderStatus.Shipped, OrderStatus.Cancelled],
    [OrderStatus.Shipped]: [OrderStatus.Delivered, OrderStatus.Returned],
    [OrderStatus.Delivered]: [OrderStatus.Returned],
    [OrderStatus.Cancelled]: [],
    [OrderStatus.Returned]: []
};

class Order {
    constructor(
        public id: string,
        public status: OrderStatus = OrderStatus.Pending
    ) {}

    canTransitionTo(newStatus: OrderStatus): boolean {
        return statusTransitions[this.status].includes(newStatus);
    }

    transitionTo(newStatus: OrderStatus): void {
        if (this.canTransitionTo(newStatus)) {
            console.log(`Order ${this.id}: ${this.status} -> ${newStatus}`);
            this.status = newStatus;
        } else {
            throw new Error(`Cannot transition from ${this.status} to ${newStatus}`);
        }
    }
}

// 使用示例
const order = new Order("ORD-001");
order.transitionTo(OrderStatus.Processing); // 成功
order.transitionTo(OrderStatus.Shipped);    // 成功
// order.transitionTo(OrderStatus.Pending);  // 错误：不能从 shipped 转换到 pending

配置管理

// 环境枚举
enum Environment {
    Development = "development",
    Testing = "testing",
    Staging = "staging",
    Production = "production"
}

// 日志级别枚举
enum LogLevel {
    Debug = 0,
    Info = 1,
    Warn = 2,
    Error = 3
}

// 配置接口
interface Config {
    environment: Environment;
    logLevel: LogLevel;
    apiUrl: string;
    enableDebug: boolean;
}

// 配置工厂
class ConfigFactory {
    static create(env: Environment): Config {
        const baseConfig = {
            environment: env,
            logLevel: LogLevel.Info,
            apiUrl: "",
            enableDebug: false
        };

        switch (env) {
            case Environment.Development:
                return {
                    ...baseConfig,
                    logLevel: LogLevel.Debug,
                    apiUrl: "http://localhost:3000/api",
                    enableDebug: true
                };
            
            case Environment.Testing:
                return {
                    ...baseConfig,
                    logLevel: LogLevel.Debug,
                    apiUrl: "http://test-api.example.com/api",
                    enableDebug: true
                };
            
            case Environment.Staging:
                return {
                    ...baseConfig,
                    logLevel: LogLevel.Warn,
                    apiUrl: "http://staging-api.example.com/api",
                    enableDebug: false
                };
            
            case Environment.Production:
                return {
                    ...baseConfig,
                    logLevel: LogLevel.Error,
                    apiUrl: "https://api.example.com/api",
                    enableDebug: false
                };
            
            default:
                throw new Error(`Unsupported environment: ${env}`);
        }
    }
}

// 使用示例
const config = ConfigFactory.create(Environment.Development);
console.log(config);

枚举的最佳实践

1. 优先使用字符串枚举

// 推荐：字符串枚举
enum Theme {
    Light = "light",
    Dark = "dark",
    Auto = "auto"
}

// 不推荐：数字枚举（除非有特殊需求）
enum ThemeNumeric {
    Light,  // 0
    Dark,   // 1
    Auto    // 2
}

2. 使用 const 枚举优化性能

// 对于不需要反向映射的枚举，使用 const
const enum HttpMethod {
    GET = "GET",
    POST = "POST",
    PUT = "PUT",
    DELETE = "DELETE"
}

// 编译后会被内联，减少运行时开销

3. 枚举命名规范

// 好的命名
enum UserStatus {
    Active = "active",
    Inactive = "inactive",
    Suspended = "suspended"
}

enum PaymentMethod {
    CreditCard = "credit_card",
    PayPal = "paypal",
    BankTransfer = "bank_transfer"
}

// 避免过于简短或模糊的命名
enum Status {  // 太模糊
    A = "a",   // 不清楚含义
    B = "b"
}

4. 枚举与联合类型的选择

// 当值是固定的字符串时，可以考虑使用联合类型
type Theme = "light" | "dark" | "auto";

// 当需要更多功能（如反向映射、位运算）时，使用枚举
enum Permission {
    Read = 1,
    Write = 2,
    Execute = 4
}

总结

枚举是 TypeScript 中组织相关常量的强大工具：

  1. 数字枚举：默认从 0 开始，支持反向映射
  2. 字符串枚举：更加明确和可读，推荐使用
  3. 常量枚举：编译时内联，性能更好
  4. 位标志枚举：适用于权限系统等场景

选择合适的枚举类型可以让代码更加清晰、类型安全和易于维护。
