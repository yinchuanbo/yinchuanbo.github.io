三斜线指令 (Triple-Slash Directives)

TypeScript | 2025.07.31

1. 什么是三斜线指令？

三斜线指令是 TypeScript 中的特殊注释，以 /// 开头，用于向编译器提供额外的编译指令和元数据。这些指令必须出现在文件的顶部，在任何代码或注释之前。

2. 基本语法

/// <directive attribute="value" />

// 示例
/// <reference path="./types.d.ts" />
/// <reference types="node" />
/// <amd-module name="MyModule" />

3. 常用三斜线指令

3.1 reference path 指令

用于声明文件之间的依赖关系。

// main.ts
/// <reference path="./utils.ts" />
/// <reference path="./types.d.ts" />

// 现在可以使用 utils.ts 和 types.d.ts 中定义的内容
const result = calculateSum(1, 2); // 来自 utils.ts
const user: User = { name: "Alice", age: 30 }; // 来自 types.d.ts

// utils.ts
function calculateSum(a: number, b: number): number {
  return a + b;
}

function formatDate(date: Date): string {
  return date.toISOString().split('T')[0];
}

// types.d.ts
interface User {
  name: string;
  age: number;
}

interface Product {
  id: string;
  name: string;
  price: number;
}

3.2 reference types 指令

用于声明对 npm 包类型定义的依赖。

/// <reference types="node" />
/// <reference types="jquery" />
/// <reference types="lodash" />

// 现在可以使用这些库的类型定义
import * as fs from 'fs';
import * as path from 'path';

// 使用 Node.js 类型
const buffer: Buffer = fs.readFileSync('file.txt');
const dirname: string = path.dirname(__filename);

// 使用 jQuery 类型（如果在浏览器环境）
// $('body').addClass('loaded');

// 使用 Lodash 类型
// const result = _.map([1, 2, 3], x => x * 2);

3.3 reference lib 指令

用于包含内置库的类型定义。

/// <reference lib="es2015" />
/// <reference lib="dom" />
/// <reference lib="webworker" />

// 使用 ES2015 特性
const map = new Map<string, number>();
const set = new Set<string>();
const promise = Promise.resolve(42);

// 使用 DOM API（在浏览器环境）
const element: HTMLElement = document.getElementById('app')!;
const canvas: HTMLCanvasElement = document.createElement('canvas');

// 使用 Web Worker API
// const worker = new Worker('worker.js');

3.4 amd-module 指令

用于为 AMD 模块指定模块名称。

/// <amd-module name="MyCustomModule" />

export function greet(name: string): string {
  return `Hello, ${name}!`;
}

export class Calculator {
  add(a: number, b: number): number {
    return a + b;
  }
}

// 编译后的 AMD 模块将使用 "MyCustomModule" 作为模块名

3.5 amd-dependency 指令

用于声明对非 TypeScript AMD 模块的依赖。

/// <amd-dependency path="legacy-library" />
/// <amd-dependency path="jquery" name="$" />

// 声明全局变量（对应 amd-dependency）
declare var legacyLibrary: any;
declare var $: any;

export function useLibraries() {
  // 使用传统库
  legacyLibrary.doSomething();
  
  // 使用 jQuery
  $('body').fadeIn();
}

4. 高级三斜线指令

4.1 no-default-lib 指令

禁用默认库的包含。

/// <reference no-default-lib="true"/>
/// <reference lib="es5" />

// 只包含 ES5 库，不包含默认的库
// 这在创建自定义运行时环境时很有用

interface Array<T> {
  length: number;
  [n: number]: T;
}

interface Object {
  constructor: Function;
  toString(): string;
  valueOf(): Object;
}

// 基本的全局函数
declare function parseInt(s: string, radix?: number): number;
declare function parseFloat(string: string): number;

4.2 条件编译指令

虽然不是标准的三斜线指令，但可以结合使用实现条件编译。

/// <reference path="./debug.d.ts" />

// debug.d.ts
declare const DEBUG: boolean;

// main.ts
if (DEBUG) {
  console.log("Debug mode enabled");
}

// 在不同环境下，DEBUG 可以有不同的值

    id: string;
    name: string;
    email: string;
    roles: string[];
  }
  
  interface ApiResponse<T> {
    data: T;
    status: number;
    message: string;
  }
}

// core/utils.ts
namespace Core {
  export class HttpClient {
    constructor(private config: Config) {}
    
    async get<T>(url: string): Promise<ApiResponse<T>> {
      // 实现 HTTP GET
      return {} as ApiResponse<T>;
    }
    
    async post<T>(url: string, data: any): Promise<ApiResponse<T>> {
      // 实现 HTTP POST
      return {} as ApiResponse<T>;
    }
  }
  
  export function formatDate(date: Date): string {
    return date.toISOString().split('T')[0];
  }
  
  export function generateId(): string {
    return Math.random().toString(36).substr(2, 9);

5.2 插件系统的类型定义

// plugins/base.d.ts
/// <reference types="node" />

declare namespace PluginSystem {
  interface Plugin {
    name: string;
    version: string;
    dependencies?: string[];
    init(context: PluginContext): Promise<void>;
    destroy?(): Promise<void>;
  }
  
  interface PluginContext {
    config: any;
    logger: Logger;
    events: EventEmitter;
  }
  
  interface Logger {
    info(message: string): void;
    warn(message: string): void;
    error(message: string): void;
    debug(message: string): void;
  }
  
  interface EventEmitter {
    on(event: string, listener: Function): void;
    emit(event: string, ...args: any[]): void;
    off(event: string, listener: Function): void;
  }
}

// plugins/auth-plugin.ts
/// <reference path="./base.d.ts" />

class AuthPlugin implements PluginSystem.Plugin {
  name = "auth-plugin";
  version = "1.0.0";
  dependencies = ["database-plugin"];
  
  async init(context: PluginSystem.PluginContext): Promise<void> {
    context.logger.info("Initializing auth plugin");
    
    context.events.on("user:login", this.handleLogin.bind(this));
    context.events.on("user:logout", this.handleLogout.bind(this));
  }
  
  private handleLogin(user: any) {
    // 处理用户登录
  }
  
  private handleLogout(user: any) {
    // 处理用户登出
  }
}

export = AuthPlugin;

5.3 多环境配置系统

// config/base.d.ts
declare namespace Config {

    host: string;

6. 与现代模块系统的关系

6.1 三斜线指令 vs ES6 模块

// 传统三斜线指令方式
/// <reference path="./utils.ts" />
/// <reference path="./types.d.ts" />

// 使用全局命名空间
const result = MyNamespace.calculateSum(1, 2);

// 现代 ES6 模块方式
import { calculateSum } from './utils';
import { User } from './types';

// 显式导入
const result = calculateSum(1, 2);
const user: User = { name: "Alice", age: 30 };

6.2 何时使用三斜线指令

// ✅ 适合使用三斜线指令的场景

// 1. 声明文件中的依赖
/// <reference types="node" />
declare module "my-module" {
  export function doSomething(): void;
}

// 2. 全局类型定义
/// <reference path="./global.d.ts" />
declare global {
  interface Window {
    myGlobalFunction(): void;
  }
}

// 3. AMD 模块配置
/// <amd-module name="MyModule" />
export class MyClass {}

// ❌ 不推荐使用三斜线指令的场景

// 1. 常规模块导入（使用 import 代替）
// /// <reference path="./module.ts" />
import { MyFunction } from './module';

// 2. npm 包导入（使用 import 代替）
// /// <reference types="lodash" />
import * as _ from 'lodash';

7.2 编译顺序和依赖解析

// 文件 A
/// <reference path="./B.ts" />
/// <reference path="./C.ts" />

// TypeScript 编译器会：
// 1. 首先编译 B.ts
// 2. 然后编译 C.ts  
// 3. 最后编译 A.ts

// 这确保了依赖关系的正确解析

8. 最佳实践

8.1 使用原则

// ✅ 好的实践

// 1. 将三斜线指令放在文件顶部
/// <reference types="node" />
/// <reference path="./types.d.ts" />

// 2. 按类型分组
/// <reference types="node" />
/// <reference types="express" />
/// <reference path="./utils.ts" />
/// <reference path="./models.ts" />

// 3. 使用相对路径
/// <reference path="./relative/path.ts" />

// ❌ 避免的做法

// 1. 不要在代码中间使用
const x = 1;
/// <reference path="./types.d.ts" /> // 错误位置

// 2. 避免循环引用
// A.ts: /// <reference path="./B.ts" />
// B.ts: /// <reference path="./A.ts" /> // 循环引用

// 3. 不要使用绝对路径
/// <reference path="/absolute/path/file.ts" /> // 不推荐

8.2 迁移策略

// 从三斜线指令迁移到 ES6 模块

// 旧方式
/// <reference path="./utils.ts" />
namespace MyApp {
  export function main() {
    const result = Utils.calculate(1, 2);
  }
}

// 新方式
import { calculate } from './utils';

export function main() {
  const result = calculate(1, 2);
}

10. 总结

三斜线指令是 TypeScript 的重要特性：

  1. 文件依赖 - 声明文件间的依赖关系
  2. 类型引用 - 引用外部类型定义
  3. 库包含 - 控制内置库的包含
  4. 模块配置 - 配置 AMD 模块

使用场景：

  • 声明文件编写
  • 全局类型定义
  • 传统项目迁移
  • 特殊编译需求

现代开发中，推荐优先使用 ES6 模块系统，三斜线指令主要用于特殊场景和向后兼容。
