TypeScript 模板字面量类型 (Template Literal Types)

什么是模板字面量类型？

模板字面量类型是 TypeScript 4.1 引入的强大特性，它允许使用模板字面量语法来创建新的字符串字面量类型。这个特性结合了 JavaScript 的模板字面量语法和 TypeScript 的类型系统。

基本语法

// 基本模板字面量类型语法
type TemplateType = `prefix_${string}_suffix`;

// 使用联合类型
type Color = 'red' | 'green' | 'blue';
type ColoredButton = `${Color}Button`;
// 结果：'redButton' | 'greenButton' | 'blueButton'

// 多个插值
type Size = 'small' | 'medium' | 'large';
type StyledButton = `${Color}_${Size}_button`;
// 结果：'red_small_button' | 'red_medium_button' | 'red_large_button' | 
//       'green_small_button' | 'green_medium_button' | 'green_large_button' |
//       'blue_small_button' | 'blue_medium_button' | 'blue_large_button'

字符串操作

字符串转换工具类型

// 驼峰命名转换
type CamelCase<S extends string> = S extends `${infer P1}_${infer P2}${infer P3}`
  ? `${P1}${Uppercase<P2>}${CamelCase<P3>}`
  : S;

type CamelCased = CamelCase<'hello_world_example'>;
// 结果：'helloWorldExample'

// 下划线命名转换
type SnakeCase<S extends string> = S extends `${infer T}${infer U}`
  ? `${T extends Capitalize<T> ? '_' : ''}${Lowercase<T>}${SnakeCase<U>}`
  : S;

type SnakeCased = SnakeCase<'HelloWorldExample'>;
// 结果：'hello_world_example'

// 短横线命名转换
type KebabCase<S extends string> = S extends `${infer T}${infer U}`
  ? `${T extends Capitalize<T> ? '-' : ''}${Lowercase<T>}${KebabCase<U>}`
  : S;

type KebabCased = KebabCase<'HelloWorldExample'>;
// 结果：'hello-world-example'

// 首字母大写
type PascalCase<S extends string> = S extends `${infer First}${infer Rest}`
  ? `${Uppercase<First>}${Rest}`
  : S;

type PascalCased = PascalCase<'hello'>;
// 结果：'Hello'

字符串解析

// 解析版本号
type ParseVersion<V extends string> = V extends `${infer Major}.${infer Minor}.${infer Patch}`
  ? {
      major: Major;
      minor: Minor;
      patch: Patch;
    }
  : never;

type Version = ParseVersion<'1.2.3'>;
// 结果：{ major: '1'; minor: '2'; patch: '3'; }

// 解析 URL 路径参数
type ExtractParams<T extends string> = 
  T extends `${string}:${infer Param}/${infer Rest}`
    ? Param | ExtractParams<Rest>
    : T extends `${string}:${infer Param}`
    ? Param
    : never;

type RouteParams = ExtractParams<'/users/:id/posts/:postId/comments/:commentId'>;
// 结果：'id' | 'postId' | 'commentId'

// 解析查询字符串
type ParseQuery<T extends string> = 
  T extends `${infer Key}=${infer Value}&${infer Rest}`
    ? { [K in Key]: Value } & ParseQuery<Rest>
    : T extends `${infer Key}=${infer Value}`
    ? { [K in Key]: Value }
    : {};

type QueryParams = ParseQuery<'name=john&age=25&active=true'>;
// 结果：{ name: 'john'; age: '25'; active: 'true'; }

// 解析文件路径
type ParsePath<T extends string> = T extends `${infer Dir}/${infer File}`
  ? { directory: Dir; filename: ParsePath<File> }
  : T extends `${infer Name}.${infer Ext}`
  ? { name: Name; extension: Ext }
  : { name: T; extension: never };

type FilePath = ParsePath<'src/components/Button.tsx'>;
// 结果：{
//   directory: 'src';
//   filename: {
//     directory: 'components';
//     filename: { name: 'Button'; extension: 'tsx' };
//   };
// }

API 路由类型系统

// HTTP 方法
type HttpMethod = 'GET' | 'POST' | 'PUT' | 'DELETE' | 'PATCH';

// 路由定义
type Route = 
  | '/users'
  | '/users/:id'
  | '/users/:id/posts'
  | '/users/:id/posts/:postId'
  | '/posts'
  | '/posts/:id'
  | '/auth/login'
  | '/auth/register';

// 组合 HTTP 方法和路由
type ApiEndpoint = `${HttpMethod} ${Route}`;

// 提取路由参数
type ExtractRouteParams<T extends string> = 
  T extends `${string} ${infer Path}`
    ? ExtractParams<Path>
    : never;

// 路由参数映射
type RouteParamsMap = {
  [K in ApiEndpoint]: ExtractRouteParams<K> extends never
    ? {}
    : { [P in ExtractRouteParams<K>]: string };
};

// 类型安全的 API 客户端
class TypedApiClient {
  async request<T extends ApiEndpoint>(
    endpoint: T,
    params: RouteParamsMap[T],
    body?: any
  ): Promise<any> {
    const [method, path] = endpoint.split(' ');
    
    // 替换路径参数
    let finalPath = path;
    for (const [key, value] of Object.entries(params)) {
      finalPath = finalPath.replace(`:${key}`, value as string);
    }
    
    return fetch(finalPath, {
      method,
      body: body ? JSON.stringify(body) : undefined,
      headers: {
        'Content-Type': 'application/json'
      }
    }).then(res => res.json());
  }
}

const client = new TypedApiClient();

// 类型安全的使用
client.request('GET /users/:id', { id: '123' });
client.request('POST /users/:id/posts', { id: '123' }, { title: 'Hello' });
// client.request('GET /users/:id', {}); // Error: 缺少 id 参数

  title: string;

// 查询操作符
type QueryOperator = 'eq' | 'ne' | 'gt' | 'gte' | 'lt' | 'lte' | 'like' | 'in';

// 生成查询方法名
type QueryMethod<T extends string, O extends string> = `${T}${Capitalize<O>}`;

// 查询构建器类型
type QueryBuilder<T> = {
  [K in keyof T as K extends string 
    ? QueryMethod<K, QueryOperator>
    : never
  ]: T[K] extends number
    ? (value: number) => QueryBuilder<T>
    : T[K] extends string
    ? (value: string) => QueryBuilder<T>
    : (value: T[K]) => QueryBuilder<T>;
} & {
  build(): string;
  execute(): Promise<T[]>;
};

// 使用示例（概念性）
declare function createQueryBuilder<T>(): QueryBuilder<T>;

const userQuery = createQueryBuilder<User>()
  .nameEq('John')
  .ageGt(18)
  .emailLike('%@gmail.com')
  .build();

// SQL 查询类型
type SQLQuery<T extends string> = `SELECT * FROM ${T}`;
type SQLInsert<T extends string> = `INSERT INTO ${T}`;
type SQLUpdate<T extends string> = `UPDATE ${T} SET`;
type SQLDelete<T extends string> = `DELETE FROM ${T}`;

type SQLOperation<T extends string, O extends string> = 
  O extends 'select' ? SQLQuery<T> :
  O extends 'insert' ? SQLInsert<T> :
  O extends 'update' ? SQLUpdate<T> :
  O extends 'delete' ? SQLDelete<T> :
  never;

type UserSelectQuery = SQLOperation<'users', 'select'>;
// 结果：'SELECT * FROM users'

状态机类型系统

// 状态定义
type UserState = 'idle' | 'loading' | 'success' | 'error';
type UserAction = 'fetch' | 'reset' | 'retry';

// 状态转换类型
type StateTransition<S extends string, A extends string> = `${S}_${A}`;

// 生成所有可能的状态转换
type AllTransitions = StateTransition<UserState, UserAction>;
// 结果：'idle_fetch' | 'idle_reset' | 'idle_retry' | 'loading_fetch' | 'loading_reset' | 
//       'loading_retry' | 'success_fetch' | 'success_reset' | 'success_retry' | 
//       'error_fetch' | 'error_reset' | 'error_retry'

// 有效的状态转换
type ValidTransitions = 
  | 'idle_fetch'
  | 'loading_reset'
  | 'success_reset'
  | 'error_retry'
  | 'error_reset';

// 状态机定义
type StateMachine<S extends string, A extends string> = {
  [K in StateTransition<S, A>]: K extends ValidTransitions ? S : never;
};

// 状态机实现
class TypedStateMachine<S extends string, A extends string> {
  constructor(private currentState: S) {}

  transition<T extends A>(action: T): S | never {
    const transitionKey = `${this.currentState}_${action}` as StateTransition<S, A>;
    
    // 这里需要运行时逻辑来处理状态转换
    switch (transitionKey) {
      case 'idle_fetch':
        this.currentState = 'loading' as S;
        break;
      case 'loading_reset':
      case 'success_reset':
      case 'error_reset':
        this.currentState = 'idle' as S;
        break;
      case 'error_retry':
        this.currentState = 'loading' as S;
        break;
      default:
        throw new Error(`Invalid transition: ${transitionKey}`);
    }
    
    return this.currentState;
  }

  getCurrentState(): S {
    return this.currentState;
  }
}

类型安全的本地化

// 支持的语言
type Locale = 'en' | 'zh' | 'ja' | 'ko';

// 翻译键
type TranslationKey = 
  | 'common.save'
  | 'common.cancel'
  | 'common.delete'
  | 'user.name'
  | 'user.email'
  | 'user.profile'
  | 'error.required'
  | 'error.invalid_email';

// 生成本地化资源键
type LocalizedKey<L extends Locale, K extends TranslationKey> = `${L}.${K}`;

// 翻译资源类型
type TranslationResources = {
  [K in TranslationKey]: string;
};

// 多语言资源
type MultiLanguageResources = {
  [L in Locale]: TranslationResources;
};

// 翻译函数类型
type TranslateFunction = <K extends TranslationKey>(
  key: K,
  params?: Record<string, string | number>
) => string;

// 带参数的翻译键
type ParameterizedTranslationKey = 
  | 'user.welcome'  // "Welcome, {name}!"
  | 'user.age'      // "Age: {age}"
  | 'error.min_length'; // "Minimum length is {min}"

// 提取参数
type ExtractParams<T extends string> = 
  T extends `${string}{${infer Param}}${infer Rest}`
    ? Param | ExtractParams<Rest>
    : never;

type WelcomeParams = ExtractParams<'Welcome, {name}!'>;
// 结果：'name'

type AgeParams = ExtractParams<'Age: {age}'>;
// 结果：'age'

  key: K,
  ...args: ExtractParams<K> extends never

): string;

模板字面量类型最佳实践

1. 避免过度复杂的模板

// 不好的做法：过度复杂
type OverlyComplex<A, B, C, D> = `${A}_${B}_${C}_${D}_suffix`;

// 好的做法：分步构建
type Step1<A, B> = `${A}_${B}`;
type Step2<T, C> = `${T}_${C}`;
type FinalType<T, D> = `${T}_${D}_suffix`;

type WellStructured<A, B, C, D> = FinalType<Step2<Step1<A, B>, C>, D>;

2. 使用辅助类型

// 辅助类型
type Join<T extends string[], S extends string = '_'> = 
  T extends [infer First, ...infer Rest]
    ? First extends string
      ? Rest extends string[]
        ? Rest['length'] extends 0
          ? First
          : `${First}${S}${Join<Rest, S>}`
        : never
      : never
    : '';

type JoinedType = Join<['hello', 'world', 'example']>;
// 结果：'hello_world_example'

3. 提供类型约束

// 约束模板参数
type SafeTemplate<T extends string> = T extends `${string}${'script' | 'iframe'}${string}`
  ? never
  : `prefix_${T}_suffix`;

type SafeType = SafeTemplate<'button'>; // OK
// type UnsafeType = SafeTemplate<'script'>; // Error

4. 性能考虑

// 限制联合类型的大小
type LimitedUnion = 'a' | 'b' | 'c'; // 好：小的联合类型
// type HugeUnion = 'a1' | 'a2' | ... | 'z999'; // 避免：巨大的联合类型

// 使用条件类型优化
type OptimizedTemplate<T extends string> = T extends 'special'
  ? 'special_case'
  : `normal_${T}`;

总结

模板字面量类型是 TypeScript 的强大特性：

  1. 字符串操作：强大的字符串类型操作能力
  2. 类型生成：动态生成复杂的字符串类型
  3. 模式匹配：支持字符串模式匹配和解析
  4. 类型安全：编译时的字符串类型检查

掌握模板字面量类型有助于：

  • 创建类型安全的 API
  • 构建强大的配置系统
  • 实现复杂的字符串操作
  • 提高代码的类型表达能力
