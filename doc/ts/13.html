TypeScript 类型推断 (Type Inference)

什么是类型推断？

类型推断是 TypeScript 编译器自动推导变量、函数返回值、表达式等类型的能力。这使得我们可以在不显式声明类型的情况下，仍然享受到类型安全的好处。

基本类型推断

变量类型推断

// 基本类型推断
let message = "Hello, World!"; // 推断为 string
let count = 42;                // 推断为 number
let isActive = true;           // 推断为 boolean
let items = [1, 2, 3];         // 推断为 number[]
let mixed = [1, "hello", true]; // 推断为 (string | number | boolean)[]

// 对象类型推断
let person = {
    name: "Alice",
    age: 30,
    isEmployed: true
};
// 推断为 { name: string; age: number; isEmployed: boolean; }

// 嵌套对象推断
let config = {
    database: {
        host: "localhost",
        port: 5432,
        credentials: {
            username: "admin",
            password: "secret"
        }
    },
    features: ["auth", "logging"]
};
// TypeScript 会推断出完整的嵌套结构类型

函数返回类型推断

// 函数返回类型推断
function add(a: number, b: number) {
    return a + b; // 推断返回类型为 number
}

function greet(name: string) {
    return `Hello, ${name}!`; // 推断返回类型为 string
}

function getUser(id: number) {
    return {
        id,
        name: "User " + id,
        createdAt: new Date()
    };
    // 推断返回类型为 { id: number; name: string; createdAt: Date; }
}

// 条件返回类型推断
function processValue(value: string | number) {
    if (typeof value === "string") {
        return value.toUpperCase(); // 这个分支推断为 string
    } else {
        return value * 2; // 这个分支推断为 number
    }
}
// 整个函数推断返回类型为 string | number

上下文类型推断

函数参数类型推断

// 上下文类型推断
const numbers = [1, 2, 3, 4, 5];

// 回调函数参数类型从上下文推断
const doubled = numbers.map(n => n * 2);        // n 推断为 number
const filtered = numbers.filter(n => n > 2);   // n 推断为 number
const sum = numbers.reduce((acc, n) => acc + n, 0); // acc 和 n 都推断为 number

// 事件处理器类型推断

fetchUser(1).then(user => {

泛型上下文推断

// 泛型上下文推断
function identity<T>(arg: T): T {
    return arg;
}

let result1 = identity("hello");    // T 推断为 string
let result2 = identity(42);         // T 推断为 number
let result3 = identity([1, 2, 3]);  // T 推断为 number[]

// 复杂泛型推断
function map<T, U>(array: T[], fn: (item: T) => U): U[] {
    return array.map(fn);
}

const strings = ["1", "2", "3"];
const numbers = map(strings, s => parseInt(s)); // T 推断为 string，U 推断为 number

// 约束泛型推断
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
    return obj[key];
}

const person = { name: "Alice", age: 30, city: "New York" };
const name = getProperty(person, "name");  // 推断为 string
const age = getProperty(person, "age");    // 推断为 number

条件类型推断

基本条件类型推断

// 条件类型推断
type IsArray<T> = T extends any[] ? true : false;

type Test1 = IsArray<string[]>;  // true
type Test2 = IsArray<number>;    // false
type Test3 = IsArray<any[]>;     // true

// 提取类型信息
type ArrayElement<T> = T extends (infer U)[] ? U : never;

type StringElement = ArrayElement<string[]>;  // string
type NumberElement = ArrayElement<number[]>;  // number
type NeverType = ArrayElement<string>;        // never

// 函数类型推断
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : any;
type Parameters<T> = T extends (...args: infer P) => any ? P : never;

function example(a: string, b: number): boolean {
    return true;
}

type ExampleReturn = ReturnType<typeof example>;    // boolean
type ExampleParams = Parameters<typeof example>;    // [string, number]

复杂条件类型推断

// 深度推断
type DeepReadonly<T> = {
    readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P];
};

interface User {
    id: number;
    profile: {
        name: string;
        settings: {
            theme: string;
            notifications: boolean;
        };
    };
}

type ReadonlyUser = DeepReadonly<User>;
// 所有嵌套属性都变为只读

// 联合类型分发
type ToArray<T> = T extends any ? T[] : never;

type StringOrNumberArray = ToArray<string | number>;  // string[] | number[]

// 递归类型推断
type Flatten<T> = T extends (infer U)[] 
    ? U extends any[] 
        ? Flatten<U> 
        : U 
    : T;

type NestedArray = number[][][];
type FlatType = Flatten<NestedArray>;  // number

模板字面量类型推断

基本模板推断

// 模板字面量类型推断
type Greeting<T extends string> = `Hello, ${T}!`;

type AliceGreeting = Greeting<"Alice">;  // "Hello, Alice!"
type BobGreeting = Greeting<"Bob">;      // "Hello, Bob!"

// 提取模板字面量类型
type ExtractName<T> = T extends `Hello, ${infer Name}!` ? Name : never;

type ExtractedAlice = ExtractName<"Hello, Alice!">;  // "Alice"
type ExtractedBob = ExtractName<"Hello, Bob!">;      // "Bob"
type ExtractedNone = ExtractName<"Hi there!">;       // never

// 复杂模板推断
type ParseRoute<T extends string> = T extends `/${infer Path}` 
    ? Path extends `${infer Segment}/${infer Rest}`
        ? [Segment, ...ParseRoute<`/${Rest}`>]
        : [Path]
    : [];

type Route1 = ParseRoute<"/users/123/posts">;  // ["users", "123", "posts"]
type Route2 = ParseRoute<"/api/v1/users">;     // ["api", "v1", "users"]

状态管理类型推断

// 状态管理器的类型推断
interface State {
    user: {
        id: number;
        name: string;
        email: string;
    } | null;
    posts: Array<{
        id: number;
        title: string;
        content: string;
        authorId: number;
    }>;
    loading: boolean;
    error: string | null;
}

type ActionType = 
    | "SET_USER"
    | "CLEAR_USER"
    | "ADD_POST"
    | "SET_LOADING"
    | "SET_ERROR";

type Action<T extends ActionType, P = any> = {
    type: T;
    payload: P;
};

// 动作创建器的类型推断
function createAction<T extends ActionType>(type: T) {
    return <P>(payload: P): Action<T, P> => ({ type, payload });
}

const setUser = createAction("SET_USER");
const clearUser = createAction("CLEAR_USER");
const addPost = createAction("ADD_POST");

// 使用时类型会被正确推断
const userAction = setUser({ id: 1, name: "Alice", email: "alice@example.com" });
// userAction 的类型为 Action<"SET_USER", { id: number; name: string; email: string; }>

const clearAction = clearUser(null);
// clearAction 的类型为 Action<"CLEAR_USER", null>

// Reducer 类型推断
type Reducer<S, A> = (state: S, action: A) => S;

function createReducer<S>(initialState: S) {
    return function<A>(
        handlers: {
            [K in A extends { type: infer T } ? T : never]?: (
                state: S,
                action: A extends { type: K } ? A : never
            ) => S;
        }
    ): Reducer<S, A> {
        return (state = initialState, action) => {
            const handler = (handlers as any)[action.type];
            return handler ? handler(state, action) : state;
        };
    };
}

type AppAction = 
    | Action<"SET_USER", State["user"]>
    | Action<"CLEAR_USER", null>
    | Action<"ADD_POST", State["posts"][0]>
    | Action<"SET_LOADING", boolean>
    | Action<"SET_ERROR", string>;

const appReducer = createReducer<State>({
    user: null,
    posts: [],
    loading: false,
    error: null
})({
    SET_USER: (state, action) => ({
        ...state,
        user: action.payload  // payload 类型被正确推断
    }),
    CLEAR_USER: (state) => ({
        ...state,
        user: null
    }),
    ADD_POST: (state, action) => ({
        ...state,
        posts: [...state.posts, action.payload]  // payload 类型被正确推断
    })
});

表单验证类型推断

// 表单验证的类型推断
type ValidationRule<T> = (value: T) => string | null;

interface FormField<T> {
    value: T;
    rules: ValidationRule<T>[];
    error: string | null;
}

type FormSchema<T> = {
    [K in keyof T]: FormField<T[K]>;
};

function createForm<T extends Record<string, any>>(schema: {
    [K in keyof T]: {
        initialValue: T[K];
        rules?: ValidationRule<T[K]>[];
    };
}) {
    const form = {} as FormSchema<T>;
    
    for (const key in schema) {
        form[key] = {
            value: schema[key].initialValue,
            rules: schema[key].rules || [],
            error: null
        };
    }
    
    return {
        form,
        setValue<K extends keyof T>(field: K, value: T[K]) {
            form[field].value = value;
            form[field].error = null;
        },
        validate(): boolean {
            let isValid = true;
            
            for (const key in form) {
                const field = form[key];
                for (const rule of field.rules) {
                    const error = rule(field.value);
                    if (error) {
                        field.error = error;
                        isValid = false;
                        break;
                    }
                }
            }
            
            return isValid;
        },
        getValues(): T {
            const values = {} as T;
            for (const key in form) {
                values[key] = form[key].value;
            }
            return values;
        }
    };
}

// 使用表单，类型会被正确推断
const userForm = createForm({
    name: {
        initialValue: "",
        rules: [
            (value: string) => value.length < 2 ? "Name must be at least 2 characters" : null
        ]
    },
    email: {
        initialValue: "",
        rules: [
            (value: string) => !/\S+@\S+\.\S+/.test(value) ? "Invalid email format" : null
        ]
    },
    age: {
        initialValue: 0,
        rules: [
            (value: number) => value < 18 ? "Must be at least 18 years old" : null
        ]
    }
});

// TypeScript 知道这些方法的参数类型
userForm.setValue("name", "Alice");     // value 参数推断为 string
userForm.setValue("age", 25);           // value 参数推断为 number
// userForm.setValue("age", "25");      // 错误：类型不匹配

const values = userForm.getValues();    // 推断为 { name: string; email: string; age: number; }

            : ApiEndpoints[T] extends { body: infer B }
                ? { body: B }
                : {}

// 类型安全的 API 调用

类型推断的最佳实践

1. 适当使用显式类型注解

// 好的做法：在需要时使用显式类型
let items: string[] = [];  // 空数组需要显式类型
let callback: (x: number) => string;  // 函数类型需要显式声明

// 推断已经足够的情况
let message = "Hello";     // 推断为 string，无需显式类型
let count = items.length;  // 推断为 number，无需显式类型

2. 利用上下文类型推断

// 好的做法：利用上下文推断
interface EventHandler {
    onClick: (event: MouseEvent) => void;
    onKeyPress: (event: KeyboardEvent) => void;
}

const handler: EventHandler = {
    onClick: (e) => {        // e 自动推断为 MouseEvent
        console.log(e.clientX);
    },
    onKeyPress: (e) => {     // e 自动推断为 KeyboardEvent
        console.log(e.key);
    }
};

3. 合理使用泛型推断

// 好的做法：让 TypeScript 推断泛型类型
function createArray<T>(item: T): T[] {
    return [item];
}

const numbers = createArray(1);        // T 推断为 number
const strings = createArray("hello");  // T 推断为 string

// 只在必要时显式指定泛型类型
const mixed = createArray<string | number>("hello");

4. 避免过度依赖推断

// 不好的做法：过度依赖推断导致类型不明确
function processData(data: any) {  // any 类型失去了类型安全
    return data.someProperty;
}

// 好的做法：提供足够的类型信息
function processData<T extends { someProperty: any }>(data: T): T["someProperty"] {
    return data.someProperty;
}

总结

TypeScript 的类型推断系统非常强大：

  1. 基本推断：变量、函数返回值的自动类型推导
  2. 上下文推断：根据使用环境推断类型
  3. 泛型推断：自动推导泛型参数类型
  4. 条件类型推断：基于条件的复杂类型推导
  5. 模板字面量推断：字符串模板的类型推导

合理利用类型推断可以让代码更简洁，同时保持类型安全。
