TypeScript 命名空间 (Namespaces)

什么是命名空间？

命名空间（Namespace）是 TypeScript 早期用于组织代码的方式，它提供了一种将相关的类、接口、函数和变量组织在一起的方法。虽然现在推荐使用 ES6 模块，但命名空间在某些场景下仍然有用。

基本命名空间语法

简单命名空间

// 基本命名空间定义
namespace Geometry {
    export interface Point {
        x: number;
        y: number;
    }

    export class Circle {
        constructor(
            public center: Point,
            public radius: number
        ) {}

        getArea(): number {
            return Math.PI * this.radius * this.radius;
        }

        getCircumference(): number {
            return 2 * Math.PI * this.radius;
        }
    }

    export function distance(p1: Point, p2: Point): number {
        const dx = p1.x - p2.x;
        const dy = p1.y - p2.y;
        return Math.sqrt(dx * dx + dy * dy);
    }

    // 内部函数（不导出）
    function validatePoint(point: Point): boolean {
        return typeof point.x === 'number' && typeof point.y === 'number';
    }
}

// 使用命名空间
const point1: Geometry.Point = { x: 0, y: 0 };
const point2: Geometry.Point = { x: 3, y: 4 };

const circle = new Geometry.Circle(point1, 5);
console.log(circle.getArea()); // 78.54

const dist = Geometry.distance(point1, point2);
console.log(dist); // 5

// 嵌套命名空间
namespace Graphics {
    export namespace Shapes {
        export interface Shape {
            area(): number;
            perimeter(): number;
        }

        export class Rectangle implements Shape {
            constructor(
                public width: number,
                public height: number
            ) {}

            area(): number {
                return this.width * this.height;
            }

            perimeter(): number {
                return 2 * (this.width + this.height);
            }
        }

        export class Triangle implements Shape {
            constructor(
                public base: number,
                public height: number,
                public side1: number,
                public side2: number
            ) {}

            area(): number {
                return 0.5 * this.base * this.height;
            }

            perimeter(): number {
                return this.base + this.side1 + this.side2;
            }
        }
    }

    export namespace Colors {
        export enum RGB {
            Red = '#FF0000',
            Green = '#00FF00',
            Blue = '#0000FF'
        }

        export interface Color {
            name: string;
            hex: string;
            rgb: [number, number, number];
        }

        export function hexToRgb(hex: string): [number, number, number] {
            const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
            return result ? [
                parseInt(result[1], 16),
                parseInt(result[2], 16),
                parseInt(result[3], 16)
            ] : [0, 0, 0];
        }
    }
}

const redColor = Graphics.Colors.RGB.Red;

/// <reference path="geometry-point.ts" />
/// <reference path="geometry-shapes.ts" />

namespace Geometry {
    export namespace Utils {
        export function getShapeInfo(shape: Shape): string {
            return `Area: ${shape.area()}, Perimeter: ${shape.perimeter()}`;
        }

        export function findClosestPoint(target: Point, points: Point[]): Point {
            let closest = points[0];
            let minDistance = distance(target, closest);

            for (const point of points) {
                const dist = distance(target, point);
                if (dist < minDistance) {
                    minDistance = dist;
                    closest = point;
                }
            }

            return closest;
        }
    }
}

命名空间与模块的结合

模块内的命名空间

// api-client.ts
export namespace ApiClient {
    export interface RequestConfig {
        url: string;
        method: 'GET' | 'POST' | 'PUT' | 'DELETE';
        headers?: Record<string, string>;
        data?: any;
    }

    export interface Response<T> {
        data: T;
        status: number;
        statusText: string;
    }

    export class HttpClient {
        private baseUrl: string;

        constructor(baseUrl: string) {
            this.baseUrl = baseUrl;
        }

        async request<T>(config: RequestConfig): Promise<Response<T>> {
            const url = config.url.startsWith('http') 
                ? config.url 
                : `${this.baseUrl}${config.url}`;

            const response = await fetch(url, {
                method: config.method,
                headers: config.headers,
                body: config.data ? JSON.stringify(config.data) : undefined
            });

            const data = await response.json();

            return {
                data,
                status: response.status,
                statusText: response.statusText
            };
        }
    }

    export namespace Interceptors {
        export type RequestInterceptor = (config: RequestConfig) => RequestConfig;
        export type ResponseInterceptor<T> = (response: Response<T>) => Response<T>;

        export class InterceptorManager {
            private requestInterceptors: RequestInterceptor[] = [];
            private responseInterceptors: ResponseInterceptor<any>[] = [];

            addRequestInterceptor(interceptor: RequestInterceptor): void {
                this.requestInterceptors.push(interceptor);
            }

            addResponseInterceptor<T>(interceptor: ResponseInterceptor<T>): void {
                this.responseInterceptors.push(interceptor);
            }

            processRequest(config: RequestConfig): RequestConfig {
                return this.requestInterceptors.reduce(
                    (acc, interceptor) => interceptor(acc),
                    config
                );
            }

            processResponse<T>(response: Response<T>): Response<T> {
                return this.responseInterceptors.reduce(
                    (acc, interceptor) => interceptor(acc),
                    response
                );
            }
        }
    }
}

// 使用模块内的命名空间
import { ApiClient } from './api-client';

const client = new ApiClient.HttpClient('https://api.example.com');
const interceptorManager = new ApiClient.Interceptors.InterceptorManager();

// 添加请求拦截器
interceptorManager.addRequestInterceptor((config) => {
    config.headers = {
        ...config.headers,
        'Authorization': 'Bearer token'
    };
    return config;
});

// game.ts
namespace Game {
    export namespace Entities {
        export interface Entity {
            id: string;
            position: { x: number; y: number };
            health: number;
            update(deltaTime: number): void;
            render(context: CanvasRenderingContext2D): void;
        }

        export class Player implements Entity {
            id: string;
            position: { x: number; y: number };
            health: number;
            speed: number;

            constructor(id: string, x: number, y: number) {
                this.id = id;
                this.position = { x, y };
                this.health = 100;
                this.speed = 5;
            }

            update(deltaTime: number): void {
                // 更新玩家逻辑
            }

            render(context: CanvasRenderingContext2D): void {
                context.fillStyle = 'blue';
                context.fillRect(this.position.x, this.position.y, 32, 32);
            }

            move(direction: { x: number; y: number }): void {
                this.position.x += direction.x * this.speed;
                this.position.y += direction.y * this.speed;
            }
        }

        export class Enemy implements Entity {
            id: string;
            position: { x: number; y: number };
            health: number;
            damage: number;

            constructor(id: string, x: number, y: number) {
                this.id = id;
                this.position = { x, y };
                this.health = 50;
                this.damage = 10;
            }

            update(deltaTime: number): void {
                // 更新敌人 AI
            }

            render(context: CanvasRenderingContext2D): void {
                context.fillStyle = 'red';
                context.fillRect(this.position.x, this.position.y, 24, 24);
            }
        }
    }

    export namespace Systems {
        export class EntityManager {
            private entities: Map<string, Entities.Entity> = new Map();

            addEntity(entity: Entities.Entity): void {
                this.entities.set(entity.id, entity);
            }

            removeEntity(id: string): void {
                this.entities.delete(id);
            }

            getEntity(id: string): Entities.Entity | undefined {
                return this.entities.get(id);
            }

            updateAll(deltaTime: number): void {
                for (const entity of this.entities.values()) {
                    entity.update(deltaTime);
                }
            }

            renderAll(context: CanvasRenderingContext2D): void {
                for (const entity of this.entities.values()) {
                    entity.render(context);
                }
            }
        }

        export class CollisionSystem {
            static checkCollision(
                entity1: Entities.Entity,
                entity2: Entities.Entity
            ): boolean {
                const dx = entity1.position.x - entity2.position.x;
                const dy = entity1.position.y - entity2.position.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                return distance < 32; // 简单的圆形碰撞检测
            }

            static resolveCollisions(entities: Entities.Entity[]): void {
                for (let i = 0; i < entities.length; i++) {
                    for (let j = i + 1; j < entities.length; j++) {
                        if (this.checkCollision(entities[i], entities[j])) {
                            // 处理碰撞
                            console.log(`Collision between ${entities[i].id} and ${entities[j].id}`);
                        }
                    }
                }
            }
        }
    }

    export namespace Utils {
        export function randomBetween(min: number, max: number): number {
            return Math.random() * (max - min) + min;
        }

        export function clamp(value: number, min: number, max: number): number {
            return Math.max(min, Math.min(max, value));
        }

        export function lerp(start: number, end: number, t: number): number {
            return start + (end - start) * t;
        }
    }

    export class GameEngine {
        private entityManager: Systems.EntityManager;
        private canvas: HTMLCanvasElement;
        private context: CanvasRenderingContext2D;
        private lastTime: number = 0;

        constructor(canvasId: string) {
            this.entityManager = new Systems.EntityManager();
            this.canvas = document.getElementById(canvasId) as HTMLCanvasElement;
            this.context = this.canvas.getContext('2d')!;
        }

        start(): void {
            // 创建游戏实体
            const player = new Entities.Player('player', 100, 100);
            const enemy = new Entities.Enemy('enemy1', 200, 200);

            this.entityManager.addEntity(player);
            this.entityManager.addEntity(enemy);

            // 开始游戏循环
            this.gameLoop(0);
        }

        private gameLoop(currentTime: number): void {
            const deltaTime = currentTime - this.lastTime;
            this.lastTime = currentTime;

            // 清空画布
            this.context.clearRect(0, 0, this.canvas.width, this.canvas.height);

            // 更新所有实体
            this.entityManager.updateAll(deltaTime);

            // 渲染所有实体
            this.entityManager.renderAll(this.context);

            // 继续游戏循环
            requestAnimationFrame((time) => this.gameLoop(time));
        }
    }
}

数学库命名空间

// math-library.ts
namespace MathLib {
    export namespace Vector {
        export interface Vector2D {
            x: number;
            y: number;
        }

        export interface Vector3D {
            x: number;
            y: number;
            z: number;
        }

        export function create2D(x: number, y: number): Vector2D {
            return { x, y };
        }

        export function create3D(x: number, y: number, z: number): Vector3D {
            return { x, y, z };
        }

        export function add2D(v1: Vector2D, v2: Vector2D): Vector2D {
            return { x: v1.x + v2.x, y: v1.y + v2.y };
        }

        export function add3D(v1: Vector3D, v2: Vector3D): Vector3D {
            return { x: v1.x + v2.x, y: v1.y + v2.y, z: v1.z + v2.z };
        }

        export function magnitude2D(v: Vector2D): number {
            return Math.sqrt(v.x * v.x + v.y * v.y);
        }

        export function magnitude3D(v: Vector3D): number {
            return Math.sqrt(v.x * v.x + v.y * v.y + v.z * v.z);
        }

        export function normalize2D(v: Vector2D): Vector2D {
            const mag = magnitude2D(v);
            return mag > 0 ? { x: v.x / mag, y: v.y / mag } : { x: 0, y: 0 };
        }

        export function dot2D(v1: Vector2D, v2: Vector2D): number {
            return v1.x * v2.x + v1.y * v2.y;
        }

        export function cross3D(v1: Vector3D, v2: Vector3D): Vector3D {
            return {
                x: v1.y * v2.z - v1.z * v2.y,
                y: v1.z * v2.x - v1.x * v2.z,
                z: v1.x * v2.y - v1.y * v2.x
            };
        }
    }

    export namespace Matrix {
        export type Matrix2x2 = [[number, number], [number, number]];
        export type Matrix3x3 = [[number, number, number], [number, number, number], [number, number, number]];

        export function identity2x2(): Matrix2x2 {
            return [[1, 0], [0, 1]];
        }

        export function identity3x3(): Matrix3x3 {
            return [[1, 0, 0], [0, 1, 0], [0, 0, 1]];
        }

        export function multiply2x2(m1: Matrix2x2, m2: Matrix2x2): Matrix2x2 {
            return [
                [
                    m1[0][0] * m2[0][0] + m1[0][1] * m2[1][0],
                    m1[0][0] * m2[0][1] + m1[0][1] * m2[1][1]
                ],
                [
                    m1[1][0] * m2[0][0] + m1[1][1] * m2[1][0],
                    m1[1][0] * m2[0][1] + m1[1][1] * m2[1][1]
                ]
            ];
        }

        export function rotation2D(angle: number): Matrix2x2 {
            const cos = Math.cos(angle);
            const sin = Math.sin(angle);
            return [[cos, -sin], [sin, cos]];
        }
    }

    export namespace Geometry {
        export function circleArea(radius: number): number {
            return Math.PI * radius * radius;
        }

        export function sphereVolume(radius: number): number {
            return (4 / 3) * Math.PI * radius * radius * radius;
        }

        export function triangleArea(base: number, height: number): number {
            return 0.5 * base * height;
        }

        export function distanceBetweenPoints(
            p1: Vector.Vector2D,
            p2: Vector.Vector2D
        ): number {
            const dx = p1.x - p2.x;
            const dy = p1.y - p2.y;
            return Math.sqrt(dx * dx + dy * dy);
        }
    }

    export namespace Random {
        export function between(min: number, max: number): number {
            return Math.random() * (max - min) + min;
        }

        export function integer(min: number, max: number): number {
            return Math.floor(Math.random() * (max - min + 1)) + min;
        }

        export function boolean(probability: number = 0.5): boolean {
            return Math.random() < probability;
        }

        export function choice<T>(array: T[]): T {
            return array[Math.floor(Math.random() * array.length)];
        }

        export function shuffle<T>(array: T[]): T[] {
            const result = [...array];
            for (let i = result.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [result[i], result[j]] = [result[j], result[i]];
            }
            return result;
        }
    }
}

        export namespace C {

            }

    }

}

总结

TypeScript 命名空间提供了代码组织的另一种方式：

  1. 基本语法：使用 namespace 关键字定义
  2. 嵌套支持：可以创建多层嵌套的命名空间
  3. 别名功能：简化长命名空间的访问
  4. 多文件支持：可以跨文件扩展命名空间
  5. 全局增强：扩展全局类型和第三方库

虽然现代开发推荐使用 ES6 模块，但命名空间在特定场景下仍然有其价值。
