类型操作符 (Type Operators)

TypeScript | 2025.07.31

1. 什么是类型操作符？

类型操作符是 TypeScript 提供的一组用于操作和转换类型的工具。它们允许我们基于现有类型创建新类型，提供了强大的类型编程能力。

2. keyof 操作符

2.1 基本用法

keyof 操作符用于获取对象类型的所有键名，返回一个联合类型。

interface Person {
  name: string;
  age: number;
  email: string;
}

type PersonKeys = keyof Person; // "name" | "age" | "email"

// 使用示例
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];
}

const person: Person = {
  name: "Alice",
  age: 30,
  email: "alice@example.com"
};

const name = getProperty(person, "name"); // string
const age = getProperty(person, "age");   // number

2.2 与数组和元组的使用

// 数组类型
type ArrayKeys = keyof string[]; // number | "length" | "toString" | ...

// 元组类型
type TupleKeys = keyof [string, number]; // "0" | "1" | "length" | "toString" | ...

// 字符串字面量类型
type StringKeys = keyof "hello"; // number | "toString" | "charAt" | ...

2.3 实际应用

// 类型安全的对象属性访问
function pick<T, K extends keyof T>(obj: T, keys: K[]): Pick<T, K> {
  const result = {} as Pick<T, K>;
  keys.forEach(key => {
    result[key] = obj[key];
  });
  return result;
}

const user = {
  id: 1,
  name: "John",
  email: "john@example.com",
  password: "secret"
};

const publicInfo = pick(user, ["id", "name", "email"]);
// 类型: { id: number; name: string; email: string; }

3. typeof 操作符

3.1 基本用法

typeof 操作符用于获取变量或属性的类型。

const message = "Hello, TypeScript!";
type MessageType = typeof message; // string

const numbers = [1, 2, 3, 4, 5];
type NumbersType = typeof numbers; // number[]

const config = {
  apiUrl: "https://api.example.com",
  timeout: 5000,
  retries: 3
} as const;

type ConfigType = typeof config;
// {
//   readonly apiUrl: "https://api.example.com";
//   readonly timeout: 5000;
//   readonly retries: 3;
// }

3.2 与函数的使用

function createUser(name: string, age: number) {
  return {
    id: Math.random(),
    name,
    age,
    createdAt: new Date()
  };
}

type CreateUserFunction = typeof createUser;
// (name: string, age: number) => { id: number; name: string; age: number; createdAt: Date; }

type User = ReturnType<typeof createUser>;
// { id: number; name: string; age: number; createdAt: Date; }

3.3 与类的使用

class ApiClient {
  constructor(public baseUrl: string) {}
  
  async get(endpoint: string) {
    // 实现
  }
  
  async post(endpoint: string, data: any) {
    // 实现
  }
}

type ApiClientClass = typeof ApiClient;
// new (baseUrl: string) => ApiClient

type ApiClientInstance = InstanceType<typeof ApiClient>;
// ApiClient

4. in 操作符

4.1 基本用法

in 操作符用于检查对象是否包含某个属性，常用于类型守卫。

interface Bird {
  fly(): void;
  layEggs(): void;
}

interface Fish {
  swim(): void;
  layEggs(): void;
}

function move(animal: Bird | Fish) {
  if ("fly" in animal) {
    // animal 被推断为 Bird 类型
    animal.fly();
  } else {
    // animal 被推断为 Fish 类型
    animal.swim();
  }
}

4.2 映射类型中的使用

type Optional<T> = {
  [K in keyof T]?: T[K];
};

type RequiredFields<T, K extends keyof T> = T & {
  [P in K]-?: T[P];
};

interface User {
  id?: number;
  name?: string;
  email?: string;
  age?: number;
}

type RequiredUser = RequiredFields<User, "id" | "name">;
// {
//   id: number;
//   name: string;
//   email?: string;
//   age?: number;
// }

4.3 条件类型中的应用

type HasProperty<T, K extends string | number | symbol> = K extends keyof T ? true : false;

type Test1 = HasProperty<{ name: string; age: number }, "name">; // true
type Test2 = HasProperty<{ name: string; age: number }, "email">; // false

// 更复杂的示例
type FilterByType<T, U> = {
  [K in keyof T]: T[K] extends U ? K : never;
}[keyof T];

interface Example {
  name: string;
  age: number;
  isActive: boolean;
  count: number;
}

type StringKeys = FilterByType<Example, string>; // "name"
type NumberKeys = FilterByType<Example, number>; // "age" | "count"

5. 索引访问类型

5.1 基本语法

interface Person {
  name: string;
  age: number;
  address: {
    street: string;
    city: string;
    country: string;
  };
}

type PersonName = Person["name"]; // string
type PersonAge = Person["age"];   // number
type PersonAddress = Person["address"]; // { street: string; city: string; country: string; }
type PersonStreet = Person["address"]["street"]; // string

// 使用联合类型访问多个属性
type PersonNameOrAge = Person["name" | "age"]; // string | number

5.2 与数组和元组的使用

const fruits = ["apple", "banana", "orange"] as const;
type Fruit = typeof fruits[number]; // "apple" | "banana" | "orange"

type FirstFruit = typeof fruits[0]; // "apple"

// 数组元素类型
type ArrayElement<T> = T extends readonly (infer U)[] ? U : never;
type FruitType = ArrayElement<typeof fruits>; // "apple" | "banana" | "orange"

5.3 动态属性访问

function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];
}

function getNestedProperty<T, K1 extends keyof T, K2 extends keyof T[K1]>(
  obj: T,
  key1: K1,
  key2: K2
): T[K1][K2] {
  return obj[key1][key2];
}

const user = {
  profile: {
    personal: {
      name: "John",
      age: 30
    },
    professional: {
      title: "Developer",
      company: "Tech Corp"
    }
  }
};

const name = getNestedProperty(user, "profile", "personal"); // { name: string; age: number; }

6. 条件类型操作符

6.1 extends 关键字

type IsString<T> = T extends string ? true : false;

type Test1 = IsString<string>; // true
type Test2 = IsString<number>; // false

// 更复杂的条件类型
type NonNullable<T> = T extends null | undefined ? never : T;

type Example1 = NonNullable<string | null>; // string
type Example2 = NonNullable<number | undefined>; // number

6.2 infer 关键字

// 提取函数返回类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never;

// 提取数组元素类型
type ElementType<T> = T extends (infer U)[] ? U : never;

// 提取 Promise 的值类型
type PromiseType<T> = T extends Promise<infer U> ? U : never;

type AsyncFunction = () => Promise<string>;
type Result = PromiseType<ReturnType<AsyncFunction>>; // string

7. 实际应用示例

7.1 API 响应类型生成器

interface ApiResponse<T> {
  data: T;
  status: number;
  message: string;
}

interface User {
  id: number;
  name: string;
  email: string;
}

// 自动生成 API 响应类型
type UserResponse = ApiResponse<User>;
type UsersResponse = ApiResponse<User[]>;

// 提取数据类型
type ExtractData<T> = T extends ApiResponse<infer U> ? U : never;

type UserData = ExtractData<UserResponse>; // User
type UsersData = ExtractData<UsersResponse>; // User[]

7.2 表单验证类型系统

interface FormField {
  value: string;
  error?: string;
  touched: boolean;
}

type FormState<T> = {
  [K in keyof T]: FormField;
};

type FormErrors<T> = {
  [K in keyof T]?: string;
};

type FormValues<T> = {
  [K in keyof T]: T[K] extends FormField ? string : never;
};

interface LoginForm {
  username: FormField;
  password: FormField;
}

type LoginErrors = FormErrors<LoginForm>; // { username?: string; password?: string; }
type LoginValues = FormValues<LoginForm>; // { username: string; password: string; }

7.3 状态管理类型系统

interface State {
  user: {
    id: number;
    name: string;
    preferences: {
      theme: "light" | "dark";
      language: string;
    };
  };
  posts: Array<{
    id: number;
    title: string;
    content: string;
  }>;
}

// 深度路径类型
type DeepKeys<T> = T extends object
  ? {
      [K in keyof T]: K extends string
        ? T[K] extends object
          ? `${K}` | `${K}.${DeepKeys<T[K]>}`
          : `${K}`
        : never;
    }[keyof T]
  : never;

type StatePaths = DeepKeys<State>;
// "user" | "posts" | "user.id" | "user.name" | "user.preferences" | 
// "user.preferences.theme" | "user.preferences.language"

// 根据路径获取值类型
type GetByPath<T, P extends string> = P extends `${infer K}.${infer Rest}`
  ? K extends keyof T
    ? GetByPath<T[K], Rest>
    : never
  : P extends keyof T
  ? T[P]
  : never;

type UserName = GetByPath<State, "user.name">; // string
type Theme = GetByPath<State, "user.preferences.theme">; // "light" | "dark"

8. 类型操作符最佳实践

8.1 性能考虑

// ❌ 避免过度复杂的类型操作
type OverlyComplex<T> = {
  [K in keyof T]: T[K] extends object
    ? {
        [P in keyof T[K]]: T[K][P] extends object
          ? {
              [Q in keyof T[K][P]]: T[K][P][Q];
            }
          : T[K][P];
      }
    : T[K];
};

// ✅ 使用辅助类型简化
type DeepReadonly<T> = {
  readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P];
};

8.2 可读性优化

// ❌ 难以理解的类型操作
type Weird<T> = T extends (...args: infer A) => infer R
  ? A extends [infer F, ...infer Rest]
    ? (first: F, ...rest: Rest) => R
    : never
  : never;

// ✅ 清晰的类型定义
type ExtractFirstParameter<T> = T extends (first: infer F, ...rest: any[]) => any ? F : never;
type ExtractRestParameters<T> = T extends (first: any, ...rest: infer R) => any ? R : never;
type ExtractReturnType<T> = T extends (...args: any[]) => infer R ? R : never;

type ReorganizedFunction<T> = T extends (...args: any[]) => any
  ? (first: ExtractFirstParameter<T>, ...rest: ExtractRestParameters<T>) => ExtractReturnType<T>
  : never;

8.3 错误处理

// 类型安全的错误处理
type SafeGet<T, K> = K extends keyof T ? T[K] : undefined;

interface User {
  id: number;
  name: string;
}

type UserId = SafeGet<User, "id">; // number
type UserAge = SafeGet<User, "age">; // undefined

// 运行时类型检查
function hasProperty<T, K extends string>(
  obj: T,
  key: K
): obj is T & Record<K, unknown> {
  return typeof obj === "object" && obj !== null && key in obj;
}

function safeAccess<T, K extends string>(obj: T, key: K): SafeGet<T, K> {
  if (hasProperty(obj, key)) {
    return (obj as any)[key];
  }
  return undefined as SafeGet<T, K>;
}

9. 总结

类型操作符是 TypeScript 类型系统的核心工具：

  1. keyof - 获取对象类型的键名联合类型
  2. typeof - 获取变量或表达式的类型
  3. in - 检查属性存在性，用于类型守卫
  4. 索引访问 - 通过键名获取对应的类型
  5. 条件类型 - 基于条件创建类型分支
  6. infer - 在条件类型中推断类型

这些操作符为我们提供了强大的类型编程能力，使得 TypeScript 能够表达复杂的类型关系和约束。
