Vite 的配置文件是项目的核心，提供了丰富的配置选项来定制构建行为。本章将详细介绍 Vite 配置文件的各个方面。

## 配置文件基础

### 配置文件类型
```javascript
// vite.config.js - 基础配置
export default {
  // 配置选项
}

// vite.config.js - 使用 defineConfig
import { defineConfig } from 'vite'

export default defineConfig({
  // 配置选项，提供类型提示
})

// vite.config.ts - TypeScript 配置
import { defineConfig } from 'vite'

export default defineConfig({
  // 完整的类型支持
})
```

### 条件配置
```javascript
// 根据命令和模式进行条件配置
export default defineConfig(({ command, mode, ssrBuild }) => {
  if (command === 'serve') {
    // 开发环境配置
    return {
      plugins: [vue()],
      server: {
        port: 3000
      }
    }
  } else {
    // 生产环境配置
    return {
      plugins: [vue()],
      build: {
        minify: 'terser'
      }
    }
  }
})
```

### 异步配置
```javascript
// 异步配置函数
export default defineConfig(async ({ command, mode }) => {
  const data = await loadConfigFromFile()
  
  return {
    plugins: [vue()],
    define: {
      __CONFIG__: JSON.stringify(data)
    }
  }
})
```

## 核心配置选项

### 基础选项
```javascript
export default defineConfig({
  // 项目根目录
  root: process.cwd(),
  
  // 开发或生产环境服务的公共基础路径
  base: '/',
  
  // 环境变量模式
  mode: 'development',
  
  // 全局常量替换
  define: {
    __VERSION__: JSON.stringify('1.0.0'),
    __DEV__: JSON.stringify(true),
    'process.env.NODE_ENV': JSON.stringify('development')
  },
  
  // 插件配置
  plugins: [
    vue(),
    react()
  ],
  
  // 公共目录
  publicDir: 'public',
  
  // 缓存目录
  cacheDir: 'node_modules/.vite',
  
  // 模块解析配置
  resolve: {
    alias: {
      '@': path.resolve(__dirname, 'src'),
      '~': path.resolve(__dirname, 'src'),
      'components': path.resolve(__dirname, 'src/components')
    },
    extensions: ['.mjs', '.js', '.ts', '.jsx', '.tsx', '.json', '.vue']
  }
})
```

### 服务器配置
```javascript
export default defineConfig({
  server: {
    // 服务器主机名
    host: 'localhost', // 或 '0.0.0.0' 或 true
    
    // 端口号
    port: 5173,
    
    // 端口被占用时是否直接退出
    strictPort: false,
    
    // 启用 HTTPS
    https: false,
    // 或者提供证书
    https: {
      key: fs.readFileSync('path/to/key.pem'),
      cert: fs.readFileSync('path/to/cert.pem')
    },
    
    // 自动打开浏览器
    open: true,
    // 或指定打开的路径
    open: '/docs/index.html',
    
    // 代理配置
    proxy: {
      '/api': {
        target: 'http://jsonplaceholder.typicode.com',
        changeOrigin: true,
        rewrite: (path) => path.replace(/^\/api/, '')
      },
      '/socket.io': {
        target: 'ws://localhost:5174',
        ws: true
      }
    },
    
    // CORS 配置
    cors: true,
    
    // 预热文件
    warmup: {
      clientFiles: ['./src/components/*.vue', './src/utils/helper.js']
    },
    
    // 文件系统访问限制
    fs: {
      strict: true,
      allow: ['..'],
      deny: ['.env', '.env.*', '*.{pem,crt}']
    }
  }
})
```

### 构建配置
```javascript
export default defineConfig({
  build: {
    // 构建目标
    target: 'modules', // 'es2015' | 'es2017' | 'es2018' | 'es2019' | 'es2020' | 'esnext'
    
    // 输出目录
    outDir: 'dist',
    
    // 静态资源目录
    assetsDir: 'assets',
    
    // 小于此阈值的导入或引用资源将内联为 base64 编码
    assetsInlineLimit: 4096,
    
    // 启用/禁用 CSS 代码拆分
    cssCodeSplit: true,
    
    // CSS 目标
    cssTarget: 'chrome80',
    
    // 构建后是否生成 source map
    sourcemap: false, // true | false | 'inline' | 'hidden'
    
    // 自定义底层的 Rollup 打包配置
    rollupOptions: {
      input: {
        main: resolve(__dirname, 'index.html'),
        nested: resolve(__dirname, 'nested/index.html')
      },
      output: {
        manualChunks: {
          vendor: ['vue', 'vue-router'],
          utils: ['lodash-es']
        }
      }
    },
    
    // 库模式
    lib: {
      entry: resolve(__dirname, 'lib/main.js'),
      name: 'MyLib',
      fileName: 'my-lib'
    },
    
    // 压缩器
    minify: 'esbuild', // 'terser' | 'esbuild' | false
    
    // 传递给 Terser 的更多 minify 选项
    terserOptions: {
      compress: {
        drop_console: true,
        drop_debugger: true
      }
    },
    
    // 构建时清空输出目录
    emptyOutDir: true,
    
    // chunk 大小警告的限制
    chunkSizeWarningLimit: 500,
    
    // 启用/禁用 gzip 压缩大小报告
    reportCompressedSize: true
  }
})
```

## 多环境配置

### 环境特定配置
```javascript
// vite.config.base.js - 基础配置
export const baseConfig = {
  plugins: [vue()],
  resolve: {
    alias: {
      '@': path.resolve(__dirname, 'src')
    }
  }
}

// vite.config.dev.js - 开发环境
import { mergeConfig } from 'vite'
import { baseConfig } from './vite.config.base.js'

export default mergeConfig(baseConfig, {
  mode: 'development',
  server: {
    port: 3000,
    open: true
  },
  define: {
    __DEV__: true
  }
})

// vite.config.prod.js - 生产环境
import { mergeConfig } from 'vite'
import { baseConfig } from './vite.config.base.js'

export default mergeConfig(baseConfig, {
  mode: 'production',
  build: {
    minify: 'terser',
    sourcemap: false
  },
  define: {
    __DEV__: false
  }
})
```

### 动态配置选择
```javascript
// vite.config.js
export default defineConfig(({ mode }) => {
  const config = {
    plugins: [vue()]
  }
  
  if (mode === 'development') {
    config.server = {
      port: 3000,
      proxy: {
        '/api': 'http://localhost:8080'
      }
    }
  }
  
  if (mode === 'production') {
    config.build = {
      minify: 'terser',
      rollupOptions: {
        output: {
          manualChunks: {
            vendor: ['vue', 'vue-router']
          }
        }
      }
    }
  }
  
  if (mode === 'testing') {
    config.test = {
      environment: 'jsdom'
    }
  }
  
  return config
})
```

## 插件配置

### 插件条件应用
```javascript
export default defineConfig({
  plugins: [
    vue(),
    
    // 仅在开发环境使用
    ...(process.env.NODE_ENV === 'development' ? [
      mockPlugin()
    ] : []),
    
    // 仅在生产环境使用
    ...(process.env.NODE_ENV === 'production' ? [
      compressionPlugin(),
      bundleAnalyzerPlugin()
    ] : []),
    
    // 条件插件
    {
      ...somePlugin(),
      apply: 'build' // 'serve' | 'build' | (config, { command }) => boolean
    }
  ]
})
```

## 配置验证和调试

### 配置验证
```javascript
import { defineConfig } from 'vite'
import { z } from 'zod'

const configSchema = z.object({
  server: z.object({
    port: z.number().min(1000).max(65535)
  }).optional(),
  build: z.object({
    outDir: z.string().min(1)
  }).optional()
})

export default defineConfig((env) => {
  const config = {
    server: {
      port: 3000
    },
    build: {
      outDir: 'dist'
    }
  }
  
  // 验证配置
  const result = configSchema.safeParse(config)
  if (!result.success) {
    throw new Error(`Invalid config: ${result.error.message}`)
  }
  
  return config
})
```

### 配置调试
```javascript
export default defineConfig((env) => {
  const config = {
    // ... 配置选项
  }
  
  // 调试输出
  if (process.env.DEBUG_CONFIG) {
    console.log('Vite config:', JSON.stringify(config, null, 2))
  }
  
  return config
})
```

## 总结

Vite 配置文件提供了：
- 灵活的配置方式和条件配置
- 丰富的服务器和构建选项
- 强大的插件系统配置
- 完善的多环境支持

掌握这些配置选项，能够让你充分发挥 Vite 的强大功能，构建出符合项目需求的开发和构建环境。
