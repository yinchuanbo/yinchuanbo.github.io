### 函数式分包策略
```javascript
// vite.config.js
export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        manualChunks: (id) => {
          // node_modules 中的包
          if (id.includes('node_modules')) {
            // Vue 生态
            if (id.includes('vue') || id.includes('pinia')) {
              return 'vue-ecosystem'
            }
            
            // UI 组件库
            if (id.includes('element-plus') || id.includes('ant-design')) {
              return 'ui-library'
            }
            
            // 工具库
            if (id.includes('lodash') || id.includes('dayjs') || id.includes('axios')) {
              return 'utilities'
            }
            
            // 图表和可视化
            if (id.includes('echarts') || id.includes('d3') || id.includes('chart')) {
              return 'visualization'
            }
            
            // 其他第三方库
            return 'vendor'
          }
          
          // 业务代码分包
          if (id.includes('/src/views/')) {
            // 按页面分包
            const match = id.match(/\/views\/([^\/]+)/)
            if (match) {
              return `page-${match[1].toLowerCase()}`
            }
          }
          
          if (id.includes('/src/components/')) {
            // 公共组件
            if (id.includes('/src/components/common/')) {
              return 'common-components'
            }
            // 业务组件
            return 'business-components'
          }
          
          if (id.includes('/src/utils/')) {
            return 'app-utils'
          }
        }
      }
    }
  }
})
```

```javascript
// src/utils/moduleLoader.js
class ModuleLoader {
  constructor() {
    this.cache = new Map()
    this.loading = new Map()
  }
  
  async loadModule(moduleName) {
    // 检查缓存
    if (this.cache.has(moduleName)) {
      return this.cache.get(moduleName)
    }
    
    // 检查是否正在加载
    if (this.loading.has(moduleName)) {
      return this.loading.get(moduleName)
    }
    
    // 开始加载
    const loadPromise = this.importModule(moduleName)
    this.loading.set(moduleName, loadPromise)
    
    try {
      const module = await loadPromise
      this.cache.set(moduleName, module)
      this.loading.delete(moduleName)
      return module
    } catch (error) {
      this.loading.delete(moduleName)
      throw error
    }
  }
  
  async importModule(moduleName) {
    switch (moduleName) {
      case 'charts':
        return import('./charts/index.js')
      
      case 'editor':
        return import('./editor/index.js')
      
      case 'pdf':
        return import('./pdf/index.js')
      
      case 'excel':
        return import('./excel/index.js')
      
      case 'image-processor':
        return import('./imageProcessor/index.js')
      
      default:
        throw new Error(`Unknown module: ${moduleName}`)
    }
  }
  
  // 预加载模块
  async preloadModule(moduleName) {
    try {
      await this.loadModule(moduleName)
      console.log(`Module ${moduleName} preloaded successfully`)
    } catch (error) {
      console.warn(`Failed to preload module ${moduleName}:`, error)
    }
  }
  
  // 批量预加载
  async preloadModules(moduleNames) {
    const promises = moduleNames.map(name => this.preloadModule(name))
    await Promise.allSettled(promises)
  }
}

export const moduleLoader = new ModuleLoader()

// 使用示例
export async function useCharts() {
  const chartsModule = await moduleLoader.loadModule('charts')
  return chartsModule
}

export async function useEditor() {
  const editorModule = await moduleLoader.loadModule('editor')
  return editorModule
}
```

```javascript
// src/utils/libraryLoader.js
class LibraryLoader {
  constructor() {
    this.loadedLibraries = new Set()
    this.loadingPromises = new Map()
  }
  
  async loadLibrary(libraryName) {
    if (this.loadedLibraries.has(libraryName)) {
      return this.getLibraryExports(libraryName)
    }
    
    if (this.loadingPromises.has(libraryName)) {
      return this.loadingPromises.get(libraryName)
    }
    
    const loadPromise = this.importLibrary(libraryName)
    this.loadingPromises.set(libraryName, loadPromise)
    
    try {
      const library = await loadPromise
      this.loadedLibraries.add(libraryName)
      this.loadingPromises.delete(libraryName)
      return library
    } catch (error) {
      this.loadingPromises.delete(libraryName)
      throw error
    }
  }
  
  async importLibrary(libraryName) {
    switch (libraryName) {
      case 'echarts':
        const echarts = await import('echarts')
        return echarts
      
      case 'monaco-editor':
        const monaco = await import('monaco-editor')
        return monaco
      
      case 'pdf-lib':
        const pdfLib = await import('pdf-lib')
        return pdfLib
      
      case 'xlsx':
        const xlsx = await import('xlsx')
        return xlsx
      
      case 'fabric':
        const fabric = await import('fabric')
        return fabric
      
      default:
        throw new Error(`Unknown library: ${libraryName}`)
    }
  }
  
  getLibraryExports(libraryName) {
    // 返回已加载库的导出
    switch (libraryName) {
      case 'echarts':
        return window.echarts
      case 'monaco-editor':
        return window.monaco
      default:
        return null
    }
  }
}

export const libraryLoader = new LibraryLoader()

// 使用示例
export async function createChart(container, options) {

  const chart = echarts.init(container)

  const monaco = await libraryLoader.loadLibrary('monaco-editor')

  return editor
}
```

### 智能预加载
```javascript
// src/utils/preloader.js
class IntelligentPreloader {
  constructor() {
    this.preloadQueue = []
    this.isPreloading = false
    this.userBehavior = {
      visitedRoutes: new Set(),
      clickedComponents: new Set(),
      timeSpent: new Map()
    }
  }
  
  // 基于用户行为的预加载
  analyzeAndPreload() {
    // 分析用户行为模式
    const patterns = this.analyzeUserBehavior()
    
    // 根据模式预加载可能需要的资源
    patterns.forEach(pattern => {
      this.schedulePreload(pattern.resource, pattern.priority)
    })
  }
  
  analyzeUserBehavior() {
    const patterns = []
    
    // 分析访问过的路由
    this.userBehavior.visitedRoutes.forEach(route => {
      const relatedRoutes = this.getRelatedRoutes(route)
      relatedRoutes.forEach(relatedRoute => {
        patterns.push({
          resource: relatedRoute,
          priority: this.calculatePriority(route, relatedRoute),
          type: 'route'
        })
      })
    })
    
    // 分析点击过的组件
    this.userBehavior.clickedComponents.forEach(component => {
      const relatedComponents = this.getRelatedComponents(component)
      relatedComponents.forEach(relatedComponent => {
        patterns.push({
          resource: relatedComponent,
          priority: this.calculateComponentPriority(component, relatedComponent),
          type: 'component'
        })
      })
    })
    
    return patterns.sort((a, b) => b.priority - a.priority)
  }
  
  schedulePreload(resource, priority) {
    this.preloadQueue.push({ resource, priority })
    this.preloadQueue.sort((a, b) => b.priority - a.priority)
    
    if (!this.isPreloading) {
      this.startPreloading()
    }
  }
  
  async startPreloading() {
    this.isPreloading = true
    
    while (this.preloadQueue.length > 0) {
      const { resource } = this.preloadQueue.shift()
      
      try {
        await this.preloadResource(resource)
        console.log(`Preloaded: ${resource}`)
      } catch (error) {
        console.warn(`Failed to preload ${resource}:`, error)
      }
      
      // 避免阻塞主线程
      await this.sleep(100)
    }
    
    this.isPreloading = false
  }
  
  async preloadResource(resource) {
    if (resource.startsWith('/')) {
      // 路由预加载
      return import(/* webpackChunkName: "preload" */ `../views${resource}.vue`)
    } else {
      // 组件预加载
      return import(/* webpackChunkName: "preload" */ `../components/${resource}.vue`)
    }
  }
  
  getRelatedRoutes(route) {
    // 根据路由关系返回相关路由
    const routeMap = {
      '/dashboard': ['/dashboard/analytics', '/dashboard/reports'],
      '/profile': ['/profile/settings', '/profile/security'],
      '/admin': ['/admin/users', '/admin/settings']
    }
    
    return routeMap[route] || []
  }
  
  getRelatedComponents(component) {
    // 根据组件关系返回相关组件
    const componentMap = {
      'DataTable': ['DataTablePagination', 'DataTableFilter'],
      'Chart': ['ChartLegend', 'ChartTooltip'],
      'Form': ['FormValidation', 'FormSubmit']
    }
    
    return componentMap[component] || []
  }
  
  calculatePriority(route, relatedRoute) {
    // 计算路由优先级
    const timeSpent = this.userBehavior.timeSpent.get(route) || 0
    const visitCount = this.userBehavior.visitedRoutes.has(route) ? 1 : 0
    
    return timeSpent * 0.6 + visitCount * 0.4
  }
  
  calculateComponentPriority(component, relatedComponent) {
    // 计算组件优先级
    return Math.random() * 100 // 简化实现
  }
  
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms))
  }
  
  // 记录用户行为
  recordRouteVisit(route) {
    this.userBehavior.visitedRoutes.add(route)
    const startTime = Date.now()
    
    return () => {
      const timeSpent = Date.now() - startTime
      this.userBehavior.timeSpent.set(route, timeSpent)
    }
  }
  
  recordComponentClick(component) {
    this.userBehavior.clickedComponents.add(component)
  }
}

export const preloader = new IntelligentPreloader()

// 在路由守卫中使用
router.beforeEach((to, from, next) => {
  const stopRecording = preloader.recordRouteVisit(to.path)
  
  // 在路由离开时停止记录
  router.afterEach(() => {
    stopRecording()
    preloader.analyzeAndPreload()
  })
  
  next()
})
```

```javascript
// src/utils/networkAwarePreloader.js
class NetworkAwarePreloader {
  constructor() {
    this.connection = navigator.connection || navigator.mozConnection || navigator.webkitConnection
    this.preloadStrategy = this.determineStrategy()
  }
  
  determineStrategy() {
    if (!this.connection) {
      return 'conservative' // 保守策略
    }
    
    const { effectiveType, downlink, rtt } = this.connection
    
    // 根据网络条件确定预加载策略
    if (effectiveType === '4g' && downlink > 2) {
      return 'aggressive' // 激进策略
    } else if (effectiveType === '3g' || downlink > 1) {
      return 'moderate' // 适中策略
    } else {
      return 'conservative' // 保守策略
    }
  }
  
  async preloadBasedOnNetwork(resources) {
    const strategy = this.preloadStrategy
    
    switch (strategy) {
      case 'aggressive':
        // 预加载所有资源
        return this.preloadAll(resources)
      
      case 'moderate':
        // 预加载高优先级资源
        return this.preloadHighPriority(resources)
      
      case 'conservative':
        // 只预加载关键资源
        return this.preloadCritical(resources)
      
      default:
        return Promise.resolve()
    }
  }
  
  async preloadAll(resources) {
    const promises = resources.map(resource => this.preloadResource(resource))
    return Promise.allSettled(promises)
  }
  
  async preloadHighPriority(resources) {
    const highPriority = resources.filter(r => r.priority > 0.7)
    const promises = highPriority.map(resource => this.preloadResource(resource))
    return Promise.allSettled(promises)
  }
  
  async preloadCritical(resources) {
    const critical = resources.filter(r => r.critical)
    const promises = critical.map(resource => this.preloadResource(resource))
    return Promise.allSettled(promises)
  }
  
  async preloadResource(resource) {
    try {
      if (resource.type === 'route') {
        return import(/* webpackChunkName: "route-preload" */ `../views/${resource.path}.vue`)
      } else if (resource.type === 'component') {
        return import(/* webpackChunkName: "component-preload" */ `../components/${resource.name}.vue`)
      } else if (resource.type === 'library') {
        return import(/* webpackChunkName: "library-preload" */ resource.path)
      }
    } catch (error) {
      console.warn(`Failed to preload ${resource.path}:`, error)
    }
  }
  
  // 监听网络变化
  startNetworkMonitoring() {
    if (this.connection) {
      this.connection.addEventListener('change', () => {
        const newStrategy = this.determineStrategy()
        if (newStrategy !== this.preloadStrategy) {
          console.log(`Network strategy changed: ${this.preloadStrategy} -> ${newStrategy}`)
          this.preloadStrategy = newStrategy
        }
      })
    }
  }
}

export const networkAwarePreloader = new NetworkAwarePreloader()
```

```javascript
// src/utils/performanceMonitor.js
class LoadingPerformanceMonitor {
  constructor() {
    this.metrics = {
      chunkLoadTimes: new Map(),
      failedLoads: new Map(),
      cacheHits: new Map()
    }
  }
  
  startChunkLoadTimer(chunkName) {
    const startTime = performance.now()
    
    return {
      end: () => {
        const loadTime = performance.now() - startTime
        this.recordChunkLoadTime(chunkName, loadTime)
        return loadTime
      }
    }
  }
  
  recordChunkLoadTime(chunkName, loadTime) {
    if (!this.metrics.chunkLoadTimes.has(chunkName)) {
      this.metrics.chunkLoadTimes.set(chunkName, [])
    }
    
    this.metrics.chunkLoadTimes.get(chunkName).push({
      loadTime,
      timestamp: Date.now()
    })
    
    // 保持最近100次记录
    const records = this.metrics.chunkLoadTimes.get(chunkName)
    if (records.length > 100) {
      records.shift()
    }
  }
  
  recordFailedLoad(chunkName, error) {
    if (!this.metrics.failedLoads.has(chunkName)) {
      this.metrics.failedLoads.set(chunkName, []
    }
    
    this.metrics.failedLoads.get(chunkName).push({
      error: error.message,
      timestamp: Date.now()
    })
  }
  
  getAverageLoadTime(chunkName) {
    const records = this.metrics.chunkLoadTimes.get(chunkName)
    if (!records || records.length === 0) return 0
    
    const total = records.reduce((sum, record) => sum + record.loadTime, 0)
    return total / records.length
  }
  
  getFailureRate(chunkName) {
    const loadTimes = this.metrics.chunkLoadTimes.get(chunkName) || []
    const failures = this.metrics.failedLoads.get(chunkName) || []
    
    const totalAttempts = loadTimes.length + failures.length
    return totalAttempts > 0 ? failures.length / totalAttempts : 0
  }
  
  generateReport() {
    const report = {
      timestamp: new Date().toISOString(),
      chunks: {}
    }
    
    for (const [chunkName] of this.metrics.chunkLoadTimes) {
      report.chunks[chunkName] = {
        averageLoadTime: this.getAverageLoadTime(chunkName),
        failureRate: this.getFailureRate(chunkName),
        loadCount: this.metrics.chunkLoadTimes.get(chunkName).length,
        failureCount: (this.metrics.failedLoads.get(chunkName) || []).length
      }
    }
    
    return report
  }
  
  // 自动优化建议
  getOptimizationSuggestions() {
    const suggestions = []
    
    for (const [chunkName] of this.metrics.chunkLoadTimes) {
      const avgLoadTime = this.getAverageLoadTime(chunkName)
      const failureRate = this.getFailureRate(chunkName)
      
      if (avgLoadTime > 3000) {
        suggestions.push({
          type: 'slow-loading',
          chunk: chunkName,
          message: `${chunkName} 加载时间过长 (${avgLoadTime.toFixed(2)}ms)，建议进一步分割或优化`
        })
      }
      
      if (failureRate > 0.1) {
        suggestions.push({
          type: 'high-failure-rate',
          chunk: chunkName,
          message: `${chunkName} 失败率过高 (${(failureRate * 100).toFixed(2)}%)，建议添加重试机制`
        })
      }
    }
    
    return suggestions
  }
}

export const performanceMonitor = new LoadingPerformanceMonitor()

// 包装动态导入以监控性能
export function monitoredImport(importFn, chunkName) {
  const timer = performanceMonitor.startChunkLoadTimer(chunkName)
  
  return importFn()
    .then(module => {
      timer.end()
      return module
    })
    .catch(error => {
      performanceMonitor.recordFailedLoad(chunkName, error)
      throw error
    })
}

// 使用示例
const LazyComponent = () => monitoredImport(
  () => import('./HeavyComponent.vue'),
  'HeavyComponent'
)
```
