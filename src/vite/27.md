微前端架构允许将大型前端应用拆分为多个独立的小应用。Vite 为微前端提供了良好的支持，本章将详细介绍如何使用 Vite 构建微前端应用。

## 微前端基础概念

### 微前端架构优势
- **技术栈无关**：不同团队可以使用不同的技术栈
- **独立开发部署**：各个微应用可以独立开发和部署
- **团队自治**：不同团队可以独立维护自己的应用
- **渐进式升级**：可以逐步迁移和升级应用

### 微前端实现方案
```javascript
// 1. 构建时集成 - 通过 npm 包的方式
// 2. 运行时集成 - 通过 iframe 或 JavaScript 动态加载
// 3. 服务端集成 - 通过服务端模板拼接
```

```javascript
// src/micro-apps/index.js
import { defineAsyncComponent } from 'vue'
import { MicroAppManager } from './manager'

const microAppManager = new MicroAppManager()

export async function setupMicroApps(app, router) {
  // 注册微应用
  const microApps = [
    {
      name: 'user-management',
      entry: 'micro-app-1/App',
      route: '/users',
      title: '用户管理'
    },
    {
      name: 'order-management', 
      entry: 'micro-app-2/App',
      route: '/orders',
      title: '订单管理'
    },
    {
      name: 'product-management',
      entry: 'micro-app-3/App', 
      route: '/products',
      title: '商品管理'
    }
  ]
  
  // 动态注册路由和组件
  for (const microApp of microApps) {
    await microAppManager.registerMicroApp(microApp, router)
  }
  
  // 全局错误处理
  microAppManager.setupErrorHandling(app)
}

// src/micro-apps/manager.js
export class MicroAppManager {
  constructor() {
    this.microApps = new Map()
    this.loadingStates = new Map()
    this.errorStates = new Map()
  }
  
  async registerMicroApp(config, router) {
    const { name, entry, route, title } = config
    
    try {
      // 创建异步组件
      const AsyncComponent = defineAsyncComponent({
        loader: () => this.loadMicroApp(entry),
        loadingComponent: () => import('../components/MicroAppLoading.vue'),
        errorComponent: () => import('../components/MicroAppError.vue'),
        delay: 200,
        timeout: 10000
      })
      
      // 注册路由
      router.addRoute({
        path: route,
        name: name,
        component: AsyncComponent,
        meta: {
          title,
          microApp: true,
          microAppName: name
        }
      })
      
      // 注册子路由（如果需要）
      router.addRoute({
        path: `${route}/:pathMatch(.*)*`,
        name: `${name}-child`,
        component: AsyncComponent,
        meta: {
          title,
          microApp: true,
          microAppName: name
        }
      })
      
      this.microApps.set(name, config)
      console.log(`✅ Micro app ${name} registered successfully`)
      
    } catch (error) {
      console.error(`❌ Failed to register micro app ${name}:`, error)
      this.errorStates.set(name, error)
    }
  }
  
  async loadMicroApp(entry) {
    this.loadingStates.set(entry, true)
    
    try {
      // 动态导入微应用
      const module = await import(/* @vite-ignore */ entry)
      this.loadingStates.set(entry, false)
      return module.default || module
    } catch (error) {
      this.loadingStates.set(entry, false)
      this.errorStates.set(entry, error)
      throw error
    }
  }
  
  setupErrorHandling(app) {
    app.config.errorHandler = (error, instance, info) => {
      console.error('Micro app error:', error, info)
      
      // 发送错误报告
      this.reportError(error, instance, info)
    }
  }
  
  reportError(error, instance, info) {
    // 错误上报逻辑
    const errorReport = {
      message: error.message,
      stack: error.stack,
      componentName: instance?.$options.name,
      info,
      timestamp: new Date().toISOString(),
      userAgent: navigator.userAgent,
      url: window.location.href
    }
    
    // 发送到错误监控服务
    fetch('/api/errors', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(errorReport)
    }).catch(err => {
      console.error('Failed to report error:', err)
    })
  }
  
  getMicroAppStatus(name) {
    return {
      registered: this.microApps.has(name),
      loading: this.loadingStates.get(name) || false,
      error: this.errorStates.get(name) || null
    }
  }
}
```

## 应用间通信

### 事件总线通信
```javascript
// src/shared/eventBus.js
class MicroAppEventBus {
  constructor() {
    this.events = new Map()
    this.setupMessageListener()
  }
  
  // 订阅事件
  on(event, callback) {
    if (!this.events.has(event)) {
      this.events.set(event, [])
    }
    
    this.events.get(event).push(callback)
    
    // 返回取消订阅函数
    return () => {
      const callbacks = this.events.get(event)
      const index = callbacks.indexOf(callback)
      if (index > -1) {
        callbacks.splice(index, 1)
      }
    }
  }
  
  // 发送事件
  emit(event, data) {
    const callbacks = this.events.get(event) || []
    callbacks.forEach(callback => {
      try {
        callback(data)
      } catch (error) {
        console.error(`Error in event callback for ${event}:`, error)
      }
    })
    
    // 通过 postMessage 发送给其他微应用
    this.broadcastToMicroApps(event, data)
  }
  
  // 广播给所有微应用
  broadcastToMicroApps(event, data) {
    const message = {
      type: 'MICRO_APP_EVENT',
      event,
      data,
      timestamp: Date.now(),
      source: this.getCurrentAppName()
    }
    
    // 发送给父窗口
    if (window.parent !== window) {
      window.parent.postMessage(message, '*')
    }
    
    // 发送给所有 iframe
    const iframes = document.querySelectorAll('iframe')
    iframes.forEach(iframe => {
      try {
        iframe.contentWindow.postMessage(message, '*')
      } catch (error) {
        // 跨域限制，忽略错误
      }
    })
  }
  
  // 设置消息监听
  setupMessageListener() {
    window.addEventListener('message', (event) => {
      if (event.data.type === 'MICRO_APP_EVENT') {
        const { event: eventName, data, source } = event.data
        
        // 避免处理自己发送的消息
        if (source !== this.getCurrentAppName()) {
          this.handleRemoteEvent(eventName, data)
        }
      }
    })
  }
  
  // 处理远程事件
  handleRemoteEvent(event, data) {
    const callbacks = this.events.get(event) || []
    callbacks.forEach(callback => {
      try {
        callback(data)
      } catch (error) {
        console.error(`Error in remote event callback for ${event}:`, error)
      }
    })
  }
  
  getCurrentAppName() {
    return window.__MICRO_APP_NAME__ || 'unknown'
  }
  
  // 清理所有事件监听
  destroy() {
    this.events.clear()
  }
}

export const eventBus = new MicroAppEventBus()

// 使用示例
// 在主应用中
eventBus.on('user-updated', (userData) => {
  console.log('User updated:', userData)
  // 更新全局状态
})

// 在微应用中
eventBus.emit('user-updated', {
  id: 123,
  name: 'John Doe',
  email: 'john@example.com'
})
```

### 共享状态管理
```javascript
// src/shared/globalStore.js
import { reactive, readonly } from 'vue'
import { eventBus } from './eventBus'

class GlobalStateManager {
  constructor() {
    this.state = reactive({
      user: null,
      theme: 'light',
      language: 'zh-CN',
      permissions: [],
      notifications: []
    })
    
    this.setupEventListeners()
  }
  
  // 获取只读状态
  getState() {
    return readonly(this.state)
  }
  
  // 更新用户信息
  updateUser(userData) {
    this.state.user = userData
    this.broadcastStateChange('user', userData)
  }
  
  // 更新主题
  updateTheme(theme) {
    this.state.theme = theme
    this.broadcastStateChange('theme', theme)
    
    // 应用主题到 DOM
    document.documentElement.setAttribute('data-theme', theme)
  }
  
  // 更新语言
  updateLanguage(language) {
    this.state.language = language
    this.broadcastStateChange('language', language)
  }
  
  // 更新权限
  updatePermissions(permissions) {
    this.state.permissions = permissions
    this.broadcastStateChange('permissions', permissions)
  }
  
  // 添加通知
  addNotification(notification) {
    const id = Date.now().toString()
    const notificationWithId = { ...notification, id }
    
    this.state.notifications.push(notificationWithId)
    this.broadcastStateChange('notifications', this.state.notifications)
    
    return id
  }
  
  // 移除通知
  removeNotification(id) {
    const index = this.state.notifications.findIndex(n => n.id === id)
    if (index > -1) {
      this.state.notifications.splice(index, 1)
      this.broadcastStateChange('notifications', this.state.notifications)
    }
  }
  
  // 广播状态变化
  broadcastStateChange(key, value) {
    eventBus.emit('global-state-change', {
      key,
      value,
      timestamp: Date.now()
    })
  }
  
  // 设置事件监听
  setupEventListeners() {
    // 监听来自其他微应用的状态变化
    eventBus.on('global-state-change', ({ key, value }) => {
      if (this.state.hasOwnProperty(key)) {
        this.state[key] = value
      }
    })
    
    // 监听状态同步请求
    eventBus.on('sync-global-state', () => {
      eventBus.emit('global-state-sync', {
        state: this.state,
        timestamp: Date.now()
      })
    })
    
    // 处理状态同步响应
    eventBus.on('global-state-sync', ({ state }) => {
      Object.keys(state).forEach(key => {
        if (this.state.hasOwnProperty(key)) {
          this.state[key] = state[key]
        }
      })
    })
  }
  
  // 请求状态同步
  requestStateSync() {
    eventBus.emit('sync-global-state')
  }
}

export const globalStateManager = new GlobalStateManager()

// Vue 插件
export function createGlobalStatePlugin() {
  return {
    install(app) {
      app.config.globalProperties.$globalState = globalStateManager.getState()
      app.provide('globalState', globalStateManager.getState())
      app.provide('globalStateManager', globalStateManager)
    }
  }
}

// Composable
export function useGlobalState() {
  return {
    state: globalStateManager.getState(),
    updateUser: globalStateManager.updateUser.bind(globalStateManager),
    updateTheme: globalStateManager.updateTheme.bind(globalStateManager),
    updateLanguage: globalStateManager.updateLanguage.bind(globalStateManager),
    updatePermissions: globalStateManager.updatePermissions.bind(globalStateManager),
    addNotification: globalStateManager.addNotification.bind(globalStateManager),
    removeNotification: globalStateManager.removeNotification.bind(globalStateManager)
  }
}
```

### CSS 作用域隔离
```javascript
// src/utils/styleIsolation.js
class StyleIsolationManager {
  constructor() {
    this.microAppStyles = new Map()
    this.scopeId = 0
  }
  
  // 为微应用创建样式作用域
  createScope(microAppName) {
    const scopeId = `micro-app-${microAppName}-${++this.scopeId}`
    
    // 创建样式容器
    const styleContainer = document.createElement('div')
    styleContainer.id = scopeId
    styleContainer.className = 'micro-app-container'
    
    // 添加作用域样式
    const scopeStyle = document.createElement('style')
    scopeStyle.textContent = `
      #${scopeId} {
        /* 重置样式，避免全局样式影响 */
        all: initial;
        font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif;
        font-size: 14px;
        line-height: 1.5;
        color: #333;
      }
      
      #${scopeId} * {
        box-sizing: border-box;
      }
      
      /* 防止微应用样式泄露 */
      #${scopeId} .micro-app-content {
        isolation: isolate;
      }
    `
    
    document.head.appendChild(scopeStyle)
    
    this.microAppStyles.set(microAppName, {
      scopeId,
      container: styleContainer,
      scopeStyle
    })
    
    return { scopeId, container: styleContainer }
  }
  
  // 移除微应用样式作用域
  removeScope(microAppName) {
    const styleInfo = this.microAppStyles.get(microAppName)
    if (styleInfo) {
      const { container, scopeStyle } = styleInfo
      
      // 移除容器
      if (container.parentNode) {
        container.parentNode.removeChild(container)
      }
      
      // 移除样式
      if (scopeStyle.parentNode) {
        scopeStyle.parentNode.removeChild(scopeStyle)
      }
      
      this.microAppStyles.delete(microAppName)
    }
  }
  
  // 添加微应用特定样式
  addMicroAppStyle(microAppName, css) {
    const styleInfo = this.microAppStyles.get(microAppName)
    if (styleInfo) {
      const { scopeId } = styleInfo
      
      // 为 CSS 添加作用域前缀
      const scopedCSS = this.addScopeToCSS(css, `#${scopeId}`)
      
      const style = document.createElement('style')
      style.setAttribute('data-micro-app', microAppName)
      style.textContent = scopedCSS
      
      document.head.appendChild(style)
      
      return style
    }
  }
  
  // 为 CSS 规则添加作用域
  addScopeToCSS(css, scope) {
    return css.replace(/([^{}]+){/g, (match, selector) => {
      // 简单的作用域添加逻辑
      const trimmedSelector = selector.trim()
      if (trimmedSelector.startsWith('@')) {
        // 媒体查询等不需要作用域
        return match
      }
      
      return `${scope} ${trimmedSelector} {`
    })
  }
  
  // 清理所有微应用样式
  cleanup() {
    this.microAppStyles.forEach((styleInfo, microAppName) => {
      this.removeScope(microAppName)
    })
  }
}

export const styleIsolationManager = new StyleIsolationManager()
```

### Shadow DOM 隔离
```javascript
// src/components/MicroAppContainer.vue
<template>
  <div ref="containerRef" class="micro-app-wrapper">
    <!-- Shadow DOM 将在这里创建 -->
  </div>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue'

const props = defineProps({
  microAppName: {
    type: String,
    required: true
  },
  entry: {
    type: String,
    required: true
  }
})

const containerRef = ref(null)
let shadowRoot = null
let microAppInstance = null

onMounted(async () => {
  await setupShadowDOM()
  await loadMicroApp()
})

onUnmounted(() => {
  cleanup()
})

async function setupShadowDOM() {
  if (containerRef.value && containerRef.value.attachShadow) {
    // 创建 Shadow DOM
    shadowRoot = containerRef.value.attachShadow({ mode: 'open' })
    
    // 创建微应用容器
    const appContainer = document.createElement('div')
    appContainer.id = 'micro-app-root'
    appContainer.style.cssText = `
      width: 100%;
      height: 100%;
      font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif;
    `
    
    shadowRoot.appendChild(appContainer)
    
    // 加载基础样式
    await loadBaseStyles()
  } else {
    console.warn('Shadow DOM not supported, falling back to regular DOM')
    // 降级到普通 DOM
  }
}

async function loadBaseStyles() {
  if (!shadowRoot) return
  
  // 创建样式元素
  const style = document.createElement('style')
  style.textContent = `
    /* 重置样式 */
    * {
      margin: 0;
      padding: 0;
      box-sizing: border-box;
    }
    
    /* 基础样式 */
    #micro-app-root {
      font-size: 14px;
      line-height: 1.5;
      color: #333;
    }
    
    /* 防止样式泄露 */
    :host {
      display: block;
      contain: layout style paint;
    }
  `
  
  shadowRoot.appendChild(style)
}

async function loadMicroApp() {
  try {
    // 动态加载微应用
    const MicroApp = await import(/* @vite-ignore */ props.entry)
    
    // 在 Shadow DOM 中渲染微应用
    const appContainer = shadowRoot?.querySelector('#micro-app-root') || containerRef.value
    
    if (MicroApp.default) {
      // 如果是 Vue 组件
      const { createApp } = await import('vue')
      microAppInstance = createApp(MicroApp.default)
      microAppInstance.mount(appContainer)
    } else if (typeof MicroApp.mount === 'function') {
      // 如果是微前端应用
      await MicroApp.mount({
        container: appContainer,
        props: {
          name: props.microAppName
        }
      })
      microAppInstance = MicroApp
    }
    
  } catch (error) {
    console.error(`Failed to load micro app ${props.microAppName}:`, error)
    showError(error)
  }
}

function showError(error) {
  const container = shadowRoot?.querySelector('#micro-app-root') || containerRef.value
  if (container) {
    container.innerHTML = `
      <div style="
        padding: 20px;
        text-align: center;
        color: #dc3545;
        border: 1px solid #dc3545;
        border-radius: 4px;
        background: #f8d7da;
      ">
        <h3>微应用加载失败</h3>
        <p>${error.message}</p>
        <button onclick="location.reload()" style="
          margin-top: 10px;
          padding: 8px 16px;
          background: #dc3545;
          color: white;
          border: none;
          border-radius: 4px;
          cursor: pointer;
        ">重新加载</button>
      </div>
    `
  }
}

function cleanup() {
  if (microAppInstance) {
    if (typeof microAppInstance.unmount === 'function') {
      microAppInstance.unmount()
    } else if (microAppInstance.unmount) {
      microAppInstance.unmount()
    }
  }
  
  if (shadowRoot) {
    shadowRoot.innerHTML = ''
  }
}
</script>

<style scoped>
.micro-app-wrapper {
  width: 100%;
  height: 100%;
  overflow: hidden;
}
</style>
```

## 总结

Vite 微前端架构支持提供了：

- 完善的错误处理和监控

通过合理的架构设计和配置，可以构建出稳定、可维护的微前端应用系统，实现大型前端项目的模块化管理。
