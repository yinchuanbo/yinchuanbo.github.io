Vite 提供了强大的路径解析和别名系统，让你能够更优雅地管理项目中的模块导入。本章将详细介绍如何配置和使用路径解析功能。

## 路径解析基础

### 默认解析规则
```javascript
// Vite 默认的模块解析顺序
import Component from './Component'
// 1. ./Component.js
// 2. ./Component.ts
// 3. ./Component.jsx
// 4. ./Component.tsx
// 5. ./Component.vue
// 6. ./Component/index.js
// 7. ./Component/index.ts
// 8. ./Component/index.jsx
// 9. ./Component/index.tsx
// 10. ./Component/index.vue
```

## 别名配置

### 基础别名设置
```javascript
// vite.config.js
import { defineConfig } from 'vite'
import path from 'path'

export default defineConfig({
  resolve: {
    alias: {
      // 使用 @ 指向 src 目录
      '@': path.resolve(__dirname, 'src'),
      
      // 使用 ~ 指向 src 目录（另一种常见约定）
      '~': path.resolve(__dirname, 'src'),
      
      // 组件目录别名
      'components': path.resolve(__dirname, 'src/components'),
      'views': path.resolve(__dirname, 'src/views'),
      'utils': path.resolve(__dirname, 'src/utils'),
      'assets': path.resolve(__dirname, 'src/assets'),
      'styles': path.resolve(__dirname, 'src/styles'),
      
      // 第三方库别名
      'vue': 'vue/dist/vue.esm-bundler.js'
    }
  }
})
```

### 高级别名配置
```javascript
export default defineConfig({
  resolve: {
    alias: [
      // 对象形式配置
      {
        find: '@',
        replacement: path.resolve(__dirname, 'src')
      },
      
      // 正则表达式别名
      {
        find: /^~(.*)$/,
        replacement: path.resolve(__dirname, 'node_modules/$1')
      },
      
      // 条件别名
      {
        find: 'lodash',
        replacement: process.env.NODE_ENV === 'production' 
          ? 'lodash-es' 
          : 'lodash'
      },
      
      // 函数形式别名
      {
        find: /^@components\/(.*)$/,
        replacement: (match, p1) => {
          return path.resolve(__dirname, `src/components/${p1}`)
        }
      }
    ]
  }
})
```

## 条件解析

### 环境相关别名
```javascript
export default defineConfig(({ mode }) => {
  const alias = {
    '@': path.resolve(__dirname, 'src'),
    'components': path.resolve(__dirname, 'src/components'),
    'utils': path.resolve(__dirname, 'src/utils')
  }
  
  // 开发环境特定别名
  if (mode === 'development') {
    alias['mock'] = path.resolve(__dirname, 'src/mock')
    alias['test-utils'] = path.resolve(__dirname, 'src/test-utils')
  }
  
  // 生产环境特定别名
  if (mode === 'production') {
    alias['lodash'] = 'lodash-es'
    alias['moment'] = 'dayjs'
  }
  
  return {
    resolve: { alias }
  }
})
```

### 平台相关解析
```javascript
export default defineConfig({
  resolve: {
    alias: {
      // 根据平台选择不同的实现
      'platform': process.platform === 'win32'

        : path.resolve(__dirname, 'src/platform/unix'),
      
      // 根据构建目标选择实现
      'renderer': process.env.BUILD_TARGET === 'electron'
        ? path.resolve(__dirname, 'src/renderer/electron')
        : path.resolve(__dirname, 'src/renderer/web')
    }
  }
})
```

## 模块解析策略

### 自定义解析插件
```javascript
// plugins/customResolve.js
export function customResolvePlugin() {
  return {
    name: 'custom-resolve',
    resolveId(id, importer) {
      // 处理特殊的导入模式
      if (id.startsWith('virtual:')) {
        return id
      }
      
      // 处理动态别名
      if (id.startsWith('dynamic:')) {
        const realPath = id.replace('dynamic:', '')
        return path.resolve(__dirname, 'src/dynamic', realPath)
      }
      
      // 处理版本化导入
      if (id.includes('@v')) {
        const [moduleName, version] = id.split('@v')
        return path.resolve(__dirname, `src/versions/v${version}/${moduleName}`)
      }
      
      return null
    },
    
    load(id) {
      if (id.startsWith('virtual:config')) {
        return `export default ${JSON.stringify(getConfig())}`
      }
      
      return null
    }
  }
}

// vite.config.js
export default defineConfig({
  plugins: [
    customResolvePlugin()
  ]
})
```

import fs from 'fs'
import path from 'path'
import { loadConfigFromFile } from 'vite'

async function validateAliases() {
  const { config } = await loadConfigFromFile('vite.config.js')
  const aliases = config.resolve?.alias || {}
  
  console.log('Validating aliases...')
  
  for (const [alias, target] of Object.entries(aliases)) {
    if (typeof target === 'string') {
      const resolvedPath = path.resolve(target)
      
      if (!fs.existsSync(resolvedPath)) {
        console.error(`❌ Alias "${alias}" points to non-existent path: ${target}`)
      } else {
        console.log(`✅ Alias "${alias}" -> ${target}`)
      }
    }
  }

## 故障排除

### 常见问题解决
```javascript
// 1. 别名不生效
// 检查路径是否正确
console.log(path.resolve(__dirname, 'src')) // 确认路径

// 2. TypeScript 类型错误
// 确保 tsconfig.json 中的 paths 配置与 Vite 别名一致

// 3. 动态导入失败
// 确保动态导入的路径包含足够的静态信息
const component = await import(`@/components/${name}.vue`) // ✅
const component = await import(dynamicPath) // ❌

// 4. CSS 中的别名问题
// 在 CSS 中使用别名需要添加 ~ 前缀
@import '~styles/variables.scss'; // ✅
@import 'styles/variables.scss';  // 可能不工作
```

### 解析冲突处理
```javascript
export default defineConfig({
  resolve: {
    alias: {
      // 使用精确匹配避免冲突
      '@/components': path.resolve(__dirname, 'src/components'),
      '@': path.resolve(__dirname, 'src'), // 放在后面，避免覆盖更具体的别名
      
      // 使用正则表达式进行精确控制
      'components': {
        find: /^components\/(.*)$/,
        replacement: path.resolve(__dirname, 'src/components/$1')
      }
    }
  }
})
```
