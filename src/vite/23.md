Vite 的依赖预构建是其快速启动的关键特性之一。本章将详细介绍依赖预构建的工作原理、配置方法和优化策略。

## 预构建基础概念

### 为什么需要预构建
```javascript
// 问题：CommonJS 模块在浏览器中无法直接使用
const lodash = require('lodash') // ❌ 浏览器不支持

// 问题：深层嵌套的依赖导致大量网络请求
import { debounce } from 'lodash-es'
// 可能触发数百个模块请求

// 解决方案：预构建将依赖转换为 ESM 并打包
import { debounce } from '/node_modules/.vite/deps/lodash-es.js?v=abc123'
// 单个请求获取所有相关代码
```

### 预构建的工作流程
```javascript
// 1. 扫描阶段 - 发现依赖
// Vite 扫描入口文件和动态导入，收集依赖列表

// 2. 分析阶段 - 确定需要预构建的依赖
const needsPreBuild = (dep) => {
  // CommonJS 模块
  if (isCommonJS(dep)) return true
  
  // 有很多内部模块的 ESM 包
  if (hasDeepImports(dep)) return true
  
  // 用户明确指定的依赖
  if (isInIncludeList(dep)) return true
  
  return false
}

// 3. 构建阶段 - 使用 esbuild 预构建
// 将 CommonJS 转换为 ESM
// 将多个模块打包为单个文件
// 生成 sourcemap

// 4. 缓存阶段 - 存储构建结果
// 存储在 node_modules/.vite/deps/
// 生成元数据文件 _metadata.json
```

## 预构建配置

### 基础配置选项
```javascript
// vite.config.js
export default defineConfig({
  optimizeDeps: {
    // 强制预构建的依赖
    include: [
      'lodash-es',
      'axios',
      'dayjs',
      'vue-router'
    ],
    
    // 排除预构建的依赖
    exclude: [
      'your-local-package',
      '@your-org/internal-lib'
    ],
    
    // 预构建入口点
    entries: [
      'src/main.js',
      'src/worker.js',
      'src/pages/**/*.vue'
    ],
    
    // 强制重新预构建
    force: false,
    
    // 自定义 esbuild 选项
    esbuildOptions: {
      target: 'es2020',
      supported: {
        bigint: true
      },
      plugins: [
        // 自定义 esbuild 插件
      ]
    }
  }
})
```

### 动态依赖处理
```javascript
// 处理动态导入的依赖
export default defineConfig({
  optimizeDeps: {
    include: [
      // 动态导入的组件库
      'ant-design-vue',
      'ant-design-vue/es/**',
      
      // 动态加载的语言包
      'dayjs/locale/zh-cn',
      'dayjs/locale/en',
      
      // 条件加载的 polyfill
      'core-js/stable',
      'regenerator-runtime/runtime'
    ],
    
    // 自定义扫描插件
    plugins: [
      {
        name: 'dynamic-deps-scanner',
        setup(build) {
          build.onResolve({ filter: /^virtual:/ }, args => {
            // 处理虚拟模块的依赖
            return {
              path: args.path,
              namespace: 'virtual'
            }
          })
          
          build.onLoad({ filter: /.*/, namespace: 'virtual' }, args => {
            // 返回虚拟模块内容
            return {
              contents: generateVirtualModule(args.path),
              loader: 'js'
            }
          })
        }
      }
    ]
  }
})

function generateVirtualModule(path) {
  // 根据路径生成虚拟模块内容
  if (path === 'virtual:dynamic-imports') {
    return `
      export const loadChart = () => import('echarts')
      export const loadEditor = () => import('monaco-editor')
      export const loadPDF = () => import('pdfjs-dist')
    `
  }
  return ''
}
```

### 缓存优化
```javascript
// vite.config.js
import { createHash } from 'crypto'

export default defineConfig({
  optimizeDeps: {
    // 自定义缓存键生成
    cacheDir: 'node_modules/.vite/deps',
    
    // 基于内容的缓存失效
    force: shouldForceRebuild(),
    
    include: getDependenciesToPreBuild()
  }
})

function shouldForceRebuild() {
  // 检查关键文件是否变化
  const criticalFiles = [
    'package.json',
    'package-lock.json',
    'yarn.lock',
    'pnpm-lock.yaml'
  ]
  
  const currentHash = createHash('md5')
  
  criticalFiles.forEach(file => {
    try {
      const content = fs.readFileSync(file, 'utf-8')
      currentHash.update(content)
    } catch {
      // 文件不存在
    }
  })
  
  const newHash = currentHash.digest('hex')
  const cacheFile = 'node_modules/.vite/deps-hash'
  
  try {
    const oldHash = fs.readFileSync(cacheFile, 'utf-8')
    if (oldHash === newHash) {
      return false
    }
  } catch {
    // 缓存文件不存在
  }
  
  fs.writeFileSync(cacheFile, newHash)
  return true
}

function getDependenciesToPreBuild() {
  const packageJson = JSON.parse(fs.readFileSync('package.json', 'utf-8'))
  const dependencies = Object.keys(packageJson.dependencies || {})
  
  // 过滤需要预构建的依赖
  return dependencies.filter(dep => {
    // 排除纯 ESM 包
    if (isESMOnly(dep)) return false
    
    // 排除 Node.js 专用包
    if (isNodeOnly(dep)) return false
    
    // 包含有复杂导入的包
    if (hasComplexImports(dep)) return true
    
    return true
  })
}
```

### 性能监控
```javascript
// vite.config.js
export default defineConfig({
  optimizeDeps: {
    esbuildOptions: {
      plugins: [
        {
          name: 'prebuild-monitor',
          setup(build) {
            const startTime = Date.now()
            const buildStats = {
              totalFiles: 0,
              totalSize: 0,
              dependencies: {}
            }
            
            build.onStart(() => {
              console.log('🔨 Starting dependency pre-bundling...')
            })
            
            build.onResolve({ filter: /.*/ }, args => {
              buildStats.totalFiles++
              
              if (!buildStats.dependencies[args.path]) {
                buildStats.dependencies[args.path] = {
                  resolveTime: Date.now() - startTime,
                  importer: args.importer
                }
              }
              
              return null
            })
            
            build.onEnd(result => {
              const duration = Date.now() - startTime
              
              console.log(`✅ Pre-bundling completed in ${duration}ms`)
              console.log(`📦 Processed ${buildStats.totalFiles} files`)
              console.log(`📊 Dependencies: ${Object.keys(buildStats.dependencies).length}`)
              
              // 输出详细统计
              if (process.env.VITE_DEBUG_PREBUILD) {
                console.table(buildStats.dependencies)
              }
              
              // 检查性能问题
              if (duration > 10000) {
                console.warn('⚠️  Pre-bundling took longer than 10s, consider optimizing dependencies')
              }
            })
          }
        }
      ]
    }
  }
})
```

## 特殊场景处理

### Monorepo 环境
```javascript
// vite.config.js
import { defineConfig } from 'vite'
import { resolve } from 'path'

export default defineConfig({
  optimizeDeps: {
    // 包含 workspace 中的包
    include: [
      // 外部依赖
      'vue',
      'lodash-es',
      
      // workspace 内的包
      '@my-org/shared-utils',
      '@my-org/ui-components'
    ],
    
    // 排除正在开发的包
    exclude: [
      '@my-org/current-package'
    ],
    
    // 自定义解析
    esbuildOptions: {
      plugins: [
        {
          name: 'monorepo-resolver',
          setup(build) {
            build.onResolve({ filter: /^@my-org\// }, args => {
              const packageName = args.path
              const packagePath = resolve(__dirname, `../packages/${packageName.split('/')[1]}`)
              
              return {
                path: resolve(packagePath, 'src/index.js')
              }
            })
          }
        }
      ]
    }
  }
})
```

```javascript
// scripts/validate-prebuild.js
import fs from 'fs'
import path from 'path'

function validatePreBuild() {
  const depsDir = 'node_modules/.vite/deps'
  const metadataFile = path.join(depsDir, '_metadata.json')
  
  if (!fs.existsSync(metadataFile)) {
    console.error('❌ Pre-build metadata not found')
    return false
  }
  
  const metadata = JSON.parse(fs.readFileSync(metadataFile, 'utf-8'))
  
  console.log('📊 Pre-build validation:')
  console.log(`  Hash: ${metadata.hash}`)
  console.log(`  Dependencies: ${Object.keys(metadata.optimized).length}`)
  
  // 验证依赖文件存在
  const missingDeps = []
  Object.entries(metadata.optimized).forEach(([dep, info]) => {
    const depFile = path.join(depsDir, info.file)
    if (!fs.existsSync(depFile)) {
      missingDeps.push(dep)
    }
  })
  
  if (missingDeps.length > 0) {
    console.error('❌ Missing pre-built dependencies:', missingDeps)
    return false
  }
  
  console.log('✅ Pre-build validation passed')
  return true
}
```

### 性能分析工具
```javascript
// scripts/analyze-prebuild.js
import fs from 'fs'
import path from 'path'

function analyzePreBuild() {
  const depsDir = 'node_modules/.vite/deps'
  const metadataFile = path.join(depsDir, '_metadata.json')
  
  if (!fs.existsSync(metadataFile)) {
    console.error('No pre-build metadata found')
    return
  }
  
  const metadata = JSON.parse(fs.readFileSync(metadataFile, 'utf-8'))
  const analysis = {
    totalDeps: 0,
    totalSize: 0,
    largestDeps: [],
    duplicates: []
  }
  
  Object.entries(metadata.optimized).forEach(([dep, info]) => {
    const depFile = path.join(depsDir, info.file)
    
    if (fs.existsSync(depFile)) {
      const stats = fs.statSync(depFile)
      const size = stats.size
      
      analysis.totalDeps++
      analysis.totalSize += size
      analysis.largestDeps.push({ dep, size })
    }
  })
  
  // 排序找出最大的依赖
  analysis.largestDeps.sort((a, b) => b.size - a.size)
  analysis.largestDeps = analysis.largestDeps.slice(0, 10)
  
  console.log('📊 Pre-build Analysis:')
  console.log(`  Total dependencies: ${analysis.totalDeps}`)
  console.log(`  Total size: ${(analysis.totalSize / 1024 / 1024).toFixed(2)} MB`)
  console.log('\n🔝 Largest dependencies:')
  
  analysis.largestDeps.forEach(({ dep, size }, index) => {
    console.log(`  ${index + 1}. ${dep}: ${(size / 1024).toFixed(2)} KB`)
  })
}

analyzePreBuild()
```

## 总结

Vite 的依赖预构建提供了：
- 自动的 CommonJS 到 ESM 转换
- 智能的依赖发现和分析
- 高效的缓存机制
- 灵活的配置选项
- 强大的性能优化能力

通过合理配置依赖预构建，可以显著提升开发服务器的启动速度和运行性能，为开发者提供更好的开发体验。
