# TypeScript 工具类型 (Utility Types)

## 什么是工具类型？

工具类型是 TypeScript 内置的一组泛型类型，用于对现有类型进行转换和操作。它们提供了强大的类型操作能力，帮助开发者创建更灵活和可重用的类型定义。

## 基础工具类型

### Partial<T>

将类型 T 的所有属性变为可选。

```typescript
interface User {
  id: number;
  name: string;
  email: string;
  age: number;
}

// 所有属性都变为可选
type PartialUser = Partial<User>;
// 等价于：
// {
//   id?: number;
//   name?: string;
//   email?: string;
//   age?: number;
// }

// 实际应用：更新用户信息
function updateUser(id: number, updates: Partial<User>): User {
  const existingUser = getUserById(id);
  return { ...existingUser, ...updates };
}

// 使用示例
updateUser(1, { name: "Alice" }); // 只更新名字
updateUser(2, { email: "bob@example.com", age: 30 }); // 更新邮箱和年龄
```

### Required<T>

将类型 T 的所有属性变为必需。

```typescript
interface Config {
  apiUrl?: string;
  timeout?: number;
  retries?: number;
  debug?: boolean;
}

// 所有属性都变为必需
type RequiredConfig = Required<Config>;
// 等价于：
// {
//   apiUrl: string;
//   timeout: number;
//   retries: number;
//   debug: boolean;
// }

// 实际应用：验证配置完整性
function validateConfig(config: Config): RequiredConfig {
  const defaultConfig: RequiredConfig = {
    apiUrl: 'https://api.example.com',
    timeout: 5000,
    retries: 3,
    debug: false
  };
  
  return { ...defaultConfig, ...config };
}
```

### Readonly<T>

将类型 T 的所有属性变为只读。

```typescript
interface MutableUser {
  id: number;
  name: string;
  email: string;
}

// 所有属性都变为只读
type ReadonlyUser = Readonly<MutableUser>;
// 等价于：
// {
//   readonly id: number;
//   readonly name: string;
//   readonly email: string;
// }

// 实际应用：不可变数据
function createImmutableUser(data: MutableUser): ReadonlyUser {
  return Object.freeze({ ...data });
}

const user = createImmutableUser({ id: 1, name: "Alice", email: "alice@example.com" });
// user.name = "Bob"; // Error: Cannot assign to 'name' because it is a read-only property
```

### Pick<T, K>

从类型 T 中选择指定的属性 K。

```typescript
interface User {
  id: number;
  name: string;
  email: string;
  password: string;
  createdAt: Date;
  updatedAt: Date;
}

// 只选择公开的用户信息
type PublicUser = Pick<User, 'id' | 'name' | 'email'>;
// 等价于：
// {
//   id: number;
//   name: string;
//   email: string;
// }

// 选择用于创建用户的字段
type CreateUserInput = Pick<User, 'name' | 'email' | 'password'>;

// 实际应用：API 响应
function getUserProfile(id: number): PublicUser {
  const fullUser = getUserFromDatabase(id);
  return {
    id: fullUser.id,
    name: fullUser.name,
    email: fullUser.email
  };
}
```

### Omit<T, K>

从类型 T 中排除指定的属性 K。

```typescript
interface User {
  id: number;
  name: string;
  email: string;
  password: string;
  createdAt: Date;
  updatedAt: Date;
}

// 排除敏感信息
type SafeUser = Omit<User, 'password'>;
// 等价于：
// {
//   id: number;
//   name: string;
//   email: string;
//   createdAt: Date;
//   updatedAt: Date;
// }

// 排除自动生成的字段
type CreateUserInput = Omit<User, 'id' | 'createdAt' | 'updatedAt'>;

// 实际应用：表单数据
function createUser(input: CreateUserInput): Promise<User> {
  const now = new Date();
  const newUser: User = {
    id: generateId(),
    ...input,
    createdAt: now,
    updatedAt: now
  };
  
  return saveUserToDatabase(newUser);
}
```

## 高级工具类型

### Record<K, T>

创建一个对象类型，其属性键为 K，属性值为 T。

```typescript
// 基本用法
type StringRecord = Record<string, number>;
// 等价于：{ [key: string]: number }

type UserRoles = Record<'admin' | 'user' | 'guest', string[]>;
// 等价于：
// {
//   admin: string[];
//   user: string[];
//   guest: string[];
// }

// 实际应用：权限管理
const rolePermissions: UserRoles = {
  admin: ['read', 'write', 'delete', 'manage'],
  user: ['read', 'write'],
  guest: ['read']
};

// 状态管理
type LoadingStates = Record<'users' | 'posts' | 'comments', boolean>;

const loadingState: LoadingStates = {
  users: false,
  posts: true,
  comments: false
};

// 动态键类型
type ApiEndpoints = Record<`/api/${string}`, () => Promise<any>>;

const endpoints: ApiEndpoints = {
  '/api/users': () => fetch('/api/users').then(r => r.json()),
  '/api/posts': () => fetch('/api/posts').then(r => r.json()),
  '/api/comments': () => fetch('/api/comments').then(r => r.json())
};
```

### Exclude<T, U>

从联合类型 T 中排除可以赋值给 U 的类型。

```typescript
type AllColors = 'red' | 'green' | 'blue' | 'yellow' | 'purple';
type PrimaryColors = 'red' | 'green' | 'blue';

// 排除主要颜色
type SecondaryColors = Exclude<AllColors, PrimaryColors>;
// 结果：'yellow' | 'purple'

// 实际应用：事件类型过滤
type AllEvents = 'click' | 'scroll' | 'resize' | 'load' | 'error';
type UserEvents = 'click' | 'scroll';
type SystemEvents = Exclude<AllEvents, UserEvents>;
// 结果：'resize' | 'load' | 'error'

// 函数参数类型过滤
type AllTypes = string | number | boolean | null | undefined;
type ValidInputTypes = Exclude<AllTypes, null | undefined>;
// 结果：string | number | boolean

function processInput(value: ValidInputTypes): string {
  return String(value);
}
```

### Extract<T, U>

从联合类型 T 中提取可以赋值给 U 的类型。

```typescript
type AllTypes = string | number | boolean | Date | RegExp;
type PrimitiveTypes = string | number | boolean;

// 提取原始类型
type ExtractedPrimitives = Extract<AllTypes, PrimitiveTypes>;
// 结果：string | number | boolean

// 实际应用：API 响应类型提取
type ApiResponse = 
  | { type: 'success'; data: any }
  | { type: 'error'; message: string }
  | { type: 'loading' }
  | { type: 'idle' };

type ErrorResponse = Extract<ApiResponse, { type: 'error' }>;
// 结果：{ type: 'error'; message: string }

type DataResponse = Extract<ApiResponse, { data: any }>;
// 结果：{ type: 'success'; data: any }
```

### NonNullable<T>

从类型 T 中排除 null 和 undefined。

```typescript
type NullableString = string | null | undefined;
type NonNullString = NonNullable<NullableString>;
// 结果：string

// 实际应用：数组过滤
function filterNonNull<T>(array: (T | null | undefined)[]): NonNullable<T>[] {
  return array.filter((item): item is NonNullable<T> => item != null);
}

const mixedArray = ['hello', null, 'world', undefined, 'typescript'];
const cleanArray = filterNonNull(mixedArray);
// cleanArray 类型为 string[]

// 可选链结果处理
interface User {
  profile?: {
    avatar?: string;
  };
}

function getAvatar(user: User): NonNullable<string> | null {
  const avatar = user.profile?.avatar;
  return avatar ? avatar : null;
}
```

## 函数相关工具类型

### Parameters<T>

获取函数类型 T 的参数类型元组。

```typescript
function createUser(name: string, age: number, email: string): User {
  return { id: generateId(), name, age, email };
}

// 获取函数参数类型
type CreateUserParams = Parameters<typeof createUser>;
// 结果：[string, number, string]

// 实际应用：函数包装器
function loggedCreateUser(...args: CreateUserParams): User {
  console.log('Creating user with params:', args);
  return createUser(...args);
}

// 泛型函数参数提取
type FetchFunction = <T>(url: string, options?: RequestInit) => Promise<T>;
type FetchParams = Parameters<FetchFunction>;
// 结果：[string, RequestInit?]

// 方法参数提取
class ApiClient {
  async get<T>(endpoint: string, params?: Record<string, any>): Promise<T> {
    // 实现...
    return {} as T;
  }
}

type GetMethodParams = Parameters<ApiClient['get']>;
// 结果：[string, Record<string, any>?]
```

### ReturnType<T>

获取函数类型 T 的返回类型。

```typescript
function getUser(id: number): Promise<User> {
  return fetch(`/api/users/${id}`).then(r => r.json());
}

// 获取函数返回类型
type GetUserReturn = ReturnType<typeof getUser>;
// 结果：Promise<User>

// 实际应用：API 响应类型推断
async function fetchData() {
  const users = await getUsers();
  const posts = await getPosts();
  
  return {
    users,
    posts,
    timestamp: Date.now()
  };
}

type FetchDataResult = ReturnType<typeof fetchData>;
// 结果：Promise<{ users: User[]; posts: Post[]; timestamp: number }>

// 条件返回类型
function processValue<T>(value: T): T extends string ? string : number {
  return typeof value === 'string' ? value.toUpperCase() as any : 42 as any;
}

type ProcessStringResult = ReturnType<typeof processValue<string>>;
// 结果：string

type ProcessNumberResult = ReturnType<typeof processValue<number>>;
// 结果：number
```

### ConstructorParameters<T>

获取构造函数类型 T 的参数类型元组。

```typescript
class User {
  constructor(
    public name: string,
    public email: string,
    public age?: number
  ) {}
}

// 获取构造函数参数类型
type UserConstructorParams = ConstructorParameters<typeof User>;
// 结果：[string, string, number?]

// 实际应用：工厂函数
function createUserInstance(...args: UserConstructorParams): User {
  return new User(...args);
}

// 泛型构造函数
interface Constructable<T = {}> {
  new (...args: any[]): T;
}

function createInstance<T extends Constructable>(
  ctor: T,
  ...args: ConstructorParameters<T>
): InstanceType<T> {
  return new ctor(...args);
}

const user = createInstance(User, 'Alice', 'alice@example.com', 25);
```

### InstanceType<T>

获取构造函数类型 T 的实例类型。

```typescript
class ApiClient {
  constructor(private baseUrl: string) {}
  
  async get<T>(endpoint: string): Promise<T> {
    // 实现...
    return {} as T;
  }
}

// 获取实例类型
type ApiClientInstance = InstanceType<typeof ApiClient>;
// 结果：ApiClient

// 实际应用：依赖注入
interface ServiceConstructor<T = {}> {
  new (...args: any[]): T;
}

class ServiceContainer {
  private services = new Map<string, any>();
  
  register<T>(name: string, constructor: ServiceConstructor<T>, ...args: any[]): void {
    this.services.set(name, new constructor(...args));
  }
  
  get<T extends ServiceConstructor>(name: string): InstanceType<T> | undefined {
    return this.services.get(name);
  }
}

const container = new ServiceContainer();
container.register('apiClient', ApiClient, 'https://api.example.com');
const client = container.get<typeof ApiClient>('apiClient');
```

## 字符串操作工具类型

### Uppercase<T>

将字符串字面量类型转换为大写。

```typescript
type UppercaseHello = Uppercase<'hello'>;
// 结果：'HELLO'

type UppercaseUnion = Uppercase<'get' | 'post' | 'put' | 'delete'>;
// 结果：'GET' | 'POST' | 'PUT' | 'DELETE'

// 实际应用：HTTP 方法类型
type HttpMethod = 'get' | 'post' | 'put' | 'delete';
type UppercaseHttpMethod = Uppercase<HttpMethod>;

const methods: Record<UppercaseHttpMethod, string> = {
  GET: 'GET',
  POST: 'POST',
  PUT: 'PUT',
  DELETE: 'DELETE'
};
```

### Lowercase<T>

将字符串字面量类型转换为小写。

```typescript
type LowercaseHello = Lowercase<'HELLO'>;
// 结果：'hello'

// 实际应用：CSS 属性名标准化
type CSSProperty = 'BACKGROUND-COLOR' | 'FONT-SIZE' | 'MARGIN-TOP';
type NormalizedCSSProperty = Lowercase<CSSProperty>;
// 结果：'background-color' | 'font-size' | 'margin-top'
```

### Capitalize<T>

将字符串字面量类型的首字母大写。

```typescript
type CapitalizedHello = Capitalize<'hello'>;
// 结果：'Hello'

// 实际应用：组件名生成
type ComponentName = 'button' | 'input' | 'select';
type ReactComponentName = Capitalize<ComponentName>;
// 结果：'Button' | 'Input' | 'Select'

type ComponentProps<T extends string> = `${Capitalize<T>}Props`;
type ButtonProps = ComponentProps<'button'>;
// 结果：'ButtonProps'
```

### Uncapitalize<T>

将字符串字面量类型的首字母小写。

```typescript
type UncapitalizedHello = Uncapitalize<'Hello'>;
// 结果：'hello'

// 实际应用：方法名生成
type ActionType = 'CREATE_USER' | 'UPDATE_USER' | 'DELETE_USER';
type MethodName<T extends string> = Uncapitalize<
  Lowercase<T> extends `${infer First}_${infer Rest}`
    ? `${First}${Capitalize<Lowercase<Rest>>}`
    : Lowercase<T>
>;

type CreateUserMethod = MethodName<'CREATE_USER'>;
// 结果：'createUser'
```

### 字符串解析工具类型

```typescript
// 解析路径参数
type ExtractParams<T extends string> = 
  T extends `${string}:${infer Param}/${infer Rest}`
    ? Param | ExtractParams<Rest>
    : T extends `${string}:${infer Param}`
    ? Param
    : never;

type RouteParams = ExtractParams<'/users/:id/posts/:postId'>;
// 结果：'id' | 'postId'

// 解析查询参数
type ParseQuery<T extends string> = 
  T extends `${infer Key}=${infer Value}&${infer Rest}`
    ? { [K in Key]: Value } & ParseQuery<Rest>
    : T extends `${infer Key}=${infer Value}`
    ? { [K in Key]: Value }
    : {};

type QueryParams = ParseQuery<'name=john&age=25&active=true'>;
// 结果：{ name: 'john'; age: '25'; active: 'true' }
```

// 基础用户类型
interface User {
  id: number;
  name: string;
  email: string;
  age: number;
  isActive: boolean;
  profile: {
    avatar: string;

type UserFormInput = Omit<User, 'id' | 'profile'> & {

type FormErrors<T> = Partial<Record<keyof T, string>>;
type UserFormErrors = FormErrors<UserFormInput>;

interface FormState<T> {

type UserFormState = FormState<UserFormInput>;

function createUserForm(): UserFormState {

### API 客户端类型

```typescript
// API 响应基础类型
interface ApiResponse<T> {
  success: boolean;
  data: T;
  message?: string;
}

// 资源类型定义
interface User {
  id: number;
  name: string;
  email: string;
}

interface Post {
  id: number;
  title: string;
  content: string;
  authorId: number;
}

// API 端点定义
type ApiEndpoints = {
  'GET /users': ApiResponse<User[]>;
  'GET /users/:id': ApiResponse<User>;
  'POST /users': ApiResponse<User>;
  'PUT /users/:id': ApiResponse<User>;
  'DELETE /users/:id': ApiResponse<void>;
  'GET /posts': ApiResponse<Post[]>;
  'GET /posts/:id': ApiResponse<Post>;
};

// 提取方法和路径
type ExtractMethod<T extends string> = T extends `${infer Method} ${string}` ? Method : never;
type ExtractPath<T extends string> = T extends `${string} ${infer Path}` ? Path : never;

type Methods = ExtractMethod<keyof ApiEndpoints>;
// 结果：'GET' | 'POST' | 'PUT' | 'DELETE'

type Paths = ExtractPath<keyof ApiEndpoints>;
// 结果：'/users' | '/users/:id' | '/posts' | '/posts/:id'

// 类型安全的 API 客户端
class TypedApiClient {
  async request<K extends keyof ApiEndpoints>(
    endpoint: K,
    options?: RequestInit
  ): Promise<ApiEndpoints[K]> {
    const [method, path] = endpoint.split(' ');
    // 实际请求实现
    return {} as ApiEndpoints[K];
  }
}

const client = new TypedApiClient();

// 类型安全的使用
client.request('GET /users').then(response => {
  // response 类型为 ApiResponse<User[]>
  console.log(response.data); // User[]
});

client.request('GET /users/:id').then(response => {
  // response 类型为 ApiResponse<User>
  console.log(response.data); // User
});
```

### 状态管理类型

```typescript
// 状态类型定义
interface AppState {
  user: {
    currentUser: User | null;
    isLoading: boolean;
    error: string | null;
  };
  posts: {
    items: Post[];
    isLoading: boolean;
    error: string | null;
  };
}

// Action 类型生成
type ActionType<T extends string> = Uppercase<T>;
type LoadingAction<T extends string> = `${ActionType<T>}_LOADING`;
type SuccessAction<T extends string> = `${ActionType<T>}_SUCCESS`;
type ErrorAction<T extends string> = `${ActionType<T>}_ERROR`;

type UserActions = 
  | LoadingAction<'fetch_user'>
  | SuccessAction<'fetch_user'>
  | ErrorAction<'fetch_user'>;
// 结果：'FETCH_USER_LOADING' | 'FETCH_USER_SUCCESS' | 'FETCH_USER_ERROR'

// Action 创建器类型
type ActionCreator<T, P = void> = P extends void 
  ? () => { type: T }
  : (payload: P) => { type: T; payload: P };

type UserActionCreators = {
  fetchUserLoading: ActionCreator<'FETCH_USER_LOADING'>;
  fetchUserSuccess: ActionCreator<'FETCH_USER_SUCCESS', User>;
  fetchUserError: ActionCreator<'FETCH_USER_ERROR', string>;
};

// Reducer 类型
type Reducer<S, A> = (state: S, action: A) => S;

type UserAction = 
  | { type: 'FETCH_USER_LOADING' }
  | { type: 'FETCH_USER_SUCCESS'; payload: User }
  | { type: 'FETCH_USER_ERROR'; payload: string };

const userReducer: Reducer<AppState['user'], UserAction> = (state, action) => {
  switch (action.type) {
    case 'FETCH_USER_LOADING':
      return { ...state, isLoading: true, error: null };
    case 'FETCH_USER_SUCCESS':
      return { ...state, isLoading: false, currentUser: action.payload };
    case 'FETCH_USER_ERROR':
      return { ...state, isLoading: false, error: action.payload };
    default:
      return state;
  }
};
```

## 工具类型最佳实践

### 1. 组合使用工具类型

```typescript
// 组合多个工具类型
interface BaseEntity {
  id: number;
  createdAt: Date;
  updatedAt: Date;
}

interface User extends BaseEntity {
  name: string;
  email: string;
  password: string;
}

// 创建用户输入：排除自动生成字段，密码可选
type CreateUserInput = Partial<Pick<User, 'password'>> & 
  Omit<User, keyof BaseEntity | 'password'>;

// 更新用户输入：所有字段可选，排除 id
type UpdateUserInput = Partial<Omit<User, 'id'>>;

// 公开用户信息：排除敏感信息
type PublicUser = Omit<User, 'password'>;
```

### 2. 创建自定义工具类型

```typescript
// 深度可选类型
type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P];
};

// 深度只读类型
type DeepReadonly<T> = {
  readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P];
};

// 可空类型
type Nullable<T> = T | null;

// 可选键类型
type OptionalKeys<T> = {
  [K in keyof T]-?: {} extends Pick<T, K> ? K : never;
}[keyof T];

// 必需键类型
type RequiredKeys<T> = {
  [K in keyof T]-?: {} extends Pick<T, K> ? never : K;
}[keyof T];

// 使用示例
interface Config {
  apiUrl: string;
  timeout?: number;
  retries?: number;
}

type ConfigOptionalKeys = OptionalKeys<Config>;
// 结果：'timeout' | 'retries'

type ConfigRequiredKeys = RequiredKeys<Config>;
// 结果：'apiUrl'
```

### 3. 条件工具类型

```typescript
// 条件类型工具
type IsArray<T> = T extends any[] ? true : false;
type IsFunction<T> = T extends (...args: any[]) => any ? true : false;
type IsPromise<T> = T extends Promise<any> ? true : false;

// 数组元素类型提取
type ArrayElement<T> = T extends (infer U)[] ? U : never;

type StringArray = string[];
type StringElement = ArrayElement<StringArray>;
// 结果：string

// Promise 值类型提取
type PromiseValue<T> = T extends Promise<infer U> ? U : never;

type UserPromise = Promise<User>;
type UserType = PromiseValue<UserPromise>;
// 结果：User

// 函数参数和返回值提取
type FunctionInfo<T> = T extends (...args: infer P) => infer R 
  ? { params: P; return: R }
  : never;

type CreateUserInfo = FunctionInfo<typeof createUser>;
// 结果：{ params: [string, number, string]; return: User }
```

## 总结

工具类型是 TypeScript 类型系统的强大功能：

1. **内置工具类型**：提供常用的类型转换操作
2. **组合能力**：可以组合使用创建复杂类型
3. **类型安全**：在编译时确保类型正确性
4. **代码复用**：减少重复的类型定义

掌握工具类型有助于：
- 创建更灵活的类型定义
- 提高代码的类型安全性
- 减少样板代码
- 构建类型安全的 API

## 练习题

1. 使用 `Pick` 和 `Omit` 创建不同的用户类型变体
2. 实现一个自定义工具类型 `DeepPartial`
3. 创建一个类型安全的事件系统
4. 使用模板字面量类型创建 API 路由类型
