# TypeScript 模块 (Modules)

## 什么是模块？

模块是 TypeScript 中组织代码的基本单位。每个文件都可以是一个模块，模块可以导出（export）和导入（import）功能，实现代码的重用和组织。

## ES6 模块语法

### 基本导出和导入

```typescript
// math.ts - 导出模块
export function add(a: number, b: number): number {
    return a + b;
}

export function subtract(a: number, b: number): number {
    return a - b;
}

export const PI = 3.14159;

export class Calculator {
    multiply(a: number, b: number): number {
        return a * b;
    }
}

// main.ts - 导入模块
import { add, subtract, PI, Calculator } from './math';

console.log(add(5, 3));        // 8
console.log(subtract(10, 4));  // 6
console.log(PI);               // 3.14159

const calc = new Calculator();
console.log(calc.multiply(4, 5)); // 20
```

### 混合导出

```typescript
// api.ts - 混合导出
export interface ApiResponse<T> {
    data: T;
    status: number;
    message: string;
}

export function createResponse<T>(data: T, status: number = 200): ApiResponse<T> {
    return {
        data,
        status,
        message: status === 200 ? "Success" : "Error"
    };
}

// 默认导出
export default class ApiClient {
    private baseUrl: string;

    constructor(baseUrl: string) {
        this.baseUrl = baseUrl;
    }

    async get<T>(endpoint: string): Promise<ApiResponse<T>> {
        // 实现 GET 请求
        const response = await fetch(`${this.baseUrl}${endpoint}`);
        const data = await response.json();
        return createResponse(data, response.status);
    }
}

// main.ts - 导入混合导出
import ApiClient, { ApiResponse, createResponse } from './api';

const client = new ApiClient("https://api.example.com");
const response = createResponse({ id: 1, name: "Test" });
```

## 动态导入

### 基本动态导入

```typescript
// heavy-module.ts
export class HeavyProcessor {
    process(data: any[]): any[] {
        // 复杂的处理逻辑
        console.log("Processing heavy data...");
        return data.map(item => ({ ...item, processed: true }));
    }
}

export function complexCalculation(n: number): number {
    // 复杂计算
    let result = 0;
    for (let i = 0; i < n; i++) {
        result += Math.sqrt(i);
    }
    return result;
}

// main.ts - 动态导入
async function processData(data: any[]) {
    if (data.length > 1000) {
        // 只有在需要时才加载重型模块
        const { HeavyProcessor } = await import('./heavy-module');
        const processor = new HeavyProcessor();
        return processor.process(data);
    } else {
        // 简单处理
        return data.map(item => ({ ...item, processed: true }));
    }
}

// 条件导入
async function loadFeature(featureName: string) {
    switch (featureName) {
        case 'charts':
            const chartModule = await import('./features/charts');
            return new chartModule.ChartRenderer();
        case 'reports':
            const reportModule = await import('./features/reports');
            return new reportModule.ReportGenerator();
        default:
            throw new Error(`Unknown feature: ${featureName}`);
    }
}
```

### 动态导入的类型处理

```typescript
// 获取动态导入模块的类型
type ChartModule = typeof import('./features/charts');
type ReportModule = typeof import('./features/reports');

// 使用条件类型处理动态导入
type FeatureModule<T extends string> = 
    T extends 'charts' ? ChartModule :
    T extends 'reports' ? ReportModule :
    never;

async function getFeatureModule<T extends string>(
    featureName: T
): Promise<FeatureModule<T>> {
    return await import(`./features/${featureName}`) as FeatureModule<T>;
}
```

## 实际应用示例

### 服务层架构

```typescript
// services/base.service.ts
export abstract class BaseService {
    protected baseUrl: string;

    constructor(baseUrl: string) {
        this.baseUrl = baseUrl;
    }

    protected async request<T>(
        endpoint: string, 
        options?: RequestInit
    ): Promise<T> {
        const response = await fetch(`${this.baseUrl}${endpoint}`, options);
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        return response.json();
    }
}

// services/user.service.ts
import { BaseService } from './base.service';

export interface User {
    id: number;
    name: string;
    email: string;
}

export class UserService extends BaseService {
    constructor() {
        super('/api/users');
    }

    async getUsers(): Promise<User[]> {
        return this.request<User[]>('');
    }

    async getUserById(id: number): Promise<User> {
        return this.request<User>(`/${id}`);
    }

    async createUser(user: Omit<User, 'id'>): Promise<User> {
        return this.request<User>('', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(user)
        });
    }
}

// services/index.ts
export { BaseService } from './base.service';
export { UserService, type User } from './user.service';
export { ProductService, type Product } from './product.service';

// main.ts
import { UserService } from './services';

const userService = new UserService();
const users = await userService.getUsers();
```

### 插件系统

```typescript
// plugins/plugin.interface.ts
export interface Plugin {
    name: string;
    version: string;
    init(): Promise<void>;
    destroy(): Promise<void>;
}

// plugins/logger.plugin.ts
import { Plugin } from './plugin.interface';

export class LoggerPlugin implements Plugin {
    name = 'logger';
    version = '1.0.0';

    async init(): Promise<void> {
        console.log('Logger plugin initialized');
    }

    async destroy(): Promise<void> {
        console.log('Logger plugin destroyed');
    }

    log(message: string, level: 'info' | 'warn' | 'error' = 'info'): void {
        console.log(`[${level.toUpperCase()}] ${message}`);
    }
}

// plugins/analytics.plugin.ts
import { Plugin } from './plugin.interface';

export class AnalyticsPlugin implements Plugin {
    name = 'analytics';
    version = '1.0.0';

    async init(): Promise<void> {
        console.log('Analytics plugin initialized');
    }

    async destroy(): Promise<void> {
        console.log('Analytics plugin destroyed');
    }

    track(event: string, data?: any): void {
        console.log(`Tracking event: ${event}`, data);
    }
}

// core/plugin-manager.ts
import { Plugin } from '../plugins/plugin.interface';

export class PluginManager {
    private plugins = new Map<string, Plugin>();

    async loadPlugin(pluginName: string): Promise<void> {
        try {
            // 动态导入插件
            const module = await import(`../plugins/${pluginName}.plugin`);
            const PluginClass = Object.values(module)[0] as new () => Plugin;
            const plugin = new PluginClass();
            
            await plugin.init();
            this.plugins.set(plugin.name, plugin);
            
            console.log(`Plugin ${plugin.name} loaded successfully`);
        } catch (error) {
            console.error(`Failed to load plugin ${pluginName}:`, error);
        }
    }

    getPlugin<T extends Plugin>(name: string): T | undefined {
        return this.plugins.get(name) as T;
    }

    async unloadPlugin(name: string): Promise<void> {
        const plugin = this.plugins.get(name);
        if (plugin) {
            await plugin.destroy();
            this.plugins.delete(name);
            console.log(`Plugin ${name} unloaded`);
        }
    }
}

// main.ts
import { PluginManager } from './core/plugin-manager';
import { LoggerPlugin } from './plugins/logger.plugin';
import { AnalyticsPlugin } from './plugins/analytics.plugin';

const pluginManager = new PluginManager();

// 加载插件
await pluginManager.loadPlugin('logger');
await pluginManager.loadPlugin('analytics');

// 使用插件
const logger = pluginManager.getPlugin<LoggerPlugin>('logger');
const analytics = pluginManager.getPlugin<AnalyticsPlugin>('analytics');

logger?.log('Application started');
analytics?.track('app_start');
```

### 配置管理模块

```typescript
// config/environment.ts
export enum Environment {
    Development = 'development',
    Testing = 'testing',
    Staging = 'staging',
    Production = 'production'
}

export interface DatabaseConfig {
    host: string;
    port: number;
    database: string;
    username: string;
    password: string;
}

export interface ApiConfig {
    baseUrl: string;
    timeout: number;
    retries: number;
}

export interface AppConfig {
    environment: Environment;
    port: number;
    database: DatabaseConfig;
    api: ApiConfig;
    features: {
        enableLogging: boolean;
        enableAnalytics: boolean;
        enableCache: boolean;
    };
}

// config/config.factory.ts
import { AppConfig, Environment } from './environment';

export class ConfigFactory {
    static create(env: Environment): AppConfig {
        const baseConfig: AppConfig = {
            environment: env,
            port: 3000,
            database: {
                host: 'localhost',
                port: 5432,
                database: 'myapp',
                username: 'user',
                password: 'password'
            },
            api: {
                baseUrl: 'http://localhost:3000',
                timeout: 5000,
                retries: 3
            },
            features: {
                enableLogging: true,
                enableAnalytics: false,
                enableCache: false
            }
        };

        switch (env) {
            case Environment.Development:
                return {
                    ...baseConfig,
                    features: {
                        ...baseConfig.features,
                        enableLogging: true,
                        enableAnalytics: false
                    }
                };

            case Environment.Production:
                return {
                    ...baseConfig,
                    port: 8080,
                    api: {
                        ...baseConfig.api,
                        baseUrl: 'https://api.myapp.com'
                    },
                    features: {
                        enableLogging: true,
                        enableAnalytics: true,
                        enableCache: true
                    }
                };

            default:
                return baseConfig;
        }
    }
}

// config/index.ts
export { Environment, type AppConfig } from './environment';
export { ConfigFactory } from './config.factory';

// 创建配置实例
const currentEnv = (process.env.NODE_ENV as Environment) || Environment.Development;
export const config = ConfigFactory.create(currentEnv);
```

### 2. 避免循环依赖

```typescript
// 不好的做法 - 循环依赖
// user.ts
import { Order } from './order';
export class User {
    orders: Order[] = [];
}

// order.ts
import { User } from './user';
export class Order {
    user: User;
}

// 好的做法 - 使用接口打破循环依赖
// types.ts
export interface IUser {
    id: number;
    name: string;
}

export interface IOrder {
    id: number;
    userId: number;
}

// user.ts
import { IUser, IOrder } from './types';
export class User implements IUser {
    id: number;
    name: string;
    orders: IOrder[] = [];
}

// order.ts
import { IOrder, IUser } from './types';
export class Order implements IOrder {
    id: number;
    userId: number;
}
```

## 总结

TypeScript 模块系统提供了强大的代码组织能力：

1. **ES6 模块语法**：标准的导入导出语法
2. **动态导入**：按需加载模块
3. **模块解析**：灵活的路径解析配置
4. **类型声明**：为第三方模块提供类型支持
5. **模块增强**：扩展现有模块的功能

合理使用模块可以让代码更加模块化、可维护和可重用。
