# TypeScript 条件类型 (Conditional Types)

## 什么是条件类型？

条件类型是 TypeScript 中的高级类型特性，它允许根据类型关系来选择类型。条件类型的语法类似于 JavaScript 中的三元运算符，但作用于类型层面。

## 基本语法

```typescript
// 基本条件类型语法
T extends U ? X : Y

// 如果类型 T 可以赋值给类型 U，则结果类型为 X，否则为 Y
```

## 简单条件类型

### 基础示例

```typescript
// 简单的条件类型
type IsString<T> = T extends string ? true : false;

type Test1 = IsString<string>;  // true
type Test2 = IsString<number>;  // false
type Test3 = IsString<'hello'>; // true (字面量类型也是 string)

// 实际应用：类型检查
type TypeName<T> = 
  T extends string ? 'string' :
  T extends number ? 'number' :
  T extends boolean ? 'boolean' :
  T extends undefined ? 'undefined' :
  T extends Function ? 'function' :
  'object';

type T1 = TypeName<string>;    // 'string'
type T2 = TypeName<42>;        // 'number'
type T3 = TypeName<true>;      // 'boolean'
type T4 = TypeName<() => void>; // 'function'
type T5 = TypeName<{}>;        // 'object'
```

### 联合类型的条件判断

```typescript
// 联合类型会分发条件类型
type ToArray<T> = T extends any ? T[] : never;

type StringOrNumberArray = ToArray<string | number>;
// 结果：string[] | number[]
// 等价于：ToArray<string> | ToArray<number>

// 阻止分发行为
type ToArrayNonDistributive<T> = [T] extends [any] ? T[] : never;

type SingleArray = ToArrayNonDistributive<string | number>;
// 结果：(string | number)[]
```

## infer 关键字

### 基本用法

```typescript
// infer 用于在条件类型中推断类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : any;

function getString(): string { return 'hello'; }
function getNumber(): number { return 42; }

type StringReturn = ReturnType<typeof getString>; // string
type NumberReturn = ReturnType<typeof getNumber>; // number

// 推断数组元素类型
type ArrayElement<T> = T extends (infer U)[] ? U : never;

type StringElement = ArrayElement<string[]>; // string
type NumberElement = ArrayElement<number[]>; // number
type MixedElement = ArrayElement<(string | number)[]>; // string | number
```

### 复杂推断

```typescript
// 推断函数参数类型
type Parameters<T> = T extends (...args: infer P) => any ? P : never;

function createUser(name: string, age: number, email: string): User {
  return { id: 1, name, age, email };
}

type CreateUserParams = Parameters<typeof createUser>;
// 结果：[string, number, string]

// 推断 Promise 的值类型
type PromiseValue<T> = T extends Promise<infer U> ? U : T;

type UserPromise = Promise<User>;
type UserType = PromiseValue<UserPromise>; // User
type StringType = PromiseValue<string>;    // string (不是 Promise)

// 推断对象属性类型
type PropertyType<T, K extends keyof T> = T extends { [P in K]: infer U } ? U : never;

interface User {
  id: number;
  name: string;
  email: string;
}

type UserIdType = PropertyType<User, 'id'>;     // number
type UserNameType = PropertyType<User, 'name'>; // string
```

### 多重推断

```typescript
// 推断函数的参数和返回值
type FunctionInfo<T> = T extends (...args: infer P) => infer R 
  ? { parameters: P; returnType: R }
  : never;

function processData(data: string[], options: { sort: boolean }): number {
  return data.length;
}

type ProcessDataInfo = FunctionInfo<typeof processData>;
// 结果：{
//   parameters: [string[], { sort: boolean }];
//   returnType: number;
// }

// 推断嵌套结构
type DeepArrayElement<T> = T extends (infer U)[][]
  ? U
  : T extends (infer U)[]
  ? U
  : never;

type NestedStringElement = DeepArrayElement<string[][]>; // string
type SimpleStringElement = DeepArrayElement<string[]>;   // string
```

## 高级条件类型

### 递归条件类型

```typescript
// 递归展开嵌套数组
type Flatten<T> = T extends (infer U)[]
  ? U extends any[]
    ? Flatten<U>
    : U
  : T;

type NestedArray = number[][][];
type FlattenedType = Flatten<NestedArray>; // number

// 深度可选类型
type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object
    ? DeepPartial<T[P]>
    : T[P];
};

interface NestedConfig {
  database: {
    host: string;
    port: number;
    credentials: {
      username: string;
      password: string;
    };
  };
  cache: {
    enabled: boolean;
    ttl: number;
  };
}

type PartialConfig = DeepPartial<NestedConfig>;
// 所有嵌套属性都变为可选
```

### 字符串操作条件类型

```typescript
// 字符串模式匹配
type StartsWith<T extends string, U extends string> = 
  T extends `${U}${string}` ? true : false;

type StartsWithHello = StartsWith<'hello world', 'hello'>; // true
type StartsWithBye = StartsWith<'hello world', 'bye'>;     // false

// 提取字符串前缀
type ExtractPrefix<T extends string, U extends string> = 
  T extends `${infer Prefix}${U}${string}` ? Prefix : never;

type Prefix = ExtractPrefix<'getUserById', 'By'>; // 'getUser'

// 路径参数提取
type ExtractParams<T extends string> = 
  T extends `${string}:${infer Param}/${infer Rest}`
    ? Param | ExtractParams<Rest>
    : T extends `${string}:${infer Param}`
    ? Param
    : never;

type RouteParams = ExtractParams<'/users/:id/posts/:postId'>;
// 结果：'id' | 'postId'

// 驼峰命名转换
type CamelCase<S extends string> = S extends `${infer P1}_${infer P2}${infer P3}`
  ? `${P1}${Uppercase<P2>}${CamelCase<P3>}`
  : S;

type CamelCased = CamelCase<'hello_world_example'>; // 'helloWorldExample'
```

### 对象操作条件类型

```typescript
// 提取可选属性
type OptionalKeys<T> = {
  [K in keyof T]-?: {} extends Pick<T, K> ? K : never;
}[keyof T];

// 提取必需属性
type RequiredKeys<T> = {
  [K in keyof T]-?: {} extends Pick<T, K> ? never : K;
}[keyof T];

interface MixedInterface {
  required1: string;
  required2: number;
  optional1?: boolean;
  optional2?: Date;
}

type OptionalProps = OptionalKeys<MixedInterface>; // 'optional1' | 'optional2'
type RequiredProps = RequiredKeys<MixedInterface>; // 'required1' | 'required2'

// 分离可选和必需属性
type SplitOptionalRequired<T> = {
  optional: Pick<T, OptionalKeys<T>>;
  required: Pick<T, RequiredKeys<T>>;
};

type SplitMixed = SplitOptionalRequired<MixedInterface>;
// 结果：{
//   optional: { optional1?: boolean; optional2?: Date };
//   required: { required1: string; required2: number };
// }
```

## 实际应用示例

### API 响应类型处理

```typescript
// API 响应基础类型
interface ApiResponse<T> {
  success: boolean;
  data: T;
  message?: string;
}

interface ApiError {
  success: false;
  error: string;
  code: number;
}

// 条件类型处理 API 响应
type ApiResult<T> = T extends { success: true }
  ? T['data']
  : T extends { success: false }
  ? never
  : T;

// 提取成功响应的数据类型
type ExtractData<T> = T extends ApiResponse<infer U> ? U : never;

type UserResponse = ApiResponse<User>;
type UserData = ExtractData<UserResponse>; // User

// 安全的 API 调用类型
type SafeApiCall<T> = T extends ApiResponse<infer U>
  ? Promise<U>
  : T extends ApiError
  ? Promise<never>
  : Promise<T>;

async function fetchUser(id: number): SafeApiCall<ApiResponse<User>> {
  const response = await fetch(`/api/users/${id}`);
  const data = await response.json();
  
  if (data.success) {
    return data.data; // 类型为 User
  }
  
  throw new Error(data.error);
}
```

### 表单验证类型

```typescript
// 验证规则类型
interface ValidationRule<T> {
  required?: boolean;
  minLength?: T extends string ? number : never;
  maxLength?: T extends string ? number : never;
  min?: T extends number ? number : never;
  max?: T extends number ? number : never;
  pattern?: T extends string ? RegExp : never;
  custom?: (value: T) => boolean | string;
}

// 根据字段类型生成验证规则
type FieldValidation<T> = {
  [K in keyof T]: ValidationRule<T[K]>;
};

interface UserForm {
  name: string;
  email: string;
  age: number;
  website?: string;
}

type UserValidation = FieldValidation<UserForm>;
// 结果：{
//   name: ValidationRule<string>;
//   email: ValidationRule<string>;
//   age: ValidationRule<number>;
//   website: ValidationRule<string | undefined>;
// }

// 验证错误类型
type ValidationErrors<T> = {
  [K in keyof T]?: string[];
};

// 表单状态类型
type FormState<T> = {
  values: T;
  errors: ValidationErrors<T>;
  touched: { [K in keyof T]?: boolean };
  isValid: boolean;
};

// 验证函数类型
type ValidateFunction<T> = (
  values: T,
  rules: FieldValidation<T>
) => ValidationErrors<T>;
```

### 状态机类型

```typescript
// 状态机状态定义
interface StateDefinition {
  [stateName: string]: {
    on?: {
      [eventName: string]: string;
    };
    entry?: string[];
    exit?: string[];
  };
}

// 提取状态名
type StateNames<T extends StateDefinition> = keyof T;

// 提取事件名
type EventNames<T extends StateDefinition> = {
  [K in keyof T]: T[K] extends { on: infer Events }
    ? Events extends Record<string, any>
      ? keyof Events
      : never
    : never;
}[keyof T];

// 状态机定义示例
const userStateMachine = {
  idle: {
    on: {
      FETCH: 'loading',
      CREATE: 'creating'
    }
  },
  loading: {
    on: {
      SUCCESS: 'success',
      ERROR: 'error'
    },
    entry: ['showSpinner']
  },
  creating: {
    on: {
      SUCCESS: 'success',
      ERROR: 'error'
    }
  },
  success: {
    on: {
      RESET: 'idle'
    },
    exit: ['hideSpinner']
  },
  error: {
    on: {
      RETRY: 'loading',
      RESET: 'idle'
    }
  }
} as const;

type UserStates = StateNames<typeof userStateMachine>;
// 结果：'idle' | 'loading' | 'creating' | 'success' | 'error'

type UserEvents = EventNames<typeof userStateMachine>;
// 结果：'FETCH' | 'CREATE' | 'SUCCESS' | 'ERROR' | 'RESET' | 'RETRY'

// 类型安全的状态机
class StateMachine<T extends StateDefinition> {
  constructor(
    private definition: T,
    private currentState: StateNames<T>
  ) {}

  transition(event: EventNames<T>): StateNames<T> | null {
    const state = this.definition[this.currentState];
    if (state.on && event in state.on) {
      this.currentState = state.on[event as string] as StateNames<T>;
      return this.currentState;
    }
    return null;
  }

  getCurrentState(): StateNames<T> {
    return this.currentState;
  }
}

const machine = new StateMachine(userStateMachine, 'idle');
machine.transition('FETCH'); // 返回 'loading'
```

### 路由类型系统

```typescript
// 路由定义类型
interface RouteDefinition {
  path: string;
  component: string;
  children?: RouteDefinition[];
}

// 提取路径参数
type ExtractRouteParams<T extends string> = 
  T extends `${string}:${infer Param}/${infer Rest}`
    ? { [K in Param]: string } & ExtractRouteParams<Rest>
    : T extends `${string}:${infer Param}`
    ? { [K in Param]: string }
    : {};

// 路由配置
const routes = [
  {
    path: '/users/:id',
    component: 'UserDetail'
  },
  {
    path: '/posts/:postId/comments/:commentId',
    component: 'CommentDetail'
  },
  {
    path: '/dashboard',
    component: 'Dashboard'
  }
] as const;

// 提取所有路由路径
type RoutePaths = typeof routes[number]['path'];
// 结果：'/users/:id' | '/posts/:postId/comments/:commentId' | '/dashboard'

// 路由参数类型映射
type RouteParamsMap = {
  [K in RoutePaths]: ExtractRouteParams<K>;
};
// 结果：{
//   '/users/:id': { id: string };
//   '/posts/:postId/comments/:commentId': { postId: string; commentId: string };
//   '/dashboard': {};
// }

// 类型安全的路由导航
function navigate<T extends RoutePaths>(
  path: T,
  params: RouteParamsMap[T]
): void {
  // 路由导航实现
  console.log(`Navigating to ${path} with params:`, params);
}

// 使用示例
navigate('/users/:id', { id: '123' }); // OK
navigate('/posts/:postId/comments/:commentId', { 
  postId: '456', 
  commentId: '789' 
}); // OK
navigate('/dashboard', {}); // OK

// navigate('/users/:id', {}); // Error: 缺少 id 参数
// navigate('/dashboard', { id: '123' }); // Error: 不需要参数
```

## 条件类型最佳实践

### 1. 避免过度复杂的条件类型

```typescript
// 不好的做法：过度复杂
type ComplexType<T> = T extends string
  ? T extends `${infer A}${infer B}`
    ? B extends string
      ? A extends 'get'
        ? `Getting ${B}`
        : A extends 'set'
        ? `Setting ${B}`
        : never
      : never
    : never
  : never;

// 好的做法：分解为多个简单类型
type ExtractAction<T extends string> = T extends `${infer Action}${string}` ? Action : never;
type ExtractTarget<T extends string> = T extends `${string}${infer Target}` ? Target : never;

type ActionType<T extends string> = ExtractAction<T> extends 'get'
  ? 'Getting'
  : ExtractAction<T> extends 'set'
  ? 'Setting'
  : never;

type SimpleType<T extends string> = `${ActionType<T>} ${ExtractTarget<T>}`;
```

### 2. 使用辅助类型提高可读性

```typescript
// 辅助类型
type IsArray<T> = T extends any[] ? true : false;
type IsFunction<T> = T extends (...args: any[]) => any ? true : false;
type IsObject<T> = T extends object ? true : false;

// 主要类型使用辅助类型
type SerializableValue<T> = 
  IsArray<T> extends true
    ? T extends (infer U)[]
      ? SerializableValue<U>[]
      : never
    : IsFunction<T> extends true
    ? never
    : IsObject<T> extends true
    ? { [K in keyof T]: SerializableValue<T[K]> }
    : T;

interface ComplexObject {
  name: string;
  age: number;
  callback: () => void;
  items: { id: number; data: string }[];
}

type SerializableObject = SerializableValue<ComplexObject>;
// 结果：{
//   name: string;
//   age: number;
//   callback: never;
//   items: { id: number; data: string }[];
// }
```

### 3. 提供默认类型参数

```typescript
// 提供合理的默认值
type ApiResponse<T = any, E = Error> = 
  | { success: true; data: T }
  | { success: false; error: E };

type SafeExtract<T, K extends keyof T, D = never> = 
  T extends Record<K, infer U> ? U : D;

// 使用示例
type UserResponse = ApiResponse<User>; // 使用默认错误类型
type CustomResponse = ApiResponse<User, string>; // 自定义错误类型

interface User {
  id: number;
  profile?: { avatar: string };
}

type Avatar = SafeExtract<User['profile'], 'avatar', string>;
// 如果 profile 存在且有 avatar，返回 string；否则返回 string（默认值）
```

## 总结

条件类型是 TypeScript 类型系统的强大特性：

1. **类型选择**：根据类型关系动态选择类型
2. **类型推断**：使用 `infer` 关键字推断类型信息
3. **递归处理**：支持递归条件类型处理复杂结构
4. **字符串操作**：强大的字符串模式匹配能力

掌握条件类型有助于：
- 创建更智能的类型系统
- 构建类型安全的 API
- 实现复杂的类型转换
- 提高代码的类型表达能力
