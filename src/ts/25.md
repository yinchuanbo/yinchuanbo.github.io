## 1. 什么是编译选项？

编译选项是 TypeScript 编译器的配置参数，用于控制编译过程的各个方面。这些选项通常在 `tsconfig.json` 文件中配置，也可以通过命令行参数传递。正确配置编译选项对于项目的类型安全、性能和兼容性至关重要。

## 2. tsconfig.json 基本结构

```json
{
  "compilerOptions": {
    // 编译器选项
  },
  "include": [
    // 包含的文件或目录
  ],
  "exclude": [
    // 排除的文件或目录
  ],
  "files": [
    // 明确指定的文件
  ],
  "extends": "./base-config.json",
  "compileOnSave": true,
  "typeAcquisition": {
    // 类型获取选项
  },
  "watchOptions": {
    // 监视选项
  }
}
```

## 3. 核心编译选项

### 3.1 目标和模块选项

```json
{
  "compilerOptions": {
    // 编译目标 JavaScript 版本
    "target": "ES2020",
    
    // 模块系统
    "module": "CommonJS",
    
    // 模块解析策略
    "moduleResolution": "node",
    
    // 库文件
    "lib": ["ES2020", "DOM", "DOM.Iterable"],
    
    // 输出目录
    "outDir": "./dist",
    
    // 根目录
    "rootDir": "./src",
    
    // 基础 URL
    "baseUrl": "./",
    
    // 路径映射
    "paths": {
      "@/*": ["src/*"],
      "@components/*": ["src/components/*"],
      "@utils/*": ["src/utils/*"]
    }
  }
}
```

### 3.2 类型检查选项

```json
{
  "compilerOptions": {
    // 严格模式（启用所有严格检查）
    "strict": true,
    
    // 严格空值检查
    "strictNullChecks": true,
    
    // 严格函数类型检查
    "strictFunctionTypes": true,
    
    // 严格绑定调用检查
    "strictBindCallApply": true,
    
    // 严格属性初始化检查
    "strictPropertyInitialization": true,
    
    // 禁止隐式 any
    "noImplicitAny": true,
    
    // 禁止隐式返回
    "noImplicitReturns": true,
    
    // 禁止隐式 this
    "noImplicitThis": true,
    
    // 禁止未使用的局部变量
    "noUnusedLocals": true,
    
    // 禁止未使用的参数
    "noUnusedParameters": true,
    
    // 精确可选属性类型
    "exactOptionalPropertyTypes": true,
    
    // 禁止不可达代码
    "allowUnreachableCode": false,
    
    // 禁止未使用的标签
    "allowUnusedLabels": false
  }
}
```

### 3.3 输出选项

```json
{
  "compilerOptions": {
    // 生成声明文件
    "declaration": true,
    
    // 声明文件输出目录
    "declarationDir": "./types",
    
    // 生成声明映射文件
    "declarationMap": true,
    
    // 生成源映射文件
    "sourceMap": true,
    
    // 内联源映射
    "inlineSourceMap": false,
    
    // 内联源代码
    "inlineSources": false,
    
    // 移除注释
    "removeComments": true,
    
    // 不生成输出文件
    "noEmit": false,
    
    // 发生错误时不生成输出
    "noEmitOnError": true,
    
    // 只生成声明文件
    "emitDeclarationOnly": false,
    
    // 保留 const 枚举
    "preserveConstEnums": true,
    
    // 新行字符
    "newLine": "lf"
  }
}
```

## 11. 编译器 API 使用

### 11.1 程序化编译

```typescript
import * as ts from 'typescript';

function compile(fileNames: string[], options: ts.CompilerOptions): void {
  const program = ts.createProgram(fileNames, options);
  const emitResult = program.emit();
  
  const allDiagnostics = ts
    .getPreEmitDiagnostics(program)
    .concat(emitResult.diagnostics);
  
  allDiagnostics.forEach(diagnostic => {
    if (diagnostic.file) {
      const { line, character } = ts.getLineAndCharacterOfPosition(
        diagnostic.file,
        diagnostic.start!
      );
      const message = ts.flattenDiagnosticMessageText(
        diagnostic.messageText,
        '\n'
      );
      console.log(
        `${diagnostic.file.fileName} (${line + 1},${character + 1}): ${message}`
      );
    } else {
      console.log(ts.flattenDiagnosticMessageText(diagnostic.messageText, '\n'));
    }
  });
  
  const exitCode = emitResult.emitSkipped ? 1 : 0;
  console.log(`Process exiting with code '${exitCode}'.`);
  process.exit(exitCode);
}

// 使用
compile(['hello.ts'], {
  noEmitOnError: true,
  noImplicitAny: true,
  target: ts.ScriptTarget.ES5,
  module: ts.ModuleKind.CommonJS
});
```

### 11.2 自定义转换器

```typescript
import * as ts from 'typescript';

const transformer: ts.TransformerFactory<ts.SourceFile> = context => {
  return sourceFile => {
    const visitor = (node: ts.Node): ts.Node => {
      // 自定义转换逻辑
      if (ts.isCallExpression(node)) {
        // 处理函数调用
      }
      return ts.visitEachChild(node, visitor, context);
    };
    return ts.visitNode(sourceFile, visitor);
  };
};

// 使用自定义转换器
const program = ts.createProgram(['input.ts'], {});
const result = program.emit(undefined, undefined, undefined, false, {
  before: [transformer]
});
```

## 12. 故障排除和最佳实践

### 12.1 常见问题

```json
// 问题 1: 模块解析失败
{
  "compilerOptions": {
    "moduleResolution": "node",
    "baseUrl": "./",
    "paths": {
      "@/*": ["src/*"]
    }
  }

// 问题 2: 类型检查过于严格
{
  "compilerOptions": {
    "strict": false,
    "noImplicitAny": false,
    "strictNullChecks": false
  }

// 问题 3: 编译性能问题
{
  "compilerOptions": {
    "skipLibCheck": true,
    "incremental": true,
    "tsBuildInfoFile": "./.tsbuildinfo"
  }

```

### 12.2 最佳实践

1. **渐进式严格模式**: 逐步启用严格检查选项
2. **合理的目标版本**: 根据运行环境选择合适的 target
3. **路径映射**: 使用路径映射简化导入
4. **增量编译**: 在大型项目中启用增量编译
5. **配置继承**: 使用配置继承避免重复

## 13. 总结

TypeScript 编译选项提供了强大的配置能力：

1. **类型安全**: 通过严格模式和各种检查选项确保类型安全
2. **性能优化**: 通过增量编译、跳过检查等选项提升编译性能
3. **环境适配**: 针对不同环境和用途配置不同的编译选项
4. **开发体验**: 通过路径映射、源映射等选项改善开发体验

正确配置编译选项是 TypeScript 项目成功的关键因素之一。
