## 1. 什么是声明文件？

声明文件（Declaration Files）是以 `.d.ts` 为扩展名的文件，它们不包含实现代码，只包含类型信息。声明文件的主要作用是为 JavaScript 库提供类型定义，使 TypeScript 能够理解这些库的 API 并提供类型检查和智能提示。

## 2. 声明文件的基本语法

### 2.1 基本结构

```typescript
// 声明全局变量
declare var myGlobal: string;

// 声明全局函数
declare function myFunction(a: number, b: string): boolean;

// 声明全局类
declare class MyClass {
  constructor(name: string);
  name: string;
  getName(): string;
}

// 声明全局命名空间
declare namespace MyNamespace {
  function someFunction(): void;
  const someValue: number;
}

// 声明全局接口
declare interface MyInterface {
  prop: string;
  method(): void;
}

// 声明全局类型
declare type MyType = string | number;
```

### 2.2 模块声明

```typescript
// 为 JavaScript 模块提供类型定义
declare module "module-name" {
  export function someFunction(): void;
  export const someValue: number;
  export default class SomeClass {}
}

// 为 CSS/SCSS 等非代码文件提供类型定义
declare module "*.css" {
  const content: { [className: string]: string };
  export default content;
}

declare module "*.png" {
  const value: string;
  export default value;
}
```

### 2.3 扩展全局对象

```typescript
// 扩展 Window 接口
declare global {
  interface Window {
    myCustomProperty: string;
    myCustomMethod(): void;
  }
  
  interface Array<T> {
    customMethod(): T[];
  }
  
  namespace NodeJS {
    interface Process {
      customEnv: string;
    }
  }
}
```

## 3. 编写声明文件的最佳实践

### 3.1 全局库声明

适用于通过全局变量或属性暴露 API 的库。

```typescript
// jquery.d.ts
declare namespace $ {
  function ajax(settings: AjaxSettings): Promise<any>;
  
  interface AjaxSettings {
    url: string;
    method?: 'GET' | 'POST' | 'PUT' | 'DELETE';
    data?: any;
    dataType?: string;
    headers?: Record<string, string>;
  }
}

// 全局函数重载
declare function $(selector: string): $.Element;
declare function $(element: HTMLElement): $.Element;

// 命名空间扩展
declare namespace $ {
  interface Element {
    css(property: string): string;
    css(property: string, value: string): this;
    on(event: string, handler: (event: Event) => void): this;
  }
}

// 全局变量
declare const jQuery: typeof $;
```

### 3.2 模块库声明

适用于使用 ES 模块、CommonJS 或 AMD 模块系统的库。

```typescript
// lodash.d.ts
declare module 'lodash' {
  // 导出函数
  export function map<T, U>(
    array: T[],
    iteratee: (value: T, index: number, array: T[]) => U
  ): U[];
  
  export function filter<T>(
    array: T[],
    predicate: (value: T, index: number, array: T[]) => boolean
  ): T[];
  
  // 导出接口
  export interface LoDashStatic {
    map: typeof map;
    filter: typeof filter;
    // 更多方法...
  }
  
  // 默认导出
  const _: LoDashStatic;
  export default _;
}
```

### 3.3 UMD 库声明

适用于同时支持全局变量和模块导入的库。

```typescript
// moment.d.ts
declare namespace moment {
  interface Moment {
    format(format?: string): string;
    add(amount: number, unit: string): Moment;
    subtract(amount: number, unit: string): Moment;
  }
  
  interface MomentStatic {
    (date?: string | Date | number): Moment;
    utc(date?: string | Date | number): Moment;
    unix(timestamp: number): Moment;
  }
}

// 全局变量
declare const moment: moment.MomentStatic;

// 模块导出
declare module 'moment' {
  export = moment;
}
```

### 3.4 插件声明

为已有库添加新功能的插件。

```typescript
// jquery-plugin.d.ts
import * as $ from 'jquery';

// 模块扩展
declare module 'jquery' {
  interface JQuery {
    myPlugin(options?: MyPluginOptions): JQuery;
    anotherPlugin(): void;
  }
}

// 插件选项接口
interface MyPluginOptions {
  speed?: number;
  timeout?: number;
  onComplete?: () => void;
}
```

## 4. 高级声明文件技巧

### 4.1 条件类型和泛型

```typescript
// 高级类型定义
declare module 'advanced-library' {
  // 条件类型
  export type ArrayOrSingle<T> = T extends any[] ? T : T[];
  
  // 泛型约束
  export function process<T extends object>(input: T): T;
  
  // 泛型默认值
  export class Container<T = any> {
    value: T;
    setValue(val: T): void;
    getValue(): T;
  }
  
  // 映射类型
  export type ReadonlyProps<T> = {
    readonly [K in keyof T]: T[K];
  };
  
  // 条件类型与 infer
  export type UnpackPromise<T> = T extends Promise<infer U> ? U : T;
}
```

### 4.2 函数重载

```typescript
// 函数重载
declare module 'http-client' {
  export function request(url: string): Promise<string>;
  export function request(url: string, options: RequestOptions): Promise<string>;
  export function request(url: string, options: RequestOptions, callback: (error: Error | null, response: string) => void): void;
  
  export interface RequestOptions {
    method?: 'GET' | 'POST' | 'PUT' | 'DELETE';
    headers?: Record<string, string>;
    body?: string | object;
    timeout?: number;
  }
}
```

### 4.3 声明合并

```typescript
// 接口合并
declare namespace API {
  interface User {
    id: number;
    name: string;
  }
}

declare namespace API {
  interface User {
    email: string;
    role: string;
  }
  
  // 合并后 User 包含所有属性
  // id, name, email, role
}

// 命名空间与类合并
declare class MyLibrary {
  constructor(config: MyLibrary.Config);
  method(): void;
}

declare namespace MyLibrary {
  interface Config {
    apiKey: string;
    timeout: number;
  }
  
  function initialize(): MyLibrary;
}
```

### 4.4 模块扩展

```typescript
// 扩展 Express 请求对象
import * as express from 'express';

    interface Request {
      user?: {
        id: string;

  }

## 6. 自动生成声明文件

### 6.1 使用 TypeScript 编译器

```bash
# 在 tsconfig.json 中启用声明文件生成
# 然后运行编译
tsc --declaration --declarationDir ./types
```

### 6.2 使用 API Extractor

Microsoft 的 API Extractor 可以从 TypeScript 项目生成干净的 API 定义。

```bash
# 安装 API Extractor
npm install --save-dev @microsoft/api-extractor

# 创建配置文件
npx api-extractor init

# 运行 API Extractor
npx api-extractor run
```

### 6.3 使用 dts-gen

自动为 JavaScript 库生成声明文件。

```bash
# 安装 dts-gen
npm install -g dts-gen

# 为 npm 包生成声明文件
dts-gen --name lodash --source npm

## 8. 实际应用示例

### 8.1 为 JavaScript 库创建声明文件

假设我们有一个简单的 JavaScript 库：

```javascript
// simple-lib.js
var SimpleLib = {
  add: function(a, b) {
    return a + b;
  },
  subtract: function(a, b) {
    return a - b;
  },
  formatName: function(first, last) {
    return first + ' ' + last;
  },
  User: function(name, age) {
    this.name = name;
    this.age = age;
    this.greet = function() {
      return 'Hello, ' + this.name;
    };
  }
};

if (typeof module !== 'undefined' && module.exports) {
  module.exports = SimpleLib;
}
```

对应的声明文件：

```typescript
// simple-lib.d.ts
declare namespace SimpleLib {
  function add(a: number, b: number): number;
  function subtract(a: number, b: number): number;
  function formatName(first: string, last: string): string;
  
  class User {
    name: string;
    age: number;
    constructor(name: string, age: number);
    greet(): string;
  }
}

// 支持 CommonJS 模块
declare module 'simple-lib' {
  export = SimpleLib;
}
```

### 8.2 React 组件库声明文件

```typescript
// my-react-components.d.ts
import * as React from 'react';

declare module 'my-react-components' {
  export interface ButtonProps {
    text: string;
    onClick?: () => void;
    disabled?: boolean;
    variant?: 'primary' | 'secondary' | 'danger';
  }
  
  export class Button extends React.Component<ButtonProps> {}
  
  export interface CardProps {
    title: string;
    subtitle?: string;
    children?: React.ReactNode;
    onClose?: () => void;
  }
  
  export class Card extends React.Component<CardProps> {}
  
  export interface ModalProps {
    isOpen: boolean;
    onClose: () => void;
    title: string;
    size?: 'small' | 'medium' | 'large';
    children: React.ReactNode;
  }
  
  export class Modal extends React.Component<ModalProps> {}
  
  // 主题定义
  export interface Theme {
    colors: {
      primary: string;
      secondary: string;
      background: string;
      text: string;
    };
    spacing: {
      small: number;
      medium: number;
      large: number;
    };
  }
  
  // 主题提供者
  export interface ThemeProviderProps {
    theme: Theme;
    children: React.ReactNode;
  }
  
  export class ThemeProvider extends React.Component<ThemeProviderProps> {}
  
  // Hooks
  export function useTheme(): Theme;
  export function useModal(): [boolean, () => void, () => void];
}
```

### 8.3 Node.js 模块声明文件

```typescript
// custom-database.d.ts
/// <reference types="node" />

declare module 'custom-database' {
  import { EventEmitter } from 'events';
  
  export interface ConnectionOptions {
    host: string;
    port: number;
    username: string;
    password: string;
    database: string;
    ssl?: boolean;
    timeout?: number;
  }
  
  export interface QueryResult<T = any> {
    rows: T[];
    rowCount: number;
    fields: {
      name: string;
      type: string;
    }[];
  }
  
  export class Connection extends EventEmitter {
    constructor(options: ConnectionOptions);
    connect(): Promise<void>;
    disconnect(): Promise<void>;
    query<T = any>(sql: string, params?: any[]): Promise<QueryResult<T>>;
    transaction<T>(callback: (client: Connection) => Promise<T>): Promise<T>;
    
    on(event: 'connect', listener: () => void): this;
    on(event: 'disconnect', listener: () => void): this;
    on(event: 'error', listener: (error: Error) => void): this;
  }
  
  export function createConnection(options: ConnectionOptions): Connection;
  export function createPool(options: ConnectionOptions & { maxConnections?: number }): Pool;
  
  export class Pool extends EventEmitter {
    getConnection(): Promise<Connection>;
    releaseConnection(connection: Connection): void;
    query<T = any>(sql: string, params?: any[]): Promise<QueryResult<T>>;
    end(): Promise<void>;
  }
  
  export const version: string;
}
```

## 9. 声明文件调试和故障排除

### 9.1 常见问题

```typescript
// 问题 1: 类型不匹配
// 声明文件
declare function process(value: string): number;

// 实际使用
process(42); // 类型错误: 参数类型 'number' 不能赋值给参数类型 'string'

// 问题 2: 缺少属性
// 声明文件
interface Config {
  apiUrl: string;
  timeout: number;
}

// 实际使用
const config: Config = {
  apiUrl: "https://api.example.com"
  // 错误: 缺少属性 'timeout'
};

// 问题 3: 命名冲突
// 声明文件 1
declare namespace Utils {
  function format(value: string): string;
}

// 声明文件 2
declare namespace Utils {
  function format(value: number): string; // 错误: 重复函数实现
}
```

### 9.2 调试技巧

```typescript
// 使用 typeof 检查类型
const myVar = someLibrary.getValue();
type MyVarType = typeof myVar; // 查看推断类型

## 10. 最佳实践总结

### 10.1 声明文件编写原则

1. **精确性**: 确保类型定义与实际 JavaScript 代码行为一致
2. **完整性**: 尽可能覆盖库的所有公共 API
3. **可读性**: 使用清晰的命名和注释
4. **模块化**: 适当拆分大型声明文件
5. **版本一致**: 确保声明文件版本与库版本匹配

### 10.2 文档和注释

```typescript
/**
 * 表示用户信息的接口
 * @since 1.0.0
 */
export interface User {
  /** 用户唯一标识 */
  id: string;
  
  /** 用户名称 */
  name: string;
  
  /**
   * 用户邮箱
   * @example "user@example.com"
   */
  email: string;
  
  /**
   * 用户角色
   * @default "user"
   */
  role?: "admin" | "user" | "guest";
}

/**
 * 创建新用户
 * @param userData - 用户数据
 * @returns 创建的用户对象
 * @throws 如果用户数据无效
 * @example
 * ```typescript
 * const user = createUser({ name: "John", email: "john@example.com" });
 * ```
 */
export function createUser(userData: Partial<User>): User;
```

## 11. 总结

声明文件是 TypeScript 生态系统的重要组成部分，它们：

1. **提供类型信息** - 为 JavaScript 库添加类型定义
2. **增强开发体验** - 提供代码补全和类型检查
3. **提高代码质量** - 减少类型错误和运行时问题
4. **促进生态系统** - 使不同库能够协同工作

掌握声明文件的编写和使用是 TypeScript 开发者的重要技能，它能够帮助你更好地与 JavaScript 生态系统集成，并提高代码的可维护性和可靠性。
