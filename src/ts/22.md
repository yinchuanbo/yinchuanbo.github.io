## 1. 什么是混入？

混入（Mixins）是一种设计模式，允许我们将多个类的功能组合到一个类中，实现类似多重继承的效果。在 TypeScript 中，由于类只能继承一个父类，混入提供了一种灵活的方式来共享和重用代码。

## 2. 基础混入模式

### 2.1 简单混入实现

```typescript
// 基础类
class Disposable {
  isDisposed: boolean = false;
  
  dispose() {
    this.isDisposed = true;
  }
}

class Activatable {
  isActive: boolean = false;
  
  activate() {
    this.isActive = true;
  }
  
  deactivate() {
    this.isActive = false;
  }
}

// 混入函数
function applyMixins(derivedCtor: any, baseCtors: any[]) {
  baseCtors.forEach(baseCtor => {
    Object.getOwnPropertyNames(baseCtor.prototype).forEach(name => {
      if (name !== 'constructor') {
        Object.defineProperty(
          derivedCtor.prototype,
          name,
          Object.getOwnPropertyDescriptor(baseCtor.prototype, name) || Object.create(null)
        );
      }
    });
  });
}

// 目标类
class SmartObject {
  constructor() {
    // 初始化逻辑
  }
  
  interact() {
    console.log("Interacting with smart object");
  }
}

// 应用混入
interface SmartObject extends Disposable, Activatable {}
applyMixins(SmartObject, [Disposable, Activatable]);

// 使用
const smartObj = new SmartObject();
smartObj.activate();
console.log(smartObj.isActive); // true
smartObj.dispose();
console.log(smartObj.isDisposed); // true
```

### 2.2 类型安全的混入

```typescript
// 混入构造函数类型
type Constructor<T = {}> = new (...args: any[]) => T;

// 混入函数类型
type Mixin<T extends Constructor> = (Base: T) => T;

// 时间戳混入
function Timestamped<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    timestamp = Date.now();
    
    getAge() {
      return Date.now() - this.timestamp;
    }
  };
}

// 标签混入
function Tagged<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    tags: string[] = [];
    
    addTag(tag: string) {
      this.tags.push(tag);
    }
    
    hasTag(tag: string): boolean {
      return this.tags.includes(tag);
    }
    
    removeTag(tag: string) {
      const index = this.tags.indexOf(tag);
      if (index !== -1) {
        this.tags.splice(index, 1);
      }
    }
  };
}

// 基础类
class User {
  constructor(public name: string) {}
  
  greet() {
    return `Hello, I'm ${this.name}`;
  }
}

// 应用混入
const TimestampedUser = Timestamped(User);
const TaggedUser = Tagged(User);
const EnhancedUser = Tagged(Timestamped(User));

// 使用
const user1 = new TimestampedUser("Alice");
console.log(user1.getAge()); // 时间差

const user2 = new TaggedUser("Bob");
user2.addTag("developer");
user2.addTag("typescript");
console.log(user2.hasTag("developer")); // true

const user3 = new EnhancedUser("Charlie");
user3.addTag("admin");
console.log(user3.getAge());
console.log(user3.tags);
```

## 3. 高级混入模式

### 3.1 条件混入

```typescript
// 条件混入类型
type ConditionalMixin<T, Condition> = Condition extends true ? T : {};

// 可配置的混入
interface MixinOptions {
  withLogging?: boolean;
  withValidation?: boolean;
  withCaching?: boolean;
}

function createConfigurableMixin<T extends Constructor>(options: MixinOptions = {}) {
  return function(Base: T) {
    class Mixed extends Base {
      // 基础功能
    }
    
    // 条件性添加日志功能
    if (options.withLogging) {
      Object.assign(Mixed.prototype, {
        log(message: string) {
          console.log(`[${this.constructor.name}] ${message}`);
        }
      });
    }
    
    // 条件性添加验证功能
    if (options.withValidation) {
      Object.assign(Mixed.prototype, {
        validate(): boolean {
          // 验证逻辑
          return true;
        }
      });
    }
    
    // 条件性添加缓存功能
    if (options.withCaching) {
      Object.assign(Mixed.prototype, {
        cache: new Map(),
        getCached(key: string) {
          return this.cache.get(key);
        },
        setCached(key: string, value: any) {
          this.cache.set(key, value);
        }
      });
    }
    
    return Mixed;
  };
}

// 使用条件混入
class DataService {
  constructor(public name: string) {}
  
  fetchData() {
    return "data";
  }
}

const EnhancedDataService = createConfigurableMixin({
  withLogging: true,
  withCaching: true
})(DataService);

const service = new EnhancedDataService("MyService");
// service.log("Starting service"); // 如果启用了日志
// service.setCached("key", "value"); // 如果启用了缓存
```

### 3.2 泛型混入

```typescript
// 泛型混入接口
interface Serializable<T> {
  serialize(): string;
  deserialize(data: string): T;
}

interface Cloneable<T> {
  clone(): T;
}

// 泛型混入实现
function Serializable<TBase extends Constructor>(Base: TBase) {
  return class extends Base implements Serializable<InstanceType<TBase>> {
    serialize(): string {
      return JSON.stringify(this);
    }
    
    deserialize(data: string): InstanceType<TBase> {
      const parsed = JSON.parse(data);
      return Object.assign(this, parsed);
    }
  };
}

function Cloneable<TBase extends Constructor>(Base: TBase) {
  return class extends Base implements Cloneable<InstanceType<TBase>> {
    clone(): InstanceType<TBase> {
      const cloned = Object.create(Object.getPrototypeOf(this));
      return Object.assign(cloned, this);
    }
  };
}

// 基础模型类
class Model {
  constructor(public id: string, public data: any) {}
}

// 应用泛型混入
const SerializableModel = Serializable(Model);
const CloneableModel = Cloneable(Model);
const FullFeaturedModel = Cloneable(Serializable(Model));

// 使用
const model = new FullFeaturedModel("1", { name: "Test" });
const serialized = model.serialize();
const cloned = model.clone();

console.log(serialized); // JSON 字符串
console.log(cloned.id);  // "1"
```

### 3.3 装饰器风格的混入

```typescript
// 装饰器混入
function mixin(...mixins: any[]) {
  return function<T extends Constructor>(target: T) {
    mixins.forEach(mixin => {
      Object.getOwnPropertyNames(mixin.prototype).forEach(name => {
        if (name !== 'constructor') {
          Object.defineProperty(
            target.prototype,
            name,
            Object.getOwnPropertyDescriptor(mixin.prototype, name) || Object.create(null)
          );
        }
      });
    });
    return target;
  };
}

// 混入类
class Flyable {
  fly() {
    console.log("Flying...");
  }
  
  land() {
    console.log("Landing...");
  }
}

class Swimmable {
  swim() {
    console.log("Swimming...");
  }
  
  dive() {
    console.log("Diving...");
  }
}

// 使用装饰器应用混入
@mixin(Flyable, Swimmable)
class Duck {
  name: string;
  
  constructor(name: string) {
    this.name = name;
  }
  
  quack() {
    console.log(`${this.name} says quack!`);
  }
}

// 类型声明
interface Duck extends Flyable, Swimmable {}

// 使用
const duck = new Duck("Donald");
duck.quack();
duck.fly();
duck.swim();
```

## 4. 实际应用示例

### 4.1 事件系统混入

```typescript
// 事件监听器接口
interface EventListener<T = any> {
  (event: T): void;
}

// 事件发射器混入
function EventEmitter<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    private listeners: Map<string, EventListener[]> = new Map();
    
    on<T>(event: string, listener: EventListener<T>): void {
      if (!this.listeners.has(event)) {
        this.listeners.set(event, []);
      }
      this.listeners.get(event)!.push(listener);
    }
    
    off<T>(event: string, listener: EventListener<T>): void {
      const eventListeners = this.listeners.get(event);
      if (eventListeners) {
        const index = eventListeners.indexOf(listener);
        if (index !== -1) {
          eventListeners.splice(index, 1);
        }
      }
    }
    
    emit<T>(event: string, data: T): void {
      const eventListeners = this.listeners.get(event);
      if (eventListeners) {
        eventListeners.forEach(listener => listener(data));
      }
    }
    
    once<T>(event: string, listener: EventListener<T>): void {
      const onceListener = (data: T) => {
        listener(data);
        this.off(event, onceListener);
      };
      this.on(event, onceListener);
    }
  };
}

// 基础组件类
class Component {
  constructor(public name: string) {}
  
  render() {
    console.log(`Rendering ${this.name}`);
  }
}

// 应用事件混入
const EventfulComponent = EventEmitter(Component);

// 使用
const component = new EventfulComponent("MyComponent");

component.on("click", (data) => {
  console.log("Component clicked:", data);
});

component.on("hover", (data) => {
  console.log("Component hovered:", data);
});

component.emit("click", { x: 100, y: 200 });
component.emit("hover", { target: "button" });
```

### 4.2 状态管理混入

```typescript
// 状态管理接口
interface StateManager<T> {
  getState(): T;
  setState(newState: Partial<T>): void;
  subscribe(listener: (state: T) => void): () => void;
}

// 状态管理混入
function Stateful<TBase extends Constructor, TState = any>(
  initialState: TState
) {
  return function(Base: TBase) {
    return class extends Base implements StateManager<TState> {
      private state: TState = initialState;
      private subscribers: Array<(state: TState) => void> = [];
      
      getState(): TState {
        return { ...this.state };
      }
      
      setState(newState: Partial<TState>): void {
        const prevState = this.state;
        this.state = { ...this.state, ...newState };
        
        // 通知订阅者
        this.subscribers.forEach(subscriber => {
          subscriber(this.state);
        });
        
        // 触发生命周期方法
        if (typeof (this as any).onStateChange === 'function') {
          (this as any).onStateChange(this.state, prevState);
        }
      }
      
      subscribe(listener: (state: TState) => void): () => void {
        this.subscribers.push(listener);
        
        // 返回取消订阅函数
        return () => {
          const index = this.subscribers.indexOf(listener);
          if (index !== -1) {
            this.subscribers.splice(index, 1);
          }
        };
      }
    };
  };
}

// 应用状态管理
interface CounterState {
  count: number;
  step: number;
}

class Counter {
  constructor(public name: string) {}
  
  increment() {
    const currentState = (this as any).getState();
    (this as any).setState({ count: currentState.count + currentState.step });
  }
  
  decrement() {
    const currentState = (this as any).getState();
    (this as any).setState({ count: currentState.count - currentState.step });
  }
  
  onStateChange(newState: CounterState, prevState: CounterState) {
    console.log(`${this.name} state changed:`, {
      from: prevState.count,
      to: newState.count
    });
  }
}

// 应用混入
const StatefulCounter = Stateful<typeof Counter, CounterState>({
  count: 0,
  step: 1
})(Counter);

// 类型声明
interface StatefulCounter extends StateManager<CounterState> {}

// 使用
const counter = new StatefulCounter("MyCounter");

const unsubscribe = counter.subscribe((state) => {
  console.log("State updated:", state);
});

counter.increment(); // count: 1
counter.increment(); // count: 2
counter.setState({ step: 5 });
counter.increment(); // count: 7

unsubscribe();
```

### 4.3 验证混入

```typescript
// 验证规则接口
interface ValidationRule<T> {
  validate(value: T): boolean;
  message: string;
}

// 验证结果接口
interface ValidationResult {
  isValid: boolean;
  errors: string[];
}

// 验证混入
function Validatable<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    private validationRules: Map<string, ValidationRule<any>[]> = new Map();
    
    addValidationRule<T>(field: string, rule: ValidationRule<T>): void {
      if (!this.validationRules.has(field)) {
        this.validationRules.set(field, []);
      }
      this.validationRules.get(field)!.push(rule);
    }
    
    validateField(field: string, value: any): ValidationResult {
      const rules = this.validationRules.get(field) || [];
      const errors: string[] = [];
      
      for (const rule of rules) {
        if (!rule.validate(value)) {
          errors.push(rule.message);
        }
      }
      
      return {
        isValid: errors.length === 0,
        errors
      };
    }
    
    validateAll(): ValidationResult {
      const allErrors: string[] = [];
      
      for (const [field, rules] of this.validationRules) {
        const fieldValue = (this as any)[field];
        const result = this.validateField(field, fieldValue);
        allErrors.push(...result.errors);
      }
      
      return {
        isValid: allErrors.length === 0,
        errors: allErrors
      };
    }
  };
}

// 常用验证规则
const ValidationRules = {
  required: <T>(message = "Field is required"): ValidationRule<T> => ({
    validate: (value: T) => value != null && value !== "",
    message
  }),
  
  minLength: (min: number, message?: string): ValidationRule<string> => ({
    validate: (value: string) => value && value.length >= min,
    message: message || `Minimum length is ${min}`
  }),
  
  email: (message = "Invalid email format"): ValidationRule<string> => ({
    validate: (value: string) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value),
    message
  }),
  
  range: (min: number, max: number, message?: string): ValidationRule<number> => ({
    validate: (value: number) => value >= min && value <= max,
    message: message || `Value must be between ${min} and ${max}`
  })
};

// 用户模型
class User {
  constructor(
    public name: string = "",
    public email: string = "",
    public age: number = 0
  ) {}
}

// 应用验证混入
const ValidatableUser = Validatable(User);

// 使用
const user = new ValidatableUser("", "invalid-email", 15);

// 添加验证规则
user.addValidationRule("name", ValidationRules.required("Name is required"));
user.addValidationRule("name", ValidationRules.minLength(2, "Name must be at least 2 characters"));
user.addValidationRule("email", ValidationRules.required("Email is required"));
user.addValidationRule("email", ValidationRules.email("Please enter a valid email"));
user.addValidationRule("age", ValidationRules.range(18, 100, "Age must be between 18 and 100"));

// 验证
const nameResult = user.validateField("name", user.name);
console.log("Name validation:", nameResult);

const allResult = user.validateAll();
console.log("All validation:", allResult);
```

## 5. 混入的最佳实践

### 5.1 类型安全

```typescript
// ✅ 使用类型安全的混入
type MixinConstructor<T = {}> = new (...args: any[]) => T;

function TypeSafeMixin<TBase extends MixinConstructor>(Base: TBase) {
  return class extends Base {
    // 确保类型安全的实现
    mixinMethod(): string {
      return "mixin method";
    }
  };
}

// ❌ 避免使用 any 类型
function UnsafeMixin(Base: any) {
  return class extends Base {
    // 类型不安全
  };
}
```

### 5.2 命名约定

```typescript
// ✅ 清晰的命名约定
function WithLogging<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    log(message: string) {
      console.log(`[${this.constructor.name}] ${message}`);
    }
  };
}

function WithCaching<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    private cache = new Map();
    
    getCached(key: string) {
      return this.cache.get(key);
    }
  };
}

// ✅ 组合混入的清晰命名
const LoggingCachingService = WithCaching(WithLogging(BaseService));
```

### 5.3 避免冲突

```typescript
// ✅ 使用命名空间避免方法冲突
function WithNamespacedLogging<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    logging = {
      log: (message: string) => {
        console.log(`[${this.constructor.name}] ${message}`);
      },
      error: (message: string) => {
        console.error(`[${this.constructor.name}] ERROR: ${message}`);
      }
    };
  };
}

// ✅ 使用前缀避免属性冲突
function WithPrefixedState<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    private _mixinState: any = {};
    
    getMixinState() {
      return this._mixinState;
    }
    
    setMixinState(state: any) {
      this._mixinState = { ...this._mixinState, ...state };
    }
  };
}
```

## 6. 混入 vs 其他模式

### 6.1 混入 vs 继承

```typescript
// 继承方式 - 单一继承链
class Animal {
  name: string;
  constructor(name: string) {
    this.name = name;
  }
}

class Mammal extends Animal {
  warmBlooded = true;
}

class Dog extends Mammal {
  bark() {
    console.log("Woof!");
  }
}

// 混入方式 - 多重能力组合
class Animal2 {
  constructor(public name: string) {}
}

function Flyable<T extends Constructor>(Base: T) {
  return class extends Base {
    fly() {
      console.log(`${(this as any).name} is flying`);
    }
  };
}

function Swimmable<T extends Constructor>(Base: T) {
  return class extends Base {
    swim() {
      console.log(`${(this as any).name} is swimming`);
    }
  };
}

// 可以组合多种能力
const FlyingSwimmingAnimal = Swimmable(Flyable(Animal2));
const duck = new FlyingSwimmingAnimal("Duck");
duck.fly();
duck.swim();
```

### 6.2 混入 vs 组合

```typescript
// 组合方式
class Logger {
  log(message: string) {
    console.log(message);
  }
}

class Cache {
  private data = new Map();
  
  get(key: string) {
    return this.data.get(key);
  }
  
  set(key: string, value: any) {
    this.data.set(key, value);
  }
}

class ServiceWithComposition {
  private logger = new Logger();
  private cache = new Cache();
  
  doSomething() {
    this.logger.log("Doing something");
    this.cache.set("key", "value");
  }
}

// 混入方式
const ServiceWithMixins = WithCaching(WithLogging(class BaseService {
  doSomething() {
    this.log("Doing something");
    this.setCached("key", "value");
  }
}));
```

## 7. 总结

混入是 TypeScript 中实现多重继承的强大模式：

1. **灵活性** - 可以组合多个功能类
2. **重用性** - 混入可以应用到不同的基类
3. **类型安全** - 通过泛型保证类型安全
4. **可组合** - 多个混入可以链式组合

混入特别适用于：
- 横切关注点（日志、缓存、验证等）
- 可选功能的组合
- 代码重用和模块化
- 避免深层继承链
