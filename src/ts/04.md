TypeScript 支持基于类的面向对象编程，提供了类、继承、访问修饰符等特性。

## 基本类定义

### 简单类
```typescript
class Greeter {
    greeting: string;
    
    constructor(message: string) {
        this.greeting = message;
    }
    
    greet() {
        return "Hello, " + this.greeting;
    }
}

let greeter = new Greeter("world");
console.log(greeter.greet()); // "Hello, world"
```

### 类的属性和方法
```typescript
class Person {
    // 属性声明
    name: string;
    age: number;
    
    // 构造函数
    constructor(name: string, age: number) {
        this.name = name;
        this.age = age;
    }
    
    // 方法
    introduce(): string {
        return `Hi, I'm ${this.name} and I'm ${this.age} years old.`;
    }
    
    // 静态方法
    static createAdult(name: string): Person {
        return new Person(name, 18);
    }
}

const person = new Person("Alice", 25);
console.log(person.introduce());

const adult = Person.createAdult("Bob");
```

## 继承

### 基本继承
```typescript
class Animal {
    name: string;
    
    constructor(name: string) {
        this.name = name;
    }
    
    move(distanceInMeters: number = 0) {
        console.log(`${this.name} moved ${distanceInMeters}m.`);
    }
}

class Dog extends Animal {
    breed: string;
    
    constructor(name: string, breed: string) {
        super(name); // 调用父类构造函数
        this.breed = breed;
    }
    
    bark() {
        console.log("Woof! Woof!");
    }
    
    // 重写父类方法
    move(distanceInMeters = 5) {
        console.log("Running...");
        super.move(distanceInMeters);
    }
}

const dog = new Dog("Rex", "German Shepherd");
dog.bark();
dog.move(10);
```

### 抽象类
```typescript
abstract class Shape {
    abstract area(): number;
    
    // 具体方法
    displayArea(): void {
        console.log(`Area: ${this.area()}`);
    }
}

class Circle extends Shape {
    constructor(private radius: number) {
        super();
    }
    
    area(): number {
        return Math.PI * this.radius * this.radius;
    }
}

class Rectangle extends Shape {
    constructor(private width: number, private height: number) {
        super();
    }
    
    area(): number {
        return this.width * this.height;
    }
}

const circle = new Circle(5);
circle.displayArea(); // Area: 78.54

const rectangle = new Rectangle(4, 6);
rectangle.displayArea(); // Area: 24
```

## 访问修饰符

### public（公共）
```typescript
class PublicExample {
    public name: string; // 默认为 public
    
    public constructor(name: string) {
        this.name = name;
    }
    
    public greet(): string {
        return `Hello, ${this.name}`;
    }
}
```

### private（私有）
```typescript
class PrivateExample {
    private secret: string;
    
    constructor(secret: string) {
        this.secret = secret;
    }
    
    private getSecret(): string {
        return this.secret;
    }
    
    public revealSecret(): string {
        return this.getSecret(); // 类内部可以访问私有成员
    }
}

const example = new PrivateExample("top secret");
// console.log(example.secret); // Error: 'secret' is private
console.log(example.revealSecret()); // OK
```

### protected（受保护）
```typescript
class ProtectedExample {
    protected name: string;
    
    constructor(name: string) {
        this.name = name;
    }
    
    protected displayName(): void {
        console.log(`Name: ${this.name}`);
    }
}

class ExtendedExample extends ProtectedExample {
    constructor(name: string, private age: number) {
        super(name);
    }
    
    public introduce(): void {
        this.displayName(); // 子类可以访问 protected 成员
        console.log(`Age: ${this.age}`);
    }
}

const extended = new ExtendedExample("Alice", 25);
extended.introduce();
// extended.displayName(); // Error: 'displayName' is protected
```

## 参数属性

简化构造函数参数的写法：

```typescript
class ParameterProperties {
    // 参数属性：在构造函数参数前加访问修饰符
    constructor(
        public name: string,
        private age: number,
        protected email: string,
        readonly id: number
    ) {
        // 不需要手动赋值，TypeScript 会自动创建属性并赋值
    }
    
    getInfo(): string {
        return `${this.name} (${this.age}) - ${this.email}`;
    }
}

const user = new ParameterProperties("Alice", 25, "alice@example.com", 1);
console.log(user.name); // "Alice"
console.log(user.getInfo());
```

## 存取器 (Getters/Setters)

```typescript
class Employee {
    private _fullName: string = "";
    private _department: string = "";
    
    get fullName(): string {
        return this._fullName;
    }
    
    set fullName(newName: string) {
        if (newName && newName.length > 0) {
            this._fullName = newName;
        } else {
            throw new Error("Name cannot be empty");
        }
    }
    
    get department(): string {
        return this._department;
    }
    
    set department(dept: string) {
        this._department = dept.toUpperCase();
    }
}

const employee = new Employee();
employee.fullName = "Bob Smith";
employee.department = "engineering";

console.log(employee.fullName); // "Bob Smith"
console.log(employee.department); // "ENGINEERING"
```

## 静态属性和方法

```typescript
class MathUtils {
    static PI: number = 3.14159;
    static readonly E: number = 2.71828;
    
    static calculateCircleArea(radius: number): number {
        return this.PI * radius * radius;
    }
    
    static max(a: number, b: number): number {
        return a > b ? a : b;
    }
}

console.log(MathUtils.PI); // 3.14159
console.log(MathUtils.calculateCircleArea(5)); // 78.5395
console.log(MathUtils.max(10, 20)); // 20
```

## 泛型类

```typescript
class GenericNumber<T> {
    zeroValue: T;
    add: (x: T, y: T) => T;
    
    constructor(zeroValue: T, addFn: (x: T, y: T) => T) {
        this.zeroValue = zeroValue;
        this.add = addFn;
    }
}

let myGenericNumber = new GenericNumber<number>(0, (x, y) => x + y);
console.log(myGenericNumber.add(5, 10)); // 15

let stringNumeric = new GenericNumber<string>("", (x, y) => x + y);
console.log(stringNumeric.add("Hello ", "World")); // "Hello World"
```

## 实际应用示例

### 用户管理系统
```typescript
abstract class User {
    protected constructor(
        public readonly id: number,
        public name: string,
        public email: string,
        protected createdAt: Date = new Date()
    ) {}
    
    abstract getPermissions(): string[];
    
    public getProfile(): object {
        return {
            id: this.id,
            name: this.name,
            email: this.email,
            type: this.constructor.name
        };
    }
}

class AdminUser extends User {
    constructor(id: number, name: string, email: string) {
        super(id, name, email);
    }
    
    getPermissions(): string[] {
        return ["read", "write", "delete", "admin"];
    }
    
    deleteUser(userId: number): void {
        console.log(`Admin ${this.name} deleted user ${userId}`);
    }
}

class RegularUser extends User {
    private _posts: string[] = [];
    
    constructor(id: number, name: string, email: string) {
        super(id, name, email);
    }
    
    getPermissions(): string[] {
        return ["read", "write"];
    }
    
    get posts(): string[] {
        return [...this._posts]; // 返回副本
    }
    
    addPost(post: string): void {
        this._posts.push(post);
    }
}

// 使用示例
const admin = new AdminUser(1, "Admin User", "admin@example.com");
const user = new RegularUser(2, "John Doe", "john@example.com");

console.log(admin.getPermissions()); // ["read", "write", "delete", "admin"]
console.log(user.getPermissions()); // ["read", "write"]

user.addPost("My first post");
console.log(user.posts); // ["My first post"]
```

### 数据存储类
```typescript
interface Serializable {
    serialize(): string;
    deserialize(data: string): void;
}

class DataStore<T extends Serializable> {
    private items: Map<string, T> = new Map();
    
    save(key: string, item: T): void {
        this.items.set(key, item);
        // 模拟保存到本地存储
        localStorage.setItem(key, item.serialize());
    }
    
    load(key: string, itemClass: new() => T): T | null {
        // 先从内存中查找
        if (this.items.has(key)) {
            return this.items.get(key)!;
        }
        
        // 从本地存储加载
        const data = localStorage.getItem(key);
        if (data) {
            const item = new itemClass();
            item.deserialize(data);
            this.items.set(key, item);
            return item;
        }
        
        return null;
    }
    
    delete(key: string): boolean {
        localStorage.removeItem(key);
        return this.items.delete(key);
    }
}

class UserData implements Serializable {
    constructor(
        public name: string = "",
        public age: number = 0
    ) {}
    
    serialize(): string {
        return JSON.stringify({ name: this.name, age: this.age });
    }
    
    deserialize(data: string): void {
        const parsed = JSON.parse(data);
        this.name = parsed.name;
        this.age = parsed.age;
    }
}

// 使用示例
const userStore = new DataStore<UserData>();
const userData = new UserData("Alice", 25);

userStore.save("user1", userData);
const loadedUser = userStore.load("user1", UserData);
console.log(loadedUser); // UserData { name: "Alice", age: 25 }
```
