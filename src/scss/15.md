### 工具函数

```scss
// abstracts/_functions.scss

// 获取颜色值
@function color($name, $variant: 500) {
  $color-map: map-get($colors, $name);
  
  @if not $color-map {
    @error "Color '#{$name}' not found in $colors map.";
  }
  
  @if type-of($color-map) == color {
    @return $color-map;
  }
  
  $color-value: map-get($color-map, $variant);
  
  @if not $color-value {
    @error "Color variant '#{$variant}' not found for color '#{$name}'.";
  }
  
  @return $color-value;
}

// 获取间距值
@function spacing($size) {
  $space: map-get($spacing, $size);
  
  @if not $space {
    @error "Spacing size '#{$size}' not found in $spacing map.";
  }
  
  @return $space;
}

// 获取字体大小
@function font-size($size) {
  $fs: map-get($font-sizes, $size);
  
  @if not $fs {
    @error "Font size '#{$size}' not found in $font-sizes map.";
  }
  
  @return $fs;
}

// 获取阴影
@function shadow($size) {
  $sh: map-get($shadows, $size);
  
  @if not $sh {
    @error "Shadow size '#{$size}' not found in $shadows map.";
  }
  
  @return $sh;
}

// 获取边框圆角
@function radius($size) {
  $r: map-get($border-radius, $size);
  
  @if not $r {
    @error "Border radius '#{$size}' not found in $border-radius map.";
  }
  
  @return $r;
}

// 流体字体大小计算
@function fluid-font-size($min-size, $max-size, $min-width: 320px, $max-width: 1200px) {
  $slope: ($max-size - $min-size) / ($max-width - $min-width);
  $intersection: (-$min-width * $slope) + $min-size;
  
  @return clamp(#{$min-size}, #{$intersection} + #{$slope * 100vw}, #{$max-size});
}
```

### 响应式混合器

```scss
// abstracts/_mixins.scss

// 响应式断点
@mixin respond-to($breakpoint) {
  $size: map-get($breakpoints, $breakpoint);
  
  @if not $size {
    @error "Breakpoint '#{$breakpoint}' not found in $breakpoints map.";
  }
  
  @if $size == 0 {
    @content;
  } @else {
    @media (min-width: #{$size}) {
      @content;
    }
  }
}

// 容器混合器
@mixin container($max-width: null) {
  width: 100%;
  margin-left: auto;
  margin-right: auto;
  padding-left: spacing(4);
  padding-right: spacing(4);
  
  @if $max-width {
    max-width: $max-width;
  } @else {
    @each $bp, $width in $container-max-widths {
      @include respond-to($bp) {
        max-width: $width;
      }
    }
  }
}

// 网格系统
@mixin row($gutter: spacing(4)) {
  display: flex;
  flex-wrap: wrap;
  margin-left: -($gutter / 2);
  margin-right: -($gutter / 2);
}

@mixin col($size: null, $bp: null, $gutter: spacing(4)) {
  padding-left: $gutter / 2;
  padding-right: $gutter / 2;
  
  @if $size {
    @if $bp {
      @include respond-to($bp) {
        flex: 0 0 percentage($size / 12);
        max-width: percentage($size / 12);
      }
    } @else {
      flex: 0 0 percentage($size / 12);
      max-width: percentage($size / 12);
    }
  } @else {
    flex: 1;
  }
}

// 文本样式
@mixin text-style($size, $weight: 400, $line-height: 1.5) {
  font-size: font-size($size);
  font-weight: $weight;
  line-height: $line-height;
}

// 按钮样式
@mixin button-style($variant: primary, $size: base) {
  display: inline-flex;
  align-items: center;
  justify-content: center;
  border: none;
  border-radius: radius(md);
  cursor: pointer;
  text-decoration: none;
  transition: all 0.2s ease;
  font-family: inherit;
  
  @if $size == sm {
    padding: spacing(2) spacing(3);
    @include text-style(sm, 500);
  } @else if $size == lg {
    padding: spacing(3) spacing(6);
    @include text-style(lg, 500);
  } @else {
    padding: spacing(2) spacing(4);
    @include text-style(base, 500);
  }
  
  @if $variant == primary {
    background-color: color(primary);
    color: white;
    
    &:hover {
      background-color: color(primary, 700);
    }
  } @else if $variant == secondary {
    background-color: color(gray, 100);
    color: color(gray, 900);
    
    &:hover {
      background-color: color(gray, 200);
    }
  } @else if $variant == outline {
    background-color: transparent;
    color: color(primary);
    border: 1px solid color(primary);
    
    &:hover {
      background-color: color(primary);
      color: white;
    }
  }
}

// 卡片样式
@mixin card($padding: spacing(6), $shadow: base) {
  background-color: white;
  border-radius: radius(lg);
  box-shadow: shadow($shadow);
  padding: $padding;
}
```

1. **完整的设计系统**：颜色、字体、间距、阴影等系统化管理
2. **响应式设计**：移动优先的响应式布局

4. **主题系统**：支持深色主题切换

6. **可维护性**：清晰的代码结构和命名约定
