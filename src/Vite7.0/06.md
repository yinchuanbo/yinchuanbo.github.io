Vite 7.0 提供了强大的构建优化功能，能够生成高性能的生产环境代码。本文将详细介绍构建配置、优化策略和最佳实践。

## 构建基础配置

### 基本构建选项
```javascript
// vite.config.js
export default defineConfig({
  build: {
    // 构建输出目录
    outDir: 'dist',
    
    // 静态资源目录
    assetsDir: 'assets',
    
    // 小于此阈值的导入或引用资源将内联为 base64 编码
    assetsInlineLimit: 4096,
    
    // 启用/禁用 CSS 代码拆分
    cssCodeSplit: true,
    
    // 构建后是否生成 source map 文件
    sourcemap: false,
    
    // 设置为 false 可以禁用最小化混淆
    minify: 'terser',
    
    // 传递给 Terser 的更多 minify 选项
    terserOptions: {
      compress: {
        drop_console: true,
        drop_debugger: true
      }
    },
    
    // 启用/禁用 gzip 压缩大小报告
    reportCompressedSize: true,
    
    // chunk 大小警告的限制
    chunkSizeWarningLimit: 500
  }
})
```

### 多环境构建配置
```javascript
export default defineConfig(({ command, mode }) => {
  const config = {
    // 基础配置
  }
  
  if (mode === 'production') {
    config.build = {
      ...config.build,
      minify: 'terser',
      sourcemap: false,
      terserOptions: {
        compress: {
          drop_console: true,
          drop_debugger: true
        }
      }
    }
  } else if (mode === 'development') {
    config.build = {
      ...config.build,
      minify: false,
      sourcemap: true
    }
  }
  
  return config
})
```

## 代码分割优化

### 手动代码分割
```javascript
export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        manualChunks: {
          // 将 React 相关库打包成一个 chunk
          'react-vendor': ['react', 'react-dom'],
          
          // 将 Vue 相关库打包成一个 chunk
          'vue-vendor': ['vue', 'vue-router', 'vuex'],
          
          // 将工具库打包成一个 chunk
          'utils': ['lodash', 'axios', 'dayjs'],
          
          // 将 UI 组件库打包成一个 chunk
          'ui-vendor': ['element-plus', 'ant-design-vue']
        }
      }
    }
  }
})
```

### 动态代码分割
```javascript
// 函数式代码分割
export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        manualChunks(id) {
          // 将 node_modules 中的包分离
          if (id.includes('node_modules')) {
            // 将大型库单独分离
            if (id.includes('lodash')) {
              return 'lodash'
            }
            if (id.includes('moment')) {
              return 'moment'
            }
            if (id.includes('echarts')) {
              return 'echarts'
            }
            // 其他第三方库
            return 'vendor'
          }
          
          // 将特定目录的文件分离
          if (id.includes('src/components')) {
            return 'components'
          }
          if (id.includes('src/utils')) {
            return 'utils'
          }
        }
      }
    }
  }
})
```

### CSS 优化
```javascript
export default defineConfig({
  css: {
    // PostCSS 配置
    postcss: {
      plugins: [
        // 自动添加浏览器前缀
        require('autoprefixer'),
        
        // CSS 压缩
        require('cssnano')({
          preset: 'default'
        }),
        
        // 移除未使用的 CSS
        require('@fullhuman/postcss-purgecss')({
          content: ['./src/**/*.{vue,js,ts,jsx,tsx}'],
          defaultExtractor: content => content.match(/[\w-/:]+(?<!:)/g) || []
        })
      ]
    }
  },
  
  build: {
    // CSS 代码分割
    cssCodeSplit: true,
    
    rollupOptions: {
      output: {
        // CSS 文件命名
        assetFileNames: (assetInfo) => {
          if (assetInfo.name.endsWith('.css')) {
            return 'css/[name]-[hash][extname]'
          }
          return 'assets/[name]-[hash][extname]'
        }
      }
    }
  }
})
```

### 性能预算
```javascript
export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        // 控制 chunk 大小
        manualChunks(id) {
          // 确保单个 chunk 不超过 500KB
          if (id.includes('node_modules')) {
            const chunks = ['react', 'lodash', 'moment']
            const chunk = chunks.find(chunk => id.includes(chunk))
            return chunk || 'vendor'
          }
        }
      }
    },
    
    // 设置 chunk 大小警告限制
    chunkSizeWarningLimit: 500
  }
})
```

## 多页面应用构建

### 多页面配置
```javascript
import { defineConfig } from 'vite'
import { resolve } from 'path'

export default defineConfig({
  build: {
    rollupOptions: {
      input: {
        main: resolve(__dirname, 'index.html'),
        admin: resolve(__dirname, 'admin.html'),
        mobile: resolve(__dirname, 'mobile.html')
      },
      
      output: {
        // 为不同页面生成不同的 chunk
        manualChunks: {
          'main-vendor': ['vue', 'vue-router'],
          'admin-vendor': ['vue', 'element-plus'],
          'mobile-vendor': ['vue', 'vant']
        }
      }
    }
  }
})
```

### 页面特定优化
```javascript
export default defineConfig({
  build: {
    rollupOptions: {
      input: {
        main: resolve(__dirname, 'index.html'),
        admin: resolve(__dirname, 'admin.html')
      },
      
      output: {
        manualChunks(id) {
          // 根据页面分离代码
          if (id.includes('src/admin')) {
            return 'admin'
          }
          if (id.includes('src/main')) {
            return 'main'
          }
          
          // 共享依赖
          if (id.includes('node_modules')) {
            return 'vendor'
          }
        }
      }
    }
  }
})
```

## 库模式构建

### 构建 JavaScript 库
```javascript
export default defineConfig({
  build: {
    lib: {
      entry: resolve(__dirname, 'lib/main.js'),
      name: 'MyLib',
      fileName: 'my-lib',
      formats: ['es', 'cjs', 'umd', 'iife']
    },
    
    rollupOptions: {
      // 确保外部化处理那些你不想打包进库的依赖
      external: ['vue', 'lodash'],
      
      output: {
        // 在 UMD 构建模式下为这些外部化的依赖提供一个全局变量
        globals: {
          vue: 'Vue',
          lodash: '_'
        }
      }
    }
  }
})
```

### 缓存优化
```javascript
export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        // 文件名包含 hash 以支持长期缓存
        entryFileNames: 'js/[name]-[hash].js',
        chunkFileNames: 'js/[name]-[hash].js',
        assetFileNames: (assetInfo) => {
          const info = assetInfo.name.split('.')
          const extType = info[info.length - 1]
          
          if (/\.(png|jpe?g|gif|svg|webp|avif)$/i.test(assetInfo.name)) {
            return `images/[name]-[hash][extname]`
          }
          
          if (extType === 'css') {
            return `css/[name]-[hash][extname]`
          }
          
          return `assets/[name]-[hash][extname]`
        }
      }
    }
  }
})
```
