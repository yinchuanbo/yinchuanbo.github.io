Vite 提供了强大的环境变量和模式管理系统，让你能够在不同环境下使用不同的配置。本章将详细介绍如何使用和管理环境变量。

## 环境变量基础

### 内置环境变量
```javascript
// Vite 内置的环境变量
console.log(import.meta.env.MODE)          // 当前模式
console.log(import.meta.env.BASE_URL)      // 基础 URL
console.log(import.meta.env.PROD)          // 是否为生产环境
console.log(import.meta.env.DEV)           // 是否为开发环境
console.log(import.meta.env.SSR)           // 是否为 SSR
```

### 自定义环境变量
```bash
# .env 文件中定义的变量必须以 VITE_ 开头
VITE_API_URL=https://api.example.com
VITE_APP_TITLE=My Awesome App
VITE_FEATURE_FLAG=true
VITE_VERSION=1.0.0
```

```javascript
// 在代码中使用环境变量
const config = {
  apiUrl: import.meta.env.VITE_API_URL,
  appTitle: import.meta.env.VITE_APP_TITLE,
  featureFlag: import.meta.env.VITE_FEATURE_FLAG === 'true',
  version: import.meta.env.VITE_VERSION
}

export default config
```

## 环境文件管理

### 环境文件优先级
```bash
# 文件加载优先级（从高到低）
.env.local          # 本地环境，被 git 忽略
.env.[mode].local   # 特定模式的本地环境
.env.[mode]         # 特定模式环境
.env                # 通用环境
```

### 不同环境的配置文件
```bash
# .env - 所有环境的默认配置
VITE_APP_TITLE=My App
VITE_API_TIMEOUT=5000

# .env.development - 开发环境
VITE_API_URL=http://localhost:8080
VITE_DEBUG=true
VITE_LOG_LEVEL=debug

# .env.production - 生产环境
VITE_API_URL=https://api.production.com
VITE_DEBUG=false
VITE_LOG_LEVEL=error

# .env.staging - 测试环境
VITE_API_URL=https://api.staging.com
VITE_DEBUG=true
VITE_LOG_LEVEL=info

# .env.local - 本地个人配置（不提交到版本控制）
VITE_API_URL=http://192.168.1.100:8080
VITE_MOCK_API=true
```

### 环境变量类型处理
```javascript
// src/utils/env.js
class EnvConfig {
  constructor() {
    this.apiUrl = import.meta.env.VITE_API_URL
    this.appTitle = import.meta.env.VITE_APP_TITLE
    this.debug = this.parseBoolean(import.meta.env.VITE_DEBUG)
    this.timeout = this.parseNumber(import.meta.env.VITE_API_TIMEOUT)
    this.features = this.parseArray(import.meta.env.VITE_FEATURES)
  }
  
  parseBoolean(value) {
    return value === 'true'
  }
  
  parseNumber(value) {
    const num = parseInt(value, 10)
    return isNaN(num) ? 0 : num
  }
  
  parseArray(value) {
    return value ? value.split(',').map(item => item.trim()) : []
  }
  
  // 验证必需的环境变量
  validate() {
    const required = ['VITE_API_URL', 'VITE_APP_TITLE']
    const missing = required.filter(key => !import.meta.env[key])
    
    if (missing.length > 0) {
      throw new Error(`Missing required environment variables: ${missing.join(', ')}`)
    }
  }
}

export const env = new EnvConfig()
```

## 模式管理

### 内置模式
```bash
# 开发模式
npm run dev
# 等同于
vite --mode development

# 生产模式
npm run build
# 等同于
vite build --mode production

# 预览模式
npm run preview
# 等同于
vite preview --mode production
```

### 自定义模式
```bash
# 创建自定义模式
npm run dev -- --mode staging
npm run build -- --mode testing

# 在 package.json 中定义脚本
{
  "scripts": {
    "dev": "vite",
    "dev:staging": "vite --mode staging",
    "build": "vite build",
    "build:staging": "vite build --mode staging",
    "build:testing": "vite build --mode testing"
  }
}
```

### 模式特定配置
```javascript
// vite.config.js
export default defineConfig(({ command, mode }) => {
  // 加载环境变量
  const env = loadEnv(mode, process.cwd(), '')
  
  const config = {
    plugins: [vue()],
    define: {
      __APP_VERSION__: JSON.stringify(process.env.npm_package_version)
    }
  }
  
  // 根据模式调整配置
  if (mode === 'development') {
    config.server = {
      port: 3000,
      proxy: {
        '/api': env.VITE_API_URL
      }
    }
  }
  
  if (mode === 'staging') {
    config.build = {
      sourcemap: true,
      minify: false
    }
    config.server = {
      port: 4000
    }
  }
  
  if (mode === 'production') {
    config.build = {
      sourcemap: false,
      minify: 'terser'
    }
  }
  
  return config
})
```

## 环境变量的安全性

### 敏感信息处理
```bash
# .env.example - 提供环境变量模板
VITE_API_URL=https://api.example.com
VITE_APP_TITLE=My App
VITE_FEATURE_FLAG=false

# 敏感信息不要放在 VITE_ 开头的变量中
# ❌ 错误 - 这些会暴露给客户端
VITE_API_SECRET=secret-key
VITE_DATABASE_PASSWORD=password

# ✅ 正确 - 服务端环境变量
API_SECRET=secret-key
DATABASE_PASSWORD=password
```

### 环境变量验证
```javascript
// src/utils/envValidator.js
const requiredEnvVars = [
  'VITE_API_URL',
  'VITE_APP_TITLE'
]

const optionalEnvVars = [
  'VITE_DEBUG',
  'VITE_ANALYTICS_ID'
]

export function validateEnvironment() {
  const missing = requiredEnvVars.filter(
    varName => !import.meta.env[varName]
  )
  
  if (missing.length > 0) {
    throw new Error(
      `Missing required environment variables: ${missing.join(', ')}\n` +
      'Please check your .env file and ensure all required variables are set.'
    )
  }
  
  // 验证 URL 格式
  if (import.meta.env.VITE_API_URL) {
    try {
      new URL(import.meta.env.VITE_API_URL)
    } catch {
      throw new Error('VITE_API_URL must be a valid URL')
    }
  }
  
  console.log('Environment validation passed')
}

// 在应用启动时验证
validateEnvironment()
```

## 动态环境配置

### 运行时配置加载
```javascript
// public/config.js - 运行时配置
window.__APP_CONFIG__ = {
  apiUrl: 'https://api.runtime.com',
  features: {
    newFeature: true,
    betaMode: false
  }
}
```

```javascript
// src/utils/runtimeConfig.js
class RuntimeConfig {
  constructor() {
    this.config = {
      // 默认配置
      apiUrl: import.meta.env.VITE_API_URL,
      features: {}
    }
    
    // 合并运行时配置
    if (window.__APP_CONFIG__) {
      this.config = {
        ...this.config,
        ...window.__APP_CONFIG__
      }
    }
  }
  
  get(key, defaultValue = null) {
    return this.config[key] || defaultValue
  }
  
  set(key, value) {
    this.config[key] = value
  }
  
  async loadRemoteConfig() {
    try {
      const response = await fetch('/api/config')
      const remoteConfig = await response.json()
      this.config = { ...this.config, ...remoteConfig }
    } catch (error) {
      console.warn('Failed to load remote config:', error)
    }
  }
}

export const runtimeConfig = new RuntimeConfig()
```

### 配置热更新
```javascript
// src/utils/configWatcher.js
class ConfigWatcher {
  constructor() {
    this.listeners = []
    this.config = {}
    
    if (import.meta.hot) {
      import.meta.hot.accept('./config.js', (newModule) => {
        this.updateConfig(newModule.default)
      })
    }
  }
  
  updateConfig(newConfig) {
    const oldConfig = { ...this.config }
    this.config = newConfig
    
    this.listeners.forEach(listener => {
      listener(newConfig, oldConfig)
    })
  }
  
  subscribe(listener) {
    this.listeners.push(listener)
    
    return () => {
      const index = this.listeners.indexOf(listener)
      if (index > -1) {
        this.listeners.splice(index, 1)
      }
    }
  }
}

export const configWatcher = new ConfigWatcher()
```

export function debugEnvironment() {

## 总结

Vite 的环境变量和模式系统提供了：
- 灵活的环境配置管理
- 安全的客户端变量暴露机制
- 强大的模式切换能力
- 完善的调试和验证工具

掌握这些特性，能够帮助你构建更加灵活和可维护的应用配置系统。
