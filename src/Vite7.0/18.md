```javascript
// ✅ 良好的配置设计
export function myPlugin(options = {}) {
  // 默认配置
  const defaultOptions = {
    include: /\.(js|ts|vue)$/,
    exclude: /node_modules/,
    transformOptions: {
      removeComments: false,
      minify: false,
      sourcemap: true
    },
    outputOptions: {
      generateReport: false,
      reportPath: 'reports/plugin-report.json'
    }
  }
  
  // 合并配置
  const config = mergeConfig(defaultOptions, options)
  
  // 验证配置
  validateConfig(config)
  
  return {
    name: 'my-plugin',
    
    configResolved(resolvedConfig) {
      // 根据 Vite 配置调整插件配置
      if (resolvedConfig.command === 'build') {
        config.transformOptions.minify = config.transformOptions.minify ?? true
      }
    },
    
    transform(code, id) {
      if (!shouldTransform(id, config.include, config.exclude)) {
        return null
      }
      
      return transformCode(code, id, config.transformOptions)
    }
  }
}

function mergeConfig(defaults, user) {
  return {
    ...defaults,
    ...user,
    transformOptions: {
      ...defaults.transformOptions,
      ...user.transformOptions
    },
    outputOptions: {
      ...defaults.outputOptions,
      ...user.outputOptions
    }
  }
}

function validateConfig(config) {
  if (!config.include || !(config.include instanceof RegExp)) {
    throw new Error('include option must be a RegExp')
  }
  
  if (typeof config.transformOptions !== 'object') {
    throw new Error('transformOptions must be an object')
  }
}

function shouldTransform(id, include, exclude) {
  if (exclude && exclude.test(id)) return false
  return include.test(id)
}
```

## 性能优化最佳实践

### 缓存策略
```javascript
// ✅ 实现高效的缓存机制
export function cachedTransformPlugin(options = {}) {
  const cache = new Map()
  const { cacheSize = 1000, ttl = 300000 } = options // 5分钟TTL
  
  return {
    name: 'cached-transform-plugin',
    
    buildStart() {
      // 清理过期缓存
      this.cleanupCache()
    },
    
    transform(code, id) {
      const cacheKey = this.getCacheKey(code, id)
      const cached = cache.get(cacheKey)
      
      // 检查缓存
      if (cached && Date.now() - cached.timestamp < ttl) {
        return cached.result
      }
      
      // 执行转换
      const result = this.performTransform(code, id)
      
      // 存储到缓存
      this.setCache(cacheKey, result)
      
      return result
    },
    
    getCacheKey(code, id) {
      // 生成缓存键
      const hash = this.hashCode(code + id)
      return `${id}:${hash}`
    },
    
    setCache(key, result) {
      // 限制缓存大小
      if (cache.size >= cacheSize) {
        const firstKey = cache.keys().next().value
        cache.delete(firstKey)
      }
      
      cache.set(key, {
        result,
        timestamp: Date.now()
      })
    },
    
    cleanupCache() {
      const now = Date.now()
      for (const [key, value] of cache.entries()) {
        if (now - value.timestamp > ttl) {
          cache.delete(key)
        }
      }
    },
    
    hashCode(str) {
      let hash = 0
      for (let i = 0; i < str.length; i++) {
        const char = str.charCodeAt(i)
        hash = ((hash << 5) - hash) + char
        hash = hash & hash // 转换为32位整数
      }
      return hash.toString()
    },
    
    performTransform(code, id) {
      // 实际的转换逻辑
      return {
        code: code.replace(/console\.log/g, 'console.debug'),
        map: null
      }
    }
  }
}
```

### 异步处理优化
```javascript
// ✅ 高效的异步处理
export function asyncProcessPlugin(options = {}) {
  const { concurrency = 4, timeout = 30000 } = options
  const processingQueue = []
  let activeProcesses = 0
  
  return {
    name: 'async-process-plugin',
    
    async transform(code, id) {
      if (!this.shouldProcess(id)) return null
      
      // 控制并发数
      if (activeProcesses >= concurrency) {
        await this.waitForSlot()
      }
      
      activeProcesses++
      
      try {
        const result = await Promise.race([
          this.processCode(code, id),
          this.createTimeout(timeout)
        ])
        
        return result
      } catch (error) {
        this.warn(`Processing failed for ${id}: ${error.message}`)
        return null
      } finally {
        activeProcesses--
        this.processNext()
      }
    },
    
    async waitForSlot() {
      return new Promise(resolve => {
        processingQueue.push(resolve)
      })
    },
    
    processNext() {
      if (processingQueue.length > 0) {
        const next = processingQueue.shift()
        next()
      }
    },
    
    createTimeout(ms) {
      return new Promise((_, reject) => {
        setTimeout(() => reject(new Error('Processing timeout')), ms)
      })
    },
    
    async processCode(code, id) {
      // 模拟异步处理
      await new Promise(resolve => setTimeout(resolve, 100))
      
      return {
        code: await this.transformAsync(code),
        map: null
      }
    },
    
    async transformAsync(code) {
      // 异步转换逻辑
      return code.toUpperCase()
    },
    
    shouldProcess(id) {
      return id.endsWith('.special')
    }
  }
}
```

### 内存管理
```javascript
// ✅ 良好的内存管理
export function memoryEfficientPlugin(options = {}) {
  const { maxMemoryUsage = 100 * 1024 * 1024 } = options // 100MB
  let memoryUsage = 0
  const resources = new Set()
  
  return {
    name: 'memory-efficient-plugin',
    
    buildStart() {
      // 监控内存使用
      this.startMemoryMonitoring()
    },
    
    transform(code, id) {
      // 检查内存使用
      if (memoryUsage > maxMemoryUsage) {
        this.cleanup()
      }
      
      const resource = this.createResource(code, id)
      resources.add(resource)
      memoryUsage += this.estimateSize(resource)
      
      return this.processResource(resource)
    },
    
    buildEnd() {
      // 清理资源
      this.cleanup()
      this.stopMemoryMonitoring()
    },
    
    createResource(code, id) {
      return {
        id,
        code,
        timestamp: Date.now(),
        processed: false
      }
    },
    
    processResource(resource) {
      const result = {
        code: resource.code.replace(/\s+/g, ' '),
        map: null
      }
      
      resource.processed = true
      return result
    },
    
    cleanup() {
      // 清理已处理的资源
      for (const resource of resources) {
        if (resource.processed) {
          resources.delete(resource)
          memoryUsage -= this.estimateSize(resource)
        }
      }
      
      // 强制垃圾回收（如果可用）
      if (global.gc) {
        global.gc()
      }
    },
    
    estimateSize(resource) {
      // 估算资源大小
      return JSON.stringify(resource).length * 2 // 粗略估算
    },
    
    startMemoryMonitoring() {
      this.memoryInterval = setInterval(() => {
        const usage = process.memoryUsage()
        if (usage.heapUsed > maxMemoryUsage) {
          this.warn('High memory usage detected, cleaning up...')
          this.cleanup()
        }
      }, 5000)
    },
    
    stopMemoryMonitoring() {
      if (this.memoryInterval) {
        clearInterval(this.memoryInterval)
      }
    }
  }
}
```

## 错误处理和调试

### 完善的错误处理
```javascript
// ✅ 健壮的错误处理
export function robustPlugin(options = {}) {
  const { strict = false, fallback = true } = options
  
  return {
    name: 'robust-plugin',
    
    transform(code, id) {
      try {
        return this.safeTransform(code, id)
      } catch (error) {
        return this.handleError(error, code, id)
      }
    },
    
    safeTransform(code, id) {
      // 输入验证
      if (!code || typeof code !== 'string') {
        throw new Error(`Invalid code input for ${id}`)
      }
      
      if (!id || typeof id !== 'string') {
        throw new Error('Invalid id input')
      }
      
      // 执行转换
      const result = this.performTransform(code, id)
      
      // 结果验证
      if (!result || typeof result.code !== 'string') {
        throw new Error(`Transform produced invalid result for ${id}`)
      }
      
      return result
    },
    
    handleError(error, code, id) {
      const errorInfo = {
        message: error.message,
        stack: error.stack,
        file: id,
        timestamp: new Date().toISOString()
      }
      
      // 记录错误
      this.logError(errorInfo)
      
      if (strict) {
        // 严格模式下抛出错误
        this.error(`Transform failed for ${id}: ${error.message}`, {
          id,
          loc: this.extractLocation(error),
          frame: this.generateFrame(code, error)
        })
      } else {
        // 非严格模式下发出警告
        this.warn(`Transform failed for ${id}: ${error.message}`)
        
        if (fallback) {
          // 返回原始代码作为后备
          return { code, map: null }
        }
      }
      
      return null
    },
    
    logError(errorInfo) {
      // 错误日志记录
      const logFile = path.join(process.cwd(), 'logs', 'plugin-errors.log')
      
      try {
        fs.appendFileSync(logFile, JSON.stringify(errorInfo) + '\n')
      } catch (logError) {
        console.error('Failed to write error log:', logError)
      }
    },
    
    extractLocation(error) {
      // 从错误中提取位置信息
      const match = error.stack?.match(/:(\d+):(\d+)/)
      if (match) {
        return {
          line: parseInt(match[1]),
          column: parseInt(match[2])
        }
      }
      return null
    },
    
    generateFrame(code, error) {
      // 生成错误上下文
      const lines = code.split('\n')
      const location = this.extractLocation(error)
      
      if (!location || location.line > lines.length) {
        return code.split('\n').slice(0, 5).join('\n')
      }
      
      const start = Math.max(0, location.line - 3)
      const end = Math.min(lines.length, location.line + 2)
      
      return lines.slice(start, end)
        .map((line, index) => {
          const lineNum = start + index + 1
          const marker = lineNum === location.line ? '>' : ' '
          return `${marker} ${lineNum} | ${line}`
        })
        .join('\n')
    },
    
    performTransform(code, id) {
      // 实际转换逻辑
      return {
        code: code.replace(/old/g, 'new'),
        map: null
      }
    }
  }
}
```

### 调试支持
```javascript
// ✅ 内置调试支持
export function debuggablePlugin(options = {}) {
  const { debug = false, verbose = false } = options
  const debugInfo = {
    transformCount: 0,
    errors: [],
    warnings: [],
    performance: {}
  }
  
  return {
    name: 'debuggable-plugin',
    
    configResolved(config) {
      // 根据环境启用调试
      if (config.command === 'serve' || process.env.DEBUG) {
        this.debug = true
      }
    },
    
    buildStart() {
      if (this.debug) {
        console.log('[debuggable-plugin] Build started')
        debugInfo.buildStartTime = Date.now()
      }
    },
    
    transform(code, id) {
      const startTime = Date.now()
      
      try {
        const result = this.performTransform(code, id)
        
        // 记录性能信息
        this.recordPerformance(id, Date.now() - startTime)
        
        if (this.debug && verbose) {
          console.log(`[debuggable-plugin] Transformed ${id} in ${Date.now() - startTime}ms`)
        }
        
        return result
      } catch (error) {
        debugInfo.errors.push({ id, error: error.message, timestamp: Date.now() })
        throw error
      }
    },
    
    buildEnd() {
      if (this.debug) {
        const duration = Date.now() - debugInfo.buildStartTime
        console.log(`[debuggable-plugin] Build completed in ${duration}ms`)
        
        if (verbose) {
          this.printDebugSummary()
        }
      }
    },
    
    recordPerformance(id, duration) {
      debugInfo.transformCount++
      debugInfo.performance[id] = duration
    },
    
    printDebugSummary() {
      console.log('\n=== Plugin Debug Summary ===')
      console.log(`Transforms: ${debugInfo.transformCount}`)
      console.log(`Errors: ${debugInfo.errors.length}`)
      console.log(`Warnings: ${debugInfo.warnings.length}`)
      
      // 性能统计
      const durations = Object.values(debugInfo.performance)
      if (durations.length > 0) {
        const avg = durations.reduce((a, b) => a + b, 0) / durations.length
        const max = Math.max(...durations)
        const min = Math.min(...durations)
        
        console.log(`Performance - Avg: ${avg.toFixed(2)}ms, Max: ${max}ms, Min: ${min}ms`)
      }
      
      // 错误详情
      if (debugInfo.errors.length > 0) {
        console.log('\nErrors:')
        debugInfo.errors.forEach(error => {
          console.log(`  ${error.id}: ${error.error}`)
        })
      }
      
      console.log('========================\n')
    },
    
    performTransform(code, id) {
      // 模拟转换
      return {
        code: code.toUpperCase(),
        map: null
      }
    }
  }
}
```

```javascript
// ✅ 处理版本兼容性
export function compatiblePlugin(options = {}) {
  let viteVersion
  let isLegacyVite = false
  
  return {
    name: 'compatible-plugin',
    
    configResolved(config) {
      // 检测 Vite 版本
      viteVersion = this.detectViteVersion()
      isLegacyVite = this.isLegacyVersion(viteVersion)
      
      if (isLegacyVite) {
        this.warn(`Using legacy Vite version ${viteVersion}. Some features may be limited.`)
      }
    },
    
    transform(code, id) {
      if (isLegacyVite) {
        return this.legacyTransform(code, id)
      } else {
        return this.modernTransform(code, id)
      }
    },
    
    detectViteVersion() {
      try {
        const vitePackage = require('vite/package.json')
        return vitePackage.version
      } catch {
        return '0.0.0'
      }
    },
    
    isLegacyVersion(version) {
      const [major, minor] = version.split('.').map(Number)
      return major < 5 || (major === 5 && minor < 0)
    },
    
    legacyTransform(code, id) {
      // 兼容旧版本的转换逻辑
      return {
        code: code.replace(/import\.meta\.env/g, 'process.env'),
        map: null
      }
    },
    
    modernTransform(code, id) {
      // 现代版本的转换逻辑
      return {
        code: code.replace(/process\.env/g, 'import.meta.env'),
        map: null
      }
    }
  }
}
```

## 配置选项

### include
- **类型**: `RegExp | RegExp[]`
- **默认值**: `/\.(js|ts|vue)$/`
- **描述**: 需要处理的文件模式

### exclude
- **类型**: `RegExp | RegExp[]`
- **默认值**: `/node_modules/`
- **描述**: 需要排除的文件模式
