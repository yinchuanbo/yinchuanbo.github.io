Vite 使用 Rollup 作为生产构建工具，提供高度优化的构建输出。本章将详细介绍如何配置和优化生产构建。

## 构建基础

### 基本构建命令
```bash
# 构建生产版本
npm run build

# 构建并分析包大小
npm run build -- --analyze

# 构建特定环境
npm run build -- --mode production
```

### 构建输出结构
```
dist/
├── assets/
│   ├── index-a1b2c3d4.js      # 主入口文件
│   ├── vendor-e5f6g7h8.js     # 第三方依赖
│   ├── style-i9j0k1l2.css     # 样式文件
│   └── logo-m3n4o5p6.png      # 静态资源
├── index.html                  # HTML 入口
└── favicon.ico                 # 网站图标
```

## 构建配置

### 基础构建选项
```javascript
// vite.config.js
export default defineConfig({
  build: {
    // 输出目录
    outDir: 'dist',
    
    // 静态资源目录
    assetsDir: 'assets',
    
    // 小于此阈值的导入或引用资源将内联为 base64 编码
    assetsInlineLimit: 4096,
    
    // 启用/禁用 CSS 代码拆分
    cssCodeSplit: true,
    
    // 构建后是否生成 source map
    sourcemap: false,
    
    // 设置为 false 可以禁用最小化混淆
    minify: 'esbuild', // 'terser' | 'esbuild' | false
    
    // 传递给 Terser 的更多 minify 选项
    terserOptions: {
      compress: {
        drop_console: true,
        drop_debugger: true
      }
    }
  }
})
```

### 高级构建配置
```javascript
export default defineConfig({
  build: {
    // 自定义底层的 Rollup 打包配置
    rollupOptions: {
      input: {
        main: resolve(__dirname, 'index.html'),
        admin: resolve(__dirname, 'admin.html')
      },
      output: {
        // 分包策略
        manualChunks: {
          vendor: ['vue', 'vue-router'],
          utils: ['lodash-es', 'axios']
        },
        // 自定义文件名
        chunkFileNames: 'js/[name]-[hash].js',
        entryFileNames: 'js/[name]-[hash].js',
        assetFileNames: 'assets/[name]-[hash].[ext]'
      }
    },
    
    // 库模式配置
    lib: {
      entry: resolve(__dirname, 'src/main.js'),
      name: 'MyLib',
      fileName: 'my-lib'
    },
    
    // 构建目标
    target: 'es2015', // 'es2015' | 'es2017' | 'es2018' | 'es2019' | 'es2020' | 'esnext'
    
    // 警告阈值
    chunkSizeWarningLimit: 500,
    
    // 启用/禁用 gzip 压缩大小报告
    reportCompressedSize: true,
    
    // 调整 chunk 大小警告的限制
    chunkSizeWarningLimit: 1000
  }
})
```

## 代码分割

### 自动代码分割
```javascript
// 动态导入自动创建分割点

// 路由级别的代码分割
const routes = [
  {
    path: '/home',
    component: () => import('./views/Home.vue')
  },
  {
    path: '/about',
    component: () => import('./views/About.vue')
  }
]
```

### 手动分包配置
```javascript
export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        manualChunks: (id) => {
          // 将 node_modules 中的包分离到 vendor chunk
          if (id.includes('node_modules')) {
            // 进一步细分第三方库
            if (id.includes('vue')) {
              return 'vue-vendor'
            }
            if (id.includes('lodash')) {
              return 'lodash-vendor'
            }
            return 'vendor'
          }
          
          // 将工具函数分离到单独的 chunk
          if (id.includes('/src/utils/')) {
            return 'utils'
          }
        }
      }
    }
  }
})
```

### 预加载策略

```javascript
// vite.config.js
```

## 资源处理

### 静态资源优化
```javascript
export default defineConfig({
  build: {
    // 小文件内联阈值
    assetsInlineLimit: 4096,
    
    rollupOptions: {
      output: {
        // 资源文件命名
        assetFileNames: (assetInfo) => {
          const info = assetInfo.name.split('.')
          const ext = info[info.length - 1]
          
          if (/\.(png|jpe?g|gif|svg)$/.test(assetInfo.name)) {
            return `images/[name]-[hash].${ext}`
          }
          if (/\.(woff2?|eot|ttf|otf)$/.test(assetInfo.name)) {
            return `fonts/[name]-[hash].${ext}`
          }
          return `assets/[name]-[hash].${ext}`
        }
      }
    }
  }
})
```

## 环境配置

### 多环境构建
```javascript
// vite.config.js
export default defineConfig(({ command, mode }) => {
  const config = {
    plugins: [vue()],
    build: {
      outDir: 'dist'
    }
  }
  
  if (mode === 'production') {
    config.build.minify = 'terser'
    config.build.terserOptions = {
      compress: {
        drop_console: true,
        drop_debugger: true
      }
    }
  }
  
  if (mode === 'staging') {
    config.build.outDir = 'dist-staging'
    config.build.sourcemap = true
  }
  
  return config
})
```

## 故障排除

### 常见构建问题
```javascript
// 1. 内存不足
export default defineConfig({
  build: {
    rollupOptions: {
      // 减少并行处理
      maxParallelFileOps: 2
    }
  }
})

// 2. 循环依赖警告
export default defineConfig({
  build: {
    rollupOptions: {
      onwarn(warning, warn) {
        // 忽略循环依赖警告
        if (warning.code === 'CIRCULAR_DEPENDENCY') return
        warn(warning)
      }
    }
  }
})
```
