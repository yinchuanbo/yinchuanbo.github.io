Vite 7.0 拥有强大的插件系统，基于 Rollup 插件架构，提供了丰富的扩展能力。本文将详细介绍插件系统的使用方法和常用插件。

## 插件系统概述

### 插件架构
Vite 插件系统基于 Rollup 插件，同时扩展了开发服务器相关的钩子：

```javascript
// vite.config.js
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import { resolve } from 'path'

export default defineConfig({
  plugins: [
    vue(),
    // 其他插件
  ]
})
```

### 插件应用条件
```javascript
export default defineConfig({
  plugins: [
    // 总是应用
    vue(),
    
    // 仅在开发环境应用
    {
      ...mockPlugin(),
      apply: 'build'
    },
    
    // 条件应用
    ...(process.env.NODE_ENV === 'development' ? [mockPlugin()] : [])
  ]
})
```

## 自定义插件开发

### 简单插件示例
```javascript
// plugins/hello-plugin.js
export function helloPlugin() {
  return {
    name: 'hello-plugin',
    
    // 构建开始时调用
    buildStart() {
      console.log('Hello from Vite plugin!')
    },
    
    // 转换代码
    transform(code, id) {
      if (id.endsWith('.hello')) {
        return `export default ${JSON.stringify(code)}`
      }
    },
    
    // 开发服务器钩子
    configureServer(server) {
      server.middlewares.use('/api/hello', (req, res, next) => {
        res.end('Hello from plugin middleware!')
      })
    }
  }
}

// 使用插件
import { helloPlugin } from './plugins/hello-plugin.js'

export default defineConfig({
  plugins: [
    helloPlugin()
  ]
})
```

### 带选项的插件
```javascript
// plugins/custom-plugin.js
export function customPlugin(options = {}) {
  const {
    include = /\.(vue|js|ts)$/,
    exclude = /node_modules/,
    transformOptions = {}
  } = options
  
  return {
    name: 'custom-plugin',
    
    // 插件配置
    config(config, { command }) {
      if (command === 'serve') {
        // 开发环境配置
        config.define = config.define || {}
        config.define.__DEV__ = true
      }
    },
    
    // 解析模块
    resolveId(id) {
      if (id === 'virtual:my-module') {
        return id
      }
    },
    
    // 加载模块
    load(id) {
      if (id === 'virtual:my-module') {
        return 'export const msg = "Hello from virtual module"'
      }
    },
    
    // 转换代码
    transform(code, id) {
      if (include.test(id) && !exclude.test(id)) {
        // 执行代码转换
        return {
          code: transformCode(code, transformOptions),
          map: null // source map
        }
      }
    }
  }
}

function transformCode(code, options) {
  // 实现代码转换逻辑
  return code.replace(/console\.log/g, 'console.debug')
}
```

### 异步插件
```javascript
export function asyncPlugin() {
  return {
    name: 'async-plugin',
    
    async buildStart() {
      // 异步初始化
      const config = await loadConfig()
      this.config = config
    },
    
    async transform(code, id) {
      if (id.endsWith('.special')) {
        // 异步转换
        const result = await processSpecialFile(code)
        return result
      }
    }
  }
}

async function loadConfig() {
  // 模拟异步配置加载
  return new Promise(resolve => {
    setTimeout(() => {
      resolve({ version: '1.0.0' })
    }, 100)
  })
}

async function processSpecialFile(code) {
  // 模拟异步文件处理
  return new Promise(resolve => {
    setTimeout(() => {
      resolve(`// Processed\n${code}`)
    }, 50)
  })
}
```

## 插件钩子详解

### 通用钩子
```javascript
export function fullFeaturedPlugin() {
  return {
    name: 'full-featured-plugin',
    
    // 构建开始
    buildStart(opts) {
      console.log('Build started with options:', opts)
    },
    
    // 解析模块 ID
    resolveId(id, importer) {
      if (id === 'virtual:config') {
        return id
      }
    },
    
    // 加载模块
    load(id) {
      if (id === 'virtual:config') {
        return `export default ${JSON.stringify(this.config)}`
      }
    },
    
    // 转换代码
    transform(code, id) {
      if (id.includes('transform-me')) {
        return {
          code: `// Transformed\n${code}`,
          map: null
        }
      }
    },
    
    // 生成 bundle
    generateBundle(options, bundle) {
      // 修改生成的 bundle
      Object.keys(bundle).forEach(fileName => {
        const file = bundle[fileName]
        if (file.type === 'chunk') {
          file.code = `/* Generated by plugin */\n${file.code}`
        }
      })
    },
    
    // 构建结束
    buildEnd(error) {
      if (error) {
        console.error('Build failed:', error)
      } else {
        console.log('Build completed successfully')
      }
    }
  }
}
```

### Vite 特有钩子
```javascript
export function viteSpecificPlugin() {
  return {
    name: 'vite-specific-plugin',
    
    // 配置解析前
    config(config, { command }) {
      if (command === 'serve') {
        config.server = config.server || {}
        config.server.port = 3000
      }
    },
    
    // 配置解析后
    configResolved(resolvedConfig) {
      this.isProduction = resolvedConfig.command === 'build'
    },
    
    // 配置开发服务器
    configureServer(server) {
      server.middlewares.use('/api', (req, res, next) => {
        if (req.url === '/api/health') {
          res.end('OK')
        } else {
          next()
        }
      })
    },
    
    // 配置预览服务器
    configurePreviewServer(server) {
      server.middlewares.use('/preview-api', (req, res, next) => {
        res.end('Preview API')
      })
    },
    
    // 处理 HMR 更新
    handleHotUpdate(ctx) {
      if (ctx.file.endsWith('.special')) {
        // 自定义 HMR 逻辑
        ctx.server.ws.send({
          type: 'full-reload'
        })
        return []
      }
    }
  }
}
```

## 插件最佳实践

### 插件组合
```javascript
// plugins/index.js
import vue from '@vitejs/plugin-vue'
import { resolve } from 'path'

export function createPlugins(env) {
  const plugins = [vue()]
  
  if (env === 'development') {
    plugins.push(
      // 开发环境插件
      mockPlugin(),
      eslintPlugin()
    )
  }
  
  if (env === 'production') {
    plugins.push(
      // 生产环境插件
      bundleAnalyzerPlugin(),
      compressionPlugin()
    )
  }
  
  return plugins
}

// vite.config.js
import { createPlugins } from './plugins'

export default defineConfig({
  plugins: createPlugins(process.env.NODE_ENV)
})
```
