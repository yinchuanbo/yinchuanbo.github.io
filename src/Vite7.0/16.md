## 插件开发基础

### 插件结构
```javascript
// plugins/my-plugin.js
export function myPlugin(options = {}) {
  // 插件选项处理
  const {
    include = /\.(vue|js|ts)$/,
    exclude = /node_modules/,
    transformOptions = {}
  } = options
  
  // 插件状态
  let config
  let isProduction = false
  
  return {
    // 插件名称（必需）
    name: 'my-plugin',
    
    // 插件应用条件
    apply: 'build', // 'serve' | 'build' | (config, { command }) => boolean
    
    // 插件执行顺序
    enforce: 'pre', // 'pre' | 'post'
    
    // 配置钩子
    config(userConfig, { command }) {
      // 修改配置
      isProduction = command === 'build'
      
      if (command === 'serve') {
        // 开发环境配置
        userConfig.server = userConfig.server || {}
        userConfig.server.port = userConfig.server.port || 3000
      }
    },
    
    configResolved(resolvedConfig) {
      // 保存解析后的配置
      config = resolvedConfig
    },
    
    // 构建钩子
    buildStart(opts) {
      console.log('Plugin build started')
    },
    
    resolveId(id, importer) {
      // 解析模块 ID
      if (id === 'virtual:my-module') {
        return id
      }
    },
    
    load(id) {
      // 加载模块
      if (id === 'virtual:my-module') {
        return 'export const message = "Hello from virtual module"'
      }
    },
    
    transform(code, id) {
      // 转换代码
      if (include.test(id) && !exclude.test(id)) {
        return transformCode(code, id, transformOptions)
      }
    },
    
    generateBundle(options, bundle) {
      // 生成构建包
      console.log('Generating bundle with', Object.keys(bundle).length, 'files')
    }
  }
}

function transformCode(code, id, options) {
  // 实际的代码转换逻辑
  return {
    code: code.replace(/console\.log/g, 'console.debug'),
    map: null // source map
  }
}
```

### 插件工厂模式
```javascript
// plugins/plugin-factory.js
export class PluginFactory {
  constructor(baseOptions = {}) {
    this.baseOptions = baseOptions
  }
  
  create(userOptions = {}) {
    const options = { ...this.baseOptions, ...userOptions }
    
    return {
      name: `${options.name || 'custom'}-plugin`,
      
      // 使用工厂方法创建的插件逻辑
      transform(code, id) {
        return this.processCode(code, id, options)
      },
      
      processCode(code, id, opts) {
        // 可复用的代码处理逻辑
        let result = code
        
        if (opts.removeComments) {
          result = this.removeComments(result)
        }
        
        if (opts.minify) {
          result = this.minifyCode(result)
        }
        
        return { code: result, map: null }
      },
      
      removeComments(code) {
        return code.replace(/\/\*[\s\S]*?\*\/|\/\/.*$/gm, '')
      },
      
      minifyCode(code) {
        // 简单的代码压缩
        return code
          .replace(/\s+/g, ' ')
          .replace(/;\s*}/g, '}')
          .trim()
      }
    }
  }
}

// 使用示例
const factory = new PluginFactory({
  removeComments: true,
  minify: false
})

export default defineConfig({
  plugins: [
    factory.create({
      name: 'my-custom',
      minify: true
    })
  ]
})
```

## 实用插件开发示例

### 环境变量注入插件
```javascript
// plugins/env-injector.js
import fs from 'fs'
import path from 'path'

export function envInjectorPlugin(options = {}) {
  const {
    envDir = process.cwd(),
    envPrefix = 'VITE_',
    envFiles = ['.env', '.env.local'],
    injectMode = 'replace' // 'replace' | 'define'
  } = options
  
  let envVars = {}
  
  return {
    name: 'env-injector',
    
    configResolved(config) {
      // 加载环境变量文件
      envFiles.forEach(file => {
        const filePath = path.resolve(envDir, file)
        if (fs.existsSync(filePath)) {
          const content = fs.readFileSync(filePath, 'utf-8')
          const vars = parseEnvFile(content)
          envVars = { ...envVars, ...vars }
        }
      })
      
      // 过滤前缀
      envVars = Object.fromEntries(
        Object.entries(envVars).filter(([key]) => key.startsWith(envPrefix))
      )
    },
    
    transform(code, id) {
      if (id.includes('node_modules')) return null
      
      let transformedCode = code
      let hasChanges = false
      
      if (injectMode === 'replace') {
        // 直接替换模式
        Object.entries(envVars).forEach(([key, value]) => {
          const pattern = new RegExp(`import\\.meta\\.env\\.${key}`, 'g')
          if (pattern.test(transformedCode)) {
            transformedCode = transformedCode.replace(pattern, JSON.stringify(value))
            hasChanges = true
          }
        })
      } else if (injectMode === 'define') {
        // 定义模式 - 在代码顶部注入
        const definitions = Object.entries(envVars)
          .map(([key, value]) => `const ${key} = ${JSON.stringify(value)};`)
          .join('\n')
        
        if (definitions && !transformedCode.includes('// ENV_INJECTED')) {
          transformedCode = `// ENV_INJECTED\n${definitions}\n${transformedCode}`
          hasChanges = true
        }
      }
      
      return hasChanges ? { code: transformedCode, map: null } : null
    }
  }
}

function parseEnvFile(content) {
  const vars = {}
  const lines = content.split('\n')
  
  lines.forEach(line => {
    line = line.trim()
    if (line && !line.startsWith('#')) {
      const [key, ...valueParts] = line.split('=')
      if (key && valueParts.length > 0) {
        let value = valueParts.join('=').trim()
        
        // 移除引号
        if ((value.startsWith('"') && value.endsWith('"')) ||
            (value.startsWith("'") && value.endsWith("'"))) {
          value = value.slice(1, -1)
        }
        
        vars[key.trim()] = value
      }
    }
  })
  
  return vars
}

// 使用示例
export default defineConfig({
  plugins: [
    envInjectorPlugin({
      envFiles: ['.env', '.env.local', '.env.development'],
      injectMode: 'replace'
    })
  ]
})
```

### 文件监听和自动生成插件
```javascript
// plugins/auto-generator.js
import fs from 'fs'
import path from 'path'
import { watch } from 'chokidar'

export function autoGeneratorPlugin(options = {}) {
  const {
    watchDirs = ['src/api'],
    outputFile = 'src/types/api.d.ts',
    patterns = ['**/*.js', '**/*.ts'],
    template = defaultTemplate
  } = options
  
  let watcher
  let server
  
  return {
    name: 'auto-generator',
    
    configureServer(devServer) {
      server = devServer
      
      // 初始生成
      generateTypes()
      
      // 监听文件变化
      watcher = watch(watchDirs, {
        ignored: /node_modules/,
        persistent: true
      })
      
      watcher.on('change', (filePath) => {
        console.log(`File changed: ${filePath}`)
        generateTypes()
        
        // 触发热更新
        const module = server.moduleGraph.getModuleByFile(outputFile)
        if (module) {
          server.reloadModule(module)
        }
      })
      
      watcher.on('add', (filePath) => {
        console.log(`File added: ${filePath}`)
        generateTypes()
      })
      
      watcher.on('unlink', (filePath) => {
        console.log(`File removed: ${filePath}`)
        generateTypes()
      })
    },
    
    buildStart() {
      // 构建时生成
      generateTypes()
    },
    
    buildEnd() {
      // 关闭监听器
      if (watcher) {
        watcher.close()
      }
    }
  }
  
  function generateTypes() {
    try {
      const apiFiles = []
      
      watchDirs.forEach(dir => {
        if (fs.existsSync(dir)) {
          const files = getAllFiles(dir, patterns)
          apiFiles.push(...files)
        }
      })
      
      const typeDefinitions = apiFiles.map(file => {
        const content = fs.readFileSync(file, 'utf-8')
        return extractTypeInfo(content, file)
      }).filter(Boolean)
      
      const output = template(typeDefinitions)
      
      // 确保输出目录存在
      const outputDir = path.dirname(outputFile)
      if (!fs.existsSync(outputDir)) {
        fs.mkdirSync(outputDir, { recursive: true })
      }
      
      fs.writeFileSync(outputFile, output)
      console.log(`Generated types: ${outputFile}`)
      
    } catch (error) {
      console.error('Failed to generate types:', error)
    }
  }
}

function getAllFiles(dir, patterns) {
  const files = []
  
  function traverse(currentDir) {
    const items = fs.readdirSync(currentDir)
    
    items.forEach(item => {
      const fullPath = path.join(currentDir, item)
      const stat = fs.statSync(fullPath)
      
      if (stat.isDirectory()) {
        traverse(fullPath)
      } else if (stat.isFile()) {
        const shouldInclude = patterns.some(pattern => {
          const regex = new RegExp(pattern.replace(/\*\*/g, '.*').replace(/\*/g, '[^/]*'))
          return regex.test(fullPath)
        })
        
        if (shouldInclude) {
          files.push(fullPath)
        }
      }
    })
  }
  
  traverse(dir)
  return files
}

function extractTypeInfo(content, filePath) {
  // 简单的类型提取逻辑
  const exports = []
  const lines = content.split('\n')
  
  lines.forEach(line => {
    // 提取导出的函数
    const functionMatch = line.match(/export\s+(?:async\s+)?function\s+(\w+)/)
    if (functionMatch) {
      exports.push({
        name: functionMatch[1],
        type: 'function',
        file: filePath
      })
    }
    
    // 提取导出的常量
    const constMatch = line.match(/export\s+const\s+(\w+)/)
    if (constMatch) {
      exports.push({
        name: constMatch[1],
        type: 'const',
        file: filePath
      })
    }
  })
  
  return exports.length > 0 ? { file: filePath, exports } : null
}

function defaultTemplate(typeDefinitions) {
  const header = `// Auto-generated file - Do not edit manually
// Generated at: ${new Date().toISOString()}

`
  
  const types = typeDefinitions.map(def => {
    const fileName = path.basename(def.file, path.extname(def.file))
    const exports = def.exports.map(exp => {
      return `  ${exp.name}: ${exp.type === 'function' ? '(...args: any[]) => any' : 'any'};`
    }).join('\n')
    
    return `export interface ${capitalize(fileName)}Api {
${exports}
}`
  }).join('\n\n')
  
  return header + types
}

function capitalize(str) {
  return str.charAt(0).toUpperCase() + str.slice(1)
}
```

### 代码分析和报告插件
```javascript
// plugins/code-analyzer.js
import fs from 'fs'
import path from 'path'

export function codeAnalyzerPlugin(options = {}) {
  const {
    outputDir = 'reports',
    includePatterns = [/\.(js|ts|vue)$/],
    excludePatterns = [/node_modules/, /\.d\.ts$/],
    metrics = ['lines', 'functions', 'complexity']
  } = options
  
  const analysisData = {
    files: [],
    summary: {
      totalFiles: 0,
      totalLines: 0,
      totalFunctions: 0,
      averageComplexity: 0
    }
  }
  
  return {
    name: 'code-analyzer',
    
    transform(code, id) {
      // 分析每个文件
      if (shouldAnalyze(id, includePatterns, excludePatterns)) {
        const analysis = analyzeCode(code, id, metrics)
        analysisData.files.push(analysis)
      }
      
      return null // 不修改代码
    },
    
    generateBundle() {
      // 生成分析报告
      generateReport(analysisData, outputDir)
    }
  }
}

function shouldAnalyze(id, includePatterns, excludePatterns) {
  // 检查是否应该分析此文件
  const shouldInclude = includePatterns.some(pattern => pattern.test(id))
  const shouldExclude = excludePatterns.some(pattern => pattern.test(id))
  
  return shouldInclude && !shouldExclude
}

function analyzeCode(code, id, metrics) {
  const analysis = {
    file: id,
    timestamp: new Date().toISOString()
  }
  
  if (metrics.includes('lines')) {
    analysis.lines = {
      total: code.split('\n').length,
      code: code.split('\n').filter(line => line.trim() && !line.trim().startsWith('//')).length,
      comments: code.split('\n').filter(line => line.trim().startsWith('//')).length
    }
  }
  
  if (metrics.includes('functions')) {
    const functionMatches = code.match(/function\s+\w+|=>\s*{|:\s*function/g) || []
    analysis.functions = {
      count: functionMatches.length,
      list: extractFunctionNames(code)
    }
  }
  
  if (metrics.includes('complexity')) {
    analysis.complexity = calculateComplexity(code)
  }
  
  return analysis
}

function extractFunctionNames(code) {
  const functions = []
  const patterns = [
    /function\s+(\w+)/g,
    /const\s+(\w+)\s*=\s*(?:async\s+)?\(/g,
    /(\w+)\s*:\s*(?:async\s+)?function/g
  ]
  
  patterns.forEach(pattern => {
    let match
    while ((match = pattern.exec(code)) !== null) {
      functions.push(match[1])
    }
  })
  
  return functions
}

function calculateComplexity(code) {
  // 简单的圈复杂度计算
  const complexityKeywords = [
    'if', 'else', 'while', 'for', 'switch', 'case', 'catch', '&&', '||', '?'
  ]
  
  let complexity = 1 // 基础复杂度
  
  complexityKeywords.forEach(keyword => {
    const regex = new RegExp(`\\b${keyword}\\b`, 'g')
    const matches = code.match(regex) || []
    complexity += matches.length
  })
  
  return complexity
}

function generateReport(data, outputDir) {
  // 计算汇总数据
  data.summary.totalFiles = data.files.length
  data.summary.totalLines = data.files.reduce((sum, file) => sum + (file.lines?.total || 0), 0)
  data.summary.totalFunctions = data.files.reduce((sum, file) => sum + (file.functions?.count || 0), 0)
  data.summary.averageComplexity = data.files.reduce((sum, file) => sum + (file.complexity || 0), 0) / data.files.length
  
  // 确保输出目录存在
  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true })
  }
  
  // 生成 JSON 报告
  const jsonReport = path.join(outputDir, 'code-analysis.json')
  fs.writeFileSync(jsonReport, JSON.stringify(data, null, 2))
  
  // 生成 HTML 报告
  const htmlReport = path.join(outputDir, 'code-analysis.html')
  const htmlContent = generateHtmlReport(data)
  fs.writeFileSync(htmlReport, htmlContent)
  
  console.log(`Code analysis report generated:`)
  console.log(`  JSON: ${jsonReport}`)
  console.log(`  HTML: ${htmlReport}`)
}

function generateHtmlReport(data) {
  return `
<!DOCTYPE html>
<html>
<head>
  <title>Code Analysis Report</title>
  <style>
    body { font-family: Arial, sans-serif; margin: 20px; }
    .summary { background: #f5f5f5; padding: 20px; border-radius: 5px; margin-bottom: 20px; }
    .file { border: 1px solid #ddd; margin: 10px 0; padding: 15px; border-radius: 5px; }
    .metric { display: inline-block; margin: 5px 10px; padding: 5px 10px; background: #e9ecef; border-radius: 3px; }
    table { width: 100%; border-collapse: collapse; }
    th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
    th { background-color: #f2f2f2; }
  </style>
</head>
<body>
  <h1>Code Analysis Report</h1>
  
  <div class="summary">
    <h2>Summary</h2>
    <div class="metric">Total Files: ${data.summary.totalFiles}</div>
    <div class="metric">Total Lines: ${data.summary.totalLines}</div>
    <div class="metric">Total Functions: ${data.summary.totalFunctions}</div>
    <div class="metric">Average Complexity: ${data.summary.averageComplexity.toFixed(2)}</div>
  </div>
  
  <h2>File Details</h2>
  <table>
    <thead>
      <tr>
        <th>File</th>

          <td>${file.functions?.count || 'N/A'}</td>
          <td>${file.complexity || 'N/A'}</td>
        </tr>
      `).join('')}
    </tbody>
  </table>
  
  <p><small>Generated at: ${new Date().toISOString()}</small></p>

</html>
  `
}
```

```javascript
// plugins/debug-helper.js
export function createDebugPlugin(pluginName, options = {}) {
  const {
    logHooks = true,
    logTransforms = false,
    logModules = false,
    logTiming = true
  } = options
  
  const timings = new Map()
  
  function logWithTime(message, startTime) {
    const duration = startTime ? Date.now() - startTime : 0
    const prefix = `[${pluginName}]`
    const timeStr = startTime ? ` (${duration}ms)` : ''
    console.log(`${prefix} ${message}${timeStr}`)
  }
  
  return {
    name: `debug-${pluginName}`,
    
    buildStart() {
      if (logHooks) logWithTime('Build started')
    },
    
    resolveId(id, importer) {
      if (logModules) {
        logWithTime(`Resolving: ${id} from ${importer || 'entry'}`)
      }
      
      if (logTiming) {
        timings.set(`resolve-${id}`, Date.now())
      }
    },
    
    load(id) {
      if (logModules) {
        const startTime = timings.get(`resolve-${id}`)
        logWithTime(`Loading: ${id}`, startTime)
      }
    },
    
    transform(code, id) {
      if (logTransforms) {
        logWithTime(`Transforming: ${id} (${code.length} chars)`)
      }
    },
    
    generateBundle(options, bundle) {
      if (logHooks) {
        const fileCount = Object.keys(bundle).length
        logWithTime(`Generated bundle with ${fileCount} files`)
      }
    },
    
    writeBundle() {
      if (logHooks) logWithTime('Bundle written')
    }
  }
}

// 使用示例
export default defineConfig({
  plugins: [
    myPlugin(),
    createDebugPlugin('my-plugin', {
      logHooks: true,
      logTransforms: true
    })
  ]
})
```

### TypeScript 类型定义
```typescript
// src/types.d.ts
import type { Plugin } from 'vite'

export interface MyPluginOptions {
  include?: RegExp | RegExp[]
  exclude?: RegExp | RegExp[]
  transformOptions?: {
    removeComments?: boolean
    minify?: boolean
  }
}

export declare function myPlugin(options?: MyPluginOptions): Plugin

export default myPlugin
```
