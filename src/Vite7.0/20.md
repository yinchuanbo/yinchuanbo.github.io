## 高级多页面配置

### 动态页面配置
```javascript
// vite.config.js
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import { resolve, join } from 'path'
import { readdirSync, statSync } from 'fs'

// 自动发现页面
function getPages() {
  const pagesDir = resolve(__dirname, 'src/pages')
  const pages = {}
  
  try {
    const entries = readdirSync(pagesDir)
    
    entries.forEach(entry => {
      const entryPath = join(pagesDir, entry)
      const stat = statSync(entryPath)
      
      if (stat.isDirectory()) {
        const htmlPath = join(entryPath, 'index.html')
        try {
          statSync(htmlPath)
          pages[entry] = htmlPath
        } catch {
          // index.html 不存在，跳过
        }
      }
    })
  } catch (error) {
    console.warn('Failed to read pages directory:', error)
  }
  
  return pages
}

export default defineConfig({
  plugins: [vue()],
  
  build: {
    rollupOptions: {
      input: getPages()
    }
  }
})
```

```javascript
// src/shared/stores/user.js
import { defineStore } from 'pinia'

export const useUserStore = defineStore('user', {
  state: () => ({
    currentUser: null,
    permissions: [],
    preferences: {
      theme: 'light',
      language: 'zh-CN'
    }
  }),
  
  getters: {
    isLoggedIn: (state) => !!state.currentUser,
    hasPermission: (state) => (permission) => {
      return state.permissions.includes(permission)
    },
    displayName: (state) => {
      return state.currentUser?.name || '游客'
    }
  },
  
  actions: {
    async login(credentials) {
      try {
        const response = await fetch('/api/auth/login', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(credentials)
        })
        
        if (!response.ok) throw new Error('登录失败')
        
        const userData = await response.json()
        this.currentUser = userData.user
        this.permissions = userData.permissions
        
        // 保存到本地存储
        localStorage.setItem('user', JSON.stringify(userData))
        
        return userData
      } catch (error) {
        throw error
      }
    },
    
    logout() {
      this.currentUser = null
      this.permissions = []
      localStorage.removeItem('user')
    },
    
    updatePreferences(newPreferences) {
      this.preferences = { ...this.preferences, ...newPreferences }
      localStorage.setItem('preferences', JSON.stringify(this.preferences))
    },
    
    loadFromStorage() {
      // 从本地存储恢复用户信息
      const userData = localStorage.getItem('user')
      if (userData) {
        const parsed = JSON.parse(userData)
        this.currentUser = parsed.user
        this.permissions = parsed.permissions
      }
      
      // 从本地存储恢复偏好设置
      const preferences = localStorage.getItem('preferences')
      if (preferences) {
        this.preferences = JSON.parse(preferences)
      }
    }
  }
})
```

```javascript
// src/shared/utils/format.js
export function formatDate(date, format = 'YYYY-MM-DD') {
  if (!date) return ''
  
  const d = new Date(date)
  const year = d.getFullYear()
  const month = String(d.getMonth() + 1).padStart(2, '0')
  const day = String(d.getDate()).padStart(2, '0')
  const hour = String(d.getHours()).padStart(2, '0')
  const minute = String(d.getMinutes()).padStart(2, '0')
  const second = String(d.getSeconds()).padStart(2, '0')
  
  return format
    .replace('YYYY', year)
    .replace('MM', month)
    .replace('DD', day)
    .replace('HH', hour)
    .replace('mm', minute)
    .replace('ss', second)
}

export function formatCurrency(amount, currency = 'CNY') {
  if (typeof amount !== 'number') return ''
  
  return new Intl.NumberFormat('zh-CN', {
    style: 'currency',
    currency
  }).format(amount)
}

export function formatFileSize(bytes) {
  if (bytes === 0) return '0 B'
  
  const k = 1024
  const sizes = ['B', 'KB', 'MB', 'GB', 'TB']
  const i = Math.floor(Math.log(bytes) / Math.log(k))
  
  return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i]
}
```

## 页面间通信

### 跨页面消息传递
```javascript
// src/shared/utils/crossPageCommunication.js
class CrossPageCommunication {
  constructor() {
    this.channel = new BroadcastChannel('app-communication')
    this.listeners = new Map()
    
    // 监听消息
    this.channel.addEventListener('message', (event) => {
      this.handleMessage(event.data)
    })
  }
  
  // 发送消息
  send(type, data, target = '*') {
    const message = {
      type,
      data,
      target,
      timestamp: Date.now(),
      source: this.getPageIdentifier()
    }
    
    this.channel.postMessage(message)
  }
  
  // 监听消息
  on(type, callback) {
    if (!this.listeners.has(type)) {
      this.listeners.set(type, [])
    }
    
    this.listeners.get(type).push(callback)
    
    // 返回取消监听的函数
    return () => {
      const callbacks = this.listeners.get(type)
      const index = callbacks.indexOf(callback)
      if (index > -1) {
        callbacks.splice(index, 1)
      }
    }
  }
  
  // 处理接收到的消息
  handleMessage(message) {
    const { type, data, target, source } = message
    
    // 检查目标
    if (target !== '*' && target !== this.getPageIdentifier()) {
      return
    }
    
    // 忽略自己发送的消息
    if (source === this.getPageIdentifier()) {
      return
    }
    
    // 触发监听器
    const callbacks = this.listeners.get(type) || []
    callbacks.forEach(callback => {
      try {
        callback(data, message)
      } catch (error) {
        console.error('Error in message callback:', error)
      }
    })
  }
  
  // 获取页面标识
  getPageIdentifier() {
    return window.location.pathname
  }
  
  // 销毁
  destroy() {
    this.channel.close()
    this.listeners.clear()
  }
}

export const crossPageComm = new CrossPageCommunication()

// 使用示例
// 在主页发送消息
crossPageComm.send('user-login', { userId: 123, name: 'John' })

// 在管理页面监听消息
crossPageComm.on('user-login', (userData) => {
  console.log('User logged in:', userData)
  // 更新管理页面的用户信息
})
```

### 共享状态同步
```javascript
// src/shared/utils/stateSync.js
import { crossPageComm } from './crossPageCommunication'

class StateSync {
  constructor() {
    this.stores = new Map()
    this.setupCommunication()
  }
  
  // 注册需要同步的 store
  register(storeName, store) {
    this.stores.set(storeName, store)
    
    // 监听 store 变化
    store.$subscribe((mutation, state) => {
      this.broadcastStateChange(storeName, mutation, state)
    })
  }
  
  // 广播状态变化
  broadcastStateChange(storeName, mutation, state) {
    crossPageComm.send('state-change', {
      storeName,
      mutation,
      state: JSON.parse(JSON.stringify(state)) // 深拷贝
    })
  }
  
  // 设置跨页面通信
  setupCommunication() {
    crossPageComm.on('state-change', (data) => {
      this.handleStateChange(data)
    })
    
    crossPageComm.on('state-request', (data) => {
      this.handleStateRequest(data)
    })
  }
  
  // 处理状态变化
  handleStateChange({ storeName, mutation, state }) {
    const store = this.stores.get(storeName)
    if (!store) return
    
    // 应用状态变化（避免循环触发）
    store.$patch(state)
  }
  
  // 处理状态请求
  handleStateRequest({ storeName, requestId }) {
    const store = this.stores.get(storeName)
    if (!store) return
    
    crossPageComm.send('state-response', {
      requestId,
      storeName,
      state: JSON.parse(JSON.stringify(store.$state))
    })
  }
  
  // 请求其他页面的状态
  requestState(storeName) {
    const requestId = Date.now().toString()
    
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        cleanup()
        reject(new Error('State request timeout'))
      }, 5000)
      
      const cleanup = crossPageComm.on('state-response', (data) => {
        if (data.requestId === requestId && data.storeName === storeName) {
          clearTimeout(timeout)
          cleanup()
          resolve(data.state)
        }
      })
      
      crossPageComm.send('state-request', { storeName, requestId })
    })
  }
}

export const stateSync = new StateSync()

// 在各个页面中注册 store
// 主页
import { useUserStore } from '@shared/stores/user'
const userStore = useUserStore()
stateSync.register('user', userStore)

// 管理页面
import { useUserStore } from '@shared/stores/user'
const userStore = useUserStore()
stateSync.register('user', userStore)
```

```javascript
// vite.config.js
import { defineConfig } from 'vite'

export default defineConfig({
  build: {
    rollupOptions: {
      input: {
        main: resolve(__dirname, 'src/pages/home/index.html'),
        admin: resolve(__dirname, 'src/pages/admin/index.html'),
        mobile: resolve(__dirname, 'src/pages/mobile/index.html')
      },
      output: {
        // 为不同页面生成不同的文件名模式
        entryFileNames: (chunkInfo) => {
          const facadeModuleId = chunkInfo.facadeModuleId
          if (facadeModuleId?.includes('/pages/admin/')) {
            return 'admin/js/[name]-[hash].js'
          }
          if (facadeModuleId?.includes('/pages/mobile/')) {
            return 'mobile/js/[name]-[hash].js'
          }
          return 'js/[name]-[hash].js'
        },
        
        chunkFileNames: (chunkInfo) => {
          if (chunkInfo.name.includes('admin')) {
            return 'admin/js/[name]-[hash].js'
          }
          if (chunkInfo.name.includes('mobile')) {
            return 'mobile/js/[name]-[hash].js'
          }
          return 'js/[name]-[hash].js'
        },
        
        assetFileNames: (assetInfo) => {
          const info = assetInfo.name.split('.')
          const ext = info[info.length - 1]
          
          // 根据资源类型分类
          if (/\.(png|jpe?g|gif|svg|webp)$/i.test(assetInfo.name)) {
            return `images/[name]-[hash].${ext}`
          }
          if (/\.(woff2?|eot|ttf|otf)$/i.test(assetInfo.name)) {
            return `fonts/[name]-[hash].${ext}`
          }
          if (/\.css$/i.test(assetInfo.name)) {
            return `css/[name]-[hash].${ext}`
          }
          return `assets/[name]-[hash].${ext}`
        }
      }
    }
  }
})
```

```javascript
// vite.config.js
export default defineConfig({
  plugins: [
    vue(),
    {
      name: 'multi-page-hmr',
      handleHotUpdate(ctx) {
        const { file, server } = ctx
        
        // 共享文件变化时，通知所有页面
        if (file.includes('/src/shared/')) {
          server.ws.send({
            type: 'full-reload'
          })
          return []
        }
        
        // 页面特定文件变化时，只通知相关页面
        if (file.includes('/src/pages/')) {
          const pageName = file.match(/\/pages\/([^\/]+)\//)?.[1]
          if (pageName) {
            // 只向特定页面发送更新
            server.ws.send({
              type: 'update',
              updates: ctx.modules.map(m => ({
                type: m.type || 'js-update',
                path: m.url,
                acceptedPath: m.url
              }))
            })
          }
        }
        
        return ctx.modules
      }
    }
  ]
})
```

## 部署策略

### 独立部署
```javascript
// scripts/build-pages.js
import { build } from 'vite'
import { resolve } from 'path'

const pages = {
  home: {
    input: resolve(__dirname, '../src/pages/home/index.html'),
    outDir: 'dist/home'
  },
  admin: {
    input: resolve(__dirname, '../src/pages/admin/index.html'),
    outDir: 'dist/admin'
  },
  mobile: {
    input: resolve(__dirname, '../src/pages/mobile/index.html'),
    outDir: 'dist/mobile'
  }
}

async function buildPages() {
  for (const [name, config] of Object.entries(pages)) {
    console.log(`Building ${name} page...`)
    
    await build({
      build: {
        rollupOptions: {
          input: config.input
        },
        outDir: config.outDir,
        emptyOutDir: true
      }
    })
    
    console.log(`${name} page built successfully!`)
  }
}

buildPages().catch(console.error)
```

### 统一部署配置
```javascript
// vite.config.js
export default defineConfig({
  build: {
    rollupOptions: {
      input: {
        main: resolve(__dirname, 'src/pages/home/index.html'),
        admin: resolve(__dirname, 'src/pages/admin/index.html'),
        mobile: resolve(__dirname, 'src/pages/mobile/index.html')
      }
    },
    
    // 生成部署配置
    generateBundle(options, bundle) {
      // 生成 nginx 配置
      const nginxConfig = generateNginxConfig(bundle)
      this.emitFile({
        type: 'asset',
        fileName: 'nginx.conf',
        source: nginxConfig
      })
      
      // 生成部署脚本
      const deployScript = generateDeployScript(bundle)
      this.emitFile({
        type: 'asset',
        fileName: 'deploy.sh',
        source: deployScript
      })
    }
  }
})

function generateNginxConfig(bundle) {
  return `
server {
    listen 80;
    server_name example.com;
    root /var/www/html;
    index index.html;
    
    # 主页
    location / {
        try_files $uri $uri/ /index.html;
    }
    
    # 管理后台
    location /admin {
        try_files $uri $uri/ /admin.html;
    }
    
    # 移动端
    location /mobile {
        try_files $uri $uri/ /mobile.html;
    }
    
    # 静态资源
    location ~* \\.(js|css|png|jpg|jpeg|gif|ico|svg)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
}
  `
}

function generateDeployScript(bundle) {
  return `#!/bin/bash
# 部署脚本
echo "Starting deployment..."

# 备份旧版本
if [ -d "/var/www/html" ]; then
    cp -r /var/www/html /var/www/html.backup.$(date +%Y%m%d_%H%M%S)
fi

# 部署新版本
cp -r dist/* /var/www/html/

# 重启 nginx
systemctl reload nginx

echo "Deployment completed!"
  `
}
```

多页面应用配置为大型项目提供了：
