### 模块热更新优化
```javascript
// src/utils/hmrOptimizer.js
class HMROptimizer {
  constructor() {
    this.updateQueue = []
    this.isProcessing = false
    this.debounceTime = 100
  }
  
  // 批量处理 HMR 更新
  batchUpdate(callback) {
    this.updateQueue.push(callback)
    
    if (!this.isProcessing) {
      this.isProcessing = true
      setTimeout(() => {
        this.processQueue()
      }, this.debounceTime)
    }
  }
  
  processQueue() {
    const callbacks = [...this.updateQueue]
    this.updateQueue = []
    this.isProcessing = false
    
    callbacks.forEach(callback => {
      try {
        callback()
      } catch (error) {
        console.error('HMR update error:', error)
      }
    })
  }
}

export const hmrOptimizer = new HMROptimizer()

// 在组件中使用
if (import.meta.hot) {
  import.meta.hot.accept(() => {
    hmrOptimizer.batchUpdate(() => {
      // 执行更新逻辑
      console.log('Component updated')
    })
  })
}
```

```javascript
         id.includes('/components/ui/')
}
```

### 动态导入优化
```javascript
// src/utils/dynamicImport.js
class DynamicImportManager {
  constructor() {
    this.cache = new Map()
    this.loading = new Map()
    this.retryCount = new Map()
    this.maxRetries = 3
  }
  
  async import(importFn, chunkName) {
    // 检查缓存
    if (this.cache.has(chunkName)) {
      return this.cache.get(chunkName)
    }
    
    // 检查是否正在加载
    if (this.loading.has(chunkName)) {
      return this.loading.get(chunkName)
    }
    
    // 开始加载
    const loadPromise = this.loadWithRetry(importFn, chunkName)
    this.loading.set(chunkName, loadPromise)
    
    try {
      const module = await loadPromise
      this.cache.set(chunkName, module)
      this.loading.delete(chunkName)
      return module
    } catch (error) {
      this.loading.delete(chunkName)
      throw error
    }
  }
  
  async loadWithRetry(importFn, chunkName) {
    const retries = this.retryCount.get(chunkName) || 0
    
    try {
      return await importFn()
    } catch (error) {
      if (retries < this.maxRetries) {
        this.retryCount.set(chunkName, retries + 1)
        
        // 延迟重试
        await new Promise(resolve => setTimeout(resolve, 1000 * (retries + 1)))
        
        return this.loadWithRetry(importFn, chunkName)
      } else {
        throw error
      }
    }
  }
  
  // 预加载
  preload(importFn, chunkName) {
    if (!this.cache.has(chunkName) && !this.loading.has(chunkName)) {
      this.import(importFn, chunkName).catch(error => {
        console.warn(`Failed to preload ${chunkName}:`, error)
      })
    }
  }
  
  // 清理缓存
  clearCache() {
    this.cache.clear()
    this.retryCount.clear()
  }
}

export const dynamicImportManager = new DynamicImportManager()

// 使用示例
export const loadComponent = (componentName) => {
  return dynamicImportManager.import(
    () => import(`../components/${componentName}.vue`),
    componentName
  )
}

// 预加载组件
export const preloadComponent = (componentName) => {
  dynamicImportManager.preload(
    () => import(`../components/${componentName}.vue`),
    componentName
  )
}
```

```vue
    index: start + index
  }))
```

```javascript
// src/utils/lazyComponent.js
import { defineAsyncComponent, ref } from 'vue'

export function createLazyComponent(importFn, options = {}) {
  const {
    delay = 200,
    timeout = 10000,
    retries = 3,
    loadingComponent = null,
    errorComponent = null
  } = options
  
  return defineAsyncComponent({
    loader: async () => {
      let lastError
      
      for (let i = 0; i <= retries; i++) {
        try {
          const module = await importFn()
          return module
        } catch (error) {
          lastError = error
          
          if (i < retries) {
            // 延迟重试
            await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)))
          }
        }
      }
      
      throw lastError
    },
    
    loadingComponent,
    errorComponent,
    delay,
    timeout
  })
}
```

### 内存优化
```javascript
// src/utils/memoryOptimizer.js
class MemoryOptimizer {
  constructor() {
    this.observers = new Set()
    this.timers = new Set()
    this.eventListeners = new Map()
    this.subscriptions = new Set()
  }
  
  // 注册观察者
  addObserver(observer) {
    this.observers.add(observer)
    
    return () => {
      if (observer.disconnect) {
        observer.disconnect()
      }
      this.observers.delete(observer)
    }
  }
  
  // 注册定时器
  addTimer(timer) {
    this.timers.add(timer)
    
    return () => {
      clearTimeout(timer)
      clearInterval(timer)
      this.timers.delete(timer)
    }
  }
  
  // 注册事件监听器
  addEventListener(element, event, handler, options) {
    if (!this.eventListeners.has(element)) {
      this.eventListeners.set(element, new Map())
    }
    
    const elementListeners = this.eventListeners.get(element)
    if (!elementListeners.has(event)) {
      elementListeners.set(event, new Set())
    }
    
    elementListeners.get(event).add(handler)
    element.addEventListener(event, handler, options)
    
    return () => {
      element.removeEventListener(event, handler, options)
      elementListeners.get(event).delete(handler)
    }
  }
  
  // 注册订阅
  addSubscription(subscription) {
    this.subscriptions.add(subscription)
    
    return () => {
      if (typeof subscription === 'function') {
        subscription()
      } else if (subscription.unsubscribe) {
        subscription.unsubscribe()
      }
      this.subscriptions.delete(subscription)
    }
  }
  
  // 清理所有资源
  cleanup() {
    // 清理观察者
    this.observers.forEach(observer => {
      if (observer.disconnect) {
        observer.disconnect()
      }
    })
    this.observers.clear()
    
    // 清理定时器
    this.timers.forEach(timer => {
      clearTimeout(timer)
      clearInterval(timer)
    })
    this.timers.clear()
    
    // 清理事件监听器
    this.eventListeners.forEach((elementListeners, element) => {
      elementListeners.forEach((handlers, event) => {
        handlers.forEach(handler => {
          element.removeEventListener(event, handler)
        })
      })
    })
    this.eventListeners.clear()
    
    // 清理订阅
    this.subscriptions.forEach(subscription => {
      if (typeof subscription === 'function') {
        subscription()
      } else if (subscription.unsubscribe) {
        subscription.unsubscribe()
      }
    })
    this.subscriptions.clear()
  }
}

// 在组件中使用
export function useMemoryOptimizer() {
  const optimizer = new MemoryOptimizer()
  
  // 组件卸载时自动清理
  onUnmounted(() => {
    optimizer.cleanup()
  })
  
  return optimizer
}
```

```javascript
// src/utils/performanceMonitor.js
class PerformanceMonitor {
  constructor() {
    this.metrics = {
      navigation: {},
      resources: [],
      userTiming: [],
      vitals: {}
    }
    
    this.init()
  }
  
  init() {
    // 监听页面加载完成
    if (document.readyState === 'complete') {
      this.collectNavigationMetrics()
    } else {
      window.addEventListener('load', () => {
        this.collectNavigationMetrics()
      })
    }
    
    // 监听资源加载
    this.observeResources()
    
    // 监听核心 Web 指标
    this.observeWebVitals()
  }
  
  collectNavigationMetrics() {
    const navigation = performance.getEntriesByType('navigation')[0]
    
    if (navigation) {
      this.metrics.navigation = {
        // DNS 查询时间
        dnsLookup: navigation.domainLookupEnd - navigation.domainLookupStart,
        
        // TCP 连接时间
        tcpConnect: navigation.connectEnd - navigation.connectStart,
        
        // SSL 握手时间
        sslHandshake: navigation.secureConnectionStart > 0 
          ? navigation.connectEnd - navigation.secureConnectionStart 
          : 0,
        
        // 请求响应时间
        requestResponse: navigation.responseEnd - navigation.requestStart,
        
        // DOM 解析时间
        domParse: navigation.domContentLoadedEventEnd - navigation.responseEnd,
        
        // 资源加载时间
        resourceLoad: navigation.loadEventEnd - navigation.domContentLoadedEventEnd,
        
        // 总加载时间
        totalLoad: navigation.loadEventEnd - navigation.navigationStart,
        
        // 首次内容绘制
        fcp: this.getFCP(),
        
        // 最大内容绘制
        lcp: this.getLCP()
      }
    }
  }
  
  observeResources() {
    const observer = new PerformanceObserver((list) => {
      list.getEntries().forEach(entry => {
        if (entry.entryType === 'resource') {
          this.metrics.resources.push({
            name: entry.name,
            type: this.getResourceType(entry.name),
            size: entry.transferSize,
            duration: entry.duration,
            startTime: entry.startTime
          })
        }
      })
    })
    
    observer.observe({ entryTypes: ['resource'] })
  }
  
  observeWebVitals() {
    // 观察 LCP
    new PerformanceObserver((list) => {
      const entries = list.getEntries()
      const lastEntry = entries[entries.length - 1]
      this.metrics.vitals.lcp = lastEntry.startTime
    }).observe({ entryTypes: ['largest-contentful-paint'] })
    
    // 观察 FID
    new PerformanceObserver((list) => {
      list.getEntries().forEach(entry => {
        this.metrics.vitals.fid = entry.processingStart - entry.startTime
      })
    }).observe({ entryTypes: ['first-input'] })
    
    // 观察 CLS
    let clsValue = 0
    new PerformanceObserver((list) => {
      list.getEntries().forEach(entry => {
        if (!entry.hadRecentInput) {
          clsValue += entry.value
          this.metrics.vitals.cls = clsValue
        }
      })
    }).observe({ entryTypes: ['layout-shift'] })
  }
  
  getFCP() {
    const fcpEntry = performance.getEntriesByName('first-contentful-paint')[0]
    return fcpEntry ? fcpEntry.startTime : null
  }
  
  getLCP() {
    const lcpEntries = performance.getEntriesByType('largest-contentful-paint')
    return lcpEntries.length > 0 ? lcpEntries[lcpEntries.length - 1].startTime : null
  }
  
  getResourceType(url) {
    if (url.match(/\.(js|mjs)$/)) return 'script'
    if (url.match(/\.css$/)) return 'stylesheet'
    if (url.match(/\.(png|jpg|jpeg|gif|svg|webp)$/)) return 'image'
    if (url.match(/\.(woff|woff2|ttf|eot)$/)) return 'font'
    return 'other'
  }
  
  // 自定义性能标记
  mark(name) {
    performance.mark(name)
  }
  
  // 测量性能
  measure(name, startMark, endMark) {
    performance.measure(name, startMark, endMark)
    
    const measure = performance.getEntriesByName(name)[0]
    this.metrics.userTiming.push({
      name,
      duration: measure.duration,
      startTime: measure.startTime
    })
  }
  
  // 获取性能报告
  getReport() {
    return {
      ...this.metrics,
      timestamp: Date.now(),
      userAgent: navigator.userAgent,
      url: window.location.href
    }
  }
  
  // 发送性能数据
  sendReport(endpoint) {
    const report = this.getReport()
    
    if (navigator.sendBeacon) {
      navigator.sendBeacon(endpoint, JSON.stringify(report))
    } else {
      fetch(endpoint, {
        method: 'POST',
        body: JSON.stringify(report),
        headers: {
          'Content-Type': 'application/json'
        }
      }).catch(error => {
        console.warn('Failed to send performance report:', error)
      })
    }
  }
}
```

```javascript
// vite.config.js
export default defineConfig({
  plugins: [
    {
      name: 'performance-budget',
      generateBundle(options, bundle) {
        const budgets = {
          // 总包大小限制
          totalSize: 2 * 1024 * 1024, // 2MB
          
          // 单个文件大小限制
          chunkSize: 500 * 1024, // 500KB
          
          // 资源数量限制
          assetCount: 50,
          
          // 特定类型限制
          jsSize: 1 * 1024 * 1024, // 1MB
          cssSize: 200 * 1024, // 200KB
          imageSize: 500 * 1024 // 500KB
        }
        
        this.checkBudgets(bundle, budgets)
      },
      
      checkBudgets(bundle, budgets) {
        let totalSize = 0
        let jsSize = 0
        let cssSize = 0
        let imageSize = 0
        let assetCount = 0
        
        Object.values(bundle).forEach(chunk => {
          const size = chunk.type === 'chunk' 
            ? chunk.code.length 
            : chunk.source.length
          
          totalSize += size
          assetCount++
          
          // 检查单个文件大小
          if (size > budgets.chunkSize) {
            this.warn(`File ${chunk.fileName} (${this.formatSize(size)}) exceeds chunk size budget (${this.formatSize(budgets.chunkSize)})`)
          }
          
          // 按类型统计
          if (chunk.fileName.endsWith('.js')) {
            jsSize += size
          } else if (chunk.fileName.endsWith('.css')) {
            cssSize += size
          } else if (/\.(png|jpg|jpeg|gif|svg|webp)$/.test(chunk.fileName)) {
            imageSize += size
          }
        })
        
        // 检查总大小
        if (totalSize > budgets.totalSize) {
          this.warn(`Total bundle size (${this.formatSize(totalSize)}) exceeds budget (${this.formatSize(budgets.totalSize)})`)
        }
        
        // 检查 JS 大小
        if (jsSize > budgets.jsSize) {
          this.warn(`JavaScript size (${this.formatSize(jsSize)}) exceeds budget (${this.formatSize(budgets.jsSize)})`)
        }
        
        // 检查 CSS 大小
        if (cssSize > budgets.cssSize) {
          this.warn(`CSS size (${this.formatSize(cssSize)}) exceeds budget (${this.formatSize(budgets.cssSize)})`)
        }
        
        // 检查图片大小
        if (imageSize > budgets.imageSize) {
          this.warn(`Image size (${this.formatSize(imageSize)}) exceeds budget (${this.formatSize(budgets.imageSize)})`)
        }
        
        // 检查资源数量
        if (assetCount > budgets.assetCount) {
          this.warn(`Asset count (${assetCount}) exceeds budget (${budgets.assetCount})`)
        }
        
        console.log(`\n📊 Performance Budget Report:`)
        console.log(`Total Size: ${this.formatSize(totalSize)} / ${this.formatSize(budgets.totalSize)}`)
        console.log(`JavaScript: ${this.formatSize(jsSize)} / ${this.formatSize(budgets.jsSize)}`)
        console.log(`CSS: ${this.formatSize(cssSize)} / ${this.formatSize(budgets.cssSize)}`)
        console.log(`Images: ${this.formatSize(imageSize)} / ${this.formatSize(budgets.imageSize)}`)
        console.log(`Assets: ${assetCount} / ${budgets.assetCount}`)
      },
      
      formatSize(bytes) {
        const sizes = ['B', 'KB', 'MB', 'GB']
        if (bytes === 0) return '0 B'
        const i = Math.floor(Math.log(bytes) / Math.log(1024))
        return Math.round(bytes / Math.pow(1024, i) * 100) / 100 + ' ' + sizes[i]
      }
    }
  ]
})
```
