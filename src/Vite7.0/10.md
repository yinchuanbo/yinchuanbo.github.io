```javascript
import { defineConfig } from 'vite'
import { resolve } from 'path'

export default defineConfig({
  build: {
    rollupOptions: {
      // 自定义输入
      input: {
        main: resolve(__dirname, 'index.html'),
        admin: resolve(__dirname, 'admin/index.html'),
        mobile: resolve(__dirname, 'mobile/index.html')
      },
      
      // 外部依赖
      external: ['vue', 'lodash'],
      
      // 输出配置
      output: {
        // 自定义 chunk 分割
        manualChunks: (id) => {
          // 将 node_modules 中的包分组
          if (id.includes('node_modules')) {
            if (id.includes('vue')) {
              return 'vue-vendor'
            }
            if (id.includes('lodash')) {
              return 'lodash-vendor'
            }
            if (id.includes('chart')) {
              return 'chart-vendor'
            }
            return 'vendor'
          }
          
          // 将特定目录的文件分组
          if (id.includes('src/components')) {
            return 'components'
          }
          if (id.includes('src/utils')) {
            return 'utils'
          }
        },
        
        // 自定义文件名
        chunkFileNames: (chunkInfo) => {
          const facadeModuleId = chunkInfo.facadeModuleId
          if (facadeModuleId) {
            const fileName = facadeModuleId.split('/').pop().replace('.js', '')
            return `js/${fileName}-[hash].js`
          }
          return 'js/[name]-[hash].js'
        },
        
        // 全局变量
        globals: {
          vue: 'Vue',
          lodash: '_'
        }
      },
      
      // 插件
      plugins: [
        // 自定义 Rollup 插件
        {
          name: 'custom-plugin',
          generateBundle(options, bundle) {
            // 自定义构建逻辑
            console.log('Bundle generated:', Object.keys(bundle))
          }
        }
      ]
    }
  }
})
```

```javascript
// plugins/virtual-module-plugin.js
export function virtualModulePlugin(modules) {
  const virtualModules = new Map()
  
  // 预处理虚拟模块
  Object.entries(modules).forEach(([id, content]) => {
    virtualModules.set(id, content)
  })
  
  return {
    name: 'virtual-module',
    
    // 解析虚拟模块 ID
    resolveId(id) {
      if (virtualModules.has(id)) {
        return id
      }
    },
    
    // 加载虚拟模块内容
    load(id) {
      if (virtualModules.has(id)) {
        return virtualModules.get(id)
      }
    },
    
    // 开发服务器钩子
    configureServer(server) {
      server.middlewares.use('/api/virtual-modules', (req, res) => {
        res.setHeader('Content-Type', 'application/json')
        res.end(JSON.stringify(Array.from(virtualModules.keys())))
      })
    },
    
    // HMR 支持
    handleHotUpdate(ctx) {
      if (virtualModules.has(ctx.file)) {
        // 重新加载虚拟模块
        ctx.server.ws.send({
          type: 'full-reload'
        })
        return []
      }
    }
  }
}

// 使用插件
export default defineConfig({
  plugins: [
    virtualModulePlugin({
      'virtual:config': `
        export default {
          apiUrl: '${process.env.VITE_API_URL}',
          version: '${process.env.npm_package_version}'
        }
      `,
      'virtual:features': `
        export const features = {
          darkMode: true,
          analytics: ${process.env.NODE_ENV === 'production'}
        }
      `
    })
  ]
})
```

### 资源预加载策略
```javascript
// 智能预加载
class ResourcePreloader {
  constructor() {
    this.preloadedResources = new Set()
    this.observer = new IntersectionObserver(this.handleIntersection.bind(this))
  }
  
  // 预加载关键资源
  preloadCritical() {
    const criticalResources = [
      '/fonts/main.woff2',
      '/images/hero.webp',
      '/js/vendor.js'
    ]
    
    criticalResources.forEach(resource => {
      this.preloadResource(resource)
    })
  }
  
  // 预加载资源
  preloadResource(href, as = 'script') {
    if (this.preloadedResources.has(href)) return
    
    const link = document.createElement('link')
    link.rel = 'preload'
    link.href = href
    link.as = as
    
    if (as === 'font') {
      link.crossOrigin = 'anonymous'
    }
    
    document.head.appendChild(link)
    this.preloadedResources.add(href)
  }
  
  // 观察元素进入视口
  observeElement(element, resource) {
    element.dataset.preloadResource = resource
    this.observer.observe(element)
  }
  
  handleIntersection(entries) {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const resource = entry.target.dataset.preloadResource
        if (resource) {
          this.preloadResource(resource)
          this.observer.unobserve(entry.target)
        }
      }
    })
  }
}

// 使用预加载器
const preloader = new ResourcePreloader()
preloader.preloadCritical()

// 在组件中使用
export default {
  mounted() {
    // 预加载下一页面的资源
    preloader.observeElement(this.$refs.nextPageLink, '/js/next-page.js')
  }
}
```

### 缓存优化
```javascript
// Service Worker 缓存策略
// sw.js
const CACHE_NAME = 'vite-app-v1'
const STATIC_CACHE = 'static-v1'
const DYNAMIC_CACHE = 'dynamic-v1'

const STATIC_ASSETS = [
  '/',
  '/index.html',
  '/manifest.json',
  // 其他静态资源
]

// 安装事件
self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(STATIC_CACHE)
      .then(cache => cache.addAll(STATIC_ASSETS))
  )
})

// 激活事件
self.addEventListener('activate', event => {
  event.waitUntil(
    caches.keys().then(cacheNames => {
      return Promise.all(
        cacheNames.map(cacheName => {
          if (cacheName !== STATIC_CACHE && cacheName !== DYNAMIC_CACHE) {
            return caches.delete(cacheName)
          }
        })
      )
    })
  )
})

// 拦截请求
self.addEventListener('fetch', event => {
  event.respondWith(
    caches.match(event.request)
      .then(response => {
        if (response) {
          return response
        }
        
        return fetch(event.request)
          .then(response => {
            if (response.status === 200) {
              const responseClone = response.clone()
              caches.open(DYNAMIC_CACHE)
                .then(cache => cache.put(event.request, responseClone))
            }
            return response
          })
      })
  )
})
```

## 监控和调试

### 性能监控
```javascript
// src/utils/performance.js
export class PerformanceMonitor {
  constructor() {
    this.metrics = new Map()
    this.observers = new Map()
  }
  
  // 监控页面加载性能
  monitorPageLoad() {
    if ('PerformanceObserver' in window) {
      const observer = new PerformanceObserver((list) => {
        list.getEntries().forEach((entry) => {
          if (entry.entryType === 'navigation') {
            this.metrics.set('pageLoad', {
              domContentLoaded: entry.domContentLoadedEventEnd - entry.domContentLoadedEventStart,
              loadComplete: entry.loadEventEnd - entry.loadEventStart,
              firstPaint: entry.responseEnd - entry.requestStart
            })
          }
        })
      })
      
      observer.observe({ entryTypes: ['navigation'] })
      this.observers.set('navigation', observer)
    }
  }
  
  // 监控资源加载
  monitorResourceLoad() {
    if ('PerformanceObserver' in window) {
      const observer = new PerformanceObserver((list) => {
        list.getEntries().forEach((entry) => {
          if (entry.duration > 1000) { // 超过1秒的资源
            console.warn('Slow resource:', entry.name, entry.duration)
          }
        })
      })
      
      observer.observe({ entryTypes: ['resource'] })
      this.observers.set('resource', observer)
    }
  }
  
  // 监控长任务
  monitorLongTasks() {
    if ('PerformanceObserver' in window) {
      const observer = new PerformanceObserver((list) => {
        list.getEntries().forEach((entry) => {
          console.warn('Long task detected:', entry.duration, entry.startTime)
        })
      })
      
      observer.observe({ entryTypes: ['longtask'] })
      this.observers.set('longtask', observer)
    }
  }
  
  // 获取性能指标
  getMetrics() {
    return Object.fromEntries(this.metrics)
  }
  
  // 清理观察器
  cleanup() {
    this.observers.forEach(observer => observer.disconnect())
    this.observers.clear()
  }
}

// 使用监控器
const monitor = new PerformanceMonitor()
monitor.monitorPageLoad()
monitor.monitorResourceLoad()
monitor.monitorLongTasks()

// 在应用卸载时清理
window.addEventListener('beforeunload', () => {
  monitor.cleanup()
})
```

### 错误边界和日志
```javascript
// src/utils/errorHandler.js
export class ErrorHandler {
  constructor(options = {}) {
    this.apiEndpoint = options.apiEndpoint
    this.enableConsoleLog = options.enableConsoleLog ?? true
    this.maxRetries = options.maxRetries ?? 3
    
    this.setupGlobalHandlers()
  }
  
  setupGlobalHandlers() {
    // 捕获 JavaScript 错误
    window.addEventListener('error', (event) => {
      this.handleError({
        type: 'javascript',
        message: event.message,
        filename: event.filename,
        lineno: event.lineno,
        colno: event.colno,
        stack: event.error?.stack
      })
    })
    
    // 捕获 Promise 拒绝
    window.addEventListener('unhandledrejection', (event) => {
      this.handleError({
        type: 'promise',
        message: event.reason?.message || 'Unhandled promise rejection',
        stack: event.reason?.stack
      })
    })
    
    // 捕获资源加载错误
    window.addEventListener('error', (event) => {
      if (event.target !== window) {
        this.handleError({
          type: 'resource',
          message: `Failed to load resource: ${event.target.src || event.target.href}`,
          element: event.target.tagName
        })
      }
    }, true)
  }
  
  handleError(error) {
    if (this.enableConsoleLog) {
      console.error('Error captured:', error)
    }
    
    // 发送错误到服务器
    if (this.apiEndpoint) {
      this.sendErrorToServer(error)
    }
  }
  
  async sendErrorToServer(error, retryCount = 0) {
    try {
      await fetch(this.apiEndpoint, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          ...error,
          timestamp: new Date().toISOString(),
          userAgent: navigator.userAgent,
          url: window.location.href
        })
      })
    } catch (err) {
      if (retryCount < this.maxRetries) {
        setTimeout(() => {
          this.sendErrorToServer(error, retryCount + 1)
        }, Math.pow(2, retryCount) * 1000)
      }
    }
  }
}

// 初始化错误处理器
const errorHandler = new ErrorHandler({
  apiEndpoint: '/api/errors',
  enableConsoleLog: import.meta.env.DEV
})
```
