TypeScript 中的函数提供了强大的类型检查功能，包括参数类型、返回值类型、可选参数、默认参数、剩余参数等特性。函数是 TypeScript 应用程序的基本构建块。

## 基本函数类型

### 函数声明
```typescript
// 具名函数
function add(x: number, y: number): number {
    return x + y;
}

// 匿名函数
let myAdd = function(x: number, y: number): number {
    return x + y;
};

// 箭头函数
let myAdd2 = (x: number, y: number): number => {
    return x + y;
};

// 简化的箭头函数
let myAdd3 = (x: number, y: number): number => x + y;
```

### 完整函数类型
```typescript
let myAdd: (x: number, y: number) => number = 
    function(x: number, y: number): number {
        return x + y;
    };

// 类型推断
let myAdd2 = function(x: number, y: number): number {
    return x + y;
}; // TypeScript 可以推断出完整的函数类型
```

## 可选参数和默认参数

### 可选参数
```typescript
function buildName(firstName: string, lastName?: string): string {
    if (lastName) {
        return firstName + " " + lastName;
    } else {
        return firstName;
    }
}

let result1 = buildName("Bob");                  // 正确
let result2 = buildName("Bob", "Adams");         // 正确
// let result3 = buildName("Bob", "Adams", "Sr."); // 错误，参数过多
```

### 默认参数
```typescript
function buildName2(firstName: string, lastName = "Smith"): string {
    return firstName + " " + lastName;
}

let result1 = buildName2("Bob");                  // "Bob Smith"
let result2 = buildName2("Bob", undefined);       // "Bob Smith"
let result3 = buildName2("Bob", "Adams");         // "Bob Adams"

// 默认参数可以在前面
function buildName3(firstName = "Will", lastName: string): string {
    return firstName + " " + lastName;
}

// let result4 = buildName3("Bob"); // 错误，必须提供 lastName
let result5 = buildName3("Bob", "Adams");         // "Bob Adams"
let result6 = buildName3(undefined, "Adams");     // "Will Adams"
```

## 函数重载

```typescript
// 重载签名
function pickCard(x: { suit: string; card: number }[]): number;
function pickCard(x: number): { suit: string; card: number };

// 实现签名
function pickCard(x: any): any {
    if (typeof x == "object") {
        let pickedCard = Math.floor(Math.random() * x.length);
        return pickedCard;
    } else if (typeof x == "number") {
        let pickedSuit = Math.floor(x / 13);
        return { suit: suits[pickedSuit], card: x % 13 };
    }
}

let suits = ["hearts", "spades", "clubs", "diamonds"];

let myDeck = [
    { suit: "diamonds", card: 2 },
    { suit: "spades", card: 10 },
    { suit: "hearts", card: 4 }
];

let pickedCard1 = myDeck[pickCard(myDeck)];
console.log("card: " + pickedCard1.card + " of " + pickedCard1.suit);

let pickedCard2 = pickCard(15);
console.log("card: " + pickedCard2.card + " of " + pickedCard2.suit);
```

### 更复杂的重载示例
```typescript
interface Calculator {
    add(a: number, b: number): number;
    add(a: string, b: string): string;
    add(a: number[], b: number[]): number[];
}

class SimpleCalculator implements Calculator {
    add(a: number, b: number): number;
    add(a: string, b: string): string;
    add(a: number[], b: number[]): number[];
    add(a: any, b: any): any {
        if (typeof a === 'number' && typeof b === 'number') {
            return a + b;
        }
        if (typeof a === 'string' && typeof b === 'string') {
            return a + b;
        }
        if (Array.isArray(a) && Array.isArray(b)) {
            return [...a, ...b];
        }
        throw new Error('Invalid arguments');
    }
}

const calc = new SimpleCalculator();
console.log(calc.add(1, 2));           // 3
console.log(calc.add("Hello", " World")); // "Hello World"
console.log(calc.add([1, 2], [3, 4]));   // [1, 2, 3, 4]
```

## this 参数

### 指定 this 类型
```typescript
interface Card {
    suit: string;
    card: number;
}

interface Deck {
    suits: string[];
    cards: number[];
    createCardPicker(this: Deck): () => Card;
}

let deck: Deck = {
    suits: ["hearts", "spades", "clubs", "diamonds"],
    cards: Array(52),
    createCardPicker: function(this: Deck) {
        return () => {
            let pickedCard = Math.floor(Math.random() * 52);
            let pickedSuit = Math.floor(pickedCard / 13);
            
            return { suit: this.suits[pickedSuit], card: pickedCard % 13 };
        }
    }
}

let cardPicker = deck.createCardPicker();
let pickedCard = cardPicker();
console.log("card: " + pickedCard.card + " of " + pickedCard.suit);
```

### this 参数在回调函数中
```typescript
interface UIElement {
    addClickListener(onclick: (this: void, e: Event) => void): void;
}

class Handler {
    info: string = "Handler info";
    
    onClickBad(this: Handler, e: Event) {
        // 这里使用了 this，但在回调中 this 会是 void
        this.info = e.type; // 错误！
    }
    
    onClickGood = (e: Event) => {
        // 箭头函数不会改变 this 的指向
        this.info = e.type; // 正确
    }
}

let h = new Handler();
// uiElement.addClickListener(h.onClickBad); // 错误
// uiElement.addClickListener(h.onClickGood); // 正确
```

## 高阶函数

### 函数作为参数
```typescript
function applyOperation(
    x: number, 
    y: number, 
    operation: (a: number, b: number) => number
): number {
    return operation(x, y);
}

const add = (a: number, b: number): number => a + b;
const multiply = (a: number, b: number): number => a * b;

console.log(applyOperation(5, 3, add));      // 8
console.log(applyOperation(5, 3, multiply)); // 15
```

### 函数作为返回值
```typescript
function createMultiplier(multiplier: number): (x: number) => number {
    return function(x: number): number {
        return x * multiplier;
    };
}

const double = createMultiplier(2);
const triple = createMultiplier(3);

console.log(double(5)); // 10
console.log(triple(5)); // 15
```

### 柯里化函数
```typescript
function curry<T, U, V>(
    fn: (arg1: T, arg2: U) => V
): (arg1: T) => (arg2: U) => V {
    return function(arg1: T) {
        return function(arg2: U) {
            return fn(arg1, arg2);
        };
    };
}

const add = (x: number, y: number): number => x + y;
const curriedAdd = curry(add);

const add5 = curriedAdd(5);
console.log(add5(3)); // 8
console.log(add5(7)); // 12
```

## 泛型函数

### 基本泛型函数
```typescript
function identity<T>(arg: T): T {
    return arg;
}

let output1 = identity<string>("myString");
let output2 = identity<number>(100);

// 类型推断
let output3 = identity("myString"); // TypeScript 推断 T 为 string
let output4 = identity(100);        // TypeScript 推断 T 为 number
```

### 泛型约束
```typescript
interface Lengthwise {
    length: number;
}

function loggingIdentity<T extends Lengthwise>(arg: T): T {
    console.log(arg.length); // 现在我们知道 arg 有 length 属性
    return arg;
}

loggingIdentity("hello");     // 正确，string 有 length 属性
loggingIdentity([1, 2, 3]);   // 正确，array 有 length 属性
// loggingIdentity(3);        // 错误，number 没有 length 属性
```

## 实例演示：数据处理管道

```typescript
// 定义数据类型
interface User {
    id: number;
    name: string;
    email: string;
    age: number;
    isActive: boolean;
    tags: string[];
}

interface ProcessingResult<T> {
    data: T[];
    count: number;
    processingTime: number;
}

// 过滤器函数类型
type FilterFunction<T> = (item: T) => boolean;
type MapFunction<T, U> = (item: T) => U;
type ReduceFunction<T, U> = (accumulator: U, current: T, index: number) => U;

// 数据处理管道类
class DataPipeline<T> {
    private data: T[];
    
    constructor(initialData: T[]) {
        this.data = [...initialData];
    }
    
    // 过滤数据
    filter(predicate: FilterFunction<T>): DataPipeline<T> {
        this.data = this.data.filter(predicate);
        return this;
    }
    
    // 映射数据
    map<U>(mapper: MapFunction<T, U>): DataPipeline<U> {
        const mappedData = this.data.map(mapper);
        return new DataPipeline(mappedData);
    }
    
    // 排序数据
    sort(compareFn?: (a: T, b: T) => number): DataPipeline<T> {
        this.data = [...this.data].sort(compareFn);
        return this;
    }
    
    // 获取前 n 个元素
    take(count: number): DataPipeline<T> {
        this.data = this.data.slice(0, count);
        return this;
    }
    
    // 跳过前 n 个元素
    skip(count: number): DataPipeline<T> {
        this.data = this.data.slice(count);
        return this;
    }
    
    // 归约操作
    reduce<U>(reducer: ReduceFunction<T, U>, initialValue: U): U {
        return this.data.reduce(reducer, initialValue);
    }
    
    // 获取结果
    toArray(): T[] {
        return [...this.data];
    }
    
    // 获取处理结果
    getResult(): ProcessingResult<T> {
        const startTime = performance.now();
        const result = this.toArray();
        const endTime = performance.now();
        
        return {
            data: result,
            count: result.length,
            processingTime: endTime - startTime
        };
    }
}

// 工具函数集合
class DataUtils {
    // 创建用户过滤器
    static createUserFilter(options: {
        minAge?: number;
        maxAge?: number;
        isActive?: boolean;
        hasTag?: string;
    }): FilterFunction<User> {
        return (user: User) => {
            if (options.minAge !== undefined && user.age < options.minAge) {
                return false;
            }
            if (options.maxAge !== undefined && user.age > options.maxAge) {
                return false;
            }
            if (options.isActive !== undefined && user.isActive !== options.isActive) {
                return false;
            }
            if (options.hasTag && !user.tags.includes(options.hasTag)) {
                return false;
            }
            return true;
        };
    }
    
    // 创建排序函数
    static createSorter<T>(
        key: keyof T,
        direction: 'asc' | 'desc' = 'asc'
    ): (a: T, b: T) => number {
        return (a: T, b: T) => {
            const aVal = a[key];
            const bVal = b[key];
            
            if (aVal < bVal) return direction === 'asc' ? -1 : 1;
            if (aVal > bVal) return direction === 'asc' ? 1 : -1;
            return 0;
        };
    }
    
    // 分组函数
    static groupBy<T, K extends string | number>(
        array: T[],
        keySelector: (item: T) => K
    ): Record<K, T[]> {
        return array.reduce((groups, item) => {
            const key = keySelector(item);
            if (!groups[key]) {
                groups[key] = [];
            }
            groups[key].push(item);
            return groups;
        }, {} as Record<K, T[]>);
    }
    
    // 统计函数
    static getStatistics(numbers: number[]): {
        count: number;
        sum: number;
        average: number;
        min: number;
        max: number;
    } {
        if (numbers.length === 0) {
            return { count: 0, sum: 0, average: 0, min: 0, max: 0 };
        }
        
        const sum = numbers.reduce((acc, num) => acc + num, 0);
        const min = Math.min(...numbers);
        const max = Math.max(...numbers);
        
        return {
            count: numbers.length,
            sum,
            average: sum / numbers.length,
            min,
            max
        };
    }
}

// 异步数据处理函数
async function processDataAsync<T, U>(
    data: T[],
    processor: (item: T) => Promise<U>,
    batchSize: number = 10
): Promise<U[]> {
    const results: U[] = [];
    
    for (let i = 0; i < data.length; i += batchSize) {
        const batch = data.slice(i, i + batchSize);
        const batchResults = await Promise.all(
            batch.map(item => processor(item))
        );
        results.push(...batchResults);
    }
    
    return results;
}

// 使用示例
const users: User[] = [
    { id: 1, name: "Alice", email: "alice@example.com", age: 25, isActive: true, tags: ["developer", "typescript"] },
    { id: 2, name: "Bob", email: "bob@example.com", age: 30, isActive: false, tags: ["designer", "ui"] },
    { id: 3, name: "Charlie", email: "charlie@example.com", age: 35, isActive: true, tags: ["developer", "javascript"] },
    { id: 4, name: "Diana", email: "diana@example.com", age: 28, isActive: true, tags: ["manager", "typescript"] },
    { id: 5, name: "Eve", email: "eve@example.com", age: 22, isActive: false, tags: ["intern", "learning"] }
];

// 数据处理管道示例
console.log("=== 数据处理管道示例 ===");

const activeTypescriptDevelopers = new DataPipeline(users)
    .filter(DataUtils.createUserFilter({ isActive: true, hasTag: "typescript" }))
    .sort(DataUtils.createSorter('age', 'desc'))
    .map(user => ({
        name: user.name,
        email: user.email,
        age: user.age,
        experience: user.age > 30 ? 'Senior' : 'Junior'
    }))
    .getResult();

console.log("Active TypeScript Developers:", activeTypescriptDevelopers);

// 统计分析
const ages = users.map(user => user.age);
const ageStats = DataUtils.getStatistics(ages);
console.log("Age Statistics:", ageStats);

// 分组分析
const usersByStatus = DataUtils.groupBy(users, user => user.isActive ? 'active' : 'inactive');
console.log("Users by Status:", usersByStatus);

// 异步处理示例
async function enrichUserData(user: User): Promise<User & { enrichedData: string }> {
    // 模拟异步操作
    await new Promise(resolve => setTimeout(resolve, 100));
    return {
        ...user,
        enrichedData: `Processed user ${user.name} at ${new Date().toISOString()}`
    };
}

// 异步处理用户数据
processDataAsync(users.slice(0, 3), enrichUserData, 2)
    .then(enrichedUsers => {
        console.log("Enriched Users:", enrichedUsers);
    });

// 函数组合示例
const compose = <T, U, V>(f: (x: U) => V, g: (x: T) => U) => (x: T): V => f(g(x));

const addOne = (x: number): number => x + 1;
const double = (x: number): number => x * 2;
const addOneThenDouble = compose(double, addOne);

console.log("Composed function result:", addOneThenDouble(5)); // (5 + 1) * 2 = 12
```

这个例子展示了 TypeScript 函数的高级特性，包括泛型函数、高阶函数、函数重载、异步处理等，构建了一个完整的数据处理管道系统。
