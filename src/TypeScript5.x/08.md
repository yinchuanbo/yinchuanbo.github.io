TypeScript 的高级类型系统提供了强大的类型操作和组合能力，包括联合类型、交叉类型、条件类型、映射类型等。这些特性让我们能够创建更精确、更灵活的类型定义。

## 联合类型 (Union Types)

### 基本联合类型
```typescript
type StringOrNumber = string | number;

function formatValue(value: StringOrNumber): string {
    if (typeof value === 'string') {
        return value.toUpperCase();
    } else {
        return value.toString();
    }
}

console.log(formatValue("hello")); // "HELLO"
console.log(formatValue(42));      // "42"
```

### 字面量联合类型
```typescript
type Theme = 'light' | 'dark' | 'auto';
type Size = 'small' | 'medium' | 'large';
type Status = 'pending' | 'approved' | 'rejected';

interface Button {
    text: string;
    theme: Theme;
    size: Size;
    disabled?: boolean;
}

const button: Button = {
    text: 'Click me',
    theme: 'dark',
    size: 'medium'
};
```

### 可辨识联合类型
```typescript
interface Square {
    kind: 'square';
    size: number;
}

interface Rectangle {
    kind: 'rectangle';
    width: number;
    height: number;
}

interface Circle {
    kind: 'circle';
    radius: number;
}

type Shape = Square | Rectangle | Circle;

function calculateArea(shape: Shape): number {
    switch (shape.kind) {
        case 'square':
            return shape.size * shape.size;
        case 'rectangle':
            return shape.width * shape.height;
        case 'circle':
            return Math.PI * shape.radius * shape.radius;
        default:
            // 确保所有情况都被处理
            const _exhaustiveCheck: never = shape;
            return _exhaustiveCheck;
    }
}

const square: Square = { kind: 'square', size: 10 };
const rectangle: Rectangle = { kind: 'rectangle', width: 10, height: 5 };
const circle: Circle = { kind: 'circle', radius: 5 };

console.log(calculateArea(square));    // 100
console.log(calculateArea(rectangle)); // 50
console.log(calculateArea(circle));    // 78.54
```

## 交叉类型 (Intersection Types)

### 基本交叉类型
```typescript
interface Flyable {
    fly(): void;
    altitude: number;
}

interface Swimmable {
    swim(): void;
    depth: number;
}

type FlyingFish = Flyable & Swimmable;

class Duck implements FlyingFish {
    altitude: number = 0;
    depth: number = 0;
    
    fly(): void {
        this.altitude = 100;
        console.log(`Flying at altitude ${this.altitude}m`);
    }
    
    swim(): void {
        this.depth = 5;
        console.log(`Swimming at depth ${this.depth}m`);
    }
}

const duck = new Duck();
duck.fly();  // Flying at altitude 100m
duck.swim(); // Swimming at depth 5m
```

### 混合对象类型
```typescript
interface Person {
    name: string;
    age: number;
}

interface Employee {
    employeeId: string;
    department: string;
    salary: number;
}

type PersonEmployee = Person & Employee;

const employee: PersonEmployee = {
    name: 'Alice',
    age: 30,
    employeeId: 'EMP001',
    department: 'Engineering',
    salary: 75000
};

// 工具函数
function createEmployee(
    person: Person,
    employment: Employee
): PersonEmployee {
    return { ...person, ...employment };
}
```

## 条件类型 (Conditional Types)

### 基本条件类型
```typescript
type IsString<T> = T extends string ? true : false;

type Test1 = IsString<string>;  // true
type Test2 = IsString<number>;  // false
type Test3 = IsString<'hello'>; // true
```

### 实用条件类型
```typescript
// 排除 null 和 undefined
type NonNullable<T> = T extends null | undefined ? never : T;

type Test4 = NonNullable<string | null>;      // string
type Test5 = NonNullable<number | undefined>; // number

// 获取函数返回类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never;

function getString(): string { return 'hello'; }
function getNumber(): number { return 42; }

type StringReturn = ReturnType<typeof getString>; // string
type NumberReturn = ReturnType<typeof getNumber>; // number

// 获取函数参数类型
type Parameters<T> = T extends (...args: infer P) => any ? P : never;

function add(a: number, b: number): number { return a + b; }
type AddParams = Parameters<typeof add>; // [number, number]
```

### 分布式条件类型
```typescript
type ToArray<T> = T extends any ? T[] : never;

type Test6 = ToArray<string | number>; // string[] | number[]

// 过滤类型
type Filter<T, U> = T extends U ? T : never;

type Test7 = Filter<'a' | 'b' | 'c' | 1 | 2, string>; // 'a' | 'b' | 'c'
```

## 映射类型 (Mapped Types)

### 基本映射类型
```typescript
type Readonly<T> = {
    readonly [P in keyof T]: T[P];
};

type Partial<T> = {
    [P in keyof T]?: T[P];
};

type Required<T> = {
    [P in keyof T]-?: T[P];
};

interface User {
    id: number;
    name: string;
    email?: string;
}

type ReadonlyUser = Readonly<User>;
// { readonly id: number; readonly name: string; readonly email?: string; }

type PartialUser = Partial<User>;
// { id?: number; name?: string; email?: string; }

type RequiredUser = Required<User>;
// { id: number; name: string; email: string; }
```

### 高级映射类型
```typescript
// 选择特定属性
type Pick<T, K extends keyof T> = {
    [P in K]: T[P];
};

// 排除特定属性
type Omit<T, K extends keyof T> = Pick<T, Exclude<keyof T, K>>;

// 创建记录类型
type Record<K extends keyof any, T> = {
    [P in K]: T;
};

interface Product {
    id: string;
    name: string;
    price: number;
    description: string;
    category: string;
}

type ProductSummary = Pick<Product, 'id' | 'name' | 'price'>;
// { id: string; name: string; price: number; }

type CreateProduct = Omit<Product, 'id'>;
// { name: string; price: number; description: string; category: string; }

type ProductCategories = Record<string, Product[]>;
// { [key: string]: Product[]; }
```

### 键重映射
```typescript
// TypeScript 4.1+ 特性
type Getters<T> = {
    [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K];
};

interface Person {
    name: string;
    age: number;
    email: string;
}

type PersonGetters = Getters<Person>;
// {
//     getName: () => string;
//     getAge: () => number;
//     getEmail: () => string;
// }

// 过滤属性
type PickByType<T, U> = {
    [K in keyof T as T[K] extends U ? K : never]: T[K];
};

interface Mixed {
    id: number;
    name: string;
    count: number;
    active: boolean;
    tags: string[];
}

type StringProps = PickByType<Mixed, string>;
// { name: string; }

type NumberProps = PickByType<Mixed, number>;
// { id: number; count: number; }
```

## 模板字面量类型

### 基本模板字面量类型
```typescript
type World = 'world';
type Greeting = `hello ${World}`; // 'hello world'

type EmailLocaleIDs = 'welcome_email' | 'email_heading';
type FooterLocaleIDs = 'footer_title' | 'footer_sendoff';

type AllLocaleIDs = `${EmailLocaleIDs | FooterLocaleIDs}_id`;
// 'welcome_email_id' | 'email_heading_id' | 'footer_title_id' | 'footer_sendoff_id'
```

### 模板字面量类型的实际应用
```typescript
// 事件系统
type EventNames = 'click' | 'focus' | 'blur';
type EventHandlers = {
    [K in EventNames as `on${Capitalize<K>}`]: (event: Event) => void;
};
// {
//     onClick: (event: Event) => void;
//     onFocus: (event: Event) => void;
//     onBlur: (event: Event) => void;
// }

// CSS 属性
type CSSProperties = 'margin' | 'padding' | 'border';
type CSSDirections = 'top' | 'right' | 'bottom' | 'left';
type DetailedCSSProperties = `${CSSProperties}-${CSSDirections}`;
// 'margin-top' | 'margin-right' | ... | 'border-left'
```

## 实例演示：类型安全的状态管理系统

```typescript
// 状态类型定义
interface AppState {
    user: {
        id: string;
        name: string;
        email: string;
        isAuthenticated: boolean;
    } | null;
    products: {
        items: Product[];
        loading: boolean;
        error: string | null;
    };
    cart: {
        items: CartItem[];
        total: number;
    };
    ui: {
        theme: 'light' | 'dark';
        sidebarOpen: boolean;
        notifications: Notification[];
    };
}

interface Product {
    id: string;
    name: string;
    price: number;
    category: string;
}

interface CartItem {
    productId: string;
    quantity: number;
    price: number;
}

interface Notification {
    id: string;
    type: 'info' | 'success' | 'warning' | 'error';
    message: string;
    timestamp: number;
}

// 动作类型系统
type ActionType = 
    | 'SET_USER'
    | 'LOGOUT_USER'
    | 'SET_PRODUCTS'
    | 'SET_PRODUCTS_LOADING'
    | 'SET_PRODUCTS_ERROR'
    | 'ADD_TO_CART'
    | 'REMOVE_FROM_CART'
    | 'UPDATE_CART_QUANTITY'
    | 'CLEAR_CART'
    | 'SET_THEME'
    | 'TOGGLE_SIDEBAR'
    | 'ADD_NOTIFICATION'
    | 'REMOVE_NOTIFICATION';

// 动作创建器类型
type ActionCreators = {
    [K in ActionType as `${Lowercase<K>}`]: K extends 'SET_USER'
        ? (user: AppState['user']) => Action<K, AppState['user']>
        : K extends 'SET_PRODUCTS'
        ? (products: Product[]) => Action<K, Product[]>
        : K extends 'SET_PRODUCTS_LOADING'
        ? (loading: boolean) => Action<K, boolean>
        : K extends 'SET_PRODUCTS_ERROR'
        ? (error: string | null) => Action<K, string | null>
        : K extends 'ADD_TO_CART'
        ? (item: CartItem) => Action<K, CartItem>
        : K extends 'REMOVE_FROM_CART'
        ? (productId: string) => Action<K, string>
        : K extends 'UPDATE_CART_QUANTITY'
        ? (payload: { productId: string; quantity: number }) => Action<K, { productId: string; quantity: number }>
        : K extends 'SET_THEME'
        ? (theme: AppState['ui']['theme']) => Action<K, AppState['ui']['theme']>
        : K extends 'ADD_NOTIFICATION'
        ? (notification: Notification) => Action<K, Notification>
        : K extends 'REMOVE_NOTIFICATION'
        ? (id: string) => Action<K, string>
        : () => Action<K, undefined>;
};

// 基础动作接口
interface Action<T extends ActionType, P = undefined> {
    type: T;
    payload: P;
}

// 所有可能的动作类型
type AppAction = 
    | Action<'SET_USER', AppState['user']>
    | Action<'LOGOUT_USER', undefined>
    | Action<'SET_PRODUCTS', Product[]>
    | Action<'SET_PRODUCTS_LOADING', boolean>
    | Action<'SET_PRODUCTS_ERROR', string | null>
    | Action<'ADD_TO_CART', CartItem>
    | Action<'REMOVE_FROM_CART', string>
    | Action<'UPDATE_CART_QUANTITY', { productId: string; quantity: number }>
    | Action<'CLEAR_CART', undefined>
    | Action<'SET_THEME', AppState['ui']['theme']>
    | Action<'TOGGLE_SIDEBAR', undefined>
    | Action<'ADD_NOTIFICATION', Notification>
    | Action<'REMOVE_NOTIFICATION', string>;

// 状态选择器类型
type StateSelector<T> = (state: AppState) => T;

// 深度路径类型
type DeepKeys<T> = T extends object
    ? {
        [K in keyof T]: K extends string
            ? T[K] extends object
                ? `${K}` | `${K}.${DeepKeys<T[K]>}`
                : `${K}`
            : never;
    }[keyof T]
    : never;

type AppStateKeys = DeepKeys<AppState>;
// 'user' | 'user.id' | 'user.name' | 'products' | 'products.items' | 'cart.total' | ...

// 根据路径获取类型
type GetByPath<T, P extends string> = P extends `${infer K}.${infer Rest}`
    ? K extends keyof T
        ? GetByPath<T[K], Rest>
        : never
    : P extends keyof T
    ? T[P]
    : never;

// 状态管理器实现
class StateManager {
    private state: AppState;
    private listeners: Array<(state: AppState) => void> = [];
    
    constructor(initialState: AppState) {
        this.state = initialState;
    }
    
    // 获取当前状态
    getState(): AppState {
        return this.state;
    }
    
    // 派发动作
    dispatch(action: AppAction): void {
        this.state = this.reducer(this.state, action);
        this.notifyListeners();
    }
    
    // 订阅状态变化
    subscribe(listener: (state: AppState) => void): () => void {
        this.listeners.push(listener);
        return () => {
            const index = this.listeners.indexOf(listener);
            if (index > -1) {
                this.listeners.splice(index, 1);
            }
        };
    }
    
    // 选择器
    select<T>(selector: StateSelector<T>): T {
        return selector(this.state);
    }
    
    // 根据路径选择状态
    selectByPath<P extends AppStateKeys>(path: P): GetByPath<AppState, P> {
        const keys = path.split('.');
        let result: any = this.state;
        
        for (const key of keys) {
            result = result[key];
            if (result === undefined) break;
        }
        
        return result;
    }
    
    private reducer(state: AppState, action: AppAction): AppState {
        switch (action.type) {
            case 'SET_USER':
                return {
                    ...state,
                    user: action.payload
                };
                
            case 'LOGOUT_USER':
                return {
                    ...state,
                    user: null
                };
                
            case 'SET_PRODUCTS':
                return {
                    ...state,
                    products: {
                        ...state.products,
                        items: action.payload,
                        loading: false,
                        error: null
                    }
                };
                
            case 'SET_PRODUCTS_LOADING':
                return {
                    ...state,
                    products: {
                        ...state.products,
                        loading: action.payload
                    }
                };
                
            case 'ADD_TO_CART':
                const existingItem = state.cart.items.find(
                    item => item.productId === action.payload.productId
                );
                
                let newItems: CartItem[];
                if (existingItem) {
                    newItems = state.cart.items.map(item =>
                        item.productId === action.payload.productId
                            ? { ...item, quantity: item.quantity + action.payload.quantity }
                            : item
                    );
                } else {
                    newItems = [...state.cart.items, action.payload];
                }
                
                return {
                    ...state,
                    cart: {
                        items: newItems,
                        total: newItems.reduce((sum, item) => sum + (item.price * item.quantity), 0)
                    }
                };
                
            case 'SET_THEME':
                return {
                    ...state,
                    ui: {
                        ...state.ui,
                        theme: action.payload
                    }
                };
                
            case 'ADD_NOTIFICATION':
                return {
                    ...state,
                    ui: {
                        ...state.ui,
                        notifications: [...state.ui.notifications, action.payload]
                    }
                };
                
            default:
                return state;
        }
    }
    
    private notifyListeners(): void {
        this.listeners.forEach(listener => listener(this.state));
    }
}

// 动作创建器
const actionCreators = {
    setUser: (user: AppState['user']): Action<'SET_USER', AppState['user']> => ({
        type: 'SET_USER',
        payload: user
    }),
    
    logoutUser: (): Action<'LOGOUT_USER', undefined> => ({
        type: 'LOGOUT_USER',
        payload: undefined
    }),
    
    setProducts: (products: Product[]): Action<'SET_PRODUCTS', Product[]> => ({
        type: 'SET_PRODUCTS',
        payload: products
    }),
    
    addToCart: (item: CartItem): Action<'ADD_TO_CART', CartItem> => ({
        type: 'ADD_TO_CART',
        payload: item
    }),
    
    setTheme: (theme: AppState['ui']['theme']): Action<'SET_THEME', AppState['ui']['theme']> => ({
        type: 'SET_THEME',
        payload: theme
    }),
    
    addNotification: (notification: Notification): Action<'ADD_NOTIFICATION', Notification> => ({
        type: 'ADD_NOTIFICATION',
        payload: notification
    })
};

// 使用示例
const initialState: AppState = {
    user: null,
    products: {
        items: [],
        loading: false,
        error: null
    },
    cart: {
        items: [],
        total: 0
    },
    ui: {
        theme: 'light',
        sidebarOpen: false,
        notifications: []
    }
};

const stateManager = new StateManager(initialState);

// 订阅状态变化
const unsubscribe = stateManager.subscribe((state) => {
    console.log('State updated:', state);
});

// 派发动作
stateManager.dispatch(actionCreators.setUser({
    id: '1',
    name: 'John Doe',
    email: 'john@example.com',
    isAuthenticated: true
}));

stateManager.dispatch(actionCreators.addToCart({
    productId: 'prod1',
    quantity: 2,
    price: 29.99
}));

stateManager.dispatch(actionCreators.setTheme('dark'));

// 使用选择器
const currentUser = stateManager.select(state => state.user);
const cartTotal = stateManager.selectByPath('cart.total');
const isAuthenticated = stateManager.select(state => state.user?.isAuthenticated ?? false);

console.log('Current user:', currentUser);
console.log('Cart total:', cartTotal);
console.log('Is authenticated:', isAuthenticated);

// 清理
unsubscribe();
```

这个例子展示了 TypeScript 高级类型的强大功能，包括联合类型、交叉类型、条件类型、映射类型、模板字面量类型等，构建了一个类型安全的状态管理系统。
