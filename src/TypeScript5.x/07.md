TypeScript 中的模块系统基于 ES6 模块标准，提供了强大的代码组织和重用机制。模块可以导出和导入变量、函数、类、接口等，帮助我们构建大型、可维护的应用程序。

## 基本导出和导入

### 命名导出
```typescript
// math.ts
export const PI = 3.14159;

export function add(x: number, y: number): number {
    return x + y;
}

export function multiply(x: number, y: number): number {
    return x * y;
}

export class Calculator {
    add(x: number, y: number): number {
        return x + y;
    }
}

export interface MathOperation {
    (x: number, y: number): number;
}
```

### 命名导入
```typescript
// main.ts
import { PI, add, multiply, Calculator } from './math';

console.log(PI); // 3.14159
console.log(add(2, 3)); // 5
console.log(multiply(4, 5)); // 20

const calc = new Calculator();
console.log(calc.add(1, 2)); // 3
```

### 重命名导入
```typescript
import { add as sum, multiply as product } from './math';

console.log(sum(2, 3)); // 5
console.log(product(4, 5)); // 20
```

### 命名空间导入
```typescript
import * as MathUtils from './math';

console.log(MathUtils.PI);
console.log(MathUtils.add(2, 3));
const calc = new MathUtils.Calculator();
```

## 混合导出

```typescript
// utils.ts
export const VERSION = '1.0.0';

export function formatDate(date: Date): string {
    return date.toISOString().split('T')[0];
}

export class EventEmitter {
    private events: { [key: string]: Function[] } = {};
    
    on(event: string, callback: Function): void {
        if (!this.events[event]) {
            this.events[event] = [];
        }
        this.events[event].push(callback);
    }
    
    emit(event: string, ...args: any[]): void {
        if (this.events[event]) {
            this.events[event].forEach(callback => callback(...args));
        }
    }
}

// 默认导出
export default class Config {
    private settings: { [key: string]: any } = {};
    
    set(key: string, value: any): void {
        this.settings[key] = value;
    }
    
    get(key: string): any {
        return this.settings[key];
    }
}
```

## 动态导入

### 基本动态导入
```typescript
async function loadMathModule() {
    const mathModule = await import('./math');
    console.log(mathModule.add(2, 3));
    
    const calc = new mathModule.Calculator();
    console.log(calc.add(1, 2));
}

loadMathModule();
```

### 条件动态导入
```typescript
async function loadFeature(featureName: string) {
    let module;
    
    switch (featureName) {
        case 'math':
            module = await import('./math');
            break;
        case 'utils':
            module = await import('./utils');
            break;
        default:
            throw new Error(`Unknown feature: ${featureName}`);
    }
    
    return module;
}

// 使用示例
loadFeature('math').then(mathModule => {
    console.log(mathModule.add(5, 3));
});
```

## 命名空间

### 定义命名空间
```typescript
// geometry.ts
namespace Geometry {
    export interface Point {
        x: number;
        y: number;
    }
    
    export class Circle {
        constructor(
            public center: Point,
            public radius: number
        ) {}
        
        area(): number {
            return Math.PI * this.radius * this.radius;
        }
    }
    
    export class Rectangle {
        constructor(
            public topLeft: Point,
            public width: number,
            public height: number
        ) {}
        
        area(): number {
            return this.width * this.height;
        }
    }
    
    export namespace Utils {
        export function distance(p1: Point, p2: Point): number {
            return Math.sqrt(
                Math.pow(p2.x - p1.x, 2) + Math.pow(p2.y - p1.y, 2)
            );
        }
    }
}

// 使用命名空间
const point1: Geometry.Point = { x: 0, y: 0 };
const point2: Geometry.Point = { x: 3, y: 4 };

const circle = new Geometry.Circle(point1, 5);
console.log(circle.area());

const distance = Geometry.Utils.distance(point1, point2);
console.log(distance); // 5
```

### 跨文件命名空间
```typescript
// shapes.ts
namespace Geometry {
    export class Triangle {
        constructor(
            public a: Point,
            public b: Point,
            public c: Point
        ) {}
        
        area(): number {
            // 使用海伦公式计算三角形面积
            const sideA = Utils.distance(this.b, this.c);
            const sideB = Utils.distance(this.a, this.c);
            const sideC = Utils.distance(this.a, this.b);
            
            const s = (sideA + sideB + sideC) / 2;
            return Math.sqrt(s * (s - sideA) * (s - sideB) * (s - sideC));
        }
    }
}
```

## 模块增强

### 扩展现有模块
```typescript
// 扩展内置模块
declare global {
    interface Array<T> {
        first(): T | undefined;
        last(): T | undefined;
    }
}

Array.prototype.first = function<T>(this: T[]): T | undefined {
    return this.length > 0 ? this[0] : undefined;
};

Array.prototype.last = function<T>(this: T[]): T | undefined {
    return this.length > 0 ? this[this.length - 1] : undefined;
};

// 现在可以使用扩展的方法
const numbers = [1, 2, 3, 4, 5];
console.log(numbers.first()); // 1
console.log(numbers.last());  // 5
```

### 扩展第三方模块
```typescript
// 假设扩展 lodash
declare module 'lodash' {
    interface LoDashStatic {
        customMethod(value: any): any;
    }
}

// 实现扩展方法
import * as _ from 'lodash';

(_.as any).customMethod = function(value: any): any {
    return `Custom: ${value}`;
};
```

```typescript
// utils/validation.ts - 验证工具
export class ValidationError extends Error {
    constructor(message: string, public field?: string) {
        super(message);
        this.name = 'ValidationError';
    }
}

export const validators = {
    email: (email: string): boolean => {
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        return emailRegex.test(email);
    },
    
    required: (value: any): boolean => {
        return value !== null && value !== undefined && value !== '';
    },
    
    minLength: (value: string, min: number): boolean => {
        return value && value.length >= min;
    },
    
    positive: (value: number): boolean => {
        return value > 0;
    }
};

export function validateUser(user: Partial<User>): void {
    if (!validators.required(user.username)) {
        throw new ValidationError('Username is required', 'username');
    }
    
    if (!validators.minLength(user.username!, 3)) {
        throw new ValidationError('Username must be at least 3 characters', 'username');
    }
    
    if (!validators.required(user.email)) {
        throw new ValidationError('Email is required', 'email');
    }
    
    if (!validators.email(user.email!)) {
        throw new ValidationError('Invalid email format', 'email');
    }
}

export function validateProduct(product: Partial<Product>): void {
    if (!validators.required(product.name)) {
        throw new ValidationError('Product name is required', 'name');
    }
    
    if (!validators.positive(product.price!)) {
        throw new ValidationError('Price must be positive', 'price');
    }
    
    if (!validators.positive(product.stock!)) {
        throw new ValidationError('Stock must be positive', 'stock');
    }
}
```

```typescript
// services/base.ts - 基础服务类
import { BaseEntity, ApiResponse } from '../types';

export abstract class BaseService<T extends BaseEntity> {
    protected baseUrl: string;
    
    constructor(baseUrl: string) {
        this.baseUrl = baseUrl;
    }
    
    protected async request<U>(
        endpoint: string,
        options: RequestInit = {}
    ): Promise<ApiResponse<U>> {
        try {
            const response = await fetch(`${this.baseUrl}${endpoint}`, {
                headers: {
                    'Content-Type': 'application/json',
                    ...options.headers
                },
                ...options
            });
            
            const data = await response.json();
            
            if (!response.ok) {
                return {
                    success: false,
                    error: data.message || 'Request failed'
                };
            }
            
            return {
                success: true,
                data
            };
        } catch (error) {
            return {
                success: false,
                error: error instanceof Error ? error.message : 'Unknown error'
            };
        }
    }
    
    abstract create(entity: Omit<T, keyof BaseEntity>): Promise<ApiResponse<T>>;
    abstract getById(id: string): Promise<ApiResponse<T>>;
    abstract update(id: string, updates: Partial<T>): Promise<ApiResponse<T>>;
    abstract delete(id: string): Promise<ApiResponse<boolean>>;
}
```

```typescript
// services/user.ts - 用户服务
import { BaseService } from './base';
import { User, ApiResponse } from '../types';
import { validateUser } from '../utils/validation';

export class UserService extends BaseService<User> {
    constructor() {
        super('/api/users');
    }
    
    async create(userData: Omit<User, keyof BaseEntity>): Promise<ApiResponse<User>> {
        try {
            validateUser(userData);
            return await this.request<User>('/', {
                method: 'POST',
                body: JSON.stringify(userData)
            });
        } catch (error) {
            return {
                success: false,
                error: error instanceof Error ? error.message : 'Validation failed'
            };
        }
    }
    
    async getById(id: string): Promise<ApiResponse<User>> {
        return await this.request<User>(`/${id}`);
    }
    
    async update(id: string, updates: Partial<User>): Promise<ApiResponse<User>> {
        try {
            if (Object.keys(updates).length > 0) {
                validateUser(updates);
            }
            return await this.request<User>(`/${id}`, {
                method: 'PUT',
                body: JSON.stringify(updates)
            });
        } catch (error) {
            return {
                success: false,
                error: error instanceof Error ? error.message : 'Validation failed'
            };
        }
    }
    
    async delete(id: string): Promise<ApiResponse<boolean>> {
        return await this.request<boolean>(`/${id}`, {
            method: 'DELETE'
        });
    }
    
    async getByEmail(email: string): Promise<ApiResponse<User>> {
        return await this.request<User>(`/email/${email}`);
    }
    
    async searchUsers(query: string): Promise<ApiResponse<User[]>> {
        return await this.request<User[]>(`/search?q=${encodeURIComponent(query)}`);
    }
}

// 创建单例实例
export const userService = new UserService();
```

```typescript
// app.ts - 应用程序使用示例
import { api, initializeApp, User, Product, ValidationError } from './index';

// 初始化应用
initializeApp({ apiBaseUrl: 'https://api.example.com' });

// 使用示例
async function demonstrateApp() {
    try {
        // 创建用户
        const userResult = await api.users.create({
            username: 'john_doe',
            email: 'john@example.com',
            firstName: 'John',
            lastName: 'Doe'
        });
        
        if (userResult.success && userResult.data) {
            console.log('User created:', userResult.data);
            
            // 创建产品
            const productResult = await api.products.create({
                name: 'Laptop',
                description: 'High-performance laptop',
                price: 999.99,
                category: 'Electronics',
                stock: 10,
                images: ['laptop1.jpg', 'laptop2.jpg']
            });
            
            if (productResult.success && productResult.data) {
                console.log('Product created:', productResult.data);
                
                // 搜索产品
                const searchResult = await api.products.searchProducts('laptop');
                if (searchResult.success) {
                    console.log('Search results:', searchResult.data);
                }
            }
        }
    } catch (error) {
        if (error instanceof ValidationError) {
            console.error(`Validation error in ${error.field}: ${error.message}`);
        } else {
            console.error('Unexpected error:', error);
        }
    }
}
```
