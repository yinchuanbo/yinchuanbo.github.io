TypeScript 的类支持面向对象编程的所有特性，包括继承、封装、多态等，并添加了类型注解和访问修饰符。

## 基本类语法

### 类的定义
```typescript
class Person {
    // 属性声明
    name: string;
    age: number;
    
    // 构造函数
    constructor(name: string, age: number) {
        this.name = name;
        this.age = age;
    }
    
    // 方法
    greet(): string {
        return `Hello, I'm ${this.name} and I'm ${this.age} years old.`;
    }
    
    // 获取器
    get info(): string {
        return `${this.name} (${this.age})`;
    }
    
    // 设置器
    set info(value: string) {
        const [name, ageStr] = value.split(' (');
        this.name = name;
        this.age = parseInt(ageStr.replace(')', ''));
    }
}

// 使用类
const person = new Person("Alice", 30);
console.log(person.greet()); // "Hello, I'm Alice and I'm 30 years old."
console.log(person.info); // "Alice (30)"
person.info = "Bob (25)";
console.log(person.name); // "Bob"
```

## 访问修饰符

### public、private、protected
```typescript
class Animal {
    public name: string; // 公共属性，默认修饰符
    private age: number; // 私有属性，只能在类内部访问
    protected species: string; // 受保护属性，可在子类中访问
    
    constructor(name: string, age: number, species: string) {
        this.name = name;
        this.age = age;
        this.species = species;
    }
    
    public makeSound(): void {
        console.log(`${this.name} makes a sound`);
    }
    
    private getAge(): number {
        return this.age;
    }
    
    protected getSpecies(): string {
        return this.species;
    }
    
    // 公共方法访问私有属性
    public getInfo(): string {
        return `${this.name} is ${this.getAge()} years old`;
    }
}

class Dog extends Animal {
    constructor(name: string, age: number) {
        super(name, age, "Canine");
    }
    
    public bark(): void {
        console.log(`${this.name} barks!`);
        // 可以访问 protected 属性
        console.log(`Species: ${this.getSpecies()}`);
        // console.log(this.age); // Error: 无法访问私有属性
    }
}

const dog = new Dog("Buddy", 3);
console.log(dog.name); // OK: public
// console.log(dog.age); // Error: private
// console.log(dog.species); // Error: protected
```

## 继承

### 基本继承
```typescript
class Vehicle {
    protected speed: number = 0;
    
    constructor(public brand: string) {}
    
    accelerate(amount: number): void {
        this.speed += amount;
        console.log(`${this.brand} accelerated to ${this.speed} km/h`);
    }
    
    brake(): void {
        this.speed = 0;
        console.log(`${this.brand} stopped`);
    }
}

class Car extends Vehicle {
    constructor(brand: string, public model: string) {
        super(brand); // 调用父类构造函数
    }
    
    // 重写父类方法
    accelerate(amount: number): void {
        super.accelerate(amount); // 调用父类方法
        console.log(`Car ${this.model} is now moving`);
    }
    
    // 新增方法
    honk(): void {
        console.log(`${this.brand} ${this.model} honks!`);
    }
}

class Motorcycle extends Vehicle {
    constructor(brand: string, public engineSize: number) {
        super(brand);
    }
    
    wheelie(): void {
        console.log(`${this.brand} motorcycle does a wheelie!`);
    }
}

const car = new Car("Toyota", "Camry");
car.accelerate(50); // 调用重写的方法
car.honk();

const bike = new Motorcycle("Honda", 600);
bike.accelerate(30);
bike.wheelie();
```

### 抽象类
```typescript
abstract class Shape {
    abstract area(): number; // 抽象方法，必须在子类中实现
    abstract perimeter(): number;
    
    // 具体方法，可以在子类中使用
    display(): void {
        console.log(`Area: ${this.area()}, Perimeter: ${this.perimeter()}`);
    }
}

class Rectangle extends Shape {
    constructor(private width: number, private height: number) {
        super();
    }
    
    area(): number {
        return this.width * this.height;
    }
    
    perimeter(): number {
        return 2 * (this.width + this.height);
    }
}

class Circle extends Shape {
    constructor(private radius: number) {
        super();
    }
    
    area(): number {
        return Math.PI * this.radius * this.radius;
    }
    
    perimeter(): number {
        return 2 * Math.PI * this.radius;
    }
}

const rectangle = new Rectangle(5, 3);
const circle = new Circle(4);

rectangle.display(); // Area: 15, Perimeter: 16
circle.display(); // Area: 50.26..., Perimeter: 25.13...

// const shape = new Shape(); // Error: 无法实例化抽象类
```

## 泛型类

### 基本泛型类
```typescript
class Container<T> {
    private items: T[] = [];
    
    add(item: T): void {
        this.items.push(item);
    }
    
    get(index: number): T | undefined {
        return this.items[index];
    }
    
    getAll(): T[] {
        return [...this.items];
    }
    
    size(): number {
        return this.items.length;
    }
    
    clear(): void {
        this.items = [];
    }
}

// 使用泛型类
const stringContainer = new Container<string>();
stringContainer.add("hello");
stringContainer.add("world");
console.log(stringContainer.getAll()); // ["hello", "world"]

const numberContainer = new Container<number>();
numberContainer.add(1);
numberContainer.add(2);
console.log(numberContainer.get(0)); // 1
```

### 泛型约束
```typescript
interface Identifiable {
    id: number;
}

class Repository<T extends Identifiable> {
    private items: T[] = [];
    
    add(item: T): void {
        this.items.push(item);
    }
    
    findById(id: number): T | undefined {
        return this.items.find(item => item.id === id);
    }
    
    update(id: number, updates: Partial<T>): boolean {
        const index = this.items.findIndex(item => item.id === id);
        if (index !== -1) {
            this.items[index] = { ...this.items[index], ...updates };
            return true;
        }
        return false;
    }
    
    delete(id: number): boolean {
        const index = this.items.findIndex(item => item.id === id);
        if (index !== -1) {
            this.items.splice(index, 1);
            return true;
        }
        return false;
    }
}

interface User extends Identifiable {
    name: string;
    email: string;
}

const userRepo = new Repository<User>();
userRepo.add({ id: 1, name: "Alice", email: "alice@example.com" });
userRepo.add({ id: 2, name: "Bob", email: "bob@example.com" });

const user = userRepo.findById(1);
console.log(user); // { id: 1, name: "Alice", email: "alice@example.com" }
```

## 装饰器（实验性功能）

### 类装饰器
```typescript
function sealed(constructor: Function) {
    Object.seal(constructor);
    Object.seal(constructor.prototype);
}

@sealed
class BugReport {
    type = "report";
    title: string;
    
    constructor(t: string) {
        this.title = t;
    }
}

// 装饰器工厂
function classLogger<T extends { new (...args: any[]): {} }>(constructor: T) {
    return class extends constructor {
        constructor(...args: any[]) {
            console.log(`Creating instance of ${constructor.name}`);
            super(...args);
        }
    };
}

@classLogger
class Service {
    constructor(public name: string) {}
}

const service = new Service("MyService"); // 输出: Creating instance of Service
```

### 方法装饰器
```typescript
function log(target: any, propertyName: string, descriptor: PropertyDescriptor) {
    const method = descriptor.value;
    
    descriptor.value = function (...args: any[]) {
        console.log(`Calling ${propertyName} with arguments:`, args);
        const result = method.apply(this, args);
        console.log(`${propertyName} returned:`, result);
        return result;
    };
}

class Calculator {
    @log
    add(a: number, b: number): number {
        return a + b;
    }
    
    @log
    multiply(a: number, b: number): number {
        return a * b;
    }
}

const calc = new Calculator();
calc.add(2, 3); // 输出日志信息
```

### 数据模型类
```typescript
interface Serializable {
    serialize(): string;
    deserialize(data: string): void;
}

abstract class BaseModel implements Serializable {
    protected createdAt: Date;
    protected updatedAt: Date;
    
    constructor() {
        this.createdAt = new Date();
        this.updatedAt = new Date();
    }
    
    serialize(): string {
        return JSON.stringify(this);
    }
    
    deserialize(data: string): void {
        const parsed = JSON.parse(data);
        Object.assign(this, parsed);
    }
    
    touch(): void {
        this.updatedAt = new Date();
    }
    
    abstract validate(): boolean;
}

class UserModel extends BaseModel {
    constructor(
        public id: number,
        public username: string,
        public email: string,
        private password: string
    ) {
        super();
    }
    
    validate(): boolean {
        return this.username.length > 0 && 
               this.email.includes('@') && 
               this.password.length >= 8;
    }
    
    changePassword(newPassword: string): boolean {
        if (newPassword.length >= 8) {
            this.password = newPassword;
            this.touch();
            return true;
        }
        return false;
    }
    
    // 重写序列化，排除密码
    serialize(): string {
        const { password, ...safeData } = this;
        return JSON.stringify(safeData);
    }
}
```

### 事件发射器类
```typescript
type EventListener<T = any> = (data: T) => void;

class EventEmitter<T extends Record<string, any> = Record<string, any>> {
    private listeners: Map<keyof T, EventListener<T[keyof T]>[]> = new Map();
    
    on<K extends keyof T>(event: K, listener: EventListener<T[K]>): void {
        if (!this.listeners.has(event)) {
            this.listeners.set(event, []);
        }
        this.listeners.get(event)!.push(listener);
    }
    
    off<K extends keyof T>(event: K, listener: EventListener<T[K]>): void {
        const eventListeners = this.listeners.get(event);
        if (eventListeners) {
            const index = eventListeners.indexOf(listener);
            if (index > -1) {
                eventListeners.splice(index, 1);
            }
        }
    }
    
    emit<K extends keyof T>(event: K, data: T[K]): void {
        const eventListeners = this.listeners.get(event);
        if (eventListeners) {
            eventListeners.forEach(listener => listener(data));
        }
    }
    
    once<K extends keyof T>(event: K, listener: EventListener<T[K]>): void {
        const onceListener = (data: T[K]) => {
            listener(data);
            this.off(event, onceListener);
        };
        this.on(event, onceListener);
    }
}

// 使用示例
interface AppEvents {
    userLogin: { userId: number; username: string };
    userLogout: { userId: number };
    dataUpdate: { table: string; id: number };
}

const eventEmitter = new EventEmitter<AppEvents>();

eventEmitter.on('userLogin', (data) => {
    console.log(`User ${data.username} logged in`);
});

eventEmitter.emit('userLogin', { userId: 1, username: 'alice' });
```

## 最佳实践

1. **合理使用访问修饰符**：遵循封装原则，适当隐藏内部实现
2. **优先组合而非继承**：避免过深的继承层次
3. **使用抽象类定义契约**：为相关类提供共同的基础结构
4. **善用泛型类**：提高代码复用性和类型安全性
5. **谨慎使用装饰器**：确保团队了解装饰器的工作原理

TypeScript 的类系统为面向对象编程提供了强大的类型支持，合理使用可以编写出更加健壮和可维护的代码。
