```typescript
type PathsToStringProps<T> = T extends string 
  ? [] 
  : {
      [K in Extract<keyof T, string>]: T[K] extends string 
        ? [K] | [K, ...PathsToStringProps<T[K]>]
        : [K, ...PathsToStringProps<T[K]>];
    }[Extract<keyof T, string>];

      ? R extends readonly string[]
        ? R['length'] extends 0
          ? F
          : `${F}${D}${Join<R, D>}`

      : never
    : '';
```

```typescript
// 方法装饰器 - 性能监控
function performance(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
  const originalMethod = descriptor.value;
  
  descriptor.value = function (...args: any[]) {
    const start = performance.now();
    const result = originalMethod.apply(this, args);
    const end = performance.now();
    
    console.log(`${propertyKey} 执行时间: ${end - start}ms`);
    return result;
  };
  
  return descriptor;
}

// 类装饰器 - 自动注册
const registry = new Map<string, any>();

function autoRegister<T extends { new (...args: any[]): {} }>(constructor: T) {
  registry.set(constructor.name, constructor);
  return constructor;
}

// 属性装饰器 - 验证
function validate(validationFn: (value: any) => boolean) {
  return function (target: any, propertyKey: string) {
    let value = target[propertyKey];
    
    const getter = () => value;
    const setter = (newValue: any) => {
      if (!validationFn(newValue)) {
        throw new Error(`Invalid value for ${propertyKey}: ${newValue}`);
      }
      value = newValue;
    };
    
    Object.defineProperty(target, propertyKey, {
      get: getter,
      set: setter,
      enumerable: true,
      configurable: true
    });
  };
}

// 使用示例
@autoRegister
class UserService {
  @validate((value: string) => value.length > 0)
  private name: string = '';
  
  @performance
  async fetchUser(id: string): Promise<User> {
    // 模拟异步操作
    await new Promise(resolve => setTimeout(resolve, 100));
    return { id, name: this.name } as User;
  }
}
```

### 2.2 反射和元数据

```typescript
import 'reflect-metadata';

// 元数据键
const METADATA_KEY = {
  INJECTABLE: Symbol('injectable'),
  PARAM_TYPES: Symbol('paramTypes'),
  DESIGN_TYPE: Symbol('design:type'),
  DESIGN_PARAM_TYPES: Symbol('design:paramtypes'),
  DESIGN_RETURN_TYPE: Symbol('design:returntype')
};

// 依赖注入装饰器
function Injectable<T extends { new (...args: any[]): {} }>(constructor: T) {
  Reflect.defineMetadata(METADATA_KEY.INJECTABLE, true, constructor);
  return constructor;
}

function Inject(token: string) {
  return function (target: any, propertyKey: string | symbol | undefined, parameterIndex: number) {
    const existingTokens = Reflect.getMetadata(METADATA_KEY.PARAM_TYPES, target) || [];
    existingTokens[parameterIndex] = token;
    Reflect.defineMetadata(METADATA_KEY.PARAM_TYPES, existingTokens, target);
  };
}

// 简单的 IoC 容器
class Container {
  private services = new Map<string, any>();
  private instances = new Map<string, any>();
  
  register<T>(token: string, constructor: new (...args: any[]) => T): void {
    this.services.set(token, constructor);
  }
  
  resolve<T>(token: string): T {
    if (this.instances.has(token)) {
      return this.instances.get(token);
    }
    
    const constructor = this.services.get(token);
    if (!constructor) {
      throw new Error(`Service ${token} not found`);
    }
    
    const paramTypes = Reflect.getMetadata(METADATA_KEY.PARAM_TYPES, constructor) || [];
    const dependencies = paramTypes.map((paramToken: string) => this.resolve(paramToken));
    
    const instance = new constructor(...dependencies);
    this.instances.set(token, instance);
    
    return instance;
  }
}

// 使用示例
@Injectable
class DatabaseService {
  connect() {
    console.log('Connected to database');
  }
}

@Injectable
class UserRepository {
  constructor(@Inject('DatabaseService') private db: DatabaseService) {}
  
  findUser(id: string) {
    this.db.connect();
    return { id, name: 'User' };
  }
}

const container = new Container();
container.register('DatabaseService', DatabaseService);
container.register('UserRepository', UserRepository);

const userRepo = container.resolve<UserRepository>('UserRepository');
```

```typescript
// 类型安全的 Builder 模式
interface QueryBuilder<T = {}> {
  select<K extends string>(fields: K[]): QueryBuilder<T & Record<K, any>>;
  where<K extends keyof T>(field: K, value: T[K]): QueryBuilder<T>;
  orderBy<K extends keyof T>(field: K, direction?: 'ASC' | 'DESC'): QueryBuilder<T>;
  limit(count: number): QueryBuilder<T>;
  build(): Query<T>;
}

interface Query<T> {
  fields: (keyof T)[];
  conditions: Array<{ field: keyof T; value: any }>;
  ordering: Array<{ field: keyof T; direction: 'ASC' | 'DESC' }>;
  limitCount?: number;
}

class SqlQueryBuilder<T = {}> implements QueryBuilder<T> {
  private query: Partial<Query<T>> = {
    fields: [],
    conditions: [],
    ordering: []
  };
  
  select<K extends string>(fields: K[]): QueryBuilder<T & Record<K, any>> {
    this.query.fields = [...(this.query.fields || []), ...fields];
    return this as any;
  }
  
  where<K extends keyof T>(field: K, value: T[K]): QueryBuilder<T> {
    this.query.conditions = [...(this.query.conditions || []), { field, value }];
    return this;
  }
  
  orderBy<K extends keyof T>(field: K, direction: 'ASC' | 'DESC' = 'ASC'): QueryBuilder<T> {
    this.query.ordering = [...(this.query.ordering || []), { field, direction }];
    return this;
  }
  
  limit(count: number): QueryBuilder<T> {
    this.query.limitCount = count;
    return this;
  }
  
  build(): Query<T> {
    return this.query as Query<T>;
  }
}

// 使用示例
interface User {
  id: number;
  name: string;
  email: string;
  age: number;
}

const query = new SqlQueryBuilder<User>()
  .select(['id', 'name', 'email'])
  .where('age', 25)
  .orderBy('name', 'ASC')
  .limit(10)
  .build();
```

### 3.2 状态机模式

```typescript
// 类型安全的状态机
type StateTransitions<States extends string, Events extends string> = {
  [S in States]: {
    [E in Events]?: States;
  };
};

type StateMachineConfig<States extends string, Events extends string> = {
  initial: States;
  transitions: StateTransitions<States, Events>;
};

class StateMachine<States extends string, Events extends string> {
  private currentState: States;
  
  constructor(private config: StateMachineConfig<States, Events>) {
    this.currentState = config.initial;
  }
  
  getCurrentState(): States {
    return this.currentState;
  }
  
  canTransition(event: Events): boolean {
    const transitions = this.config.transitions[this.currentState];
    return transitions && event in transitions;
  }
  
  transition(event: Events): boolean {
    if (!this.canTransition(event)) {
      return false;
    }
    
    const nextState = this.config.transitions[this.currentState][event];
    if (nextState) {
      this.currentState = nextState;
      return true;
    }
    
    return false;
  }
}

// 使用示例 - 订单状态机
type OrderState = 'pending' | 'confirmed' | 'shipped' | 'delivered' | 'cancelled';
type OrderEvent = 'confirm' | 'ship' | 'deliver' | 'cancel';

const orderStateMachine = new StateMachine<OrderState, OrderEvent>({
  initial: 'pending',
  transitions: {
    pending: {
      confirm: 'confirmed',
      cancel: 'cancelled'
    },
    confirmed: {
      ship: 'shipped',
      cancel: 'cancelled'
    },
    shipped: {
      deliver: 'delivered'
    },
    delivered: {},
    cancelled: {}
  }
});

console.log(orderStateMachine.getCurrentState()); // 'pending'
orderStateMachine.transition('confirm'); // true
console.log(orderStateMachine.getCurrentState()); // 'confirmed'
```

### 3.3 观察者模式

```typescript
// 类型安全的观察者模式
type EventMap = Record<string, any>;

interface Observer<T> {
  update(data: T): void;
}

interface Observable<T extends EventMap> {
  subscribe<K extends keyof T>(event: K, observer: Observer<T[K]>): void;
  unsubscribe<K extends keyof T>(event: K, observer: Observer<T[K]>): void;
  notify<K extends keyof T>(event: K, data: T[K]): void;
}

class EventEmitter<T extends EventMap> implements Observable<T> {
  private observers: {
    [K in keyof T]?: Set<Observer<T[K]>>;
  } = {};
  
  subscribe<K extends keyof T>(event: K, observer: Observer<T[K]>): void {
    if (!this.observers[event]) {
      this.observers[event] = new Set();
    }
    this.observers[event]!.add(observer);
  }
  
  unsubscribe<K extends keyof T>(event: K, observer: Observer<T[K]>): void {
    this.observers[event]?.delete(observer);
  }
  
  notify<K extends keyof T>(event: K, data: T[K]): void {
    this.observers[event]?.forEach(observer => observer.update(data));
  }
}

// 使用示例
interface AppEvents {
  userLogin: { userId: string; timestamp: Date };
  userLogout: { userId: string };
  dataUpdate: { table: string; id: string; data: any };
}

class UserService implements Observer<AppEvents['userLogin']> {
  update(data: AppEvents['userLogin']): void {
    console.log(`User ${data.userId} logged in at ${data.timestamp}`);
  }
}

class AuditService implements Observer<AppEvents['userLogin']> {
  update(data: AppEvents['userLogin']): void {
    console.log(`Audit: Login event for user ${data.userId}`);
  }
}

const eventEmitter = new EventEmitter<AppEvents>();
const userService = new UserService();
const auditService = new AuditService();

eventEmitter.subscribe('userLogin', userService);
eventEmitter.subscribe('userLogin', auditService);

eventEmitter.notify('userLogin', {
  userId: 'user123',
  timestamp: new Date()
});
```

```typescript
type HttpMethod = 'GET' | 'POST' | 'PUT' | 'DELETE' | 'PATCH';

interface ApiEndpoint<
  Method extends HttpMethod,
  Path extends string,
  RequestBody = never,
  ResponseBody = unknown,
  QueryParams = never
> {
  method: Method;
  path: Path;
  requestBody: RequestBody;
  responseBody: ResponseBody;
  queryParams: QueryParams;
}

// 具体 API 定义
type UserApi = {
  getUsers: ApiEndpoint<'GET', '/users', never, User[], { page?: number; limit?: number }>;
  getUser: ApiEndpoint<'GET', '/users/:id', never, User>;
  createUser: ApiEndpoint<'POST', '/users', CreateUserRequest, User>;
  updateUser: ApiEndpoint<'PUT', '/users/:id', UpdateUserRequest, User>;
  deleteUser: ApiEndpoint<'DELETE', '/users/:id', never, void>;
};

// 类型安全的 API 客户端
class TypedApiClient<T extends Record<string, ApiEndpoint<any, any, any, any, any>>> {
  constructor(private baseUrl: string) {}
  
  async call<K extends keyof T>(
    endpoint: K,
    options: {
      params?: ExtractPathParams<T[K]['path']>;
      query?: T[K]['queryParams'];
      body?: T[K]['requestBody'];
    } = {}
  ): Promise<T[K]['responseBody']> {
    const api = this.getEndpointConfig(endpoint);
    const url = this.buildUrl(api.path, options.params, options.query);
    
    const response = await fetch(url, {
      method: api.method,
      headers: { 'Content-Type': 'application/json' },
      body: options.body ? JSON.stringify(options.body) : undefined
    });
    
    if (!response.ok) {
      throw new Error(`API call failed: ${response.statusText}`);
    }
    
    return response.json();
  }
  
  private getEndpointConfig(endpoint: keyof T): any {
    // 实际实现中需要配置映射
    return {};
  }
  
  private buildUrl(path: string, params?: any, query?: any): string {
    let url = `${this.baseUrl}${path}`;
    
    // 替换路径参数
    if (params) {
      Object.entries(params).forEach(([key, value]) => {
        url = url.replace(`:${key}`, String(value));
      });
    }
    
    // 添加查询参数
    if (query) {
      const searchParams = new URLSearchParams();
      Object.entries(query).forEach(([key, value]) => {
        if (value !== undefined) {
          searchParams.append(key, String(value));
        }
      });
      const queryString = searchParams.toString();
      if (queryString) {
        url += `?${queryString}`;
      }
    }
    
    return url;
  }
}

// 路径参数提取工具类型
type ExtractPathParams<T extends string> = 
  T extends `${string}:${infer Param}/${infer Rest}`
    ? { [K in Param]: string } & ExtractPathParams<Rest>
    : T extends `${string}:${infer Param}`
    ? { [K in Param]: string }
    : {};

// 使用示例
const apiClient = new TypedApiClient<UserApi>('https://api.example.com');

// 类型安全的 API 调用

  query: { page: 1, limit: 10 }
});

const user = await apiClient.call('getUser', {
  params: { id: '123' }
});

  body: {
    name: 'John Doe',
    email: 'john@example.com'
  }
```

```typescript
// GraphQL 查询类型定义
interface GraphQLQuery<TVariables = {}, TResult = any> {
  query: string;
  variables?: TVariables;
  result: TResult;
}

// 查询构建器
class GraphQLQueryBuilder<TResult = {}> {
  private fields: string[] = [];
  private fragments: string[] = [];
  
  field<K extends string>(name: K): GraphQLQueryBuilder<TResult & Record<K, any>> {
    this.fields.push(name);
    return this as any;
  }
  
  fragment<T>(name: string, on: string, fields: (keyof T)[]): GraphQLQueryBuilder<TResult> {
    this.fragments.push(`
      fragment ${name} on ${on} {
        ${fields.join('\n        ')}
      }
    `);
    return this;
  }
  
  build<TVariables = {}>(
    operationName: string,
    variables?: TVariables
  ): GraphQLQuery<TVariables, TResult> {
    const query = `
      ${this.fragments.join('\n')}
      query ${operationName} {
        ${this.fields.join('\n        ')}
      }
    `;
    
    return {
      query: query.trim(),
      variables,
      result: {} as TResult
    };
  }
}

// 使用示例
interface User {
  id: string;
  name: string;
  email: string;
  posts: Post[];
}

interface Post {
  id: string;
  title: string;
  content: string;
}

const userQuery = new GraphQLQueryBuilder()
  .field('id')
  .field('name')
  .field('email')
  .build('GetUser');

// 类型安全的 GraphQL 客户端
class GraphQLClient {
  constructor(private endpoint: string) {}
  
  async execute<TVariables, TResult>(
    query: GraphQLQuery<TVariables, TResult>
  ): Promise<TResult> {
    const response = await fetch(this.endpoint, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        query: query.query,
        variables: query.variables
      })
    });
    
    const result = await response.json();
    
    if (result.errors) {
      throw new Error(`GraphQL errors: ${JSON.stringify(result.errors)}`);
    }
    
    return result.data;
  }
}
```

1. **类型系统深度利用** - 掌握模板字面量、递归类型等高级特性
2. **元编程能力** - 使用装饰器和反射实现强大的框架功能
3. **设计模式实现** - 用类型系统确保设计模式的正确性
4. **API 类型安全** - 构建端到端的类型安全 API
5. **性能优化** - 平衡类型安全和编译性能
6. **调试技巧** - 掌握类型调试和开发工具的使用

通过这些进阶技巧，您可以充分发挥 TypeScript 的威力，构建更加健壮和可维护的应用程序。
