## 1. 什么是声明合并？

声明合并是 TypeScript 的一个重要特性，它允许编译器将多个同名的声明合并成一个定义。这个特性使得我们可以扩展现有的类型定义，为库添加新的功能，或者在不同的文件中分别定义同一个类型的不同部分。

## 2. 接口合并

### 2.1 基本接口合并

```typescript
interface User {
  name: string;
}

interface User {
  age: number;
}

interface User {
  email: string;
}

// 合并后的接口等价于：
// interface User {
//   name: string;
//   age: number;
//   email: string;
// }

const user: User = {
  name: "Alice",
  age: 30,
  email: "alice@example.com"
};
```

### 2.2 方法合并

```typescript
interface Calculator {
  add(a: number, b: number): number;
}

interface Calculator {
  add(a: string, b: string): string;
  subtract(a: number, b: number): number;
}

// 合并后支持方法重载
const calc: Calculator = {
  add: (a: any, b: any) => {
    if (typeof a === "string" && typeof b === "string") {
      return a + b;
    }
    return Number(a) + Number(b);
  },
  subtract: (a: number, b: number) => a - b
};

const result1 = calc.add(1, 2);        // number
const result2 = calc.add("Hello", " World"); // string
```

### 2.3 泛型接口合并

```typescript
interface Container<T> {
  value: T;
}

interface Container<T> {
  getValue(): T;
  setValue(value: T): void;
}

class StringContainer implements Container<string> {
  constructor(public value: string) {}
  
  getValue(): string {
    return this.value;
  }
  
  setValue(value: string): void {
    this.value = value;
  }
}
```

### 2.4 接口合并的限制

```typescript
interface Config {
  apiUrl: string;

interface Config {
```

## 3. 命名空间合并

### 3.1 基本命名空间合并

```typescript
namespace Utilities {
  export function formatDate(date: Date): string {
    return date.toISOString().split('T')[0];
  }
}

namespace Utilities {
  export function formatTime(date: Date): string {
    return date.toTimeString().split(' ')[0];
  }
}

namespace Utilities {
  export const DEFAULT_LOCALE = "en-US";
}

// 使用合并后的命名空间
const formattedDate = Utilities.formatDate(new Date());
const formattedTime = Utilities.formatTime(new Date());
console.log(Utilities.DEFAULT_LOCALE);
```

### 3.2 命名空间与类合并

```typescript

  }

namespace Album {

  }

// 使用
```

### 3.3 命名空间与函数合并

```typescript
function buildLabel(name: string): string {
  return buildLabel.prefix + name + buildLabel.suffix;
}

namespace buildLabel {
  export let suffix = "";
  export let prefix = "Hello, ";
  
  export function setPrefix(newPrefix: string) {
    prefix = newPrefix;
  }
}

// 使用
console.log(buildLabel("TypeScript")); // "Hello, TypeScript"
buildLabel.setPrefix("Hi, ");
buildLabel.suffix = "!";
console.log(buildLabel("World")); // "Hi, World!"
```

### 3.4 命名空间与枚举合并

```typescript
enum Color {
  Red,
  Green,
  Blue
}

namespace Color {
  export function mix(color1: Color, color2: Color): string {
    return `Mixed ${Color[color1]} and ${Color[color2]}`;

// 使用
```

## 4. 模块增强

### 4.1 全局模块增强

```typescript
// 扩展全局 Array 接口
declare global {
  interface Array<T> {
    first(): T | undefined;
    last(): T | undefined;
  }
}

// 实现扩展方法
Array.prototype.first = function<T>(this: T[]): T | undefined {
  return this.length > 0 ? this[0] : undefined;
};

Array.prototype.last = function<T>(this: T[]): T | undefined {
  return this.length > 0 ? this[this.length - 1] : undefined;
};

// 使用
const numbers = [1, 2, 3, 4, 5];
console.log(numbers.first()); // 1
console.log(numbers.last());  // 5
```

## 5. 实际应用示例

### 5.1 插件系统设计

```typescript
// 核心接口
interface PluginSystem {
  plugins: Plugin[];
  register(plugin: Plugin): void;
}

interface Plugin {
  name: string;
  version: string;
  init(): void;
}

// 基础实现
const pluginSystem: PluginSystem = {
  plugins: [],
  register(plugin: Plugin) {
    this.plugins.push(plugin);
    plugin.init();
  }
};

// 通过声明合并扩展插件系统
interface PluginSystem {
  // 添加新的方法
  unregister(pluginName: string): boolean;
  getPlugin(name: string): Plugin | undefined;
  listPlugins(): string[];
}

// 扩展实现
Object.assign(pluginSystem, {
  unregister(pluginName: string): boolean {
    const index = this.plugins.findIndex(p => p.name === pluginName);
    if (index !== -1) {
      this.plugins.splice(index, 1);
      return true;
    }
    return false;
  },
  
  getPlugin(name: string): Plugin | undefined {
    return this.plugins.find(p => p.name === name);
  },
  
  listPlugins(): string[] {
    return this.plugins.map(p => p.name);
  }
});

// 进一步扩展插件接口
interface Plugin {
  // 添加可选的生命周期方法
  destroy?(): void;
  configure?(options: any): void;
}
```

### 5.2 API 客户端扩展

```typescript
// 基础 API 客户端
interface ApiClient {
  baseUrl: string;
  get<T>(endpoint: string): Promise<T>;
  post<T>(endpoint: string, data: any): Promise<T>;
}

class HttpClient implements ApiClient {
  constructor(public baseUrl: string) {}
  
  async get<T>(endpoint: string): Promise<T> {
    const response = await fetch(`${this.baseUrl}${endpoint}`);
    return response.json();
  }
  
  async post<T>(endpoint: string, data: any): Promise<T> {
    const response = await fetch(`${this.baseUrl}${endpoint}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    });
    return response.json();
  }
}

// 通过声明合并添加认证功能
interface ApiClient {
  token?: string;
  setToken(token: string): void;
  clearToken(): void;
}

// 扩展实现
Object.assign(HttpClient.prototype, {
  setToken(token: string) {
    this.token = token;
  },
  
  clearToken() {
    this.token = undefined;
  }
});

// 重写原有方法以支持认证
const originalGet = HttpClient.prototype.get;
const originalPost = HttpClient.prototype.post;

  // 添加认证头
  if (this.token) {
    // 实现认证逻辑
  }
  return originalGet.call(this, endpoint);
};

HttpClient.prototype.post = function<T>(endpoint: string, data: any): Promise<T> {
  // 添加认证头
  if (this.token) {
    // 实现认证逻辑
  }
  return originalPost.call(this, endpoint, data);
```

## 6. 声明合并最佳实践

### 6.1 合并顺序和优先级

```typescript
// 文件 1
interface User {
  name: string;
  age: number;
}

// 文件 2
interface User {
  email: string;
  age: number; // 必须与之前的声明类型一致
}

// 文件 3
interface User {
  // 方法重载按声明顺序排列
  greet(): string;
  greet(formal: boolean): string;
}
```

### 6.2 避免冲突

```typescript
// ❌ 避免属性类型冲突
interface Config {
  timeout: number;
}

interface Config {
  // timeout: string; // 错误：类型不匹配
  timeout: number; // 正确：类型一致
}

// ✅ 使用命名空间避免全局污染
namespace MyLibrary {
  export interface Config {
    apiKey: string;
  }
}

namespace MyLibrary {
  export interface Config {
    endpoint: string;
  }
}
```

## 8. 总结

声明合并是 TypeScript 的强大特性，它允许我们：

1. **接口合并** - 在多个地方定义同一个接口的不同部分
2. **命名空间合并** - 扩展命名空间的功能
3. **模块增强** - 为现有模块添加新的类型定义
4. **全局增强** - 扩展全局类型和接口

声明合并特别适用于：
- 扩展第三方库的类型定义
- 创建可扩展的插件系统
- 模块化的类型定义
- 渐进式的 API 设计
